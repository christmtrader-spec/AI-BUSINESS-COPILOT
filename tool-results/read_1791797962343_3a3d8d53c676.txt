  1100→    const line = await tx.salesQuotationLine.findFirst({
  1101→      where: { id: input.lineId, salesQuotationId: header.id }, // scoped via parent
  1102→      select: { id: true, qty: true, unitPriceMinor: true },
  1103→    });
  1104→    if (!line) {
  1105→      throw new ApiError(404, SALES_QUOTATION_ERROR_CODES.LINE_NOT_FOUND, "Quotation line not found.");
  1106→    }
  1107→
  1108→    const qty = input.qty ?? line.qty;
  1109→    const unitPriceMinor = input.unitPriceMinor ?? line.unitPriceMinor;
  1110→
  1111→    // §11 — the new line total must be exact.
  1112→    try {
  1113→      computeQuotationLineTotalMinor(qty, unitPriceMinor);
  1114→    } catch (error) {
  1115→      if (error instanceof Error && error.name === "SalesQuotationTotalOverflowError") {
  1116→        throw new ApiError(
  1117→          422,
  1118→          SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW,
  1119→          "The requested quantity × price exceeds the supported total range.",
  1120→        );
  1121→      }
  1122→      throw error;
  1123→    }
  1124→
  1125→    const updated = await tx.salesQuotationLine.update({
  1126→      where: { id: line.id },
  1127→      data: { qty, unitPriceMinor },
  1128→      select: {
  1129→        id: true,
  1130→        productId: true,
  1131→        productNameAtQuotation: true,
  1132→        skuAtQuotation: true,
  1133→        unitAtQuotation: true,
  1134→        qty: true,
  1135→        unitPriceMinor: true,
  1136→        createdAt: true,
  1137→        updatedAt: true,
  1138→      },
  1139→    });
  1140→
  1141→    // §11 — discount law re-check against the new subtotal.
  1142→    const lines = await loadCompleteLines(tx, header.id, cap);
  1143→    const subtotalMinor = computeQuotationSubtotalMinor(lines).subtotalMinor;
  1144→    if (header.discountMinor > subtotalMinor) {
  1145→      throw new ApiError(
  1146→        422,
  1147→        SALES_QUOTATION_ERROR_CODES.INVALID_DISCOUNT,
  1148→        "Discount cannot exceed the quotation subtotal.",
  1149→      );
  1150→    }
  1151→
  1152→    await audit(
  1153→      auditInput(
  1154→        input.businessId,
  1155→        input.userId,
  1156→        "sales_quotation.line_update",
  1157→        header.id,
  1158→        header.reference,
  1159→        {
  1160→          lineId: line.id,
  1161→          qty: { from: line.qty, to: updated.qty },
  1162→          unitPriceMinor: { from: line.unitPriceMinor, to: updated.unitPriceMinor },
  1163→        },
  1164→      ),
  1165→      tx,
  1166→    );
  1167→
  1168→    return { id: updated.id };
  1169→  });
  1170→}

  1231→/* ------------------------------ lifecycle transitions --------------------- */
  1232→
  1233→/**
  1234→ * The shared in-transaction lifecycle gate (§8): reload + canonical matrix
  1235→ * + optional validity precondition. Every transition endpoint funnels
  1236→ * through here so the law lives in exactly ONE place.
  1237→ */
  1238→async function gateTransition(
  1239→  tx: Tx,
  1240→  input: { businessId: string; id: string; timezone: string },
  1241→  to: SalesQuotationStatus,
  1242→): Promise<{
  1243→  header: Awaited<ReturnType<typeof loadQuotationHeader>>;
  1244→  lines: Awaited<ReturnType<typeof loadCompleteLines>>;
  1245→}> {
  1246→  const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
  1247→  // Reload INSIDE the transaction; never trust an earlier GET (§8).
  1248→  const header = await loadQuotationHeader(tx, input.businessId, input.id);
  1249→  if (!canTransitionSalesQuotation(header.status, to)) {
  1250→    throw new ApiError(
  1251→      409,
  1252→      SALES_QUOTATION_ERROR_CODES.INVALID_TRANSITION,
  1253→      `Cannot move a sales quotation from ${header.status} to ${to}.`,
  1254→    );
  1255→  }
  1256→  const lines = await loadCompleteLines(tx, header.id, cap);
  1257→  return { header, lines };
  1258→}

  1261→ * §10/§D — mark a DRAFT quotation SENT. Requires at least one line (an
  1262→ * empty offer can never be sent §10). sentAt is SERVER-generated inside

  1264→ * offer was sent OUTSIDE the application — the application does NOT
  1265→ * deliver email (§2.7) and the audit/UI never claims it did. After send,

  1280→        422,
  1281→        SALES_QUOTATION_ERROR_CODES.EMPTY,
  1282→        "Add at least one line before marking this quotation sent.",
  1283→      );
  1284→    }

  1516→/* --------------------------- conversion (§13) ----------------------------- */
  1517→
  1518→/**
  1519→ * §13 — THE conversion: an ACCEPTED quotation becomes exactly ONE
  1520→ * CONFIRMED SalesOrder. ONE atomic transaction:
  1521→ *   1. reload the quotation tenant-scoped;
  1522→ *   2. already-converted guard FIRST (the ONE stable conflict for every
  1523→ *      loser of any race — the P45 D-5 lesson), backed by the DB UNIQUE
  1524→ *      constraint on SalesOrder.sourceQuotationId (§7);
  1525→ *   3. persisted status law (ACCEPTED only — invalid → controlled 500,
  1526→ *      wrong-but-valid status → 409);
  1527→ *   4. complete, materially valid line set (§11);
  1528→ *   5. canonical totals recomputed from authoritative persisted lines;
  1529→ *   6. every snapshot validated (§12);
  1530→ *   7. ONE SalesOrder created CONFIRMED — customer snapshot, currency
  1531→ *      snapshot, expected date, discount copied EXACTLY; orderDate =
  1532→ *      current business-local date (established midnoon rule);
  1533→ *      internalNotes → SalesOrder.notes (staff-visible order notes —
  1534→ *      never the customer print view §16);
  1535→ *   8. ONE SalesOrderLine per quotation line — productId + frozen
  1536→ *      snapshots + exact qty/unitPriceMinor (NEVER re-priced from live
  1537→ *      Products §12/§13);
  1538→ *   9. order totals derived via the EXISTING SalesOrder domain functions;
  1539→ *  10. quotation → CONVERTED + convertedAt;
  1540→ *  11. audit ×2 (sales_order.create + sales_quotation.convert) in-tx;
  1541→ *  12. commit both documents together.
  1542→ *
  1543→ * NO Sale invoice, NO Payment, NO receivable, NO InventoryAdjustment, NO
  1544→ * stock reservation, NO Product.stockQty change (§13 invariants). NO
  1545→ * notification claims an invoice or payment occurred.
  1546→ */
  1547→export async function convertSalesQuotationToOrder(input: {
  1548→  businessId: string;
  1549→  userId: string;
  1550→  id: string;
  1551→  timezone: string;
  1552→  ip?: string | null;
  1553→  userAgent?: string | null;
  1554→}) {
  1555→  return withWriteTx(async (tx) => {
  1556→    const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
  1557→    // 1. tenant-scoped reload INSIDE the transaction (§13 step 1).
  1558→    const header = await loadQuotationHeader(tx, input.businessId, input.id);
  1559→
  1560→    // 2. already-converted guard BEFORE the status gate (§13/D-5 lesson):
  1561→    //    every loser — including a retry after the winner committed —
  1562→    //    receives the ONE documented stable conflict code.
  1563→    const existingOrder = await tx.salesOrder.findFirst({
  1564→      where: { sourceQuotationId: header.id },
  1565→      select: { id: true, reference: true },
  1566→    });
  1567→    if (existingOrder) {
  1568→      throw new ApiError(
  1569→        409,
  1570→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1571→        "This quotation has already been converted to a sales order.",
  1572→      );
  1573→    }
  1574→
  1575→    // 3. persisted status law: an ACCEPTED quotation converts; a
  1576→    //    valid-but-wrong status is an ordinary conflict; a corrupted
  1577→    //    persisted status is the controlled integrity 500 (parseDbStatus).
  1578→    if (header.status !== "ACCEPTED") {
  1579→      if (
  1580→        header.status === "REJECTED" ||
  1581→        header.status === "EXPIRED" ||
  1582→        header.status === "CANCELLED"
  1583→      ) {
  1584→        throw new ApiError(
  1585→          409,
  1586→          SALES_QUOTATION_ERROR_CODES.NOT_ACCEPTED,
  1587→          "Only ACCEPTED sales quotations can be converted.",
  1588→        );
  1589→      }
  1590→      throw new ApiError(
  1591→        409,
  1592→        SALES_QUOTATION_ERROR_CODES.INVALID_STATUS,
  1593→        "Only ACCEPTED sales quotations can be converted.",
  1594→      );
  1595→    }
  1596→    // §9 — ACCEPTED never expires; conversion remains legal after the
  1597→    // original validity date passes because acceptance was already
  1598→    // recorded (documented §9 law).
  1599→
  1600→    // 4–5. complete line set + canonical totals from persisted lines.
  1601→    const lines = await loadCompleteLines(tx, header.id, cap);
  1602→    if (lines.length === 0) {
  1603→      // An ACCEPTED quotation with zero lines is corruption (it could
  1604→      // never have passed mark-sent) — integrity, not a user error.
  1605→      mapPureSalesQuotationDomainError(
  1606→        new SalesQuotationIntegrityError(
  1607→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: accepted quotation without lines`,
  1608→        ),
  1609→      );
  1610→    }
  1611→    const quotationTotals = computeQuotationSubtotalMinor(lines);
  1612→    try {
  1613→      validateQuotationDiscount(header.discountMinor, quotationTotals.subtotalMinor);
  1614→    } catch {
  1615→      mapPureSalesQuotationDomainError(
  1616→        new SalesQuotationIntegrityError(
  1617→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: discount exceeds subtotal`,
  1618→        ),
  1619→      );
  1620→    }
  1621→    const quotationTotalMinor = computeQuotationTotalMinor(
  1622→      quotationTotals.subtotalMinor,
  1623→      header.discountMinor,
  1624→    );
  1625→
  1626→    // 6. every snapshot validated (§12/§13 step 6).
  1627→    validateHeaderSnapshots(header);
  1628→
  1629→    // Order reference: the established SO generator, distinct sequence
  1630→    // domain (§10 — quotation and order references never share a sequence).
  1631→    let orderReference: string | undefined;
  1632→    for (let attempt = 0; attempt < 5; attempt += 1) {
  1633→      const candidate = generateSalesOrderReference(new Date(), (max) => randomInt(max));
  1634→      const clash = await tx.salesOrder.findFirst({
  1635→        where: { businessId: input.businessId, reference: candidate },
  1636→        select: { id: true },
  1637→      });
  1638→      if (!clash) {
  1639→        orderReference = candidate;
  1640→        break;
  1641→      }
  1642→    }
  1643→    if (orderReference === undefined) {
  1644→      throw new ApiError(
  1645→        409,
  1646→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1647→        "Could not generate a unique order reference. Please try again.",
  1648→      );
  1649→    }
  1650→
  1651→    // §13 step 11 — orderDate is the CURRENT business-local date
  1652→    // normalized with the established UTC-midnoon rule (server clock).
  1653→    const orderDate = new Date(
  1654→      `${businessLocalCalendarDate(new Date(), input.timezone)}T12:00:00.000Z`,
  1655→    );
  1656→    const confirmedAt = new Date();
  1657→
  1658→    // 7. ONE SalesOrder — CONFIRMED, source link established (§13 steps
  1659→    //    8–12/19/20). Customer/currency snapshots copied EXACTLY.
  1660→    let order;
  1661→    try {
  1662→      order = await tx.salesOrder.create({
  1663→        data: {
  1664→          businessId: input.businessId, // server-derived
  1665→          customerId: header.customerId,
  1666→          customerNameAtOrder: header.customerNameAtQuotation, // frozen §12
  1667→          reference: orderReference,
  1668→          status: "CONFIRMED", // acceptance = commercial commitment (§13)
  1669→          currencyCodeAtOrder: header.currencyCodeAtQuotation,
  1670→          currencySymbolAtOrder: header.currencySymbolAtQuotation,
  1671→          decimalPrecisionAtOrder: header.decimalPrecisionAtQuotation,
  1672→          orderDate,
  1673→          expectedDate: header.expectedDeliveryDate,
  1674→          discountMinor: header.discountMinor, // copied EXACTLY (§13 step 13)
  1675→          notes: header.internalNotes, // staff notes continuity (§13 step 14)
  1676→          createdByUserId: input.userId,
  1677→          confirmedAt, // §13 step 20 — server-side transaction time
  1678→          sourceQuotationId: header.id, // §13 step 23 — the ONE authority
  1679→        },
  1680→        select: { id: true, reference: true, status: true, confirmedAt: true },
  1681→      });
  1682→    } catch (error) {
  1683→      // The DB unique constraint is the FINAL race authority (§7/§13):
  1684→      // a concurrent winner (or an impossible prior conversion) surfaces
  1685→      // as the ONE stable conflict — never a second order.
  1686→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
  1687→        throw new ApiError(
  1688→          409,
  1689→          SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1690→          "This quotation has already been converted to a sales order.",
  1691→        );
  1692→      }
  1693→      throw error;
  1694→    }
  1695→
  1696→    // 8. ONE SalesOrderLine per quotation line — the FROZEN snapshots and
  1697→    //    exact agreed prices (§13 steps 15–17; never re-read/re-priced).
  1698→    //    The order subtotal is re-derived via the EXISTING order domain
  1699→    //    functions so both documents provably agree (§13 step 18).
  1700→    let orderSubtotalMinor = 0;
  1701→    try {
  1702→      for (const line of lines) {
  1703→        await tx.salesOrderLine.create({
  1704→          data: {
  1705→            salesOrderId: order.id,
  1706→            productId: line.productId,
  1707→            orderedQty: line.qty, // exact quantity (§13 step 17)
  1708→            unitPriceMinor: line.unitPriceMinor, // exact agreed price
  1709→            productNameAtOrder: line.productNameAtQuotation,
  1710→            skuAtOrder: line.skuAtQuotation,
  1711→            unitAtOrder: line.unitAtQuotation,
  1712→          },
  1713→          select: { id: true },
  1714→        });
  1715→        orderSubtotalMinor += computeQuotationLineTotalMinor(line.qty, line.unitPriceMinor);
  1716→      }
  1717→      if (!Number.isSafeInteger(orderSubtotalMinor)) {
  1718→        throw new SalesQuotationIntegrityError(
  1719→          `${SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW}: order subtotal overflow during conversion`,
  1720→        );
  1721→      }
  1722→      // §13 step 18 — derive the order totals via the EXISTING domain
  1723→      // functions; quotation and order totals MUST agree exactly.
  1724→      const orderTotals = computeOrderSubtotalMinor(
  1725→        lines.map((l) => ({ orderedQty: l.qty, unitPriceMinor: l.unitPriceMinor })),
  1726→      );
  1727→      if (orderTotals.subtotalMinor !== quotationTotals.subtotalMinor) {
  1728→        mapPureSalesQuotationDomainError(
  1729→          new SalesQuotationIntegrityError(
  1730→            `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: quotation/order subtotal disagreement`,
  1731→          ),
  1732→        );
  1733→      }
  1734→    } catch (error) {
  1735→      // §13/H — a failure here aborts the WHOLE transaction: no partial
  1736→      // order, no partially copied lines, no CONVERTED quotation, no
  1737→      // stray audit (withWriteTx rolls back everything).
  1738→      throw error;
  1739→    }
  1740→
  1741→    // 10. quotation → CONVERTED + convertedAt (§13 steps 21–22).
  1742→    const convertedAt = new Date();
  1743→    await tx.salesQuotation.update({
  1744→      where: { id: header.id },
  1745→      data: { status: "CONVERTED", convertedAt },
  1746→      select: { id: true },
  1747→    });
  1748→
  1749→    // 11. audit ×2, SAME transaction (§13 step 24). Minimal safe payloads;
  1750→    //     never the notes contents.
  1751→    await audit(
  1752→      {
  1753→        businessId: input.businessId,
  1754→        actorId: input.userId,
  1755→        action: "sales_order.create",
  1756→        entityType: "SalesOrder",
  1757→        entityId: order.id,
  1758→        after: {
  1759→          reference: order.reference,
  1760→          sourceQuotationId: header.id,
  1761→          sourceQuotationReference: header.reference,
  1762→          status: order.status,
  1763→          lineCount: lines.length,
  1764→          subtotalMinor: quotationTotals.subtotalMinor,
  1765→          discountMinor: header.discountMinor,
  1766→          totalMinor: quotationTotalMinor,
  1767→        },
  1768→      },
  1769→      tx,
  1770→    );
  1771→    await audit(
  1772→      auditInput(input.businessId, input.userId, "sales_quotation.convert", header.id, header.reference, {
  1773→        salesOrderId: order.id,
  1774→        salesOrderReference: order.reference,
  1775→        lineCount: lines.length,
  1776→        subtotalMinor: quotationTotals.subtotalMinor,
  1777→        discountMinor: header.discountMinor,
  1778→        totalMinor: quotationTotalMinor,
  1779→        previousStatus: header.status,
  1780→        newStatus: "CONVERTED",
  1781→        convertedAt: convertedAt.toISOString(),
  1782→      }),
  1783→      tx,
  1784→    );
  1785→
  1786→    // 12. both documents commit together (§13 step 25).
  1787→    return {
  1788→      quotationId: header.id,
  1789→      quotationReference: header.reference,
  1790→      salesOrder: order,
  1791→    };
  1792→  });
  1793→}
