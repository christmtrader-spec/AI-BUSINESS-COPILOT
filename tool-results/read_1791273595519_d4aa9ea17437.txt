     3→/**
     4→ * Prompt 40 — stocktake service (server-only, §118).
     5→ *
     6→ * LAWS:
     7→ *  - Every query is tenant-scoped by ctx.businessId (§54) — never a client
     8→ *    businessId. Foreign stocktake/product → 404, no existence leakage (§55).
     9→ *  - Status transitions are server-authoritative through dedicated
    10→ *    operations (§9); invalid transitions → 422 INVALID_STOCKTAKE_TRANSITION.
    11→ *  - systemQtyAtCount is ALWAYS the server-read Product.stockQty at count
    12→ *    time (§15); the client supplies ONLY countedQty (§16); variance is
    13→ *    server-derived (§16/§18).
    14→ *  - Finalization is ONE atomic transaction (§39): stale-count check against
    15→ *    CURRENT stockQty (§40/§41 — never auto-rebased), InventoryAdjustment
    16→ *    rows for non-zero variances (§43), zero-variance lines skip the
    17→ *    adjustment (§44), no partial finalization (§156).
    18→ *  - FINALIZED/CANCELLED are immutable (§50/§51).
    19→ *  - Audit through the single sanctioned helper inside the same transaction
    20→ *    (§71); NO audit on reads (§69); safe metadata only (§70/§140).
    21→ */

    45→/* ---------------------------- shared internals ----------------------------- */
    46→
    47→function assertTransition(from: StocktakeStatus, to: StocktakeStatus): void {
    48→  if (!canTransitionStocktake(from, to)) {
    49→    throw new ApiError(
    50→      422,
    51→      STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
    52→      `Cannot move a stocktake from ${from} to ${to}.`,
    53→    );
    54→  }
    55→}

    57→async function loadStocktake(
    58→  tx: Tx,
    59→  businessId: string,
    60→  id: string,
    61→): Promise<{
    62→  id: string;
    63→  reference: string;
    64→  status: StocktakeStatus;
    65→  countType: string;
    66→  notes: string | null;
    67→  createdByUserId: string;
    68→  startedAt: Date | null;
    69→  reviewedAt: Date | null;
    70→  finalizedAt: Date | null;
    71→  canceledAt: Date | null;
    72→  createdAt: Date;
    73→}> {
    74→  const row = await tx.inventoryStocktake.findFirst({
    75→    where: { id, businessId }, // §54 — tenant scope from auth context only
    76→    select: {
    77→      id: true,
    78→      reference: true,
    79→      status: true,
    80→      countType: true,
    81→      notes: true,
    82→      createdByUserId: true,
    83→      startedAt: true,
    84→      reviewedAt: true,
    85→      finalizedAt: true,
    86→      canceledAt: true,
    87→      createdAt: true,
    88→    },
    89→  });
    90→  if (!row) throw new ApiError(404, "NOT_FOUND", "Stocktake not found.");
    91→  return { ...row, status: isStocktakeStatus(row.status) ? row.status : "DRAFT" };
    92→}

    94→function auditInput(
    95→  businessId: string,
    96→  actorId: string,
    97→  action: string,
    98→  stocktakeId: string,
    99→  reference: string,
   100→  after: Record<string, unknown>,
   101→  before?: Record<string, unknown>,
   102→) {
   103→  return {
   104→    businessId,
   105→    actorId,
   106→    action,
   107→    entityType: "InventoryStocktake",
   108→    entityId: stocktakeId,
   109→    ...(before ? { before } : {}),
   110→    after: { reference, ...after }, // §70 — safe metadata; never notes content
   111→  };
   112→}

   116→export async function createStocktake(input: {
   117→  businessId: string;
   118→  userId: string;
   119→  reference: string;
   120→  notes: string | null;
   121→  countType?: string;
   122→  ip?: string | null;
   123→  userAgent?: string | null;
   124→}) {
   125→  return db.$transaction(async (tx) => {
   126→    try {
   127→      const row = await tx.inventoryStocktake.create({
   128→        data: {
   129→          businessId: input.businessId,
   130→          reference: input.reference,
   131→          notes: input.notes,
   132→          countType: "SELECTED", // §24 — only SELECTED exists
   133→          status: "DRAFT",
   134→          createdByUserId: input.userId,
   135→        },
   136→        select: { id: true, reference: true, status: true },
   137→      });
   138→      await audit(
   139→        auditInput(input.businessId, input.userId, "inventory.stocktake.create", row.id, row.reference, {
   140→          status: row.status,
   141→          countType: "SELECTED",
   142→        }),
   143→        tx,
   144→      );
   145→      return row;
   146→    } catch (error) {
   147→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   148→        throw new ApiError(409, "DUPLICATE_REFERENCE", "A stocktake with this reference already exists.");
   149→      }
   150→      throw error;
   151→    }
   152→  });
   153→}

   156→
   157→export async function listStocktakes(input: {
   158→  businessId: string;
   159→  q?: string;
   160→  status?: string;
   161→  from?: string;
   162→  to?: string;
   163→  page: number;
   164→  pageSize: number;
   165→}) {
   166→  const where: Prisma.InventoryStocktakeWhereInput = { businessId: input.businessId };
   167→  if (input.q) where.reference = { contains: input.q };
   168→  if (input.status && isStocktakeStatus(input.status)) where.status = input.status;
   169→  if (input.from || input.to) {
   170→    where.createdAt = {};
   171→    if (input.from) where.createdAt.gte = new Date(`${input.from}T00:00:00.000Z`);
   172→    if (input.to) where.createdAt.lte = new Date(`${input.to}T23:59:59.999Z`);
   173→  }
   174→
   175→  const [rows, total] = await Promise.all([
   176→    db.inventoryStocktake.findMany({
   177→      where,
   178→      orderBy: { createdAt: "desc" }, // deterministic newest-first list (§134)
   179→      skip: (input.page - 1) * input.pageSize,
   180→      take: input.pageSize,
   181→      // §67 — list select minimization; a bounded line slice feeds the
   182→      // counted/net columns without loading full line tables.
   183→      select: {
   184→        id: true,
   185→        reference: true,
   186→        status: true,
   187→        countType: true,
   188→        createdAt: true,
   189→        createdByUser: { select: { id: true, name: true, email: true } },
   190→        _count: { select: { lines: true } },
   191→        lines: { select: { countedAt: true, varianceQty: true } },
   192→      },
   193→    }),
   194→    db.inventoryStocktake.count({ where }),
   195→  ]);
   196→
   197→  return {
   198→    items: rows.map(stocktakeListItemDto),
   199→    total,
   200→    page: input.page,
   201→    pageSize: input.pageSize,
   202→  };
   203→}

   205→/* -------------------------------- detail ---------------------------------- */
   206→
   207→export async function getStocktake(input: {
   208→  businessId: string;
   209→  id: string;
   210→  page: number;
   211→  pageSize: number;
   212→}) {
   213→  const header = await loadStocktake(db, input.businessId, input.id);
   214→  // §36 — summary over ALL lines (aggregate-light select, §66).
   215→  const [summaryRows, lineRows, lineTotal] = await Promise.all([
   216→    db.inventoryStocktakeLine.findMany({
   217→      where: { stocktakeId: header.id },
   218→      select: { countedAt: true, varianceQty: true },
   219→    }),
   220→    db.inventoryStocktakeLine.findMany({
   221→      where: { stocktakeId: header.id },
   222→      orderBy: [{ countedAt: "asc" }, { id: "asc" }], // deterministic ordering
   223→      skip: (input.page - 1) * input.pageSize,
   224→      take: input.pageSize,
   225→      select: {
   226→        id: true,
   227→        productId: true,
   228→        productNameAtCount: true,
   229→        skuAtCount: true,
   230→        unitAtCount: true,
   231→        systemQtyAtCount: true,
   232→        countedQty: true,
   233→        varianceQty: true,
   234→        countedAt: true,
   235→        countedByUser: { select: { id: true, name: true, email: true } },
   236→        // Current-product join: display fallback for UNCOUNTED preparation
   237→        // lines (their snapshot is still the empty placeholder). Counted
   238→        // lines always render their historical snapshot (dto decides).
   239→        product: { select: { name: true, sku: true, unit: true, stockQty: true } },
   240→      },
   241→    }),
   242→    db.inventoryStocktakeLine.count({ where: { stocktakeId: header.id } }),
   243→  ]);
   244→
   245→  return stocktakeDetailDto({
   246→    ...header,
   247→    createdByUser: await db.user.findUnique({
   248→      where: { id: header.createdByUserId },
   249→      select: { id: true, name: true, email: true },
   250→    }),
   251→    summary: computeStocktakeSummary(summaryRows),
   252→    lines: lineRows,
   253→    linesPage: { page: input.page, pageSize: input.pageSize, total: lineTotal },
   254→  });
   255→}

   257→/* ------------------------------- add line --------------------------------- */
   258→
   259→export async function addStocktakeLine(input: {
   260→  businessId: string;
   261→  userId: string;
   262→  stocktakeId: string;
   263→  productId: string;
   264→  ip?: string | null;
   265→  userAgent?: string | null;
   266→}) {
   267→  const cap = stocktakeLineCap(process.env.INVENTORY_STOCKTAKE_MAX_LINES);
   268→  return db.$transaction(async (tx) => {
   269→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   270→    if (stocktake.status !== "DRAFT") {
   271→      // §27 — DRAFT-only line preparation.
   272→      throw new ApiError(
   273→        422,
   274→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   275→        "Products can only be added while the stocktake is a draft.",
   276→      );
   277→    }
   278→    const lineCount = await tx.inventoryStocktakeLine.count({
   279→      where: { stocktakeId: stocktake.id },
   280→    });
   281→    if (lineCount >= cap) {
   282→      // §61 — bounded transaction sizes; surfaced to the UI (lines/max).
   283→      throw new ApiError(
   284→        422,
   285→        STOCKTAKE_ERROR_CODES.LINE_LIMIT,
   286→        `Stocktake line limit reached (${cap}).`,
   287→      );
   288→    }
   289→    // §20/§21/§56/§57 — same business, not deleted, active. Foreign/deleted
   290→    // product → 404 (no existence leakage); inactive → documented 422.
   291→    const product = await tx.product.findFirst({
   292→      where: { id: input.productId, businessId: input.businessId },
   293→      select: { id: true, deletedAt: true, isActive: true },
   294→    });
   295→    if (!product || product.deletedAt) {
   296→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   297→    }
   298→    if (!product.isActive) {
   299→      throw new ApiError(
   300→        422,
   301→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   302→        "Inactive products cannot be added to a stocktake.",
   303→      );
   304→    }
   305→    try {
   306→      const line = await tx.inventoryStocktakeLine.create({
   307→        data: {
   308→          stocktakeId: stocktake.id,
   309→          productId: product.id,
   310→          // Line is PREPARATION data until counted (§29) — zero placeholders.
   311→          systemQtyAtCount: 0,
   312→          countedQty: 0,
   313→          varianceQty: 0,
   314→          countedAt: new Date(0),
   315→          countedByUserId: input.userId,
   316→          productNameAtCount: "",
   317→          skuAtCount: null,
   318→          unitAtCount: null,
   319→        },
   320→        select: { id: true, productId: true },
   321→      });
   322→      await audit(
   323→        auditInput(input.businessId, input.userId, "inventory.stocktake.line_add", stocktake.id, stocktake.reference, {
   324→          lineId: line.id,
   325→          productId: product.id,
   326→        }),
   327→        tx,
   328→      );
   329→      return line;
   330→    } catch (error) {
   331→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   332→        // §28/§48 — DB uniqueness is the final protection.
   333→        throw new ApiError(
   334→          409,
   335→          STOCKTAKE_ERROR_CODES.DUPLICATE_PRODUCT,
   336→          "This product is already in the stocktake.",
   337→        );
   338→      }
   339→      throw error;
   340→    }
   341→  });
   342→}

   380→/* --------------------------------- count ---------------------------------- */
   381→
   382→export async function countStocktakeLine(input: {
   383→  businessId: string;
   384→  userId: string;
   385→  stocktakeId: string;
   386→  lineId: string;
   387→  countedQty: number;
   388→  ip?: string | null;
   389→  userAgent?: string | null;
   390→}) {
   391→  return db.$transaction(async (tx) => {
   392→    // §30 — load stocktake → verify status (COUNTING|REVIEW, §31) → verify
   393→    // line → verify product tenant/active → read current stock → snapshot.
   394→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   395→    if (stocktake.status !== "COUNTING" && stocktake.status !== "REVIEW") {
   396→      throw new ApiError(
   397→        422,
   398→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   399→        "Counting is only possible while the stocktake is counting or in review.",
   400→      );
   401→    }
   402→    const line = await tx.inventoryStocktakeLine.findFirst({
   403→      where: { id: input.lineId, stocktakeId: stocktake.id },
   404→      select: {
   405→        id: true,
   406→        productId: true,
   407→        countedAt: true,
   408→        systemQtyAtCount: true,
   409→        countedQty: true,
   410→        varianceQty: true,
   411→      },
   412→    });
   413→    if (!line) throw new ApiError(404, "NOT_FOUND", "Stocktake line not found.");
   414→
   415→    // §15 — server reads the CURRENT stock; client never supplies it.
   416→    const product = await tx.product.findFirst({
   417→      where: { id: line.productId, businessId: input.businessId },
   418→      select: { id: true, name: true, sku: true, unit: true, stockQty: true, deletedAt: true, isActive: true },
   419→    });
   420→    if (!product || product.deletedAt) {
   421→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   422→    }
   423→    if (!product.isActive) {
   424→      throw new ApiError(
   425→        422,
   426→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   427→        "Inactive products cannot be counted.",
   428→      );
   429→    }
   430→
   431→    const systemQtyAtCount = product.stockQty; // §15 snapshot at count time
   432→    const varianceQty = computeVarianceQty(input.countedQty, systemQtyAtCount); // §16/§18
   433→    const now = new Date();
   434→    const isRecount = line.countedAt !== null && line.countedAt.getTime() !== 0;
   435→
   436→    const updated = await tx.inventoryStocktakeLine.update({
   437→      where: { id: line.id },
   438→      data: {
   439→        systemQtyAtCount,
   440→        countedQty: input.countedQty,
   441→        varianceQty,
   442→        countedAt: now, // §30 step 14
   443→        countedByUserId: input.userId, // last successful count is authoritative (§47)
   444→        productNameAtCount: product.name, // §19 metadata snapshot
   445→        skuAtCount: product.sku,
   446→        unitAtCount: product.unit,
   447→      },
   448→      select: { id: true, productId: true, systemQtyAtCount: true, countedQty: true, varianceQty: true },
   449→    });
   450→
   451→    if (isRecount) {
   452→      // §32/§33 — replace in place; audit the safe before/after metadata.
   453→      await audit(
   454→        auditInput(
   455→          input.businessId,
   456→          input.userId,
   457→          "inventory.stocktake.recount",
   458→          stocktake.id,
   459→          stocktake.reference,
   460→          {
   461→            lineId: line.id,
   462→            productId: line.productId,
   463→            newSystemQty: updated.systemQtyAtCount,
   464→            newCountedQty: updated.countedQty,
   465→            newVariance: updated.varianceQty,
   466→          },
   467→          {
   468→            previousSystemQty: line.systemQtyAtCount,
   469→            previousCountedQty: line.countedQty,
   470→            previousVariance: line.varianceQty,
   471→          },
   472→        ),
   473→        tx,
   474→      );
   475→    } else {
   476→      await audit(
   477→        auditInput(input.businessId, input.userId, "inventory.stocktake.count", stocktake.id, stocktake.reference, {
   478→          lineId: line.id,
   479→          productId: line.productId,
   480→          systemQtyAtCount: updated.systemQtyAtCount,
   481→          countedQty: updated.countedQty,
   482→          variance: updated.varianceQty,
   483→        }),
   484→        tx,
   485→      );
   486→    }
   487→    return updated;
   488→  });
   489→}

   493→export async function startStocktake(input: {
   494→  businessId: string;
   495→  userId: string;
   496→  stocktakeId: string;
   497→  ip?: string | null;
   498→  userAgent?: string | null;
   499→}) {
   500→  return db.$transaction(async (tx) => {
   501→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   502→    assertTransition(stocktake.status, "COUNTING"); // §8/§34 — DRAFT only
   503→    const lineCount = await tx.inventoryStocktakeLine.count({
   504→      where: { stocktakeId: stocktake.id },
   505→    });
   506→    if (lineCount < 1) {
   507→      throw new ApiError(422, STOCKTAKE_ERROR_CODES.INVALID_TRANSITION, "Add at least one product before starting.");
   508→    }
   509→    const updated = await tx.inventoryStocktake.update({
   510→      where: { id: stocktake.id },
   511→      data: { status: "COUNTING", startedAt: new Date() },
   512→      select: { id: true, status: true, startedAt: true },
   513→    });
   514→    await audit(
   515→      auditInput(input.businessId, input.userId, "inventory.stocktake.start", stocktake.id, stocktake.reference, {
   516→        status: "COUNTING",
   517→        lineCount,
   518→      }),
   519→      tx,
   520→    );
   521→    return updated;
   522→  });
   523→}

   550→export async function reviewStocktake(input: {
   551→  businessId: string;
   552→  userId: string;
   553→  stocktakeId: string;
   554→  ip?: string | null;
   555→  userAgent?: string | null;
   556→}) {
   557→  return db.$transaction(async (tx) => {
   558→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   559→    assertTransition(stocktake.status, "REVIEW"); // §35 — COUNTING only
   560→    const lines = await tx.inventoryStocktakeLine.findMany({
   561→      where: { stocktakeId: stocktake.id },
   562→      select: { countedAt: true },
   563→    });
   564→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   565→      // §35 — every line must be counted before review.
   566→      throw new ApiError(
   567→        422,
   568→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   569→        "Count every product before sending the stocktake to review.",
   570→      );
   571→    }
   572→    const updated = await tx.inventoryStocktake.update({
   573→      where: { id: stocktake.id },
   574→      data: { status: "REVIEW", reviewedAt: new Date() },
   575→      select: { id: true, status: true, reviewedAt: true },
   576→    });
   577→    await audit(
   578→      auditInput(input.businessId, input.userId, "inventory.stocktake.review", stocktake.id, stocktake.reference, {
   579→        status: "REVIEW",
   580→        lineCount: lines.length,
   581→      }),
   582→      tx,
   583→    );
   584→    return updated;
   585→  });
   586→}
   587→
   588→/* ------------------------------- finalize --------------------------------- */
   589→
   590→export async function finalizeStocktake(input: {
   591→  businessId: string;
   592→  userId: string;
   593→  stocktakeId: string;
   594→  ip?: string | null;
   595→  userAgent?: string | null;
   596→}) {
   597→  return db.$transaction(async (tx) => {
   598→    // §39 — ONE transaction: verify → stale-check → reconcile → finalize.
   599→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   600→    assertTransition(stocktake.status, "FINALIZED"); // §38 — REVIEW only
   601→
   602→    const lines = await tx.inventoryStocktakeLine.findMany({
   603→      where: { stocktakeId: stocktake.id },
   604→      select: {
   605→        id: true,
   606→        productId: true,
   607→        systemQtyAtCount: true,
   608→        countedQty: true,
   609→        varianceQty: true,
   610→        countedAt: true,
   611→        productNameAtCount: true,
   612→      },
   613→    });
   614→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   615→      throw new ApiError(
   616→        422,
   617→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   618→        "Every product must be counted before finalization.",
   619→      );
   620→    }
   621→
   622→    // §40 stale check — every product's CURRENT stockQty must still equal
   623→    // systemQtyAtCount, and the product must still be active + not deleted
   624→    // (§57/§58). No auto-rebase (§41). Collect ALL stale/invalid lines so the
   625→    // client can recount each one (§139 — current-business products only).
   626→    const stale: Array<{
   627→      lineId: string;
   628→      productId: string;
   629→      productName: string;
   630→      systemQtyAtCount: number;
   631→      currentQty: number;
   632→      kind: "STALE" | "INACTIVE" | "DELETED";
   633→    }> = [];
   634→    const products = await tx.product.findMany({
   635→      where: { id: { in: lines.map((l) => l.productId) }, businessId: input.businessId },
   636→      select: { id: true, stockQty: true, deletedAt: true, isActive: true },
   637→    });
   638→    const productById = new Map(products.map((p) => [p.id, p]));
   639→    for (const line of lines) {
   640→      const product = productById.get(line.productId);
   641→      if (!product || product.deletedAt) {
   642→        stale.push({
   643→          lineId: line.id,
   644→          productId: line.productId,
   645→          productName: line.productNameAtCount,
   646→          systemQtyAtCount: line.systemQtyAtCount,
   647→          currentQty: product?.stockQty ?? line.systemQtyAtCount,
   648→          kind: "DELETED",
   649→        });
   650→        continue;
   651→      }
   652→      if (!product.isActive) {
   653→        stale.push({
   654→          lineId: line.id,
   655→          productId: line.productId,
   656→          productName: line.productNameAtCount,
   657→          systemQtyAtCount: line.systemQtyAtCount,
   658→          currentQty: product.stockQty,
   659→          kind: "INACTIVE",
   660→        });
   661→        continue;
   662→      }
   663→      if (product.stockQty !== line.systemQtyAtCount) {
   664→        stale.push({
   665→          lineId: line.id,
   666→          productId: line.productId,
   667→          productName: line.productNameAtCount,
   668→          systemQtyAtCount: line.systemQtyAtCount,
   669→          currentQty: product.stockQty,
   670→          kind: "STALE",
   671→        });
   672→      }
   673→    }
   674→    if (stale.length > 0) {
   675→      const hasInactive = stale.some((s) => s.kind === "INACTIVE");
   676→      const hasDeleted = stale.some((s) => s.kind === "DELETED");
   677→      throw new ApiError(
   678→        409,
   679→        hasInactive
   680→          ? STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE
   681→          : hasDeleted
   682→            ? STOCKTAKE_ERROR_CODES.STALE_COUNT
   683→            : STOCKTAKE_ERROR_CODES.STALE_COUNT,
   684→        "Inventory changed after counting. Recount the affected product(s).",
   685→        { products: stale }, // §139 — affected current-business products only
   686→      );
   687→    }
   688→
   689→    // §43/§44 — reconcile through the EXISTING InventoryAdjustment semantics.
   690→    let adjustedLineCount = 0;
   691→    for (const line of lines) {
   692→      if (line.varianceQty === 0) continue; // §44 — zero variance: no adjustment
   693→      const adjustment = await tx.inventoryAdjustment.create({
   694→        data: {
   695→          businessId: input.businessId,
   696→          productId: line.productId,
   697→          delta: line.varianceQty, // signed — §18
   698→          reason: ADJUSTMENT_REASON(stocktake.reference),
   699→          actorId: input.userId,
   700→        },
   701→        select: { id: true },
   702→      });
   703→      await tx.product.update({
   704→        where: { id: line.productId },
   705→        data: { stockQty: line.countedQty }, // §45 — stock becomes the counted value
   706→      });
   707→      adjustedLineCount += 1;
   708→      // §72/§73 — the InventoryAdjustment row itself carries the quantity
   709→      // history; no duplicate per-line audit event here. The finalize audit
   710→      // below records the workflow-level change once.
   711→      void adjustment;
   712→    }
   713→
   714→    const updated = await tx.inventoryStocktake.update({
   715→      where: { id: stocktake.id },
   716→      data: { status: "FINALIZED", finalizedAt: new Date() },
   717→      select: { id: true, status: true, finalizedAt: true },
   718→    });
   719→
   720→    // §69/§73 — ONE finalize audit for the workflow; InventoryAdjustment rows
   721→    // are the quantity-level history. Safe metadata only (§70/§140).
   722→    await audit(
   723→      auditInput(input.businessId, input.userId, "inventory.stocktake.finalize", stocktake.id, stocktake.reference, {
   724→        status: "FINALIZED",
   725→        lineCount: lines.length,
   726→        adjustedLineCount,
   727→      }),
   728→      tx,
   729→    );
   730→    return { ...updated, adjustedLineCount };
   731→  });
   732→}

   736→export async function cancelStocktake(input: {
   737→  businessId: string;
   738→  userId: string;
   739→  stocktakeId: string;
   740→  ip?: string | null;
   741→  userAgent?: string | null;

   744→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   745→    assertTransition(stocktake.status, "CANCELLED"); // §49 — DRAFT|COUNTING|REVIEW
   746→    const updated = await tx.inventoryStocktake.update({
   747→      where: { id: stocktake.id },
   748→      data: { status: "CANCELLED", canceledAt: new Date() },

   753→        status: "CANCELLED",
   754→        previousStatus: stocktake.status,
   755→      }),
   756→      tx,
   757→    );
   758→    return updated;
   759→  });

   762→/* ---------------------------- DRAFT-only edit ----------------------------- */
   763→
   764→export async function updateStocktake(input: {
   765→  businessId: string;
   766→  userId: string;
   767→  stocktakeId: string;
   768→  reference?: string;
   769→  notes?: string | null;
   770→  ip?: string | null;
   771→  userAgent?: string | null;
   772→}) {
   773→  return db.$transaction(async (tx) => {
   774→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   775→    // §50/§52 — reference/notes are editable ONLY while DRAFT.
   776→    if (stocktake.status !== "DRAFT") {
   777→      throw new ApiError(
   778→        422,
   779→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   780→        "Only draft stocktakes can be edited.",
   781→      );
   782→    }
   783→    try {
   784→      const updated = await tx.inventoryStocktake.update({
   785→        where: { id: stocktake.id },
   786→        data: {
   787→          ...(input.reference !== undefined ? { reference: input.reference } : {}),
   788→          ...(input.notes !== undefined ? { notes: input.notes } : {}),
   789→        },
   790→        select: { id: true, reference: true, notes: true },
   791→      });
   792→      await audit(
   793→        auditInput(
   794→          input.businessId,
   795→          input.userId,
   796→          "inventory.stocktake.edit",
   797→          stocktake.id,
   798→          updated.reference,
   799→          { status: stocktake.status },
   800→          { reference: stocktake.reference }, // §70 — never the notes content
   801→        ),
   802→        tx,
   803→      );
   804→      return updated;
   805→    } catch (error) {
   806→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   807→        throw new ApiError(409, "DUPLICATE_REFERENCE", "A stocktake with this reference already exists.");
   808→      }
   809→      throw error;
   810→    }
   811→  });
   812→}
