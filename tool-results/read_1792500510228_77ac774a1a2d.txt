    64→/** Sidebar label key for the current view (used for <title> too). */
    65→function labelKeyOf(view: AppView): string {
    66→  switch (view.kind) {
    67→    case "dashboard":
    68→      return "dashboard";
    69→    case "audit":
    70→      return "audit";
    71→    case "settings:business":
    72→      return "settings";
    73→    case "settings:members":
    74→      return "teamMembers";
    75→    case "settings:billing":
    76→      return "billing";
    77→    case "settings:security":
    78→      return "security";
    79→    case "settings:data":
    80→      return "dataPortability";
    81→    case "notifications":
    82→      return "notifications";
    83→    case "settings:notifications":
    84→      return "notifications";
    85→    // Prompt 33 — Search Results Center document title.
    86→    case "search":
    87→      return "search";
    88→    // Prompt 35 (§36) — CRM module document title.
    89→    case "crm":
    90→      return "crm";
    91→    // Prompt 34 — entity detail titles reuse the owning module's nav label.
    92→    case "entity":
    93→      return ENTITY_ROUTES[view.entity].labelKey;
    94→    // Prompt 39 — party statement document title (§57).
    95→    case "statements":
    96→      return "statements";
    97→    // Prompt 41 — purchase-order document titles (§71).
    98→    case "purchase-orders":
    99→      return "purchaseOrders";
   100→    case "purchase-order":
   101→      return "purchaseOrders";
   102→    // Prompt 42 — goods-receipt document titles (§117).
   103→    case "inventory-goods-receipts":
   104→      return "goodsReceipts";
   105→    case "inventory-goods-receipt":
   106→      return "goodsReceipts";
   107→    // Prompt 43 — goods-return document titles (§91).
   108→    case "inventory-goods-returns":
   109→      return "goodsReturns";
   110→    case "inventory-goods-return":
   111→      return "goodsReturns";
   112→    // Prompt 44 — goods-dispatch document titles.
   113→    case "inventory-goods-dispatches":
   114→      return "goodsDispatches";
   115→    case "inventory-goods-dispatch":
   116→      return "goodsDispatches";
   117→    // Prompt 45 — sales-order document titles.
   118→    case "sales-orders":
   119→      return "salesOrders";
   120→    case "sales-order":
   121→      return "salesOrders";
   122→    // Prompt 47 — sales-quotation document titles.
   123→    case "sales-quotations":
   124→      return "salesQuotations";
   125→    case "sales-quotation":
   126→      return "salesQuotations";
   127→    // Prompt 48 — supplier-bill document titles (§4.1).
   128→    case "supplier-bills":
   129→      return "supplierBills";
   130→    case "supplier-bill":
   131→      return "supplierBills";
   132→    // Prompt 49 — detailed-aging document titles (the owning balances module).
   133→    case "aging-details":
   134→      return view.side === "payables" ? "payables" : "receivables";
   135→    case "profile":
   136→      return "profile";
   137→    case "module": {
   138→      const map: Record<string, string> = {
   139→        customers: "customers",
   140→        suppliers: "suppliers",
   141→        products: "products",
   142→        inventory: "inventory",
   143→        sales: "sales",
   144→        expenses: "expenses",
   145→        payments: "payments",
   146→        receivables: "receivables",
   147→        payables: "payables",
   148→        reports: "reports",
   149→        copilot: "copilot",
   150→      };
   151→      return map[view.module] ?? "dashboard";
   152→    }
   153→    default:
   154→      return "dashboard";
   155→  }
   156→}

   172→
   173→  const membership =
   174→    memberships.find((m) => m.business.id === activeId) ?? memberships[0] ?? null;
   175→  const role: Role = asRole(membership?.role);
   176→
   177→  // Phase 06 — per-module document titles (localized: "Sales · Business · App").
   178→  const moduleLabel = tNav(labelKeyOf(view) as "dashboard");
   179→  useEffect(() => {
   180→    const parts = [
   181→      moduleLabel,
   182→      membership?.business.name,
   183→      tCommon("appName"),
   184→    ].filter(Boolean);
   185→    document.title = parts.join(" · ");
   186→  }, [moduleLabel, membership?.business.name, tCommon]);
   187→
   188→  // Phase 06 — idle prefetch warming: reports/summary + balances are the
   189→  // heaviest deterministic queries; warm them once per business so deep
   190→  // links into Reports/Receivables/Payables render from cache instantly.
   191→  const queryClient = useQueryClient();
   192→  const activeBusinessId = membership?.business.id;
   193→  useEffect(() => {
   194→    if (!activeBusinessId) return;
   195→    const cancel = onIdle(() => warmHeavyQueries(queryClient));
   196→    return cancel;

   200→
   201→  // Prompt 23 §41 — the provider-callback return page. `?payment_return=<id>`

   207→  // Navigation is compared by VIEW OBJECT IDENTITY, not viewKey: on a fresh
   208→  // `/?payment_return=<id>` load the underlying view is already the default
   209→  // dashboard, so clicking «Dashboard» produces the SAME viewKey and would
   210→  // never dismiss. navigate()/replaceView() always pass a fresh object.

   212→  useEffect(() => {
   213→    if (prevViewRef.current === null) {
   214→      prevViewRef.current = view;
   215→      return;
   216→    }
   217→    if (prevViewRef.current !== view) {
   218→      prevViewRef.current = view;
   219→      clearPaymentReturn();
   220→    }
   221→  }, [view, clearPaymentReturn]);

   223→  return (
   224→    <SidebarProvider>
   225→      {/* r52 ambient app background — PRESENTATION ONLY. A whisper of the
   226→          brand gradient (CSS fallback, −z-20) under the business WebGL
   227→          "growth field" (−z-10, decorative, aria-hidden, pauses when the
   228→          tab is hidden, static frame under prefers-reduced-motion, and it
   229→          hides itself entirely when WebGL is unavailable). The content
   230→          surfaces above stay readable: SidebarInset is slightly translucent
   231→          so the aurora shows through subtly, while cards/popovers remain
   232→          opaque. Nothing here affects data flow, routing or logic. */}
   233→      <div aria-hidden="true" className="bg-app-fallback pointer-events-none fixed inset-0 -z-20" />
   234→      <BusinessWebGLBackground className="opacity-35 dark:opacity-30" />
   235→      <AppSidebar membership={membership} side={dir === "rtl" ? "right" : "left"} />
   236→      <SidebarInset className="bg-background/85 dark:bg-background/80">
   237→        {/* a11y: keyboard users can jump past the navigation chrome */}
   238→        <a
   239→          href="#main-content"
   240→          className="bg-primary text-primary-foreground sr-only focus:not-sr-only focus:fixed focus:start-3 focus:top-3 focus:z-50 focus:rounded-md focus:px-3 focus:py-2 focus:text-sm focus:font-medium focus:shadow-lg"
   241→        >
   242→          {tSkip("skipToContent")}
   243→        </a>
   244→        <AppHeader />
   245→        <main id="main-content" className="flex-1 p-4 sm:p-6">
   246→          {paymentReturnSessionId ? (
   247→            <div className="mx-auto w-full max-w-6xl">
   248→              <PaymentReturnView
   249→                sessionId={paymentReturnSessionId}
   250→                onDone={clearPaymentReturn}
   251→              />
   252→            </div>

   584→export function useMembershipsQuery(enabled: boolean) {
   585→  const queryClient = useQueryClient();
   586→  return useQuery<BusinessListItem[]>({
   587→    queryKey: ["businesses"],
   588→    queryFn: async () => {
   589→      const res = await fetch("/api/v1/businesses", { credentials: "same-origin" });
   590→      let body: unknown = null;
   591→      try { body = await res.json(); } catch { /* non-JSON */ };
   592→      const err = (body as { error?: { code?: string } } | null)?.error;
   593→      if (!res.ok) {
   594→        // Prompt 30 §83/§229 — a globally invalidated session (password
   595→        // change / MFA enable / admin kill switch) must land the user on
   596→        // the sign-in view, never a permanent splash. Clear the stale
   597→        // cookie client-side; NextAuth's session state flips to signed-out.
   598→        if (res.status === 401 || err?.code === "SESSION_INVALIDATED" || err?.code === "SESSION_REVOKED") {
   599→          queryClient.clear();
   600→          void import("next-auth/react").then((m) => m.signOut({ redirect: false }));
   601→        }
   602→        throw new Error(err?.code ?? `HTTP_${res.status}`);
   603→      }
   604→      return (body as { data: BusinessListItem[] }).data;
   605→    },
   606→    enabled,
   607→    staleTime: 60_000,
   608→    retry: false,
   609→  });
   610→}

   612→/**
   613→ * Prompt 23 §41 — reads `?payment_return=<sessionId>` (set ONLY by our own
   614→ * callback API). Value is validated shape-wise (cuid-ish length bound);
   615→ * the session itself is re-authorized server-side, so a forged id simply
   616→ * renders a 404-ish "unknown" state — never another tenant's data.
   617→ * Listens to popstate so browser back/forward and the view's own
   618→ * clear() stay in sync.
   619→ */
   620→function usePaymentReturnSessionId(): [string | null, () => void] {
   621→  const read = () => {
   622→    try {
   623→      const raw = new URLSearchParams(window.location.search).get("payment_return");
   624→      return raw && raw.length >= 20 && raw.length <= 40 && /^[a-zA-Z0-9_-]+$/.test(raw) ? raw : null;
   625→    } catch {
   626→      return null;
   627→    }
   628→  };
   629→  const [sessionId, setSessionId] = useState<string | null>(read);
   630→  useEffect(() => {
   631→    const sync = () => setSessionId(read());
   632→    window.addEventListener("popstate", sync);
   633→    return () => window.removeEventListener("popstate", sync);
   634→  }, []);
   635→  const clear = useCallback(() => {
   636→    // Strip ONLY the ?payment_return query — never touch the hash (the view
   637→    // may be navigating via the app hash-router at the same time).
   638→    window.history.replaceState(null, "", `${window.location.pathname}${window.location.hash}`);
   639→    setSessionId(null);
   640→  }, []);
   641→  return [sessionId, clear];
   642→}
