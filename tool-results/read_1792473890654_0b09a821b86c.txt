   208→export type ExportRow = {
   209→  /** JSON row — IDs + relationships preserved, money as minor integers (§54/§226). */
   210→  json: Record<string, unknown>;
   211→  /** CSV row — exact decimal strings for money, ISO-8601 UTC dates (§227/§231). */
   212→  csv: Array<string | number | null>;
   213→};
   214→
   215→export type DatasetFetchResult = {
   216→  rows: ExportRow[];
   217→  /** True when the dataset hit the export record cap (job must FAIL, §26). */
   218→  truncated: boolean;
   219→};
   220→
   221→export type DatasetFetchCtx = {
   222→  businessId: string;
   223→  /** Business decimal precision for money → CSV decimal strings. */
   224→  decimalPrecision: number;
   225→  /** Hard cap applied per fetch — caller passes EXPORT_MAX_RECORDS headroom. */
   226→  maxRecords: number;
   227→};

   229→function iso(value: Date | null | undefined): string | null {
   230→  return value ? value.toISOString() : null;
   231→}

   237→type Customer = Prisma.CustomerGetPayload<object>;
   238→
   239→function customerRow(c: Customer, precision: number): ExportRow {
   240→  void precision; // customers carry no money fields
   241→  return {
   242→    json: {
   243→      id: c.id,
   244→      name: c.name,
   245→      phone: c.phone,
   246→      email: c.email,
   247→      address: c.address,
   248→      note: c.note,
   249→      deletedAt: iso(c.deletedAt),
   250→      createdAt: c.createdAt.toISOString(),
   251→      updatedAt: c.updatedAt.toISOString(),
   252→    },
   253→    csv: [c.id, c.name, c.phone, c.email, c.address, c.note, iso(c.deletedAt), c.createdAt.toISOString(), c.updatedAt.toISOString()],
   254→  };
   255→}
   256→
   257→function supplierRow(s: Prisma.SupplierGetPayload<object>, precision: number): ExportRow {
   258→  void precision;
   259→  return {
   260→    json: {
   261→      id: s.id,
   262→      name: s.name,
   263→      phone: s.phone,
   264→      email: s.email,
   265→      address: s.address,
   266→      note: s.note,
   267→      deletedAt: iso(s.deletedAt),
   268→      createdAt: s.createdAt.toISOString(),
   269→      updatedAt: s.updatedAt.toISOString(),
   270→    },
   271→    csv: [s.id, s.name, s.phone, s.email, s.address, s.note, iso(s.deletedAt), s.createdAt.toISOString(), s.updatedAt.toISOString()],
   272→  };
   273→}
   274→
   275→function productRow(p: Prisma.ProductGetPayload<object>, precision: number): ExportRow {
   276→  return {
   277→    json: {
   278→      id: p.id,
   279→      name: p.name,
   280→      sku: p.sku,
   281→      unit: p.unit,
   282→      salePriceMinor: p.salePriceMinor,
   283→      costPriceMinor: p.costPriceMinor,
   284→      stockQty: p.stockQty,
   285→      lowStockAt: p.lowStockAt,
   286→      isActive: p.isActive,
   287→      deletedAt: iso(p.deletedAt),
   288→      createdAt: p.createdAt.toISOString(),
   289→      updatedAt: p.updatedAt.toISOString(),
   290→    },
   291→    csv: [
   292→      p.id,
   293→      p.name,
   294→      p.sku,
   295→      p.unit,
   296→      formatMinorToDecimal(p.salePriceMinor, precision),
   297→      formatMinorToDecimal(p.costPriceMinor, precision),
   298→      p.stockQty,
   299→      p.lowStockAt,
   300→      p.isActive ? "true" : "false",
   301→      iso(p.deletedAt),
   302→      p.createdAt.toISOString(),
   303→      p.updatedAt.toISOString(),
   304→    ],
   305→  };
   306→}

   308→async function fetchWithCap<T>(
   309→  cap: number,
   310→  query: (take: number) => Promise<T[]>,
   311→): Promise<{ items: T[]; truncated: boolean }> {
   312→  // Fetch one row beyond the cap so truncation is DETECTABLE, never silent.
   313→  const items = await query(cap + 1);
   314→  return { items: items.slice(0, cap), truncated: items.length > cap };
   315→}
