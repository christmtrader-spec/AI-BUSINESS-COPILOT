    58→/**
    59→ * Prompt 32 §72/§76/§84 — post-commit billing notification queue.
    60→ * Domain transaction → commit → notification (NEVER inside the tx): each
    61→ * mutation collects the durable SubscriptionEvent ids it created and the
    62→ * mapper runs only after `db.$transaction` has resolved successfully. A
    63→ * rolled-back transaction therefore emits NO notification (the event rows
    64→ * it queued are discarded with it).

    66→type PendingBillingNotification = {
    67→  eventId: string;
    68→  type: SubscriptionEventType;
    69→  toPlanCode: string | null;
    70→};
    71→
    72→function flushBillingNotifications(
    73→  businessId: string,
    74→  queue: readonly PendingBillingNotification[],
    75→): void {
    76→  if (queue.length === 0) return;
    77→  void (async () => {
    78→    const { notifySubscriptionEvent } = await import("@/lib/notifications");
    79→    for (const item of queue) {
    80→      notifySubscriptionEvent({
    81→        businessId,
    82→        eventId: item.eventId,
    83→        type: item.type,
    84→        toPlanCode: item.toPlanCode,
    85→      });
    86→    }
    87→  })();
    88→}

   228→/**
   229→ * Get-or-create the deterministic FREE subscription for a business
   230→ * (§13/§14/§118). Idempotent, transactional, auditable — used at business
   231→ * creation, by the seed backfill and as the lazy safety net on read.
   232→ */
   233→export async function ensureSubscription(
   234→  businessId: string,
   235→  opts: { actorId?: string | null; source?: string; tx?: Tx } = {},
   236→): Promise<SubRow> {
   237→  const run = async (tx: Tx): Promise<SubRow> => {
   238→    const existing = await loadSubscription(tx, businessId);
   239→    if (existing) return existing;
   240→
   241→    const business = await tx.business.findUnique({ where: { id: businessId } });
   242→    if (!business) {
   243→      throw new ApiError(403, "FORBIDDEN", "You do not have access to this business.");
   244→    }
   245→
   246→    const free = await tx.plan.findUnique({ where: { code: FREE_PLAN_CODE } });
   247→    if (!free || !free.active) {
   248→      // Catalog corruption — fail closed loudly, never invent entitlements.
   249→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   250→    }
   251→
   252→    const now = new Date();
   253→    const created = await tx.subscription.create({
   254→      data: {
   255→        businessId,
   256→        planId: free.id,
   257→        status: "ACTIVE",
   258→        billingPeriod: "MONTHLY",
   259→        currentPeriodStart: now,
   260→        currentPeriodEnd: periodEnd(now, "MONTHLY"),
   261→      },
   262→      include: { plan: true },
   263→    });
   264→    await tx.subscriptionEvent.create({
   265→      data: {
   266→        businessId,
   267→        subscriptionId: created.id,
   268→        type: "CREATED",
   269→        toStatus: "ACTIVE",
   270→        toPlanCode: free.code,
   271→        dataJson: JSON.stringify({ source: opts.source ?? "ensure" }),
   272→        actorId: opts.actorId ?? null,
   273→      },
   274→    });
   275→    await audit(
   276→      {
   277→        businessId,
   278→        actorId: opts.actorId ?? null,
   279→        action: "billing.subscription_created",
   280→        entityType: "Subscription",
   281→        entityId: created.id,
   282→        after: { planCode: free.code, status: "ACTIVE", source: opts.source ?? "ensure" },
   283→      },
   284→      tx,
   285→    );
   286→    return created as SubRow;
   287→  };
   288→
   289→  if (opts.tx) return run(opts.tx);
   290→  return db.$transaction(run, { isolationLevel: "Serializable" });
   291→}
   292→
   293→/**
   294→ * Lazy expiry (§17/§91): when a read finds a subscription whose grace/
   295→ * trial/period has passed, transition it to EXPIRED with an event —
   296→ * transactional, only if still in an expirable state.
   297→ */
   298→async function expireSubscriptionIfDue(sub: SubRow): Promise<SubRow> {
   299→  const decision = decidePlan(sub);
   300→  if (!("expired" in decision) || !decision.expired) return sub;
   301→
   302→  const updated = await db.$transaction(
   303→    async (tx) => {
   304→      const current = await loadSubscription(tx, sub.businessId);
   305→      if (!current) return current;
   306→      const re = decidePlan(current);
   307→      if (!("expired" in re) || !re.expired) return current;
   308→      const row = await tx.subscription.update({
   309→        where: { id: current.id },
   310→        data: { status: "EXPIRED", endedAt: new Date() },
   311→        include: { plan: true },
   312→      });
   313→      const expiredEventId = await tx.subscriptionEvent.create({
   314→        data: {
   315→          businessId: current.businessId,
   316→          subscriptionId: current.id,
   317→          type: "EXPIRED",
   318→          fromStatus: current.status,
   319→          toStatus: "EXPIRED",
   320→          fromPlanCode: current.plan.code,
   321→        },
   322→        select: { id: true },
   323→      });
   324→      await audit(
   325→        {
   326→          businessId: current.businessId,
   327→          action: "billing.subscription_expired",
   328→          entityType: "Subscription",
   329→          entityId: current.id,
   330→          before: { status: current.status },
   331→          after: { status: "EXPIRED" },
   332→        },
   333→        tx,
   334→      );
   335→      return { row, expiredEventId: expiredEventId.id, planCode: current.plan.code };
   336→    },
   337→    { isolationLevel: "Serializable" },
   338→  );
   339→  invalidateBillingCache(sub.businessId);
   340→  if (updated && typeof updated === "object" && "expiredEventId" in updated) {
   341→    // Prompt 32 §86 — EXPIRED is a customer-visible state (post-commit).
   342→    flushBillingNotifications(sub.businessId, [
   343→      { eventId: updated.expiredEventId, type: "EXPIRED", toPlanCode: updated.planCode },
   344→    ]);
   345→  }
   346→  return updated && typeof updated === "object" && "row" in updated ? updated.row : updated;
   347→}

   349→/**
   350→ * Resolve the effective plan for a subscription row (§17 + §13 fallback).
   351→ * Returns the plan whose entitlements apply.
   352→ */
   353→async function resolveEffectivePlan(sub: SubRow, tx: Tx | typeof db) {
   354→  const decision = decidePlan(sub);
   355→  if (decision.kind === "own-plan") return { plan: sub.plan, entitled: true };
   356→  const free = await tx.plan.findUnique({ where: { code: FREE_PLAN_CODE } });
   357→  if (!free) {
   358→    throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   359→  }
   360→  return { plan: free, entitled: decision.entitled };
   361→}

   363→/** Resolve a plan's full entitlement map (fail-closed on corruption). */
   364→function entitlementsFromRows(
   365→  rows: Array<{ entitlement: { code: string; type: string; unit: string | null; description: string }; value: number | null; boolValue: boolean }>,
   366→): Record<EntitlementCode, ResolvedEntitlement> {
   367→  const map = {} as Record<EntitlementCode, ResolvedEntitlement>;
   368→  // Fail-closed defaults first (§68/§93): a MISSING catalog row can never
   369→  // behave as "allowed/unlimited". The seed guarantees completeness; if a
   370→  // row is missing anyway the gate blocks loudly.
   371→  for (const spec of ENTITLEMENTS) {
   372→    map[spec.code] = {
   373→      code: spec.code,
   374→      type: spec.type,
   375→      unit: spec.unit,
   376→      value: spec.type === "LIMIT" ? 0 : false,
   377→      description: spec.description,
   378→    };
   379→  }
   380→  for (const row of rows) {
   381→    if (!isEntitlementCode(row.entitlement.code)) continue;
   382→    const spec = ENTITLEMENTS.find((s) => s.code === row.entitlement.code)!;
   383→    map[spec.code] = {
   384→      code: spec.code,
   385→      type: spec.type,
   386→      unit: spec.unit,
   387→      value:
   388→        spec.type === "LIMIT"
   389→          ? row.value // number, or null = UNLIMITED
   390→          : row.boolValue,
   391→      description: spec.description,
   392→    };
   393→  }
   394→  return map;
   395→}

   397→/** Fresh (uncached) entitlement context — the gate-enforcement loader. */
   398→export async function getEntitlementContextFresh(businessId: string): Promise<EntitlementContext> {
   399→  try {
   400→    let sub = await loadSubscription(db, businessId);
   401→    if (!sub) {
   402→      sub = await ensureSubscription(businessId, { source: "lazy-backfill" });
   403→      invalidateBillingCache(businessId);
   404→    }
   405→    sub = await expireSubscriptionIfDue(sub);
   406→    const { plan, entitled } = await resolveEffectivePlan(sub, db);
   407→    const rows = await db.planEntitlement.findMany({
   408→      where: { planId: plan.id },
   409→      include: { entitlement: true },
   410→    });
   411→    return {
   412→      businessId,
   413→      subscription: subscriptionDto(sub, plan.code, entitled),
   414→      entitlements: entitlementsFromRows(rows),
   415→    };
   416→  } catch (error) {
   417→    if (error instanceof ApiError) throw error;
   418→    console.error("[billing] context lookup failed:", error);
   419→    throw new BillingError("BILLING_UNAVAILABLE", "Billing state is temporarily unavailable. Please try again.");
   420→  }
   421→}

   423→/** Same as getEntitlementContextFresh but inside an existing transaction. */
   424→export async function getEntitlementContextTx(tx: Tx, businessId: string): Promise<EntitlementContext> {
   425→  let sub = await loadSubscription(tx, businessId);
   426→  if (!sub) {
   427→    sub = await ensureSubscription(businessId, { source: "lazy-backfill", tx });

   431→    where: { planId: plan.id },

   436→    subscription: subscriptionDto(sub, plan.code, entitled),
   437→    entitlements: entitlementsFromRows(rows),
   438→  };

   441→/** Cached context for UI read endpoints (§67); 15s TTL, mutation-invalidated. */
   442→export async function getEntitlementContext(businessId: string): Promise<EntitlementContext> {
   443→  const hit = cache.get(businessId);
   444→  if (hit && Date.now() - hit.at < CACHE_TTL_MS) return hit.ctx;
   445→  const ctx = await getEntitlementContextFresh(businessId);
   446→  cache.set(businessId, { at: Date.now(), ctx });
   447→  return ctx;
   448→}

   454→export function hasEntitlementIn(ctx: EntitlementContext, code: EntitlementCode): boolean {

   459→export function getLimitIn(ctx: EntitlementContext, code: EntitlementCode): number | null {
   460→  const e = ctx.entitlements[code];
   461→  if (!e || e.type !== "LIMIT") return 0; // unknown/misconfigured → fail closed
   462→  return e.value === null ? null : (e.value as number);
   463→}

   465→/** §25/§27 — throws the structured error; `currentUsage` counted by caller. */
   466→export function assertWithinLimitIn(
   467→  ctx: EntitlementContext,
   468→  code: EntitlementCode,
   469→  currentUsage: number,
   470→): void {
   471→  const limit = getLimitIn(ctx, code);
   472→  if (limit === null) return; // unlimited (e.g. ENTERPRISE)
   473→  if (currentUsage >= limit) throw limitReachedError(code, limit, currentUsage);
   474→}

   512→// ---------------------------------------------------------------------------
   513→// Usage counting (§105/§106 — documented counting policies)
   514→// ---------------------------------------------------------------------------
   515→
   516→export async function countEntitlementUsage(
   517→  businessId: string,
   518→  ownerId: string,
   519→  codes: readonly EntitlementCode[],
   520→): Promise<Record<string, { used: number; limit: number | null }>> {
   521→  const ctx = await getEntitlementContext(businessId);
   522→  const out: Record<string, { used: number; limit: number | null }> = {};
   523→  for (const code of codes) {
   524→    const limit = getLimitIn(ctx, code);
   525→    let used = 0;
   526→    switch (code) {
   527→      case "CUSTOMERS_MAX":
   528→        used = await db.customer.count({ where: { businessId, deletedAt: null } });
   529→        break;
   530→      case "PRODUCTS_MAX":
   531→        // Deactivation (isActive=false) is a catalog state, NOT deletion —
   532→        // both count (documented in entitlements.md).
   533→        used = await db.product.count({ where: { businessId, deletedAt: null } });
   534→        break;
   535→      case "TEAM_MEMBERS_MAX":
   536→        // Active memberships only; owner included (§108); pending invites
   537→        // never count (§31).
   538→        used = await db.businessMember.count({ where: { businessId } });
   539→        break;
   540→      case "BUSINESSES_MAX":
   541→        // Businesses the user OWNS (§107) — staff memberships don't count.
   542→        used = await db.business.count({ where: { ownerId } });
   543→        break;
   544→      default:
   545→        used = 0;
   546→    }
   547→    out[code] = { used, limit };
   548→  }
   549→  return out;
   550→}

   552→// ---------------------------------------------------------------------------
   553→// Business-creation limit (§30/§107) — a USER-level check
   554→// ---------------------------------------------------------------------------
   555→
   556→/**
   557→ * P51-R1-FINAL: a finite BUSINESSES_MAX value is usable ONLY when it is a
   558→ * positive integer inside the supported integer domain. The documented
   559→ * contract (prisma/schema.prisma PlanEntitlement, entitlements.md,
   560→ * scripts/qa/verify-billing-catalog.mjs):
   561→ *   - LIMIT rows: value = the numeric limit; null = UNLIMITED (legitimate,
   562→ *     e.g. ENTERPRISE — preserved, never rejected here);
   563→ *   - every plan that documents a business-count limit must grant ≥ 1
   564→ *     (FREE included), so ZERO and NEGATIVE values are CATALOG CONFIGURATION
   565→ *     FAILURES — not reachable limits. The previous `limit = row.value`
   566→ *     accepted 0/-n and forged a misleading "plan limit reached" for users
   567→ *     who own nothing (defect §7 / USER-BUG3 family).
   568→ * Non-integers, NaN/Infinity (Number.isInteger rejects both) and values
   569→ * beyond Number.MAX_SAFE_INTEGER cannot persist through the Prisma `Int?`
   570→ * column (SQLite 32-bit), but are rejected here as defense-in-depth for
   571→ * any transformation that might introduce them.
   572→ */
   573→function isUsableBusinessesLimit(value: unknown): value is number {
   574→  return (
   575→    typeof value === "number" &&
   576→    Number.isInteger(value) &&
   577→    value >= 1 &&
   578→    value <= Number.MAX_SAFE_INTEGER
   579→  );
   580→}

   582→/**
   583→ * BUSINESSES_MAX is special: it caps how many businesses a USER may OWN,
   584→ * so it cannot be read from one business's subscription. Deterministic
   585→ * policy (documented in entitlements.md): the effective limit is the
   586→ * MAXIMUM BUSINESSES_MAX across the plans of businesses the user already
   587→ * owns — upgrading any owned business lifts the cap; a user with no
   588→ * businesses gets the FREE limit. Count + check run inside the caller's
   589→ * transaction (race-safe on SQLite, §112).
   590→ */
   591→export async function assertWithinBusinessesLimitTx(tx: Tx, userId: string): Promise<void> {
   592→  const owned = await tx.business.findMany({
   593→    where: { ownerId: userId },
   594→    select: {
   595→      subscription: {
   596→        include: {
   597→          plan: { include: { planEntitlements: { include: { entitlement: true } } } },
   598→        },
   599→      },
   600→    },
   601→  });
   602→
   603→  let limit = 0;
   604→  for (const b of owned) {
   605→    const sub = b.subscription;
   606→    if (!sub) continue; // grandfathered mid-backfill rows contribute nothing
   607→    const row = sub.plan.planEntitlements.find((pe) => pe.entitlement.code === "BUSINESSES_MAX");
   608→    if (!row) continue; // fail-closed: corrupt plan contributes nothing
   609→    // P51-R1-FINAL: a PRESENT row with a malformed shape is a catalog
   610→    // integrity failure — honest controlled error (§68), never silently
   611→    // zeroed and never presented as a genuine user-facing limit.
   612→    if (row.entitlement.type !== "LIMIT" || !isUsableBusinessesLimit(row.value)) {
   613→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog configuration is invalid.");
   614→    }
   615→    if (row.value === null) return; // any unlimited plan lifts the cap
   616→    limit = Math.max(limit, row.value);
   617→  }
   618→  if (limit === 0) {
   619→    // No owned business (or none with a valid subscription) → FREE limit.
   620→    const free = await tx.plan.findUnique({
   621→      where: { code: FREE_PLAN_CODE },
   622→      include: { planEntitlements: { include: { entitlement: true } } },
   623→    });
   624→    const row = free?.planEntitlements.find((pe) => pe.entitlement.code === "BUSINESSES_MAX");
   625→    if (!row) {
   626→      // BUGFIX P51-R1-USER-BUG3: an EMPTY/CORRUPT catalog is an OPERATIONAL
   627→      // failure, not a user-facing plan limit. The previous `row?.value ?? 0`
   628→      // made every fresh user (0 owned businesses) hit `0 >= 0` and receive a
   629→      // FALSE "plan limit reached" alert on their very first business. Honest
   630→      // fail-closed per §68 — same semantics as the other catalog guards above.
   631→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   632→    }
   633→    // P51-R1-FINAL: BUSINESSES_MAX must be a LIMIT-typed row. A BOOLEAN-typed
   634→    // row (value null) must NOT pass as "unlimited" — that is malformed
   635→    // catalog data (§2.3 type mismatch), fail closed with §68.
   636→    if (row.entitlement.type !== "LIMIT") {
   637→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog configuration is invalid.");
   638→    }
   639→    if (row.value === null) return; // FREE configured unlimited — mirrors getLimitIn (§472)
   640→    if (!isUsableBusinessesLimit(row.value)) {
   641→      // BUGFIX P51-R1-FINAL: FREE BUSINESSES_MAX = 0 / negative / malformed
   642→      // is a CONFIGURATION failure (documented contract: FREE ≥ 1), NOT a
   643→      // genuine limit. The previous `limit = row.value` made a fresh user
   644→      // hit `0 >= 0` again and forged "plan limit reached". Fail closed
   645→      // with the established controlled error — never fabricate a limit,
   646→      // never silently fall back, never auto-repair catalog data (§2.3).
   647→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog configuration is invalid.");
   648→    }
   649→    limit = row.value;
   650→  }
   651→
   652→  if (owned.length >= limit) {
   653→    throw limitReachedError("BUSINESSES_MAX", limit, owned.length);
   654→  }
   655→}

   661→export type SubscriptionMutationResult = {
   662→  subscription: SubscriptionDto;
   663→  /** false when the call was an idempotent no-op (§48). */
   664→  changed: boolean;
   665→};
   666→
   667→function assertTransition(from: SubscriptionStatus, to: SubscriptionStatus): void {
   668→  if (from === to) return;
   669→  if (!isValidTransition(from, to)) {
   670→    throw new BillingError(
   671→      "INVALID_SUBSCRIPTION_STATE",
   672→      `Transition ${from} → ${to} is not allowed.`,
   673→      { from, to, allowed: VALID_TRANSITIONS[from] },
   674→    );
   675→  }
   676→}

   678→async function eventAndAudit(
   679→  tx: Tx,
   680→  input: {
   681→    businessId: string;
   682→    subscriptionId: string;
   683→    type: SubscriptionEventType;
   684→    fromStatus?: string | null;
   685→    toStatus?: string | null;
   686→    fromPlanCode?: string | null;
   687→    toPlanCode?: string | null;
   688→    data?: Record<string, unknown>;
   689→    actorId?: string | null;
   690→    auditAction: string;
   691→    auditBefore?: unknown;
   692→    auditAfter?: unknown;
   693→  },
   694→): Promise<string> {
   695→  const created = await tx.subscriptionEvent.create({
   696→    data: {
   697→      businessId: input.businessId,
   698→      subscriptionId: input.subscriptionId,
   699→      type: input.type,
   700→      fromStatus: input.fromStatus ?? null,
   701→      toStatus: input.toStatus ?? null,
   702→      fromPlanCode: input.fromPlanCode ?? null,
   703→      toPlanCode: input.toPlanCode ?? null,
   704→      dataJson: input.data ? JSON.stringify(input.data) : null,
   705→      actorId: input.actorId ?? null,
   706→    },
   707→    select: { id: true },
   708→  });
   709→  await audit(
   710→    {
   711→      businessId: input.businessId,
   712→      actorId: input.actorId ?? null,
   713→      action: input.auditAction,
   714→      entityType: "Subscription",
   715→      entityId: input.subscriptionId,
   716→      before: input.auditBefore,
   717→      after: input.auditAfter,
   718→    },
   719→    tx,
   720→  );
   721→  // Prompt 32 — the durable event row id feeds the post-commit notification
   722→  // mapper (§72: domain transaction → commit → notification; the caller
   723→  // flushes AFTER this transaction resolves, §332).
   724→  return created.id;
   725→}

   727→/**
   728→ * Internal plan change (§41/§43/§82/§83): validates the target plan,
   729→ * verifies state transitions, records PLAN_CHANGED (+ REACTIVATED when it
   730→ * doubles as reactivation), all in ONE transaction. NO money changes hands
   731→ * and NOTHING is labeled as payment (§83). Downgrades never delete data —
   732→ * they only cap future growth (§42).
   733→ */
   734→export async function changePlan(
   735→  businessId: string,
   736→  targetPlanCode: string,
   737→  opts: { actorId?: string | null; source?: string } = {},
   738→): Promise<SubscriptionMutationResult> {
   739→  const billingNotifyQueue: PendingBillingNotification[] = [];
   740→  const result = await db.$transaction(
   741→    async (tx) => {
   742→      let sub = await loadSubscription(tx, businessId);
   743→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   744→
   745→      const target = await tx.plan.findUnique({ where: { code: targetPlanCode } });
   746→      if (!target || !target.active) {
   747→        throw new BillingError("INVALID_PLAN", "The requested plan does not exist or is inactive.", {
   748→          planCode: targetPlanCode,
   749→        });
   750→      }
   751→
   752→      if (sub.planId === target.id) {
   753→        // §48 idempotency — same plan → no event, no mutation.
   754→        return { sub, changed: false as const };
   755→      }
   756→
   757→      const fromStatus = asSubscriptionStatus(sub.status);
   758→      const wasCanceled = fromStatus === "CANCELED";
   759→      const reactivating = wasCanceled || fromStatus === "EXPIRED" || fromStatus === "PAUSED";
   760→      // Any plan change re-engages the subscription (a deliberate renewal
   761→      // decision by an authorized user — never automatic, §45).
   762→      assertTransition(fromStatus, "ACTIVE");
   763→
   764→      if (reactivating) {
   765→        billingNotifyQueue.push({
   766→          eventId: await eventAndAudit(tx, {
   767→            businessId,
   768→            subscriptionId: sub.id,
   769→            type: "REACTIVATED",
   770→            fromStatus,
   771→            toStatus: "ACTIVE",
   772→            fromPlanCode: sub.plan.code,
   773→            toPlanCode: target.code,
   774→            actorId: opts.actorId ?? null,
   775→            auditAction: "billing.subscription_reactivated",
   776→            auditBefore: { status: fromStatus, planCode: sub.plan.code },
   777→            auditAfter: { status: "ACTIVE", planCode: target.code },
   778→          }),
   779→          type: "REACTIVATED",
   780→          toPlanCode: target.code,
   781→        });
   782→      }
   783→
   784→      const now = new Date();
   785→      const period =
   786→        sub.currentPeriodEnd && sub.currentPeriodEnd.getTime() > now.getTime()
   787→          ? { start: sub.currentPeriodStart ?? now, end: sub.currentPeriodEnd }
   788→          : { start: now, end: periodEnd(now, (isBillingPeriod(sub.billingPeriod) ? sub.billingPeriod : "MONTHLY") as BillingPeriod) };
   789→
   790→      const updated = await tx.subscription.update({
   791→        where: { id: sub.id },
   792→        data: {
   793→          planId: target.id,
   794→          status: "ACTIVE",
   795→          cancelAtPeriodEnd: false,
   796→          canceledAt: null,
   797→          endedAt: null,
   798→          currentPeriodStart: period.start,
   799→          currentPeriodEnd: period.end,
   800→        },
   801→        include: { plan: true },
   802→      });
   803→
   804→      billingNotifyQueue.push({
   805→        eventId: await eventAndAudit(tx, {
   806→          businessId,
   807→          subscriptionId: sub.id,
   808→          type: "PLAN_CHANGED",
   809→          fromStatus,
   810→          toStatus: "ACTIVE",
   811→          fromPlanCode: sub.plan.code,
   812→          toPlanCode: target.code,
   813→          data: { source: opts.source ?? "settings" },
   814→          actorId: opts.actorId ?? null,
   815→          auditAction: "billing.plan_changed",
   816→          auditBefore: { planCode: sub.plan.code, status: fromStatus },
   817→          auditAfter: { planCode: target.code, status: "ACTIVE" },
   818→        }),
   819→        type: "PLAN_CHANGED",
   820→        toPlanCode: target.code,
   821→      });
   822→
   823→      return { sub: updated as SubRow, changed: true as const };
   824→    },
   825→    { isolationLevel: "Serializable" },
   826→  );
   827→
   828→  invalidateBillingCache(businessId);
   829→  flushBillingNotifications(businessId, billingNotifyQueue);
   830→  return {
   831→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
   832→    changed: result.changed,
   833→  };
   834→}

   836→/**
   837→ * Cancellation (§44): sets cancelAtPeriodEnd + status CANCELED — access
   838→ * continues until currentPeriodEnd, then lazy expiry moves it to EXPIRED.
   839→ * Idempotent (§48): a second cancel is a no-op. Data is never touched.
   840→ */
   841→export async function cancelSubscription(
   842→  businessId: string,
   843→  opts: { actorId?: string | null } = {},
   844→): Promise<SubscriptionMutationResult> {
   845→  const billingNotifyQueue: PendingBillingNotification[] = [];
   846→  const result = await db.$transaction(
   847→    async (tx) => {
   848→      const sub = await loadSubscription(tx, businessId);
   849→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   850→
   851→      const fromStatus = asSubscriptionStatus(sub.status);
   852→      if (fromStatus === "CANCELED" && sub.cancelAtPeriodEnd) {
   853→        return { sub, changed: false as const };
   854→      }
   855→      assertTransition(fromStatus, "CANCELED");
   856→
   857→      const updated = await tx.subscription.update({
   858→        where: { id: sub.id },
   859→        data: { status: "CANCELED", cancelAtPeriodEnd: true, canceledAt: new Date() },
   860→        include: { plan: true },
   861→      });
   862→      billingNotifyQueue.push({
   863→        eventId: await eventAndAudit(tx, {
   864→          businessId,
   865→          subscriptionId: sub.id,
   866→          type: "CANCELED",
   867→          fromStatus,
   868→          toStatus: "CANCELED",
   869→          fromPlanCode: sub.plan.code,
   870→          actorId: opts.actorId ?? null,
   871→          auditAction: "billing.subscription_canceled",
   872→          auditBefore: { status: fromStatus, planCode: sub.plan.code },
   873→          auditAfter: { status: "CANCELED", cancelAtPeriodEnd: true },
   874→        }),
   875→        type: "CANCELED",
   876→        toPlanCode: sub.plan.code,
   877→      });
   878→      return { sub: updated as SubRow, changed: true as const };
   879→    },
   880→    { isolationLevel: "Serializable" },
   881→  );
   882→  invalidateBillingCache(businessId);
   883→  flushBillingNotifications(businessId, billingNotifyQueue);
   884→  const decision = decidePlan(result.sub);
   885→  return {
   886→    subscription: subscriptionDto(
   887→      result.sub,
   888→      decision.kind === "own-plan" ? result.sub.plan.code : FREE_PLAN_CODE,
   889→      decision.entitled,
   890→    ),
   891→    changed: result.changed,
   892→  };
   893→}

   895→/**
   896→ * Reactivation (§45): only a CANCELED subscription with remaining grace can
   897→ * return to ACTIVE. EXPIRED requires the explicit changePlan renewal path.
   898→ * Idempotent: reactivating an ACTIVE subscription is a no-op.
   899→ */
   900→export async function reactivateSubscription(
   901→  businessId: string,
   902→  opts: { actorId?: string | null } = {},
   903→): Promise<SubscriptionMutationResult> {
   904→  const billingNotifyQueue: PendingBillingNotification[] = [];
   905→  const result = await db.$transaction(
   906→    async (tx) => {
   907→      const sub = await loadSubscription(tx, businessId);
   908→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   909→
   910→      const fromStatus = asSubscriptionStatus(sub.status);
   911→      if (fromStatus === "ACTIVE") return { sub, changed: false as const };
   912→      // §45/§46: reactivation is ONLY the CANCELED → ACTIVE path (undo the
   913→      // cancellation while the period still runs). An EXPIRED subscription
   914→      // returns exclusively through changePlan — a deliberate, authorized
   915→      // renewal — and PAUSED resumes through a dedicated admin flow that
   916→      // does not exist yet. Arbitrary hopping is refused.
   917→      if (fromStatus === "EXPIRED" || fromStatus === "PAUSED") {
   918→        throw new BillingError(
   919→          "INVALID_SUBSCRIPTION_STATE",
   920→          "This subscription state cannot be reactivated. Use a plan change to renew.",
   921→          { from: fromStatus, allowed: VALID_TRANSITIONS[fromStatus] },
   922→        );
   923→      }
   924→      assertTransition(fromStatus, "ACTIVE");
   925→
   926→      const updated = await tx.subscription.update({
   927→        where: { id: sub.id },
   928→        data: { status: "ACTIVE", cancelAtPeriodEnd: false, canceledAt: null, endedAt: null },
   929→        include: { plan: true },
   930→      });
   931→      billingNotifyQueue.push({
   932→        eventId: await eventAndAudit(tx, {
   933→          businessId,
   934→          subscriptionId: sub.id,
   935→          type: "REACTIVATED",
   936→          fromStatus,
   937→          toStatus: "ACTIVE",
   938→          fromPlanCode: sub.plan.code,
   939→          actorId: opts.actorId ?? null,
   940→          auditAction: "billing.subscription_reactivated",
   941→          auditBefore: { status: fromStatus },
   942→          auditAfter: { status: "ACTIVE" },
   943→        }),
   944→        type: "REACTIVATED",
   945→        toPlanCode: sub.plan.code,
   946→      });
   947→      return { sub: updated as SubRow, changed: true as const };
   948→    },
   949→    { isolationLevel: "Serializable" },
   950→  );
   951→  invalidateBillingCache(businessId);
   952→  flushBillingNotifications(businessId, billingNotifyQueue);
   953→  return {
   954→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
   955→    changed: result.changed,
   956→  };
   957→}

  1088→): Promise<SubscriptionMutationResult> {
  1089→  const billingNotifyQueue: PendingBillingNotification[] = [];
  1090→  const result = await db.$transaction(
  1091→    async (tx) => {
  1092→      let sub = await loadSubscription(tx, businessId);
  1093→      if (!sub) {
  1094→        sub = await ensureSubscription(businessId, { tx, source: "trial" });
  1095→      }
  1096→      if (sub.trialStart) {
  1097→        throw new BillingError("BUSINESS_NOT_ELIGIBLE", "This business has already used its trial.");
  1098→      }
  1099→      const fromStatus = asSubscriptionStatus(sub.status);
  1100→      assertTransition(fromStatus, "TRIALING");
  1101→
  1102→      const now = new Date();
  1103→      const trialEnd = new Date(now.getTime() + days * 24 * 60 * 60 * 1000);
  1104→      const updated = await tx.subscription.update({
  1105→        where: { id: sub.id },
  1106→        data: {
  1107→          status: "TRIALING",
  1108→          trialStart: now,
  1109→          trialEnd,
  1110→          cancelAtPeriodEnd: false,
  1111→          canceledAt: null,
  1112→        },
  1113→        include: { plan: true },
  1114→      });
  1115→      billingNotifyQueue.push({
  1116→        eventId: await eventAndAudit(tx, {
  1117→          businessId,
  1118→          subscriptionId: sub.id,
  1119→          type: "TRIAL_STARTED",
  1120→          fromStatus,
  1121→          toStatus: "TRIALING",
  1122→          fromPlanCode: sub.plan.code,
  1123→          data: { days, source: opts.source ?? "manual" },
  1124→          actorId: opts.actorId ?? null,
  1125→          auditAction: "billing.trial_started",
  1126→          auditAfter: { status: "TRIALING", trialEnd: trialEnd.toISOString() },
  1127→        }),
  1128→        type: "TRIAL_STARTED",
  1129→        toPlanCode: sub.plan.code,
  1130→      });
  1131→      return { sub: updated as SubRow, changed: true as const };
  1132→    },
  1133→    { isolationLevel: "Serializable" },
  1134→  );
  1135→  invalidateBillingCache(businessId);
  1136→  flushBillingNotifications(businessId, billingNotifyQueue);
  1137→  return {
  1138→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
  1139→    changed: result.changed,
  1140→  };
  1141→}

  1143→/** End trial → ACTIVE (TRIAL_ENDED). Keeps the plan (no payment policy yet). */
  1144→export async function endTrial(
  1145→  businessId: string,
  1146→  opts: { actorId?: string | null } = {},
  1147→): Promise<SubscriptionMutationResult> {
  1148→  const billingNotifyQueue: PendingBillingNotification[] = [];
  1149→  const result = await db.$transaction(
  1150→    async (tx) => {
  1151→      const sub = await loadSubscription(tx, businessId);
  1152→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
  1153→      const fromStatus = asSubscriptionStatus(sub.status);
  1154→      if (fromStatus !== "TRIALING") {
  1155→        throw new BillingError("INVALID_SUBSCRIPTION_STATE", "The subscription is not in a trial.", {
  1156→          status: fromStatus,
  1157→        });
  1158→      }
  1159→      const updated = await tx.subscription.update({
  1160→        where: { id: sub.id },
  1161→        data: { status: "ACTIVE" },
  1162→        include: { plan: true },
  1163→      });
  1164→      billingNotifyQueue.push({
  1165→        eventId: await eventAndAudit(tx, {
  1166→          businessId,
  1167→          subscriptionId: sub.id,
  1168→          type: "TRIAL_ENDED",
  1169→          fromStatus,
  1170→          toStatus: "ACTIVE",
  1171→          fromPlanCode: sub.plan.code,
  1172→          actorId: opts.actorId ?? null,
  1173→          auditAction: "billing.trial_ended",
  1174→          auditBefore: { status: fromStatus },
  1175→          auditAfter: { status: "ACTIVE" },
  1176→        }),
  1177→        type: "TRIAL_ENDED",
  1178→        toPlanCode: sub.plan.code,
  1179→      });
  1180→      return { sub: updated as SubRow, changed: true as const };
  1181→    },
  1182→    { isolationLevel: "Serializable" },
  1183→  );
  1184→  invalidateBillingCache(businessId);
  1185→  flushBillingNotifications(businessId, billingNotifyQueue);
  1186→  return {
  1187→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
  1188→    changed: result.changed,
  1189→  };
  1190→}

  1196→export async function listSubscriptionEvents(
  1197→  businessId: string,
  1198→  opts: { page?: number; pageSize?: number } = {},
  1199→): Promise<{ items: SubscriptionEventDto[]; total: number; page: number; pageSize: number }> {
  1200→  const page = Math.max(1, opts.page ?? 1);
  1201→  const pageSize = Math.min(100, Math.max(1, opts.pageSize ?? 20));
  1202→  const [rows, total] = await Promise.all([
  1203→    db.subscriptionEvent.findMany({
  1204→      where: { businessId },
  1205→      orderBy: { createdAt: "desc" },
  1206→      skip: (page - 1) * pageSize,
  1207→      take: pageSize,
  1208→      include: { actor: { select: { name: true, email: true } } },
  1209→    }),
  1210→    db.subscriptionEvent.count({ where: { businessId } }),
  1211→  ]);
  1212→  return {
  1213→    items: rows.map((r) => ({
  1214→      id: r.id,
  1215→      type: (isSubscriptionEventTypeSafe(r.type) ? r.type : "CREATED") as SubscriptionEventType,
  1216→      fromStatus: r.fromStatus,
  1217→      toStatus: r.toStatus,
  1218→      fromPlanCode: r.fromPlanCode,
  1219→      toPlanCode: r.toPlanCode,
  1220→      data: r.dataJson ? safeParse(r.dataJson) : null,
  1221→      actor: r.actor ? { name: r.actor.name, email: r.actor.email } : null,
  1222→      createdAt: r.createdAt.toISOString(),
  1223→    })),
  1224→    total,
  1225→    page,
  1226→    pageSize,
  1227→  };
  1228→}
  1229→
  1230→function isSubscriptionEventTypeSafe(value: string): boolean {
  1231→  return [
  1232→    "CREATED",
  1233→    "PLAN_CHANGED",
  1234→    "TRIAL_STARTED",
  1235→    "TRIAL_ENDED",
  1236→    "CANCELED",
  1237→    "REACTIVATED",
  1238→    "EXPIRED",
  1239→    "PAUSED",
  1240→    "RESUMED",
  1241→  ].includes(value);
  1242→}
  1243→
  1244→function safeParse(json: string): unknown {
  1245→  try {
  1246→    return JSON.parse(json);
  1247→  } catch {
  1248→    return null;
  1249→  }
  1250→}
