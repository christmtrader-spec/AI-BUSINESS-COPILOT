    95→    95→function inline(text: string, keyPrefix: string): ReactNode[] {
    96→    96→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
    97→    97→  return parts.map((part, i) => {
    98→    98→    const key = `${keyPrefix}-${i}`;
    99→    99→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   100→   100→      return (
   101→   101→        <strong key={key} className="font-semibold">
   102→   102→          {part.slice(2, -2)}
   103→   103→        </strong>
   104→   104→      );
   105→   105→    }
   106→   106→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   107→   107→      return (
   108→   108→        <code
   109→   109→          key={key}
   110→   110→          dir="ltr"
   111→   111→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   112→   112→        >
   113→   113→          {part.slice(1, -1)}
   114→   114→        </code>
   115→   115→      );
   116→   116→    }
   117→   117→    return <span key={key}>{part}</span>;
   118→   118→  });
   119→   119→}

   121→   121→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   122→   122→  const nodes: ReactNode[] = [];
   123→   123→  let bullets: string[] = [];
   124→   124→  let para: string[] = [];
   125→   125→
   126→   126→  const flushBullets = () => {
   127→   127→    if (bullets.length === 0) return;
   128→   128→    nodes.push(
   129→   129→      <ul
   130→   130→        key={`ul-${nodes.length}`}
   131→   131→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   132→   132→      >
   133→   133→        {bullets.map((b, i) => (
   134→   134→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   135→   135→        ))}
   136→   136→      </ul>,
   137→   137→    );
   138→   138→    bullets = [];
   139→   139→  };
   140→   140→  const flushPara = () => {
   141→   141→    if (para.length === 0) return;
   142→   142→    nodes.push(
   143→   143→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   144→   144→        {inline(para.join("\n"), `p${nodes.length}`)}
   145→   145→      </p>,
   146→   146→    );
   147→   147→    para = [];
   148→   148→  };
   149→   149→
   150→   150→  for (const raw of text.split("\n")) {
   151→   151→    const line = raw.trimEnd();
   152→   152→    if (/^\s*[-*•]\s+/.test(line)) {
   153→   153→      flushPara();
   154→   154→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   155→   155→    } else if (line === "") {
   156→   156→      flushBullets();
   157→   157→      flushPara();
   158→   158→    } else {
   159→   159→      flushBullets();
   160→   160→      para.push(line);
   161→   161→    }
   162→   162→  }
   163→   163→  flushBullets();
   164→   164→  flushPara();
   165→   165→
   166→   166→  return (
   167→   167→    <div className="space-y-1.5 text-sm">
   168→   168→      {nodes}
   169→   169→      {streaming ? (
   170→   170→        <span
   171→   171→          aria-hidden="true"
   172→   172→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   173→   173→        />
   174→   174→      ) : null}
   175→   175→    </div>
   176→   176→  );
   177→   177→}

   271→   271→export function CopilotView({
   272→   272→  business,
   273→   273→  role,
   274→   274→}: {
   275→   275→  business: BusinessSummary;
   276→   276→  role: Role;
   277→   277→}) {
   278→   278→  const t = useTranslations("copilot");
   279→   279→  const tCommon = useTranslations("common");
   280→   280→  const tErrors = useTranslations("errors");
   281→   281→
   282→   282→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   283→   283→  const [input, setInput] = useState("");
   284→   284→  const [activeStreamId, setActiveStreamId] = useState<string | null>(null);
   285→   285→  const [days, setDays] = useState<7 | 30 | 90>(30);
   286→   286→  const [copiedId, setCopiedId] = useState<string | null>(null);
   287→   287→  const [dataOpen, setDataOpen] = useState(false);
   288→   288→  const [atBottom, setAtBottom] = useState(true);
   289→   289→  const lastQuestionRef = useRef<string | null>(null);
   290→   290→  const abortRef = useRef<AbortController | null>(null);
   291→   291→  const scrollRef = useRef<HTMLDivElement>(null);
   292→   292→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   293→   293→
   294→   294→  const canUse = can(role, "copilot:use");
   295→   295→  const pending = activeStreamId !== null;
   296→   296→
   297→   297→  const suggestions = useMemo(
   298→   298→    () => [
   299→   299→      t("suggestion1"),
   300→   300→      t("suggestion2"),
   301→   301→      t("suggestion3"),
   302→   302→      t("suggestion4"),
   303→   303→      t("suggestion5"),
   304→   304→      t("suggestion6"),
   305→   305→    ],
   306→   306→    [t],
   307→   307→  );
   308→   308→
   309→   309→  // Keep the latest message in view (scroll-only effect — no setState).
   310→   310→  useEffect(() => {
   311→   311→    if (atBottom && scrollRef.current) {
   312→   312→      scrollRef.current.scrollTo({
   313→   313→        top: scrollRef.current.scrollHeight,
   314→   314→        behavior: "smooth",
   315→   315→      });
   316→   316→    }
   317→   317→  }, [messages, pending, atBottom]);
   318→   318→
   319→   319→  // Abort any in-flight stream when the module unmounts.
   320→   320→  useEffect(() => {
   321→   321→    return () => abortRef.current?.abort();
   322→   322→  }, []);
   323→   323→
   324→   324→  const patchAssistant = useCallback(
   325→   325→    (id: string, patch: (m: ChatMessage) => ChatMessage) => {
   326→   326→      setMessages((prev) => prev.map((m) => (m.id === id ? patch(m) : m)));
   327→   327→    },
   328→   328→    [],
   329→   329→  );
   330→   330→
   331→   331→  const localizedError = useCallback(
   332→   332→    (code: string, retryAfterMs?: number) => {
   333→   333→      switch (code) {
   334→   334→        case "RATE_LIMITED":
   335→   335→          return t("errorRateLimited", {
   336→   336→            seconds: String(Math.max(1, Math.ceil((retryAfterMs ?? 0) / 1000))),
   337→   337→          });
   338→   338→        case "PERMISSION_DENIED":
   339→   339→          return t("errorForbidden");
   340→   340→        case "AI_UNAVAILABLE":
   341→   341→          return t("errorAiUnavailable");
   342→   342→        default:
   343→   343→          return tErrors("generic");
   344→   344→      }
   345→   345→    },
   346→   346→    [t, tErrors],
   347→   347→  );
   348→   348→
   349→   349→  const send = useCallback(
   350→   350→    async (question: string, history: ChatMessage[]) => {
   351→   351→      const trimmed = question.trim();
   352→   352→      if (!trimmed || trimmed.length > 500 || activeStreamId !== null) return;
   353→   353→      lastQuestionRef.current = trimmed;
   354→   354→      setAtBottom(true);
   355→   355→
   356→   356→      const assistantId = nextId();
   357→   357→      setMessages((prev) => [
   358→   358→        ...prev,
   359→   359→        { id: assistantId, role: "assistant", content: "" },
   360→   360→      ]);
   361→   361→      setActiveStreamId(assistantId);
   362→   362→
   363→   363→      const controller = new AbortController();
   364→   364→      abortRef.current = controller;
   365→   365→
   366→   366→      try {
   367→   367→        const res = await fetch("/api/v1/copilot/stream", {
   368→   368→          method: "POST",
   369→   369→          credentials: "same-origin",
   370→   370→          headers: {
   371→   371→            "content-type": "application/json",
   372→   372→            "x-business-id": business.id,
   373→   373→          },
   374→   374→          body: JSON.stringify({
   375→   375→            question: trimmed,
   376→   376→            history: history
   377→   377→              .filter((m) => !m.failed)
   378→   378→              .slice(-6)
   379→   379→              .map((m) => ({ role: m.role, content: m.content })),
   380→   380→            days,
   381→   381→          }),
   382→   382→          signal: controller.signal,
   383→   383→        });
   384→   384→
   385→   385→        if (!res.ok || !res.body) {
   386→   386→          let code = "INTERNAL_ERROR";
   387→   387→          let retryAfterMs: number | undefined;
   388→   388→          try {
   389→   389→            const body = (await res.json()) as {
   390→   390→              error?: { code?: string; details?: { retryAfterMs?: number } };
   391→   391→            };
   392→   392→            if (body?.error?.code) code = body.error.code;
   393→   393→            retryAfterMs = body?.error?.details?.retryAfterMs;
   394→   394→          } catch {
   395→   395→            // keep defaults
   396→   396→          }
   397→   397→          patchAssistant(assistantId, (m) => ({
   398→   398→            ...m,
   399→   399→            content: localizedError(code, retryAfterMs),
   400→   400→            failed: true,
   401→   401→          }));
   402→   402→          return;
   403→   403→        }
   404→   404→
   405→   405→        // Parse the SSE stream incrementally.
   406→   406→        const reader = res.body.getReader();
   407→   407→        const decoder = new TextDecoder();
   408→   408→        let buffer = "";
   409→   409→        let streamError: { code?: string; message?: string } | null = null;
   410→   410→        let sawDone = false;
   411→   411→
   412→   412→        while (true) {
   413→   413→          const { done, value } = await reader.read();
   414→   414→          if (done) break;
   415→   415→          buffer += decoder.decode(value, { stream: true });
   416→   416→          const events = buffer.split("\n\n");
   417→   417→          buffer = events.pop() ?? "";
   418→   418→          for (const evt of events) {
   419→   419→            for (const line of evt.split("\n")) {
   420→   420→              if (!line.startsWith("data: ")) continue;
   421→   421→              let payload: {
   422→   422→                meta?: CopilotMeta;
   423→   423→                delta?: string;
   424→   424→                done?: boolean;
   425→   425→                error?: { code?: string; message?: string };
   426→   426→              };
   427→   427→              try {
   428→   428→                payload = JSON.parse(line.slice(6));
   429→   429→              } catch {
   430→   430→                continue;
   431→   431→              }
   432→   432→              if (payload.meta) {
   433→   433→                // First event — dataset provenance for this answer.
   434→   434→                const meta = payload.meta;
   435→   435→                patchAssistant(assistantId, (m) => ({ ...m, meta }));
   436→   436→              } else if (typeof payload.delta === "string" && payload.delta.length > 0) {
   437→   437→                const delta = payload.delta;
   438→   438→                patchAssistant(assistantId, (m) => ({
   439→   439→                  ...m,
   440→   440→                  content: m.content + delta,
   441→   441→                }));
   442→   442→              } else if (payload.error) {
   443→   443→                streamError = payload.error;
   444→   444→              } else if (payload.done) {
   445→   445→                sawDone = true;
   446→   446→              }
   447→   447→            }
   448→   448→          }
   449→   449→        }
   450→   450→
   451→   451→        if (streamError) {
   452→   452→          const code = streamError.code ?? "INTERNAL_ERROR";
   453→   453→          if (code === "RATE_LIMITED" || code === "AI_UNAVAILABLE" || !sawDone) {
   454→   454→            patchAssistant(assistantId, (m) =>
   455→   455→              m.content.trim().length === 0
   456→   456→                ? { ...m, content: localizedError(code), failed: true }
   457→   457→                : { ...m, content: `${m.content}\n\n⚠ ${localizedError(code)}` },
   458→   458→            );
   459→   459→          }
   460→   460→        }
   461→   461→      } catch (error) {
   462→   462→        if ((error as Error)?.name === "AbortError") {
   463→   463→          // User pressed Stop — keep the partial answer, drop it if empty.
   464→   464→          setMessages((prev) =>
   465→   465→            prev.flatMap((m) =>
   466→   466→              m.id === assistantId ? (m.content.trim() ? [m] : []) : [m],
   467→   467→            ),
   468→   468→          );
   469→   469→        } else if (error instanceof ApiClientError) {
   470→   470→          patchAssistant(assistantId, (m) => ({
   471→   471→            ...m,
   472→   472→            content: localizedError(error.code),
   473→   473→            failed: true,
   474→   474→          }));
   475→   475→        } else {
   476→   476→          patchAssistant(assistantId, (m) => ({
   477→   477→            ...m,
   478→   478→            content: localizedError("NETWORK_ERROR"),
   479→   479→            failed: true,
   480→   480→          }));
   481→   481→        }
   482→   482→      } finally {
   483→   483→        abortRef.current = null;
   484→   484→        setActiveStreamId((cur) => (cur === assistantId ? null : cur));
   485→   485→      }
   486→   486→    },
   487→   487→    [activeStreamId, business.id, days, localizedError, patchAssistant],
   488→   488→  );
   489→   489→
   490→   490→  const handleSubmit = () => {
   491→   491→    const trimmed = input.trim();
   492→   492→    if (!trimmed || pending) return;
   493→   493→    setInput("");
   494→   494→    if (textareaRef.current) textareaRef.current.style.height = "auto";
   495→   495→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: trimmed }]);
   496→   496→    void send(trimmed, messages);
   497→   497→  };
   498→   498→
   499→   499→  const handleStop = () => {
   500→   500→    abortRef.current?.abort();
   501→   501→  };
   502→   502→
   503→   503→  const handleRetry = () => {
