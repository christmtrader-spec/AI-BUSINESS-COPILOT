     3→/**
     4→ * Export dataset catalog — Prompt 31 §5/§6/§41-§54.
     5→ *
     6→ * LAWS:
     7→ *  - EXPLICIT allowlist: a dataset key that is not in this catalog can
     8→ *    never be exported (§39/§239 — unknown dataset rejected).
     9→ *  - Every dataset is classified (§6): PUBLIC_BUSINESS / CONFIDENTIAL /
    10→ *    HIGHLY_SENSITIVE. No dataset below contains authentication or
    11→ *    security secrets (§9/§12-§19 — enforced again by the static scanner).
    12→ *  - Tenant-owned only: fetchers filter by businessId unconditionally
    13→ *    (§32); the caller's ctx.businessId is the ONLY business key used.
    14→ *  - Per-row businessId is deliberately EXCLUDED (§10 — the manifest
    15→ *    carries the business identity; per-row copies are unnecessary).
    16→ *  - Financial export is READ-ONLY representation of the same records the
    17→ *    UI already shows (§222/§223) — amounts as exact integer minor units
    18→ *    in JSON, exact decimal strings in CSV (§226/§227).
    19→ *  - NOT exported (explicit Prompt 31 decisions, documented):
    20→ *      · AI chat transcripts (user-personal, out of portability scope §53)
    21→ *      · tenant audit trail (actor identities + history, not in catalog §5)
    22→ *      · receivables/payables (DERIVED views of sales/expenses — the same
    23→ *        facts are already inside those datasets)
    24→ *      · anything platform: subscriptions internals, provider costs, AI
    25→ *        rates, platform audit, operators (§9/§51/§52/§189).
    26→ */

   122→/* ------------------------------ row building ------------------------------- */
   123→
   124→export type ExportRow = {
   125→  /** JSON row — IDs + relationships preserved, money as minor integers (§54/§226). */
   126→  json: Record<string, unknown>;
   127→  /** CSV row — exact decimal strings for money, ISO-8601 UTC dates (§227/§231). */
   128→  csv: Array<string | number | null>;
   129→};
   130→
   131→export type DatasetFetchResult = {
   132→  rows: ExportRow[];
   133→  /** True when the dataset hit the export record cap (job must FAIL, §26). */
   134→  truncated: boolean;
   135→};
   136→
   137→export type DatasetFetchCtx = {
   138→  businessId: string;
   139→  /** Business decimal precision for money → CSV decimal strings. */
   140→  decimalPrecision: number;
   141→  /** Hard cap applied per fetch — caller passes EXPORT_MAX_RECORDS headroom. */
   142→  maxRecords: number;
   143→};

   145→function iso(value: Date | null | undefined): string | null {
   146→  return value ? value.toISOString() : null;
   147→}

   153→type Customer = Prisma.CustomerGetPayload<object>;

   155→function customerRow(c: Customer, precision: number): ExportRow {
   156→  void precision; // customers carry no money fields
   157→  return {
   158→    json: {
   159→      id: c.id,
   160→      name: c.name,
   161→      phone: c.phone,
   162→      email: c.email,
   163→      address: c.address,
   164→      note: c.note,
   165→      deletedAt: iso(c.deletedAt),
   166→      createdAt: c.createdAt.toISOString(),
   167→      updatedAt: c.updatedAt.toISOString(),
   168→    },
   169→    csv: [c.id, c.name, c.phone, c.email, c.address, c.note, iso(c.deletedAt), c.createdAt.toISOString(), c.updatedAt.toISOString()],
   170→  };
   171→}

   191→function productRow(p: Prisma.ProductGetPayload<object>, precision: number): ExportRow {
   192→  return {
   193→    json: {
   194→      id: p.id,
   195→      name: p.name,
   196→      sku: p.sku,
   197→      unit: p.unit,
   198→      salePriceMinor: p.salePriceMinor,
   199→      costPriceMinor: p.costPriceMinor,
   200→      stockQty: p.stockQty,
   201→      lowStockAt: p.lowStockAt,
   202→      isActive: p.isActive,
   203→      deletedAt: iso(p.deletedAt),
   204→      createdAt: p.createdAt.toISOString(),
   205→      updatedAt: p.updatedAt.toISOString(),
   206→    },
   207→    csv: [
   208→      p.id,
   209→      p.name,
   210→      p.sku,
   211→      p.unit,
   212→      formatMinorToDecimal(p.salePriceMinor, precision),
   213→      formatMinorToDecimal(p.costPriceMinor, precision),
   214→      p.stockQty,
   215→      p.lowStockAt,
   216→      p.isActive ? "true" : "false",
   217→      iso(p.deletedAt),
   218→      p.createdAt.toISOString(),
   219→      p.updatedAt.toISOString(),
   220→    ],
   221→  };
   222→}

   224→async function fetchWithCap<T>(
   225→  cap: number,
   226→  query: (take: number) => Promise<T[]>,
   227→): Promise<{ items: T[]; truncated: boolean }> {
   228→  // Fetch one row beyond the cap so truncation is DETECTABLE, never silent.
   229→  const items = await query(cap + 1);
   230→  return { items: items.slice(0, cap), truncated: items.length > cap };
   231→}
