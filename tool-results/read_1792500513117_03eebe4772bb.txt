    64→    64→/** Sidebar label key for the current view (used for <title> too). */
    65→    65→function labelKeyOf(view: AppView): string {
    66→    66→  switch (view.kind) {
    67→    67→    case "dashboard":
    68→    68→      return "dashboard";
    69→    69→    case "audit":
    70→    70→      return "audit";
    71→    71→    case "settings:business":
    72→    72→      return "settings";
    73→    73→    case "settings:members":
    74→    74→      return "teamMembers";
    75→    75→    case "settings:billing":
    76→    76→      return "billing";
    77→    77→    case "settings:security":
    78→    78→      return "security";
    79→    79→    case "settings:data":
    80→    80→      return "dataPortability";
    81→    81→    case "notifications":
    82→    82→      return "notifications";
    83→    83→    case "settings:notifications":
    84→    84→      return "notifications";
    85→    85→    // Prompt 33 — Search Results Center document title.
    86→    86→    case "search":
    87→    87→      return "search";
    88→    88→    // Prompt 35 (§36) — CRM module document title.
    89→    89→    case "crm":
    90→    90→      return "crm";
    91→    91→    // Prompt 34 — entity detail titles reuse the owning module's nav label.
    92→    92→    case "entity":
    93→    93→      return ENTITY_ROUTES[view.entity].labelKey;
    94→    94→    // Prompt 39 — party statement document title (§57).
    95→    95→    case "statements":
    96→    96→      return "statements";
    97→    97→    // Prompt 41 — purchase-order document titles (§71).
    98→    98→    case "purchase-orders":
    99→    99→      return "purchaseOrders";
   100→   100→    case "purchase-order":
   101→   101→      return "purchaseOrders";
   102→   102→    // Prompt 42 — goods-receipt document titles (§117).
   103→   103→    case "inventory-goods-receipts":
   104→   104→      return "goodsReceipts";
   105→   105→    case "inventory-goods-receipt":
   106→   106→      return "goodsReceipts";
   107→   107→    // Prompt 43 — goods-return document titles (§91).
   108→   108→    case "inventory-goods-returns":
   109→   109→      return "goodsReturns";
   110→   110→    case "inventory-goods-return":
   111→   111→      return "goodsReturns";
   112→   112→    // Prompt 44 — goods-dispatch document titles.
   113→   113→    case "inventory-goods-dispatches":
   114→   114→      return "goodsDispatches";
   115→   115→    case "inventory-goods-dispatch":
   116→   116→      return "goodsDispatches";
   117→   117→    // Prompt 45 — sales-order document titles.
   118→   118→    case "sales-orders":
   119→   119→      return "salesOrders";
   120→   120→    case "sales-order":
   121→   121→      return "salesOrders";
   122→   122→    // Prompt 47 — sales-quotation document titles.
   123→   123→    case "sales-quotations":
   124→   124→      return "salesQuotations";
   125→   125→    case "sales-quotation":
   126→   126→      return "salesQuotations";
   127→   127→    // Prompt 48 — supplier-bill document titles (§4.1).
   128→   128→    case "supplier-bills":
   129→   129→      return "supplierBills";
   130→   130→    case "supplier-bill":
   131→   131→      return "supplierBills";
   132→   132→    // Prompt 49 — detailed-aging document titles (the owning balances module).
   133→   133→    case "aging-details":
   134→   134→      return view.side === "payables" ? "payables" : "receivables";
   135→   135→    case "profile":
   136→   136→      return "profile";
   137→   137→    case "module": {
   138→   138→      const map: Record<string, string> = {
   139→   139→        customers: "customers",
   140→   140→        suppliers: "suppliers",
   141→   141→        products: "products",
   142→   142→        inventory: "inventory",
   143→   143→        sales: "sales",
   144→   144→        expenses: "expenses",
   145→   145→        payments: "payments",
   146→   146→        receivables: "receivables",
   147→   147→        payables: "payables",
   148→   148→        reports: "reports",
   149→   149→        copilot: "copilot",
   150→   150→      };
   151→   151→      return map[view.module] ?? "dashboard";
   152→   152→    }
   153→   153→    default:
   154→   154→      return "dashboard";
   155→   155→  }
   156→   156→}

   171→   171→  const navigate = useAppStore((s) => s.navigate);
   172→   172→
   173→   173→  const membership =
   174→   174→    memberships.find((m) => m.business.id === activeId) ?? memberships[0] ?? null;
   175→   175→  const role: Role = asRole(membership?.role);
   176→   176→
   177→   177→  // Phase 06 — per-module document titles (localized: "Sales · Business · App").
   178→   178→  const moduleLabel = tNav(labelKeyOf(view) as "dashboard");
   179→   179→  useEffect(() => {
   180→   180→    const parts = [
   181→   181→      moduleLabel,
   182→   182→      membership?.business.name,
   183→   183→      tCommon("appName"),
   184→   184→    ].filter(Boolean);

   194→   194→    if (!activeBusinessId) return;

   208→   208→  // `/?payment_return=<id>` load the underlying view is already the default
   209→   209→  // dashboard, so clicking «Dashboard» produces the SAME viewKey and would

   213→   213→    if (prevViewRef.current === null) {
   214→   214→      prevViewRef.current = view;
   215→   215→      return;
   216→   216→    }
   217→   217→    if (prevViewRef.current !== view) {
   218→   218→      prevViewRef.current = view;

   221→   221→  }, [view, clearPaymentReturn]);

   224→   224→    <SidebarProvider>

   228→   228→          tab is hidden, static frame under prefers-reduced-motion, and it
   229→   229→          hides itself entirely when WebGL is unavailable). The content
   230→   230→          surfaces above stay readable: SidebarInset is slightly translucent
   231→   231→          so the aurora shows through subtly, while cards/popovers remain
   232→   232→          opaque. Nothing here affects data flow, routing or logic. */}
   233→   233→      <div aria-hidden="true" className="bg-app-fallback pointer-events-none fixed inset-0 -z-20" />
   234→   234→      <BusinessWebGLBackground className="opacity-35 dark:opacity-30" />

   236→   236→      <SidebarInset className="bg-background/85 dark:bg-background/80">
   237→   237→        {/* a11y: keyboard users can jump past the navigation chrome */}
   238→   238→        <a
   239→   239→          href="#main-content"
   240→   240→          className="bg-primary text-primary-foreground sr-only focus:not-sr-only focus:fixed focus:start-3 focus:top-3 focus:z-50 focus:rounded-md focus:px-3 focus:py-2 focus:text-sm focus:font-medium focus:shadow-lg"
   241→   241→        >
   242→   242→          {tSkip("skipToContent")}

   584→   584→export function useMembershipsQuery(enabled: boolean) {
   585→   585→  const queryClient = useQueryClient();
   586→   586→  return useQuery<BusinessListItem[]>({
   587→   587→    queryKey: ["businesses"],
   588→   588→    queryFn: async () => {
   589→   589→      const res = await fetch("/api/v1/businesses", { credentials: "same-origin" });
   590→   590→      let body: unknown = null;
   591→   591→      try { body = await res.json(); } catch { /* non-JSON */ };
   592→   592→      const err = (body as { error?: { code?: string } } | null)?.error;
   593→   593→      if (!res.ok) {
   594→   594→        // Prompt 30 §83/§229 — a globally invalidated session (password
   595→   595→        // change / MFA enable / admin kill switch) must land the user on
   596→   596→        // the sign-in view, never a permanent splash. Clear the stale
   597→   597→        // cookie client-side; NextAuth's session state flips to signed-out.
   598→   598→        if (res.status === 401 || err?.code === "SESSION_INVALIDATED" || err?.code === "SESSION_REVOKED") {
   599→   599→          queryClient.clear();
   600→   600→          void import("next-auth/react").then((m) => m.signOut({ redirect: false }));
   601→   601→        }
   602→   602→        throw new Error(err?.code ?? `HTTP_${res.status}`);
   603→   603→      }
   604→   604→      return (body as { data: BusinessListItem[] }).data;
   605→   605→    },
   606→   606→    enabled,
   607→   607→    staleTime: 60_000,
   608→   608→    retry: false,
   609→   609→  });

   612→   612→/**
   613→   613→ * Prompt 23 §41 — reads `?payment_return=<sessionId>` (set ONLY by our own
   614→   614→ * callback API). Value is validated shape-wise (cuid-ish length bound);
   615→   615→ * the session itself is re-authorized server-side, so a forged id simply
   616→   616→ * renders a 404-ish "unknown" state — never another tenant's data.
   617→   617→ * Listens to popstate so browser back/forward and the view's own
   618→   618→ * clear() stay in sync.
   619→   619→ */
   620→   620→function usePaymentReturnSessionId(): [string | null, () => void] {
   621→   621→  const read = () => {
   622→   622→    try {
   623→   623→      const raw = new URLSearchParams(window.location.search).get("payment_return");
   624→   624→      return raw && raw.length >= 20 && raw.length <= 40 && /^[a-zA-Z0-9_-]+$/.test(raw) ? raw : null;
   625→   625→    } catch {
   626→   626→      return null;
   627→   627→    }
   628→   628→  };
   629→   629→  const [sessionId, setSessionId] = useState<string | null>(read);
   630→   630→  useEffect(() => {
   631→   631→    const sync = () => setSessionId(read());
   632→   632→    window.addEventListener("popstate", sync);
   633→   633→    return () => window.removeEventListener("popstate", sync);
   634→   634→  }, []);
   635→   635→  const clear = useCallback(() => {
   636→   636→    // Strip ONLY the ?payment_return query — never touch the hash (the view
   637→   637→    // may be navigating via the app hash-router at the same time).
   638→   638→    window.history.replaceState(null, "", `${window.location.pathname}${window.location.hash}`);
   639→   639→    setSessionId(null);
   640→   640→  }, []);
   641→   641→  return [sessionId, clear];
   642→   642→}
