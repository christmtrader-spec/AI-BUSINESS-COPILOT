   169→   169→ *   gross = SUM(POSTED GoodsReceiptLine.receivedQty)
   170→   170→ *   returned = SUM(POSTED GoodsReturnLine.returnedQty)   [DB aggregate]
   171→   171→ *   net = gross − returned                                [§79 pure law]
   172→   172→ * returned > gross is return-history corruption (GOODS_RETURN_CORRUPT_HISTORY
   173→   173→ * 500); net > ordered is receiving-history corruption
   174→   174→ * (GOODS_RECEIPT_CORRUPT_RECEIVED 500). Nothing is ever clamped. Every
   175→   175→ * remaining-check in this service (line add §66, line update §68, receipt
   176→   176→ * POST §39 step 8) and the PO receiving summary consume THIS function —
   177→   177→ * there is no second gross/return calculation anywhere (§80).

   179→   179→async function netReceivedForPoLine(
   180→   180→  tx: Tx,
   181→   181→  purchaseOrderLineId: string,
   182→   182→  orderedQty: number,

   188→   188→    net = computeNetReceivedQty(gross, returned); // §79 — returned > gross throws
   189→   189→  } catch (error) {
   190→   190→    mapPureGoodsReturnDomainError(error); // stable GOODS_RETURN_CORRUPT_HISTORY 500
   191→   191→  }
   192→   192→  const remaining = remainingQtyOrIntegrity(orderedQty, net); // §100 — net > ordered throws
   193→   193→  return { gross, returned, net, remaining };
   194→   194→}

   303→   303→ * DRAFT/CANCELLED receipts never count (§99). Tenant scope is inherited
   304→   304→ * from the receipt→PO chain which the caller has already verified.
   305→   305→ *
   306→   306→ * P42-F §5-§7 — the sum is a DATABASE aggregate (goodsReceiptLine
   307→   307→ * .aggregate _sum), never a findMany of the line's whole history into
   308→   308→ * application memory: a successful PO line can accumulate unbounded
   309→   309→ * receipt rows over its lifetime and MUST NOT be materialized on every
   310→   310→ * remaining-check. The aggregate result is re-validated (exact integer,
   311→   311→ * ≥ 0); the ordered-remaining corruption policy stays with the caller's
   312→   312→ * computeRemainingQty — nothing is silently clamped (§6).
   313→   313→ */
   314→   314→async function sumPostedQtyForPoLine(tx: Tx, purchaseOrderLineId: string): Promise<number> {
   315→   315→  const agg = await tx.goodsReceiptLine.aggregate({
   316→   316→    where: {
   317→   317→      purchaseOrderLineId,
   318→   318→      goodsReceipt: { status: "POSTED" }, // §99 — POSTED only
   319→   319→    },
   320→   320→    _sum: { receivedQty: true },
   321→   321→  });
   322→   322→  const sum = agg._sum.receivedQty ?? 0; // §6 — no rows → exactly 0
   323→   323→  if (!Number.isInteger(sum) || sum < 0) {
   324→   324→    // §9 — a corrupted persisted history (negative/fractional aggregate)
   325→   325→    // is surfaced through the ONE canonical mapping, never clamped.
   326→   326→    return mapPureGoodsReceiptDomainError(
   327→   327→      new Error("GOODS_RECEIPT_CORRUPT_RECEIVED_HISTORY"),
   328→   328→    );
   329→   329→  }
   330→   330→  return sum;
   331→   331→}

   363→   363→export async function createGoodsReceipt(input: {
   364→   364→  businessId: string;
   365→   365→  userId: string;
   366→   366→  purchaseOrderId: string;
   367→   367→  reference?: string;
   368→   368→  notes?: string | null;
   369→   369→  ip?: string | null;
   370→   370→  userAgent?: string | null;
   371→   371→}) {

   373→   373→    // §11/§23 — the PO must be tenant-scoped and CONFIRMED.
   374→   374→    const po = await tx.purchaseOrder.findFirst({
   375→   375→      where: { id: input.purchaseOrderId, businessId: input.businessId }, // §90 — foreign PO → 404
   376→   376→      select: {
   377→   377→        id: true,
   378→   378→        reference: true,
   379→   379→        status: true,
   380→   380→        supplierNameAtOrder: true, // §25 — copy the ALREADY-FROZEN snapshot
   381→   381→      },

   383→   383→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   384→   384→    assertPurchaseOrderReceivable(po.status); // §11 — DRAFT/SENT/CLOSED/CANCELLED → 422
   385→   385→
   386→   386→    // §59 — reference: user-supplied (normalized) or server-generated.
   387→   387→    let reference = input.reference;
   388→   388→    if (reference === undefined) {
   389→   389→      for (let attempt = 0; attempt < 5; attempt += 1) {
   390→   390→        const candidate = generateGoodsReceiptReference(new Date(), (max) => randomInt(max));
   391→   391→        const clash = await tx.goodsReceipt.findFirst({
   392→   392→          where: { businessId: input.businessId, reference: candidate },
   393→   393→          select: { id: true },
   394→   394→        });
   395→   395→        if (!clash) {
   396→   396→          reference = candidate;
   397→   397→          break;
   398→   398→        }
   399→   399→      }
   400→   400→      if (reference === undefined) {
   401→   401→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   402→   402→        throw new ApiError(
   403→   403→          409,
   404→   404→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   405→   405→          "Could not generate a unique reference. Please try again.",
   406→   406→        );
   407→   407→      }
   408→   408→    }
   409→   409→
   410→   410→    try {
   411→   411→      const row = await tx.goodsReceipt.create({
   412→   412→        data: {
   413→   413→          businessId: input.businessId, // §22 — server-derived, never client
   414→   414→          purchaseOrderId: po.id,
   415→   415→          reference,
   416→   416→          status: "DRAFT", // §58 — every receipt starts as preparation only
   417→   417→          supplierNameAtReceipt: po.supplierNameAtOrder, // §25 — frozen identity
   418→   418→          purchaseOrderReferenceAtReceipt: po.reference, // §24 — frozen identity
   419→   419→          notes: input.notes ?? null, // §62 — document data only
   420→   420→          createdByUserId: input.userId, // §58 — server-derived
   421→   421→        },
   422→   422→        select: { id: true, reference: true, status: true },
   423→   423→      });
   424→   424→
   425→   425→      await audit(
   426→   426→        auditInput(input.businessId, input.userId, "goods_receipt.create", row.id, row.reference, {
   427→   427→          purchaseOrderId: po.id,
   428→   428→          purchaseOrderReferenceAtReceipt: po.reference,
   429→   429→          supplierNameAtReceipt: po.supplierNameAtOrder,
   430→   430→          status: row.status,
   431→   431→        }),
   432→   432→        tx,
   433→   433→      );
   434→   434→
   435→   435→      return row;
   436→   436→    } catch (error) {
   437→   437→      // §61 — database uniqueness stays authoritative for duplicates.
   438→   438→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   439→   439→        throw new ApiError(
   440→   440→          409,
   441→   441→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   442→   442→          "A goods receipt with this reference already exists.",
   443→   443→        );
   444→   444→      }
   445→   445→      throw error;
   446→   446→    }
   447→   447→  });
   448→   448→}
