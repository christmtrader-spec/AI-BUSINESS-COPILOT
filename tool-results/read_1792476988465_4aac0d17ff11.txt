    60→function timezoneOptions(): string[] {
    61→  try {
    62→    const list = Intl.supportedValuesOf("timeZone");
    63→    if (Array.isArray(list) && list.length > 0) return list;
    64→  } catch {
    65→    // older engines
    66→  }
    67→  return FALLBACK_TIMEZONES;
    68→}

   307→  async function onRestoreFileChange(file: File | undefined) {
   308→    setRestoreReport(null);
   309→    setRestoreError(null);

   312→      return;

   315→      setRestoreFile(null);
   316→      setRestoreError(t("restoreFileError"));
   317→      return;
   318→    }
   319→    try {
   320→      const text = await file.text();
   321→      const json = JSON.parse(text) as {
   322→        $schema?: string;
   323→        business?: { currencyCode?: string };
   324→      };
   325→      if (json?.$schema !== BACKUP_SCHEMA_ID) {
   326→        setRestoreFile(null);
   327→        setRestoreError(t("restoreInvalidFile"));
   328→        return;
   329→      }
   330→      setRestoreFile({
   331→        name: file.name,
   332→        size: file.size,
   333→        body: json,
   334→        currencyCode: json.business?.currencyCode ?? "?",
   335→      });
   336→    } catch {
   337→      setRestoreFile(null);
   338→      setRestoreError(t("restoreInvalidFile"));
   339→    }

   364→        setRestoreReport(null);

   366→        else if (body?.error?.code === "PERMISSION_DENIED") setRestoreError(t("restoreOwnerOnly"));

   372→      if (mode === "merge") {
   373→        toast({ title: t("restoreDone") });
   374→        void queryClient.invalidateQueries();
   375→        resetRestore();
   376→      }
   377→    } catch {
   378→      setRestoreError(t("restoreFailed"));

   380→      setRestoreBusy(null);
