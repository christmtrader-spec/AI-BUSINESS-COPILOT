    57→function timezoneOptions(): string[] {
    58→  try {
    59→    const list = Intl.supportedValuesOf("timeZone");
    60→    if (Array.isArray(list) && list.length > 0) return list;
    61→  } catch {
    62→    // older engines
    63→  }
    64→  return FALLBACK_TIMEZONES;
    65→}

   279→  async function onRestoreFileChange(file: File | undefined) {
   280→    setRestoreReport(null);
   281→    setRestoreError(null);
   282→    if (!file) {
   283→      setRestoreFile(null);
   284→      return;
   285→    }
   286→    if (file.size > 25 * 1024 * 1024) {
   287→      setRestoreFile(null);
   288→      setRestoreError(t("restoreFileError"));
   289→      return;
   290→    }
   291→    try {
   292→      const text = await file.text();
   293→      const json = JSON.parse(text) as {
   294→        $schema?: string;
   295→        business?: { currencyCode?: string };
   296→      };
   297→      if (json?.$schema !== BACKUP_SCHEMA_ID) {
   298→        setRestoreFile(null);
   299→        setRestoreError(t("restoreInvalidFile"));
   300→        return;
   301→      }
   302→      setRestoreFile({
   303→        name: file.name,
   304→        size: file.size,
   305→        body: json,
   306→        currencyCode: json.business?.currencyCode ?? "?",
   307→      });
   308→    } catch {
   309→      setRestoreFile(null);
   310→      setRestoreError(t("restoreInvalidFile"));
   311→    }
   312→  }

   314→  async function runRestore(mode: "dry_run" | "merge") {
   315→    if (!restoreFile || !activeId) return;

   319→      const res = await fetch(
   320→        `/api/v1/businesses/${activeId}/import?mode=${mode}&ids=${restoreIdMode}`,
   321→        {
   322→        method: "POST",
   323→        credentials: "same-origin",
   324→        headers: {
   325→          "content-type": "application/json",
   326→          ...(activeId ? { "x-business-id": activeId } : {}),
   327→        },
   328→        body: JSON.stringify(restoreFile.body),
   329→      });
   330→      const body = (await res.json()) as {
   331→        ok?: boolean;
   332→        data?: { report: ImportReport };
   333→        error?: { code?: string; message?: string };
   334→      };
   335→      if (!res.ok || !body?.ok || !body.data) {
   336→        setRestoreReport(null);
   337→        if (body?.error?.code === "INVALID_BACKUP") setRestoreError(t("restoreInvalidFile"));
   338→        else if (body?.error?.code === "PERMISSION_DENIED") setRestoreError(t("restoreOwnerOnly"));
   339→        else if (body?.error?.code === "RATE_LIMITED") setRestoreError(t("exportRateLimited"));
   340→        else setRestoreError(t("restoreFailed"));
   341→        return;
   342→      }
   343→      setRestoreReport(body.data.report);
   344→      if (mode === "merge") {
   345→        toast({ title: t("restoreDone") });
   346→        void queryClient.invalidateQueries();
   347→        resetRestore();
   348→      }
   349→    } catch {
   350→      setRestoreError(t("restoreFailed"));
   351→    } finally {
   352→      setRestoreBusy(null);
   353→      setConfirmOpen(false);
   354→    }
