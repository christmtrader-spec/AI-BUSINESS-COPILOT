    69→function makeClient() {
    70→  const jar = new Map();
    71→  return async function f(path, init = {}) {
    72→    const headers = {
    73→      "content-type": "application/json",
    74→      "x-forwarded-for": init.xff ?? "10.45.7.19",
    75→      ...(init.headers ?? {}),
    76→    };
    77→    if (jar.size) headers.cookie = [...jar.entries()].map(([k, v]) => `${k}=${v}`).join("; ");
    78→    const res = await fetch(`${BASE}${path}`, {
    79→      method: init.method ?? "GET",
    80→      headers,
    81→      body: init.body,
    82→      redirect: "manual",
    83→    });
    84→    for (const sc of res.headers.getSetCookie?.() ?? []) {
    85→      const [kv] = sc.split(";");
    86→      const eqi = kv.indexOf("=");
    87→      jar.set(kv.slice(0, eqi), kv.slice(eqi + 1));
    88→    }
    89→    let body = null;
    90→    const ct = res.headers.get("content-type") ?? "";
    91→    if (ct.includes("application/json")) {
    92→      try { body = await res.json(); } catch { body = null; }
    93→    } else {
    94→      body = await res.text().catch(() => null);
    95→    }
    96→    return { status: res.status, headers: res.headers, body };
    97→  };
    98→}

   102→  const cb = await f("/api/auth/callback/credentials", {
   103→    method: "POST",
   104→    headers: { "content-type": "application/x-www-form-urlencoded" },
   105→    body: new URLSearchParams({ csrfToken: csrf, email, password: PASS, redirect: "false", json: "true" }).toString(),
   106→  });
   107→  const me = await f("/api/v1/me");
   108→  if (me.status !== 200) {
   109→    console.error(`  !! login failed for ${email}: callback=${cb.status} body=${JSON.stringify(cb.body).slice(0, 160)} me=${me.status}`);
   110→  }
   111→}

   113→async function call(f, path, init = {}, depth = 0) {
   114→  const res = await f(path, init);
   115→  if (res.status === 429 && !(init.headers ?? {})["x-no-retry"] && depth < 1) {
   116→    const wait = Math.min(res.body?.error?.details?.retryAfterMs ?? 60_000, 65_000);
   117→    console.log(`  (rate limited — waiting ${Math.round(wait / 1000)}s)`);
   118→    await new Promise((r) => setTimeout(r, wait));
   119→    return call(f, path, init, depth + 1);
   120→  }
   121→  return { status: res.status, body: res.body, headers: res.headers };
   122→}
