   115→// ---- Markdown-lite renderer (no deps, no raw HTML — safe by construction) --
   116→
   117→function inline(text: string, keyPrefix: string): ReactNode[] {
   118→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
   119→  return parts.map((part, i) => {
   120→    const key = `${keyPrefix}-${i}`;
   121→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   122→      return (
   123→        <strong key={key} className="font-semibold">
   124→          {part.slice(2, -2)}
   125→        </strong>
   126→      );
   127→    }
   128→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   129→      return (
   130→        <code
   131→          key={key}
   132→          dir="ltr"
   133→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   134→        >
   135→          {part.slice(1, -1)}
   136→        </code>
   137→      );
   138→    }
   139→    return <span key={key}>{part}</span>;
   140→  });
   141→}

   143→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   144→  const nodes: ReactNode[] = [];
   145→  let bullets: string[] = [];
   146→  let para: string[] = [];
   147→
   148→  const flushBullets = () => {
   149→    if (bullets.length === 0) return;
   150→    nodes.push(
   151→      <ul
   152→        key={`ul-${nodes.length}`}
   153→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   154→      >
   155→        {bullets.map((b, i) => (
   156→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   157→        ))}
   158→      </ul>,
   159→    );
   160→    bullets = [];
   161→  };
   162→  const flushPara = () => {
   163→    if (para.length === 0) return;
   164→    nodes.push(
   165→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   166→        {inline(para.join("\n"), `p${nodes.length}`)}
   167→      </p>,
   168→    );
   169→    para = [];
   170→  };
   171→
   172→  for (const raw of text.split("\n")) {
   173→    const line = raw.trimEnd();
   174→    if (/^\s*[-*•]\s+/.test(line)) {
   175→      flushPara();
   176→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   177→    } else if (line === "") {
   178→      flushBullets();
   179→      flushPara();
   180→    } else {
   181→      flushBullets();
   182→      para.push(line);
   183→    }
   184→  }
   185→  flushBullets();
   186→  flushPara();
   187→
   188→  return (
   189→    <div className="space-y-1.5 text-sm">
   190→      {nodes}
   191→      {streaming ? (
   192→        <span
   193→          aria-hidden="true"
   194→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   195→        />
   196→      ) : null}
   197→    </div>
   198→  );
   199→}

   291→// ---- Follow-up suggestions ---------------------------------------------------
   292→
   293→/**
   294→ * Context-aware follow-up pool. The chips are UI suggestions only — they
   295→ * never touch data. Deterministic: keyword match on the answer text picks
   296→ * the lead chip; the rest rotate by conversation length so repeated
   297→ * answers surface different prompts.
   298→ */
   299→const FOLLOW_UP_KEYS = [
   300→  "followUpOutstanding",
   301→  "followUpLowStock",
   302→  "followUpExpenses",
   303→  "followUpSales",
   304→  "followUpCash",
   305→  "followUpProducts",
   306→  "followUpPayments",
   307→  "followUpSummary",
   308→] as const;
   309→
   310→type FollowUpKey = (typeof FOLLOW_UP_KEYS)[number];
   311→
   312→const FOLLOW_UP_MATCHERS: Array<{ key: FollowUpKey; re: RegExp }> = [
   313→  { key: "followUpOutstanding", re: /owe|outstanding|receivable|بدهکار|بدهی|طلب/i },
   314→  { key: "followUpLowStock", re: /low.?stock|restock|inventory|موجودی|انبار|شارژ/i },
   315→  { key: "followUpExpenses", re: /expense|spend|cost|هزینه|خرج/i },
   316→  { key: "followUpSales", re: /sale|revenue|invoice|فروش|درآمد|فاکتور/i },
   317→  { key: "followUpCash", re: /cash|payment|collected|نقد|پرداخت|دریافتی/i },
   318→  { key: "followUpProducts", re: /product|best.?seller|top|محصول|پرفروش/i },
   319→];
   320→
   321→/** Live server counts (same deterministic aggregates as the dashboard)
   322→ *  used to RANK suggestions — a chip is only proposed when its subject
   323→ *  actually exists in the data (r10: "only propose restock when there is
   324→ *  something to restock"). */
   325→type SummaryLite = {
   326→  kpis?: { salesCount?: number; expensesCount?: number };
   327→  outstanding?: { receivableCount?: number };
   328→  inventory?: { lowStockCount?: number };
   329→  topProducts?: unknown[];
   330→};
   331→
   332→function followUpRelevant(
   333→  key: FollowUpKey,
   334→  counts: {
   335→    receivableCount: number;
   336→    lowStockCount: number;
   337→    salesCount: number;
   338→    expensesCount: number;
   339→    topProducts: number;
   340→  } | null,
   341→): boolean {
   342→  if (!counts) return true; // counts not loaded yet — keep the neutral default
   343→  switch (key) {
   344→    case "followUpOutstanding":
   345→      return counts.receivableCount > 0;
   346→    case "followUpLowStock":
   347→      return counts.lowStockCount > 0;
   348→    case "followUpExpenses":
   349→      return counts.expensesCount > 0;
   350→    case "followUpProducts":
   351→      return counts.topProducts > 0;
   352→    case "followUpSales":
   353→      return true;
   354→    case "followUpCash":
   355→    case "followUpPayments":
   356→    case "followUpSummary":
   357→      return true;
   358→  }
   359→}
   360→
   361→function pickFollowUps(
   362→  answer: string,
   363→  conversationLength: number,
   364→  counts: Parameters<typeof followUpRelevant>[1] = null,
   365→): FollowUpKey[] {
   366→  const lead = FOLLOW_UP_MATCHERS.find((m) => m.re.test(answer))?.key;
   367→  const pool = FOLLOW_UP_KEYS.filter(
   368→    (k) => k !== lead && followUpRelevant(k, counts),
   369→  );
   370→  const rotate = conversationLength % Math.max(1, pool.length);
   371→  const rotated = [...pool.slice(rotate), ...pool.slice(0, rotate)];
   372→  const picked = rotated.slice(0, lead ? 2 : 3);
   373→  return lead ? [lead, ...picked] : picked;
   374→}

   581→    setActiveSessionId(session.id);
   582→    setSessions((prev) => [session, ...prev]);
   583→    textareaRef.current?.focus();
   584→  }, [activeStreamId, sessions]);
   585→
   586→  const deleteSession = useCallback(
   587→    (id: string) => {
   588→      if (activeStreamId !== null) return;

   590→        const next = prev.filter((s) => s.id !== id);
   591→        if (activeSessionIdRef.current === id) {
   592→          const fallback = next[0]?.id ?? null;
   593→          activeSessionIdRef.current = fallback;

   628→  const send = useCallback(
   629→    async (question: string, history: ChatMessage[]) => {

   668→            const body = (await res.json()) as {
   669→              error?: { code?: string; details?: { retryAfterMs?: number } };
   670→            };

   672→            retryAfterMs = body?.error?.details?.retryAfterMs;

   674→            // keep defaults
   675→          }

   678→            content: localizedError(code, retryAfterMs),

   685→        const reader = res.body.getReader();

   689→        let sawDone = false;

   691→        while (true) {
   692→          const { done, value } = await reader.read();

   695→          const events = buffer.split("\n\n");

   701→                meta?: CopilotMeta;
   702→                delta?: string;
   703→                done?: boolean;
   704→                error?: { code?: string; message?: string };
   705→              };

   707→                payload = JSON.parse(line.slice(6));
   708→              } catch {
   709→                continue;
   710→              }
   711→              if (payload.meta) {
   712→                // First event — dataset provenance for this answer.
   713→                const meta = payload.meta;
   714→                patchAssistant(assistantId, (m) => ({ ...m, meta }));
   715→              } else if (typeof payload.delta === "string" && payload.delta.length > 0) {
   716→                const delta = payload.delta;
   717→                patchAssistant(assistantId, (m) => ({
   718→                  ...m,
   719→                  content: m.content + delta,
   720→                }));
   721→              } else if (payload.error) {
   722→                streamError = payload.error;
   723→              } else if (payload.done) {
   724→                sawDone = true;
   725→              }
   726→            }
   727→          }
   728→        }
   729→
   730→        if (streamError) {
   731→          const code = streamError.code ?? "INTERNAL_ERROR";
   732→          if (code === "RATE_LIMITED" || code === "AI_UNAVAILABLE" || !sawDone) {
   733→            patchAssistant(assistantId, (m) =>
   734→              m.content.trim().length === 0
   735→                ? { ...m, content: localizedError(code), failed: true }
   736→                : { ...m, content: `${m.content}\n\n⚠ ${localizedError(code)}` },
   737→            );
   738→          }
   739→        }
   740→      } catch (error) {
   741→        if ((error as Error)?.name === "AbortError") {
   742→          // User pressed Stop — keep the partial answer, drop it if empty.
   743→          setMessages((prev) =>
   744→            prev.flatMap((m) =>
   745→              m.id === assistantId ? (m.content.trim() ? [m] : []) : [m],
   746→            ),
   747→          );
   748→        } else if (error instanceof ApiClientError) {
   749→          patchAssistant(assistantId, (m) => ({
   750→            ...m,
   751→            content: localizedError(error.code),
   752→            failed: true,
   753→          }));
   754→        } else {
   755→          patchAssistant(assistantId, (m) => ({
   756→            ...m,
   757→            content: localizedError("NETWORK_ERROR"),
   758→            failed: true,
   759→          }));
   760→        }
   761→      } finally {
   762→        abortRef.current = null;
   763→        setActiveStreamId((cur) => (cur === assistantId ? null : cur));
   764→      }
   765→    },
   766→    [activeStreamId, business.id, days, localizedError, patchAssistant],
   767→  );
