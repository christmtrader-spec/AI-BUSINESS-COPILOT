   113→type TooltipEntry = { name?: string; value?: number; color?: string; dataKey?: string | number };
   114→
   115→function MoneyTooltip({
   116→  active,
   117→  payload,
   118→  label,
   119→  fmt,
   120→  labelFmt,
   121→}: {
   122→  active?: boolean;
   123→  payload?: TooltipEntry[];
   124→  label?: string | number;
   125→  fmt: (v: number) => string;
   126→  labelFmt?: (v: string) => string;
   127→}) {
   128→  if (!active || !payload || payload.length === 0) return null;
   129→  const labelText =
   130→    labelFmt && typeof label === "string" ? labelFmt(label) : label != null ? String(label) : "";
   131→  return (
   132→    <div className="bg-popover text-popover-foreground rounded-lg border px-3 py-2 text-xs shadow-md">
   133→      <p className="mb-1 font-medium" dir="auto">{labelText}</p>
   134→      {payload.map((entry) => (
   135→        <p key={String(entry.dataKey)} className="flex items-center gap-1.5 tabular-nums">
   136→          <span
   137→            className="size-2 shrink-0 rounded-[2px]"
   138→            style={{ background: entry.color }}
   139→            aria-hidden="true"
   140→          />
   141→          {entry.name}: <span dir="ltr">{fmt(entry.value ?? 0)}</span>
   142→        </p>
   143→      ))}
   144→    </div>
   145→  );
   146→}
