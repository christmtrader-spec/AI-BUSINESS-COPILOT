    77→    77→function ok(cond: boolean, name: string) {
    78→    78→  if (cond) passed++;
    79→    79→  else {
    80→    80→    failed++;
    81→    81→    failures.push(name);
    82→    82→  }
    83→    83→  console.log(`${cond ? "✓" : "✗"} ${name}`);
    84→    84→}
    85→    85→function eq(a: unknown, b: unknown, name: string) {
    86→    86→  const same = a === b;
    87→    87→  if (same) passed++;
    88→    88→  else {
    89→    89→    failed++;
    90→    90→    failures.push(`${name} (got ${String(a)}, want ${String(b)})`);
    91→    91→  }
    92→    92→  console.log(`${same ? "✓" : "✗"} ${name}`);
    93→    93→}
    94→    94→function throws(fn: () => unknown, name: string, errorName?: string) {
    95→    95→  try {
    96→    96→    fn();
    97→    97→    ok(false, name);
    98→    98→  } catch (e) {
    99→    99→    const matches = errorName
   100→   100→      ? (e as Error).name === errorName || String((e as Error).message).includes(errorName)
   101→   101→      : true;
   102→   102→    ok(matches, name);
   103→   103→  }
   104→   104→}
