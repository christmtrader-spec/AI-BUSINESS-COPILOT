    93→// ---- Markdown-lite renderer (no deps, no raw HTML — safe by construction) --
    94→
    95→function inline(text: string, keyPrefix: string): ReactNode[] {
    96→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
    97→  return parts.map((part, i) => {
    98→    const key = `${keyPrefix}-${i}`;
    99→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   100→      return (
   101→        <strong key={key} className="font-semibold">
   102→          {part.slice(2, -2)}
   103→        </strong>
   104→      );
   105→    }
   106→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   107→      return (
   108→        <code
   109→          key={key}
   110→          dir="ltr"
   111→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   112→        >
   113→          {part.slice(1, -1)}
   114→        </code>
   115→      );
   116→    }
   117→    return <span key={key}>{part}</span>;
   118→  });
   119→}

   121→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   122→  const nodes: ReactNode[] = [];
   123→  let bullets: string[] = [];
   124→  let para: string[] = [];
   125→
   126→  const flushBullets = () => {
   127→    if (bullets.length === 0) return;
   128→    nodes.push(
   129→      <ul
   130→        key={`ul-${nodes.length}`}
   131→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   132→      >
   133→        {bullets.map((b, i) => (
   134→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   135→        ))}
   136→      </ul>,
   137→    );
   138→    bullets = [];
   139→  };
   140→  const flushPara = () => {
   141→    if (para.length === 0) return;
   142→    nodes.push(
   143→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   144→        {inline(para.join("\n"), `p${nodes.length}`)}
   145→      </p>,
   146→    );
   147→    para = [];
   148→  };
   149→
   150→  for (const raw of text.split("\n")) {
   151→    const line = raw.trimEnd();
   152→    if (/^\s*[-*•]\s+/.test(line)) {
   153→      flushPara();
   154→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   155→    } else if (line === "") {
   156→      flushBullets();
   157→      flushPara();
   158→    } else {
   159→      flushBullets();
   160→      para.push(line);
   161→    }
   162→  }
   163→  flushBullets();
   164→  flushPara();
   165→
   166→  return (
   167→    <div className="space-y-1.5 text-sm">
   168→      {nodes}
   169→      {streaming ? (
   170→        <span
   171→          aria-hidden="true"
   172→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   173→        />
   174→      ) : null}
   175→    </div>
   176→  );
   177→}

   181→/**
   182→ * Dataset provenance chips rendered under each grounded answer — the UI
   183→ * mirror of the server snapshot: which window, which date range, and how
   184→ * many records the answer is built from. Counts come from the server
   185→ * (deterministic), never from the model.
   186→ */
   187→function CitationStrip({ meta, locale }: { meta: CopilotMeta; locale: string }) {
   188→  const t = useTranslations("copilot");
   189→  const c = meta.counts;
   190→  const chips: { key: string; icon: typeof ReceiptText; label: string; title?: string }[] = [
   191→    {
   192→      key: "window",
   193→      icon: CalendarClock,
   194→      label: t(`window_${meta.days === 7 ? "7" : meta.days === 30 ? "30" : "90"}`),
   195→    },
   196→    { key: "range", icon: BookMarked, label: `${meta.rangeFrom} → ${meta.rangeTo}` },
   197→    ...(c
   198→      ? [
   199→          { key: "sales", icon: ReceiptText, label: t("citationsSales", { count: c.sales }) },
   200→          { key: "expenses", icon: Wallet, label: t("citationsExpenses", { count: c.expenses }) },
   201→          {
   202→            key: "openInvoices",
   203→            icon: HandCoins,
   204→            label: t("citationsOpenInvoices", { count: c.openInvoices }),
   205→            title: t("citationsAllTime"),
   206→          },
   207→          ...(c.lowStock > 0
   208→            ? [
   209→                {
   210→                  key: "lowStock",
   211→                  icon: Package,
   212→                  label: t("citationsLowStock", { count: c.lowStock }),
   213→                },
   214→              ]
   215→            : []),
   216→          {
   217→            key: "payments",
   218→            icon: ArrowDown,
   219→            label: t("citationsPayments", { count: c.recentPayments }),
   220→          },
   221→        ]
   222→      : []),
   223→  ];
   224→
   225→  const generatedLabel = (() => {
   226→    try {
   227→      const time = new Intl.DateTimeFormat(locale === "fa" ? "fa-IR" : "en-GB", {
   228→        hour: "2-digit",
   229→        minute: "2-digit",
   230→      }).format(new Date(meta.generatedAt));
   231→      return t("citationsGenerated", { time });
   232→    } catch {
   233→      return null;
   234→    }
   235→  })();
   236→
   237→  return (
   238→    <div
   239→      className="animate-in fade-in slide-in-from-bottom-1 mt-1 flex flex-wrap items-center gap-x-2 gap-y-1 ps-1"
   240→      data-testid="copilot-citations"
   241→    >
   242→      <span className="text-muted-foreground/80 inline-flex items-center gap-1 text-[11px] font-medium">
   243→        <BookMarked className="size-3" aria-hidden="true" />
   244→        {t("citationsTitle")}
   245→      </span>
   246→      {chips.map((chip) => (
   247→        <span
   248→          key={chip.key}
   249→          title={chip.title}
   250→          className="border-border/70 bg-background text-muted-foreground inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[11px] leading-none"
   251→        >
   252→          <chip.icon className="size-3" aria-hidden="true" />
   253→          {chip.key === "range" ? (
   254→            <span dir="ltr" className="tabular-nums">
   255→              {chip.label}
   256→            </span>
   257→          ) : (
   258→            chip.label
   259→          )}
   260→        </span>
   261→      ))}
   262→      {generatedLabel ? (
   263→        <span className="text-muted-foreground/70 text-[11px]">· {generatedLabel}</span>
   264→      ) : null}
   265→    </div>
   266→  );
   267→}

   271→export function CopilotView({
   272→  business,
   273→  role,
   274→}: {
   275→  business: BusinessSummary;
   276→  role: Role;
   277→}) {
   278→  const t = useTranslations("copilot");
   279→  const tCommon = useTranslations("common");
   280→  const tErrors = useTranslations("errors");
   281→
   282→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   283→  const [input, setInput] = useState("");
   284→  const [activeStreamId, setActiveStreamId] = useState<string | null>(null);
   285→  const [days, setDays] = useState<7 | 30 | 90>(30);
   286→  const [copiedId, setCopiedId] = useState<string | null>(null);
   287→  const [dataOpen, setDataOpen] = useState(false);
   288→  const [atBottom, setAtBottom] = useState(true);
   289→  const lastQuestionRef = useRef<string | null>(null);
   290→  const abortRef = useRef<AbortController | null>(null);
   291→  const scrollRef = useRef<HTMLDivElement>(null);
   292→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   293→
   294→  const canUse = can(role, "copilot:use");
   295→  const pending = activeStreamId !== null;
   296→
   297→  const suggestions = useMemo(
   298→    () => [
   299→      t("suggestion1"),
   300→      t("suggestion2"),
   301→      t("suggestion3"),
   302→      t("suggestion4"),
   303→      t("suggestion5"),
   304→      t("suggestion6"),
   305→    ],
   306→    [t],
   307→  );
   308→
   309→  // Keep the latest message in view (scroll-only effect — no setState).
   310→  useEffect(() => {
   311→    if (atBottom && scrollRef.current) {
   312→      scrollRef.current.scrollTo({
   313→        top: scrollRef.current.scrollHeight,
   314→        behavior: "smooth",
   315→      });
   316→    }
   317→  }, [messages, pending, atBottom]);
   318→
   319→  // Abort any in-flight stream when the module unmounts.
   320→  useEffect(() => {
   321→    return () => abortRef.current?.abort();
   322→  }, []);
   323→
   324→  const patchAssistant = useCallback(
   325→    (id: string, patch: (m: ChatMessage) => ChatMessage) => {
   326→      setMessages((prev) => prev.map((m) => (m.id === id ? patch(m) : m)));
   327→    },
   328→    [],
   329→  );
   330→
   331→  const localizedError = useCallback(
   332→    (code: string, retryAfterMs?: number) => {
   333→      switch (code) {
   334→        case "RATE_LIMITED":
   335→          return t("errorRateLimited", {
   336→            seconds: String(Math.max(1, Math.ceil((retryAfterMs ?? 0) / 1000))),
   337→          });
   338→        case "PERMISSION_DENIED":
   339→          return t("errorForbidden");
   340→        case "AI_UNAVAILABLE":
   341→          return t("errorAiUnavailable");
   342→        default:
   343→          return tErrors("generic");
   344→      }
   345→    },
   346→    [t, tErrors],
   347→  );
   348→
   349→  const send = useCallback(
   350→    async (question: string, history: ChatMessage[]) => {
   351→      const trimmed = question.trim();
   352→      if (!trimmed || trimmed.length > 500 || activeStreamId !== null) return;
   353→      lastQuestionRef.current = trimmed;
   354→      setAtBottom(true);
   355→
   356→      const assistantId = nextId();
   357→      setMessages((prev) => [
   358→        ...prev,
   359→        { id: assistantId, role: "assistant", content: "" },
   360→      ]);
   361→      setActiveStreamId(assistantId);
   362→
   363→      const controller = new AbortController();
   364→      abortRef.current = controller;
   365→
   366→      try {
   367→        const res = await fetch("/api/v1/copilot/stream", {
   368→          method: "POST",
   369→          credentials: "same-origin",
   370→          headers: {
   371→            "content-type": "application/json",
   372→            "x-business-id": business.id,
   373→          },
   374→          body: JSON.stringify({
   375→            question: trimmed,
   376→            history: history
   377→              .filter((m) => !m.failed)
   378→              .slice(-6)
   379→              .map((m) => ({ role: m.role, content: m.content })),
   380→            days,
   381→          }),
   382→          signal: controller.signal,
   383→        });
   384→
   385→        if (!res.ok || !res.body) {
   386→          let code = "INTERNAL_ERROR";
   387→          let retryAfterMs: number | undefined;
   388→          try {
   389→            const body = (await res.json()) as {
   390→              error?: { code?: string; details?: { retryAfterMs?: number } };
   391→            };
   392→            if (body?.error?.code) code = body.error.code;
   393→            retryAfterMs = body?.error?.details?.retryAfterMs;
   394→          } catch {
   395→            // keep defaults
   396→          }
   397→          patchAssistant(assistantId, (m) => ({
   398→            ...m,
   399→            content: localizedError(code, retryAfterMs),
   400→            failed: true,
   401→          }));
   402→          return;
   403→        }
   404→
   405→        // Parse the SSE stream incrementally.
   406→        const reader = res.body.getReader();
   407→        const decoder = new TextDecoder();
   408→        let buffer = "";
   409→        let streamError: { code?: string; message?: string } | null = null;
   410→        let sawDone = false;
   411→
   412→        while (true) {
   413→          const { done, value } = await reader.read();
   414→          if (done) break;
   415→          buffer += decoder.decode(value, { stream: true });
   416→          const events = buffer.split("\n\n");
   417→          buffer = events.pop() ?? "";
   418→          for (const evt of events) {
   419→            for (const line of evt.split("\n")) {
   420→              if (!line.startsWith("data: ")) continue;
   421→              let payload: {
   422→                meta?: CopilotMeta;
   423→                delta?: string;
   424→                done?: boolean;
   425→                error?: { code?: string; message?: string };
   426→              };
   427→              try {
   428→                payload = JSON.parse(line.slice(6));
   429→              } catch {
   430→                continue;
   431→              }
   432→              if (payload.meta) {
   433→                // First event — dataset provenance for this answer.
   434→                const meta = payload.meta;
   435→                patchAssistant(assistantId, (m) => ({ ...m, meta }));
   436→              } else if (typeof payload.delta === "string" && payload.delta.length > 0) {
   437→                const delta = payload.delta;
   438→                patchAssistant(assistantId, (m) => ({
   439→                  ...m,
   440→                  content: m.content + delta,
   441→                }));
   442→              } else if (payload.error) {
   443→                streamError = payload.error;
   444→              } else if (payload.done) {
   445→                sawDone = true;
   446→              }
   447→            }
   448→          }
   449→        }
   450→
   451→        if (streamError) {
   452→          const code = streamError.code ?? "INTERNAL_ERROR";
   453→          if (code === "RATE_LIMITED" || code === "AI_UNAVAILABLE" || !sawDone) {
   454→            patchAssistant(assistantId, (m) =>
   455→              m.content.trim().length === 0
   456→                ? { ...m, content: localizedError(code), failed: true }
   457→                : { ...m, content: `${m.content}\n\n⚠ ${localizedError(code)}` },
   458→            );
   459→          }
   460→        }
   461→      } catch (error) {
   462→        if ((error as Error)?.name === "AbortError") {
   463→          // User pressed Stop — keep the partial answer, drop it if empty.
   464→          setMessages((prev) =>
   465→            prev.flatMap((m) =>
   466→              m.id === assistantId ? (m.content.trim() ? [m] : []) : [m],
   467→            ),
   468→          );
   469→        } else if (error instanceof ApiClientError) {
   470→          patchAssistant(assistantId, (m) => ({
   471→            ...m,
   472→            content: localizedError(error.code),
   473→            failed: true,
   474→          }));
   475→        } else {
   476→          patchAssistant(assistantId, (m) => ({
   477→            ...m,
   478→            content: localizedError("NETWORK_ERROR"),
   479→            failed: true,
   480→          }));
   481→        }
   482→      } finally {
   483→        abortRef.current = null;
   484→        setActiveStreamId((cur) => (cur === assistantId ? null : cur));
   485→      }
   486→    },
   487→    [activeStreamId, business.id, days, localizedError, patchAssistant],
   488→  );
   489→
   490→  const handleSubmit = () => {
   491→    const trimmed = input.trim();
   492→    if (!trimmed || pending) return;
   493→    setInput("");
   494→    if (textareaRef.current) textareaRef.current.style.height = "auto";
   495→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: trimmed }]);
   496→    void send(trimmed, messages);
   497→  };
   498→
   499→  const handleStop = () => {
   500→    abortRef.current?.abort();
   501→  };
   502→
   503→  const handleRetry = () => {
   504→    const q = lastQuestionRef.current;
   505→    if (!q || pending) return;
   506→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: q }]);
   507→    void send(q, messages);
   508→  };
   509→
   510→  const copyAnswer = async (m: ChatMessage) => {
   511→    try {
   512→      await navigator.clipboard.writeText(m.content);
   513→      setCopiedId(m.id);
   514→      window.setTimeout(() => setCopiedId((c) => (c === m.id ? null : c)), 1500);
   515→    } catch {
   516→      /* clipboard unavailable — silently ignore */
   517→    }
   518→  };
   519→
   520→  if (!canUse) {
   521→    return (
   522→      <div className="space-y-6" data-view="copilot">
   523→        <PageHeader title={t("title")} subtitle={t("subtitle")} />
   524→        <Card className="mx-auto max-w-md p-8 text-center">
   525→          <div className="bg-muted mx-auto flex size-12 items-center justify-center rounded-full">
   526→            <Lock className="text-muted-foreground size-5" aria-hidden="true" />
   527→          </div>
   528→          <h2 className="mt-4 text-base font-semibold">{t("lockedTitle")}</h2>
   529→          <p className="text-muted-foreground mt-1.5 text-sm">{t("lockedDesc")}</p>
   530→        </Card>
   531→      </div>
   532→    );
   533→  }
   534→
   535→  const empty = messages.length === 0;
   536→  const lastFailed = !pending && messages[messages.length - 1]?.failed === true;
   537→  const waitingForFirstToken =
   538→    pending && (messages[messages.length - 1]?.content ?? "").length === 0;
   539→
   540→  return (
   541→    <div className="space-y-4" data-view="copilot">
   542→      <PageHeader
   543→        title={t("title")}
   544→        subtitle={t("subtitle")}
   545→        actions={
   546→          <span className="text-muted-foreground border-emerald-500/30 bg-emerald-500/5 inline-flex items-center gap-1.5 rounded-full border px-3 py-1 text-xs font-medium text-emerald-700 dark:text-emerald-400">
   547→            <ShieldCheck className="size-3.5" aria-hidden="true" />
   548→            {t("readOnly")}
   549→          </span>
   550→        }
   551→      />
   552→
   553→      <Card className="flex h-[calc(100svh-15rem)] min-h-[520px] flex-col overflow-hidden">
   554→        {/* Toolbar */}
   555→        <div className="flex flex-wrap items-center justify-between gap-2 border-b px-4 py-2.5">
   556→          <div className="flex items-center gap-2" role="group" aria-label={t("windowLabel")}>
   557→            <CalendarClock className="text-muted-foreground size-4" aria-hidden="true" />
   558→            <span className="text-muted-foreground text-xs font-medium">{t("windowLabel")}</span>
   559→            <div className="bg-muted flex overflow-hidden rounded-lg p-0.5">
   560→              {WINDOWS.map((w) => (
   561→                <button
   562→                  key={w}
   563→                  type="button"
   564→                  onClick={() => setDays(w)}
   565→                  aria-pressed={days === w}
   566→                  className={cn(
   567→                    "rounded-md px-2.5 py-1 text-xs font-medium transition-colors",
   568→                    days === w
   569→                      ? "bg-background text-foreground shadow-sm"
   570→                      : "text-muted-foreground hover:text-foreground",
   571→                  )}
   572→                >
   573→                  {t(`window_${w}` as "window_7")}
   574→                </button>
   575→              ))}
   576→            </div>
   577→          </div>
   578→          <div className="flex items-center gap-1">
   579→            <Collapsible open={dataOpen} onOpenChange={setDataOpen}>
   580→              <CollapsibleTrigger asChild>
   581→                <Button variant="ghost" size="sm" className="text-muted-foreground h-8 gap-1.5 text-xs">
   582→                  <ListTree className="size-3.5" aria-hidden="true" />
   583→                  {t("dataIncludedTitle")}
   584→                  <ChevronDown className={cn("size-3.5 transition-transform", dataOpen && "rotate-180")} aria-hidden="true" />
   585→                </Button>
   586→              </CollapsibleTrigger>
   587→            </Collapsible>
   588→            {messages.length > 0 ? (
   589→              <Button
   590→                variant="ghost"
   591→                size="sm"
   592→                className="text-muted-foreground h-8 gap-1.5 text-xs"
   593→                onClick={() => setMessages([])}
   594→                disabled={pending}
