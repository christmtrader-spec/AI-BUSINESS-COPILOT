    35→type AuthErrors = {
    36→  invalidCredentials: string;

    46→/**
    47→ * Server 422 envelopes carry the zod issues in `details`. When the failed
    48→ * field is the password we can give the user the REAL reason instead of a
    49→ * generic "try again" (BUGFIX P51-R1: signup showed a useless generic error
    50→ * when the password policy rejected the submitted password).
    51→ */
    52→function hasValidationPath(details: unknown, field: string): boolean {
    53→  if (!Array.isArray(details)) return false;
    54→  return details.some(
    55→    (issue) =>
    56→      issue !== null &&
    57→      typeof issue === "object" &&
    58→      Array.isArray((issue as { path?: unknown }).path) &&
    59→      (issue as { path: unknown[] }).path[0] === field,
    60→  );
    61→}
    62→
    63→function mapSignInError(error: string | null | undefined, t: AuthErrors): string {
    64→  if (error === "CredentialsSignin") return t.invalidCredentials;
    65→  return t.generic;
    66→}
    67→
    68→function mapApiError(err: unknown, t: AuthErrors): string {
    69→  if (err instanceof ApiClientError) {
    70→    if (err.status === 0) return t.network;
    71→    if (err.code === "EMAIL_TAKEN") return t.emailTaken;
    72→    if (err.code === "RATE_LIMITED") return t.rateLimited;
    73→    if (err.code === "VALIDATION_ERROR") {
    74→      if (hasValidationPath(err.details, "password")) return t.passwordPolicy;
    75→      return t.invalidData;
    76→    }
    77→  }
    78→  return t.generic;
    79→}

   296→/**
   297→ * "Reset password" flow — consumes a single-use token from the deep link
   298→ * (#/reset-password/<token>). On success every previous session is dead:
   299→ * any stale local session is cleared immediately and the user is sent to
   300→ * sign-in via the success card button.
   301→ */
   302→function ResetPasswordCard({
   303→  token,
   304→  onGoToSignIn,
   305→  onRequestNewLink,
   306→}: {
   307→  token: string;
   308→  onGoToSignIn: () => void;
   309→  onRequestNewLink: () => void;
   310→}) {
   311→  const t = useTranslations("auth");
   312→  const queryClient = useQueryClient();
   313→  const [newPassword, setNewPassword] = useState("");
   314→  const [confirmPassword, setConfirmPassword] = useState("");
   315→  const [error, setError] = useState<string | null>(null);
   316→  const [tokenInvalid, setTokenInvalid] = useState(false);
   317→  const [doneMessage, setDoneMessage] = useState<string | null>(null);
   318→  // Password visibility toggles (presentation only — never affects logic).
   319→  const [showNew, setShowNew] = useState(false);
   320→  const [showConfirm, setShowConfirm] = useState(false);
   321→  const pwLabels = { show: t("showPassword"), hide: t("hidePassword") };
   322→
   323→  const resetMutation = useMutation({
   324→    mutationFn: async () =>
   325→      api<{ message?: string }>("/api/v1/auth/password-reset/confirm", {
   326→        method: "POST",
   327→        body: JSON.stringify({ token, newPassword, confirmPassword }),
   328→      }),
   329→    onSuccess: () => {
   330→      setError(null);
   331→      setTokenInvalid(false);
   332→      setDoneMessage(t("resetSuccessBody"));
   333→      // The server killed every session for this user — clear any stale
   334→      // local one right away (the success card below routes to sign-in).
   335→      void signOut({ redirect: false });
   336→      void queryClient.invalidateQueries();
   337→    },
   338→    onError: (err) => {
   339→      if (err instanceof ApiClientError) {
   340→        if (err.code === "INVALID_RESET_TOKEN") {
   341→          setTokenInvalid(true);
   342→          setError(null);
   343→        } else if (err.status === 0) {
   344→          setTokenInvalid(false);
   345→          setError(t("errors.network"));
   346→        } else if (err.code === "RATE_LIMITED") {
   347→          setTokenInvalid(false);
   348→          setError(t("errors.rateLimited"));
   349→        } else {
   350→          setTokenInvalid(false);
   351→          setError(t("errors.generic"));
   352→        }
   353→      } else {
   354→        setTokenInvalid(false);
   355→        setError(t("errors.generic"));
   356→      }
   357→    },
   358→  });
   359→
   360→  return (
   361→    <Card className={glassCardClass}>
   362→      <CardTopHairline />
   363→      <CardHeader className="space-y-1.5">
   364→        <AuthCardBadge>
   365→          <KeyRound className="size-5" />
   366→        </AuthCardBadge>
   367→        <CardTitle className="text-2xl">
   368→          {doneMessage ? t("resetSuccessTitle") : t("resetTitle")}
   369→        </CardTitle>
   370→        {doneMessage ? null : <CardDescription>{t("resetSubtitle")}</CardDescription>}
   371→      </CardHeader>
   372→      <CardContent>
   373→        {doneMessage ? (
   374→          <div className="space-y-5 py-2 text-center">
   375→            <div
   376→              className="bg-primary/10 mx-auto flex size-12 items-center justify-center rounded-full"
   377→              aria-hidden="true"
   378→            >
   379→              <KeyRound className="text-primary size-6" />
   380→            </div>
   381→            <p className="text-foreground text-sm" role="status">
   382→              {doneMessage}
   383→            </p>
   384→            <Button type="button" className="w-full" onClick={onGoToSignIn}>
   385→              <LogIn className="size-4" aria-hidden="true" />
   386→              {t("goToSignIn")}
   387→            </Button>
   388→          </div>
   389→        ) : (
   390→          <form
   391→            onSubmit={(e) => {
   392→              e.preventDefault();
   393→              setError(null);
   394→              setTokenInvalid(false);
   395→              if (newPassword !== confirmPassword) {
   396→                setError(t("errors.passwordMismatch"));
   397→                return;
   398→              }
   399→              resetMutation.mutate();
   400→            }}
   401→            className="space-y-4"
   402→            noValidate
   403→          >
   404→            <div className="space-y-2">
   405→              <Label htmlFor="reset-password">{t("newPassword")}</Label>
   406→              <div className="relative">
   407→                <FieldIcon icon={Lock} />
   408→                <Input
   409→                  id="reset-password"
   410→                  type={showNew ? "text" : "password"}
   411→                  autoComplete="new-password"
   412→                  dir="ltr"
   413→                  placeholder={t("passwordPlaceholder")}
   414→                  value={newPassword}
   415→                  onChange={(e) => setNewPassword(e.target.value)}
   416→                  required
   417→                  minLength={8}
   418→                  className="h-11 ps-9 pe-10"
   419→                />
   420→                <PasswordVisibilityButton
   421→                  show={showNew}
   422→                  onClick={() => setShowNew((v) => !v)}
   423→                  labels={pwLabels}
   424→                />
   425→              </div>
   426→              <p className="text-muted-foreground text-xs">{t("passwordHint")}</p>
   427→            </div>
   428→
   429→            <div className="space-y-2">
   430→              <Label htmlFor="reset-confirm">{t("confirmNewPassword")}</Label>
   431→              <div className="relative">
   432→                <FieldIcon icon={Lock} />
   433→                <Input
   434→                  id="reset-confirm"
   435→                  type={showConfirm ? "text" : "password"}
   436→                  autoComplete="new-password"
   437→                  dir="ltr"
   438→                  value={confirmPassword}
   439→                  onChange={(e) => setConfirmPassword(e.target.value)}
   440→                  required
   441→                  className="h-11 ps-9 pe-10"
   442→                />
   443→                <PasswordVisibilityButton
   444→                  show={showConfirm}
   445→                  onClick={() => setShowConfirm((v) => !v)}
   446→                  labels={pwLabels}
   447→                />
   448→              </div>
   449→            </div>
   450→
   451→            {tokenInvalid ? (
   452→              <div className={errorBoxClass} role="alert">
   453→                <p>{t("errors.invalidResetToken")}</p>
   454→                <button
   455→                  type="button"
   456→                  className="text-primary mt-1.5 font-medium underline-offset-4 hover:underline"
   457→                  onClick={onRequestNewLink}
   458→                >
   459→                  {t("sendResetLink")}
   460→                </button>
   461→              </div>
   462→            ) : error ? (
   463→              <p className={errorBoxClass} role="alert">
   464→                {error}
   465→              </p>
   466→            ) : null}
   467→
   468→            <Button
   469→              type="submit"
   470→              className="h-11 w-full bg-gradient-to-r from-emerald-600 to-teal-600 text-white shadow-lg shadow-emerald-600/25 transition-all hover:from-emerald-500 hover:to-teal-500 hover:shadow-emerald-600/40 dark:from-emerald-500 dark:to-teal-500 dark:hover:from-emerald-400 dark:hover:to-teal-400"
   471→              disabled={resetMutation.isPending}
   472→            >
   473→              {resetMutation.isPending ? (
   474→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   475→              ) : (
   476→                <KeyRound className="size-4" aria-hidden="true" />
   477→              )}
   478→              {t("resetSubmit")}
   479→            </Button>
   480→          </form>
   481→        )}
   482→      </CardContent>
   483→    </Card>
   484→  );
   485→}

   494→  // message map with explicit t() calls instead (BUGFIX r8).
   495→  const authErrorMessages: AuthErrors = {
   496→    invalidCredentials: t("errors.invalidCredentials"),
   497→    rateLimited: t("errors.rateLimited"),
   498→    emailTaken: t("errors.emailTaken"),
   499→    passwordMismatch: t("errors.passwordMismatch"),
   500→    passwordPolicy: t("errors.passwordPolicy"),
   501→    invalidData: t("errors.invalidData"),
   502→    network: t("errors.network"),
   503→    generic: t("errors.generic"),

   506→  const queryClient = useQueryClient();
   507→  const [mode, setMode] = useState<"signin" | "signup">("signin");

   511→  const [confirmPassword, setConfirmPassword] = useState("");

   513→
   514→  // Prompt 30 §46/§109 — the MFA login step machine. Password success with
   515→  // MFA enabled NEVER creates a session (authorize throws the fixed
   516→  // MFA_REQUIRED code); the client then obtains a SHORT-LIVED challenge,
   517→  // collects the TOTP/recovery code, and exchanges it for a ONE-TIME
   518→  // assertion that completes the real NextAuth login.
   519→  const [mfaStage, setMfaStage] = useState<"idle" | "challenge" | "code">("idle");
   520→  const [mfaChallenge, setMfaChallenge] = useState<string | null>(null);

   523→  // Prompt 27 §53/§58: explicit legal acceptance — UNCHECKED by default, with
   524→  // links to the actual documents BEFORE acceptance. Required by the signup UI.
   525→  const [acceptedLegal, setAcceptedLegal] = useState(false);
   526→  // Password visibility toggles (presentation only — never affects logic).
   527→  const [showPassword, setShowPassword] = useState(false);
   528→  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
   529→  const pwLabels = { show: t("showPassword"), hide: t("hidePassword") };
   530→  const tLegal = useTranslations("legal.auth");
   531→  const enter = useEnterMotion();
   532→
   533→  const signInMutation = useMutation({
   534→    mutationFn: async () => {
   535→      const res = await signIn("credentials", { redirect: false, email, password });
   536→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   537→      return res;
   538→    },
   539→    onSuccess: () => {
   540→      setError(null);
   541→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   542→      void queryClient.invalidateQueries({ queryKey: ["me"] });

   544→    onError: (err) => {

   556→    },
   557→  });

   584→          signInMutation.mutate();
   585→          return;
   586→        }
   587→        if (err.code === "RATE_LIMITED") {
   588→          setError(t("mfaRateLimited"));
   589→          return;
   590→        }

   608→        businessId: null,

   612→        email,
   613→        mfaAssertion: assertion.assertion,
   614→      });

   616→      return res;

   621→      setMfaCode("");

   630→          setMfaCode("");
   631→          return;
   632→        }

   634→          setError(t("mfaChallengeExpired"));
   635→          return;
   636→        }
   637→        if (err.code === "MFA_RATE_LIMITED" || err.code === "RATE_LIMITED") {
   638→          setError(t("mfaRateLimited"));
   639→          return;
   640→        }
   641→        if (err.code === "MFA_RECOVERY_CODE_INVALID" || err.code === "MFA_RECOVERY_CODE_USED") {
   642→          setError(t("mfaInvalidCode"));
   643→          setMfaCode("");
   644→          return;
   645→        }
   646→      }
   647→      // Assertion replay/expiry inside authorize(): restart the MFA step.
   648→      if (err.message === "MFA_REQUIRED") {
   649→        setError(t("mfaChallengeExpired"));
   650→        setMfaStage("idle");
   651→        setMfaChallenge(null);
   652→        return;
   653→      }
   654→      setError(authErrorMessages.generic);
   655→    },
