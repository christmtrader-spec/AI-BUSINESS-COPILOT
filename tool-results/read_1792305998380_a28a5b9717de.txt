     3→/**
     4→ * Prompt 51 (§8) — Detailed Inventory Valuation panel for the Reports
     5→ * module. The row-level truth behind the summary's operational inventory
     6→ * estimate: current stock × current cost for EVERY included product.
     7→ *
     8→ * Every figure comes from /api/v1/reports/inventory-valuation — the UI
     9→ * never recomputes anything (Rule 05). Sorting is server-side over the
    10→ * COMPLETE matching set; the client never sorts one page and implies it
    11→ * sorted the whole result (§6.3 — no client sortValue is wired).
    12→ *
    13→ * LAWS (ADR 0051):
    14→ *  - The valuation basis disclosure is always visible (§8.6) and the
    15→ *    wording never claims accounting concepts (no asset value, FIFO,
    16→ *    balance sheet, COGS, profit).
    17→ *  - Zero current cost is a neutral review flag — never an accusation,
    18→ *    never auto-corrected, no purchase price is ever suggested (§8.5).
    19→ *  - Products navigate by STABLE id (entity route) — never by name.
    20→ *  - Filters refetch server-authoritative data; no stale totals are shown
    21→ *    under a new filter selection (no placeholderData).
    22→ *  - Quantities of different units are never summed into one grand total.
    23→ */

    62→  name: string;

    88→  filtersActive: boolean;

    93→    zeroStockCount: number;

   102→    pageSize: number;
   103→    totalCount: number;
   104→    totalPages: number;
   105→    hasMore: boolean;
   106→  };
   107→  rows: InvValRow[];
   108→};

   110→const SORTS = [
   111→  "VALUE_DESC",
   112→  "VALUE_ASC",
   113→  "NAME_ASC",

   132→type CostState = "ALL" | "POSITIVE_COST" | "ZERO_COST";

   137→export function InventoryValuationPanel({ business }: { business: BusinessSummary }) {
   138→  const t = useTranslations("reports");
   139→  const tErrors = useTranslations("errors");
   140→  const billingErrorText = useBillingErrorText();
   141→  const uiLocale = useLocale();
   142→  const navigate = useAppStore((s) => s.navigate);
   143→
   144→  // Filter state — the DEFAULT state is documented: everything included,
   145→  // value descending. Reset returns exactly here.
   146→  const [searchInput, setSearchInput] = useState("");
   147→  const [q, setQ] = useState<string | null>(null);
   148→  const [costState, setCostState] = useState<CostState>("ALL");
   149→  const [stockState, setStockState] = useState<StockState>("ALL");
   150→  const [activity, setActivity] = useState<Activity>("ALL");
   151→  const [sort, setSort] = useState<Sort>("VALUE_DESC");
   152→  const [page, setPage] = useState(1);
   153→  const [pageSize, setPageSize] = useState(25);
   154→
   155→  // Debounced search — 300 ms after the last keystroke.
   156→  useEffect(() => {
   157→    const handle = setTimeout(() => {
   158→      const trimmed = searchInput.trim();
   159→      setQ(trimmed === "" ? null : trimmed);
   160→      setPage(1);
   161→    }, 300);
   162→    return () => clearTimeout(handle);
   163→  }, [searchInput]);
   164→
   165→  const changeFilter = <T,>(setter: (v: T) => void) => (v: T) => {
   166→    setter(v);
   167→    setPage(1); // a dataset change always restarts pagination
   168→  };
   169→
   170→  const params = new URLSearchParams();
   171→  if (q) params.set("q", q);
   172→  if (costState !== "ALL") params.set("costState", costState);
   173→  if (stockState !== "ALL") params.set("stockState", stockState);
   174→  if (activity !== "ALL") params.set("activity", activity);
   175→  if (sort !== "VALUE_DESC") params.set("sort", sort);
   176→  params.set("page", String(page));
   177→  params.set("pageSize", String(pageSize));
   178→
   179→  // The export carries ONLY the dataset filters — pagination state is
   180→  // meaningless for a complete-file export (the server exports every
   181→  // matching row regardless of the current page).
   182→  const exportParams = new URLSearchParams();
   183→  if (q) exportParams.set("q", q);
   184→  if (costState !== "ALL") exportParams.set("costState", costState);
   185→  if (stockState !== "ALL") exportParams.set("stockState", stockState);
   186→  if (activity !== "ALL") exportParams.set("activity", activity);
   187→  if (sort !== "VALUE_DESC") exportParams.set("sort", sort);
   188→
   189→  const { data, isLoading, error, refetch, isFetching } = useQuery<InvValReport>({
   190→    queryKey: ["reports", "inventory-valuation", q, costState, stockState, activity, sort, page, pageSize],
   191→    queryFn: () => api(`/api/v1/reports/inventory-valuation?${params.toString()}`),
   192→    staleTime: 15_000,
   193→    // No placeholderData: a NEW filter selection must never display the
   194→    // PREVIOUS selection's financial totals (§8.7).
   195→  });
   196→
   197→  const resetFilters = () => {
   198→    setSearchInput("");
   199→    setQ(null);
   200→    setCostState("ALL");
   201→    setStockState("ALL");
   202→    setActivity("ALL");
   203→    setSort("VALUE_DESC");
   204→    setPage(1);
   205→    setPageSize(25);
   206→  };
   207→
   208→  // CSV export — server-generated from the SAME calculation (single source
   209→  // of truth); the client only streams the blob to a file.
   210→  const exportCsv = useMutation({
   211→    mutationFn: async () => {
   212→      const res = await fetch(`/api/v1/reports/inventory-valuation/export?${exportParams.toString()}`, {
   213→        credentials: "same-origin",
   214→        headers: useAppStore.getState().activeBusinessId
   215→          ? { "x-business-id": useAppStore.getState().activeBusinessId ?? "" }
   216→          : undefined,
   217→      });
   218→      if (!res.ok) {
   219→        let message = t("invValExportFailed");
   220→        let code = "";
   221→        try {
   222→          const body = (await res.json()) as { error?: { code?: string; message?: string } };
   223→          code = body.error?.code ?? "";
   224→          if (code === "RATE_LIMITED") message = t("csvRateLimited");
   225→          else if (code === "EXPORT_TOO_LARGE" || code === "REPORT_TOO_LARGE") message = t("invValExportTooLarge");
   226→          else if (body.error?.message) message = body.error.message;
   227→        } catch {
   228→          // keep default
   229→        }
   230→        throw Object.assign(new Error(message), { code });
   231→      }
   232→      const blob = await res.blob();
   233→      const url = URL.createObjectURL(blob);
   234→      const a = document.createElement("a");
   235→      a.href = url;
   236→      const dispo = res.headers.get("content-disposition") ?? "";
   237→      const match = /filename="([^"]+)"/.exec(dispo);
   238→      a.download = match?.[1] ?? "inventory-valuation.csv";
   239→      document.body.appendChild(a);
   240→      a.click();
   241→      a.remove();
   242→      URL.revokeObjectURL(url);
   243→    },
   244→    onSuccess: () => toast({ title: t("invValExportedToast") }),
   245→    onError: (err) =>
   246→      toast({
   247→        title: billingErrorText(err) || (err instanceof Error ? err.message : tErrors("generic")),
   248→        variant: "destructive",
   249→      }),
   250→  });
   251→
   252→  const fmt = (minor: number) => fmtMoney(business, minor);
   253→  const count = (n: number) => fmtCount(uiLocale, n);
   254→
   255→  const columns = [
   256→    {
   257→      id: "product",
   258→      header: t("invValColProduct"),
   259→      align: "start" as const,
   260→      cell: (r: InvValRow) => (
   261→        <div className="flex min-w-0 flex-col">
   262→          <button
   263→            type="button"
   264→            className="hover:text-foreground max-w-[22rem] truncate rounded text-start font-medium transition-colors focus-visible:ring-ring focus-visible:outline-none focus-visible:ring-2"
   265→            title={r.sku ? `${r.name} — ${r.sku}` : r.name}
   266→            aria-label={`${t("invValOpenProduct")}: ${r.name}`}
   267→            onClick={() => navigate({ kind: "entity", entity: "product", id: r.id })}
   268→          >
   269→            {r.name}
   270→          </button>
   271→          {r.sku ? (
   272→            <span className="text-muted-foreground max-w-[22rem] truncate text-xs" dir="ltr">
   273→              {r.sku}
   274→            </span>
   275→          ) : null}
   276→        </div>
   277→      ),
   278→    },
   279→    {
   280→      id: "qty",
   281→      header: t("invValColQty"),
   282→      align: "end" as const,
   283→      cell: (r: InvValRow) => (
   284→        <span className="tabular-nums">
   285→          <Ltr>{count(r.stockQty)}</Ltr>
   286→          {r.unit ? <span className="text-muted-foreground ms-1 text-xs">{r.unit}</span> : null}
   287→        </span>
   288→      ),
   289→    },
   290→    {
   291→      id: "unitCost",
   292→      header: t("invValColUnitCost"),
   293→      align: "end" as const,
   294→      cell: (r: InvValRow) => (
   295→        <Ltr className="tabular-nums">{fmt(r.costPriceMinor)}</Ltr>
   296→      ),
   297→    },
   298→    {
   299→      id: "value",
   300→      header: t("invValColValue"),
   301→      align: "end" as const,
   302→      cell: (r: InvValRow) => (
   303→        <span className="font-medium tabular-nums">
   304→          <Ltr>{fmt(r.estimatedValueMinor)}</Ltr>
   305→        </span>
   306→      ),
   307→    },
   308→    {
   309→      id: "costState",
   310→      header: t("invValColStatus"),
   311→      align: "start" as const,
   312→      hideBelow: "md" as const,
   313→      cell: (r: InvValRow) => (
   314→        <div className="flex flex-wrap items-center gap-1">
   315→          {r.stockState === "NEGATIVE_STOCK" ? (
   316→            <Badge
   317→              variant="outline"
   318→              className="border-destructive/40 bg-destructive/5 shrink-0 px-1.5 py-0 text-[10px] text-destructive"
   319→            >
   320→              {t("invValStateNegativeStock")}
   321→            </Badge>
   322→          ) : null}
   323→          {r.costState === "ZERO_COST" ? (
   324→            <Badge
   325→              variant="outline"
   326→              className="shrink-0 border-amber-300/60 bg-amber-500/10 px-1.5 py-0 text-[10px] text-amber-700 dark:text-amber-400"
   327→            >
   328→              {t("invValStateZeroCost")}
   329→            </Badge>
   330→          ) : null}
   331→          <Badge variant="outline" className="shrink-0 px-1.5 py-0 text-[10px]">
   332→            {r.isActive ? t("invValStateActive") : t("invValStateInactive")}
   333→          </Badge>
   334→        </div>
   335→      ),
   336→    },
   337→  ];
   338→
   339→  const pg = data?.pagination;
   340→  const from = pg && pg.totalCount > 0 ? (pg.page - 1) * pg.pageSize + 1 : 0;
   341→  const to = pg ? Math.min(pg.page * pg.pageSize, pg.totalCount) : 0;
   342→
   343→  return (
   344→    <Card data-testid="inventory-valuation" data-view="module:reports:inventory">
   345→      <CardHeader className="gap-3">
   346→        <div className="flex flex-wrap items-start justify-between gap-2">
   347→          <div className="min-w-0">
   348→            <Button
   349→              variant="ghost"
   350→              size="sm"
   351→              className="text-muted-foreground -ms-2 mb-1 h-8 gap-1.5"
   352→              onClick={() => navigate({ kind: "module", module: "reports" }, { replace: true })}
   353→            >
   354→              <ChevronLeft className="size-4 rtl:rotate-180" aria-hidden="true" />
   355→              {t("invValBack")}
   356→            </Button>
   357→            <CardTitle className="flex items-center gap-2 text-lg">
   358→              <PackageSearch className="text-primary size-5" aria-hidden="true" />
   359→              {t("invValTitle")}
   360→            </CardTitle>
   361→            <CardDescription className="mt-1">{t("invValDesc")}</CardDescription>
   362→          </div>
   363→          <Button
   364→            variant="outline"
   365→            size="sm"
   366→            className="h-9 gap-1.5"
   367→            onClick={() => exportCsv.mutate()}
   368→            disabled={exportCsv.isPending || isLoading || Boolean(error)}
   369→            title={t("invValExport")}
   370→          >
   371→            {exportCsv.isPending ? (
   372→              <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   373→            ) : (
   374→              <Download className="size-4" aria-hidden="true" />
   375→            )}
   376→            <span aria-hidden="true">{exportCsv.isPending ? t("invValExporting") : t("invValExport")}</span>
   377→          </Button>
   378→        </div>
   379→
   380→        {/* §8.6 — the valuation-basis disclosure: always visible, never an
   381→            accounting claim. */}
   382→        <div
   383→          className="bg-muted/50 text-muted-foreground flex items-start gap-2 rounded-lg border px-3 py-2 text-xs"
   384→          data-testid="inventory-valuation-basis"
   385→        >
   386→          <Info className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   387→          <p>{t("invValBasis")}</p>
   388→        </div>
   389→      </CardHeader>
   390→
   391→      <CardContent className="space-y-4">
   392→        {/* Filters (§8.3) — closed vocabularies mirrored from the API. */}

   394→          <div className="min-w-[10rem] flex-1 sm:max-w-xs">
   395→            <label htmlFor="inv-val-search" className="text-muted-foreground mb-1 block text-xs font-medium">
   396→              {t("invValSearch")}
   397→            </label>
   398→            <Input
   399→              id="inv-val-search"
   400→              type="search"
   401→              value={searchInput}
   402→              onChange={(e) => setSearchInput(e.target.value)}
   403→              placeholder={t("invValSearchPlaceholder")}
   404→              className="h-9"
   405→              maxLength={80}
   406→            />
   407→          </div>
   408→          <div className="w-[8.5rem]">
   409→            <label className="text-muted-foreground mb-1 block text-xs font-medium" htmlFor="inv-val-cost">
   410→              {t("invValCostState")}
   411→            </label>
   412→            <Select value={costState} onValueChange={(v) => changeFilter(setCostState)(v as CostState)}>
   413→              <SelectTrigger id="inv-val-cost" className="h-9" aria-label={t("invValCostState")}>
   414→                <SelectValue />
   415→              </SelectTrigger>
   416→              <SelectContent>
   417→                <SelectItem value="ALL">{t("invValFilterAll")}</SelectItem>
   418→                <SelectItem value="POSITIVE_COST">{t("invValFilterPositiveCost")}</SelectItem>
   419→                <SelectItem value="ZERO_COST">{t("invValFilterZeroCost")}</SelectItem>
   420→              </SelectContent>
   421→            </Select>
   422→          </div>
   423→          <div className="w-[8.5rem]">
   424→            <label className="text-muted-foreground mb-1 block text-xs font-medium" htmlFor="inv-val-stock">
   425→              {t("invValStockState")}
   426→            </label>
   427→            <Select value={stockState} onValueChange={(v) => changeFilter(setStockState)(v as StockState)}>
   428→              <SelectTrigger id="inv-val-stock" className="h-9" aria-label={t("invValStockState")}>
   429→                <SelectValue />
   430→              </SelectTrigger>
   431→              <SelectContent>
   432→                <SelectItem value="ALL">{t("invValFilterAll")}</SelectItem>
   433→                <SelectItem value="POSITIVE_STOCK">{t("invValFilterPositiveStock")}</SelectItem>
   434→                <SelectItem value="ZERO_STOCK">{t("invValFilterZeroStock")}</SelectItem>
   435→                <SelectItem value="LOW_STOCK">{t("invValFilterLowStock")}</SelectItem>
   436→              </SelectContent>
   437→            </Select>
   438→          </div>
   439→          <div className="w-[8.5rem]">
   440→            <label className="text-muted-foreground mb-1 block text-xs font-medium" htmlFor="inv-val-activity">
   441→              {t("invValActivity")}
   442→            </label>
   443→            <Select value={activity} onValueChange={(v) => changeFilter(setActivity)(v as Activity)}>
   444→              <SelectTrigger id="inv-val-activity" className="h-9" aria-label={t("invValActivity")}>

   447→              <SelectContent>
   448→                <SelectItem value="ALL">{t("invValFilterAll")}</SelectItem>
   449→                <SelectItem value="ACTIVE">{t("invValFilterActive")}</SelectItem>
   450→                <SelectItem value="INACTIVE">{t("invValFilterInactive")}</SelectItem>
   451→              </SelectContent>

   453→          </div>
   454→          <div className="w-[10.5rem]">
   455→            <label className="text-muted-foreground mb-1 block text-xs font-medium" htmlFor="inv-val-sort">
   456→              {t("invValSort")}
   457→            </label>
   458→            <Select value={sort} onValueChange={(v) => changeFilter(setSort)(v as Sort)}>
   459→              <SelectTrigger id="inv-val-sort" className="h-9" aria-label={t("invValSort")}>
   460→                <SelectValue />
   461→              </SelectTrigger>
   462→              <SelectContent>
   463→                {SORTS.map((s) => (
   464→                  <SelectItem key={s} value={s}>
   465→                    {t(SORT_LABEL_KEYS[s])}

   472→            <label className="text-muted-foreground mb-1 block text-xs font-medium" htmlFor="inv-val-pagesize">
   473→              {t("invValPerPage")}
   474→            </label>
   475→            <Select
   476→              value={String(pageSize)}
   477→              onValueChange={(v) => changeFilter(setPageSize)(Number(v))}

   479→              <SelectTrigger id="inv-val-pagesize" className="h-9" aria-label={t("invValPerPage")}>
   480→                <SelectValue />
   481→              </SelectTrigger>
   482→              <SelectContent>
   483→                {PAGE_SIZES.map((s) => (
   484→                  <SelectItem key={s} value={s}>
   485→                    {s}
   486→                  </SelectItem>
   487→                ))}

   489→            </Select>
   490→          </div>
   491→          <Button
   492→            variant="ghost"
   493→            size="sm"
   494→            className="text-muted-foreground h-9 gap-1.5"
   495→            onClick={resetFilters}
   496→            title={t("invValReset")}

   498→            <RotateCcw className="size-3.5" aria-hidden="true" />
   499→            {t("invValReset")}
   500→          </Button>
   501→        </div>
   502→
   503→        {error ? (
   504→          <div className="text-destructive bg-destructive/10 rounded-md px-3 py-2 text-sm" role="alert">
   505→            {t("invValLoadFailed")}{" "}
   506→            <button className="underline" onClick={() => void refetch()}>
   507→              {t("retry")}

   510→        ) : isLoading || !data ? (
   511→          <div className="space-y-3" aria-busy="true">
   512→            <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   513→              {[0, 1, 2, 3].map((i) => (
   514→                <Skeleton key={i} className="h-20 rounded-xl" />
   515→              ))}
   516→            </div>
