    17→type AuthErrors = {
    18→  invalidCredentials: string;
    19→  rateLimited: string;
    20→  emailTaken: string;
    21→  passwordMismatch: string;
    22→  network: string;
    23→  generic: string;
    24→};
    25→
    26→function mapSignInError(error: string | null | undefined, t: AuthErrors): string {
    27→  if (error === "CredentialsSignin") return t.invalidCredentials;
    28→  return t.generic;
    29→}
    30→
    31→function mapApiError(err: unknown, t: AuthErrors): string {
    32→  if (err instanceof ApiClientError) {
    33→    if (err.status === 0) return t.network;
    34→    if (err.code === "EMAIL_TAKEN") return t.emailTaken;
    35→    if (err.code === "RATE_LIMITED") return t.rateLimited;
    36→  }
    37→  return t.generic;
    38→}

    44→/**
    45→ * "Forgot password" flow — requests a reset link. The server ALWAYS answers

    57→      api<{ message?: string }>("/api/v1/auth/password-reset/request", {

    59→        body: JSON.stringify({ email }),

   166→/**
   167→ * "Reset password" flow — consumes a single-use token from the deep link
   168→ * (#/reset-password/<token>). On success every previous session is dead:
   169→ * any stale local session is cleared immediately and the user is sent to
   170→ * sign-in via the success card button.
   171→ */
   172→function ResetPasswordCard({
   173→  token,
   174→  onGoToSignIn,
   175→  onRequestNewLink,
   176→}: {
   177→  token: string;
   178→  onGoToSignIn: () => void;
   179→  onRequestNewLink: () => void;
   180→}) {
   181→  const t = useTranslations("auth");
   182→  const queryClient = useQueryClient();
   183→  const [newPassword, setNewPassword] = useState("");
   184→  const [confirmPassword, setConfirmPassword] = useState("");
   185→  const [error, setError] = useState<string | null>(null);
   186→  const [tokenInvalid, setTokenInvalid] = useState(false);
   187→  const [doneMessage, setDoneMessage] = useState<string | null>(null);
   188→
   189→  const resetMutation = useMutation({
   190→    mutationFn: async () =>
   191→      api<{ message?: string }>("/api/v1/auth/password-reset/confirm", {
   192→        method: "POST",
   193→        body: JSON.stringify({ token, newPassword, confirmPassword }),
   194→      }),
   195→    onSuccess: () => {
   196→      setError(null);
   197→      setTokenInvalid(false);
   198→      setDoneMessage(t("resetSuccessBody"));
   199→      // The server killed every session for this user — clear any stale
   200→      // local one right away (the success card below routes to sign-in).
   201→      void signOut({ redirect: false });
   202→      void queryClient.invalidateQueries();
   203→    },
   204→    onError: (err) => {
   205→      if (err instanceof ApiClientError) {
   206→        if (err.code === "INVALID_RESET_TOKEN") {
   207→          setTokenInvalid(true);
   208→          setError(null);
   209→        } else if (err.status === 0) {
   210→          setTokenInvalid(false);
   211→          setError(t("errors.network"));
   212→        } else if (err.code === "RATE_LIMITED") {
   213→          setTokenInvalid(false);
   214→          setError(t("errors.rateLimited"));
   215→        } else {
   216→          setTokenInvalid(false);
   217→          setError(t("errors.generic"));
   218→        }
   219→      } else {
   220→        setTokenInvalid(false);
   221→        setError(t("errors.generic"));
   222→      }
   223→    },
   224→  });
   225→
   226→  return (
   227→    <Card className="w-full max-w-md shadow-sm">
   228→      <CardHeader className="space-y-1.5">
   229→        <CardTitle className="text-2xl">
   230→          {doneMessage ? t("resetSuccessTitle") : t("resetTitle")}
   231→        </CardTitle>
   232→        {doneMessage ? null : <CardDescription>{t("resetSubtitle")}</CardDescription>}
   233→      </CardHeader>
   234→      <CardContent>
   235→        {doneMessage ? (
   236→          <div className="space-y-5 py-2 text-center">
   237→            <div
   238→              className="bg-primary/10 mx-auto flex size-12 items-center justify-center rounded-full"
   239→              aria-hidden="true"
   240→            >
   241→              <KeyRound className="text-primary size-6" />
   242→            </div>
   243→            <p className="text-foreground text-sm" role="status">
   244→              {doneMessage}
   245→            </p>
   246→            <Button type="button" className="w-full" onClick={onGoToSignIn}>
   247→              <LogIn className="size-4" aria-hidden="true" />
   248→              {t("goToSignIn")}
   249→            </Button>
   250→          </div>
   251→        ) : (
   252→          <form
   253→            onSubmit={(e) => {
   254→              e.preventDefault();
   255→              setError(null);
   256→              setTokenInvalid(false);
   257→              if (newPassword !== confirmPassword) {
   258→                setError(t("errors.passwordMismatch"));
   259→                return;
   260→              }
   261→              resetMutation.mutate();
   262→            }}
   263→            className="space-y-4"
   264→            noValidate
   265→          >
   266→            <div className="space-y-2">
   267→              <Label htmlFor="reset-password">{t("newPassword")}</Label>
   268→              <Input
   269→                id="reset-password"
   270→                type="password"
   271→                autoComplete="new-password"
   272→                dir="ltr"
   273→                placeholder={t("passwordPlaceholder")}
   274→                value={newPassword}
   275→                onChange={(e) => setNewPassword(e.target.value)}
   276→                required
   277→                minLength={8}
   278→              />
   279→              <p className="text-muted-foreground text-xs">{t("passwordHint")}</p>
   280→            </div>
   281→
   282→            <div className="space-y-2">
   283→              <Label htmlFor="reset-confirm">{t("confirmNewPassword")}</Label>
   284→              <Input
   285→                id="reset-confirm"
   286→                type="password"
   287→                autoComplete="new-password"
   288→                dir="ltr"
   289→                value={confirmPassword}
   290→                onChange={(e) => setConfirmPassword(e.target.value)}
   291→                required
   292→              />
   293→            </div>
   294→
   295→            {tokenInvalid ? (
   296→              <div className={errorBoxClass} role="alert">
   297→                <p>{t("errors.invalidResetToken")}</p>
   298→                <button
   299→                  type="button"
   300→                  className="text-primary mt-1.5 font-medium underline-offset-4 hover:underline"
   301→                  onClick={onRequestNewLink}
   302→                >
   303→                  {t("sendResetLink")}
   304→                </button>
   305→              </div>
   306→            ) : error ? (
   307→              <p className={errorBoxClass} role="alert">
   308→                {error}
   309→              </p>
   310→            ) : null}
   311→
   312→            <Button type="submit" className="w-full" disabled={resetMutation.isPending}>
   313→              {resetMutation.isPending ? (
   314→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   315→              ) : (
   316→                <KeyRound className="size-4" aria-hidden="true" />
   317→              )}
   318→              {t("resetSubmit")}
   319→            </Button>
   320→          </form>
   321→        )}
   322→      </CardContent>
   323→    </Card>
   324→  );
   325→}

   327→export function AuthView() {
   328→  const t = useTranslations("auth");
   329→  const tCommon = useTranslations("common");
   330→  const view = useAppStore((s) => s.view);
   331→  const navigate = useAppStore((s) => s.navigate);
   332→  // next-intl's t is a function — nested property access (t.errors) is
   333→  // undefined at runtime, which crashed the onError handlers. Build the
   334→  // message map with explicit t() calls instead (BUGFIX r8).
   335→  const authErrorMessages: AuthErrors = {
   336→    invalidCredentials: t("errors.invalidCredentials"),
   337→    rateLimited: t("errors.rateLimited"),
   338→    emailTaken: t("errors.emailTaken"),
   339→    passwordMismatch: t("errors.passwordMismatch"),
   340→    network: t("errors.network"),
   341→    generic: t("errors.generic"),
   342→  };
   343→  const { status } = useSession();
   344→  const queryClient = useQueryClient();
   345→  const [mode, setMode] = useState<"signin" | "signup">("signin");
   346→  const [name, setName] = useState("");
   347→  const [email, setEmail] = useState("");
   348→  const [password, setPassword] = useState("");
   349→  const [confirmPassword, setConfirmPassword] = useState("");
   350→  const [error, setError] = useState<string | null>(null);
   351→
   352→  // Prompt 30 §46/§109 — the MFA login step machine. Password success with
   353→  // MFA enabled NEVER creates a session (authorize throws the fixed
   354→  // MFA_REQUIRED code); the client then obtains a SHORT-LIVED challenge,
   355→  // collects the TOTP/recovery code, and exchanges it for a ONE-TIME
   356→  // assertion that completes the real NextAuth login.
   357→  const [mfaStage, setMfaStage] = useState<"idle" | "challenge" | "code">("idle");
   358→  const [mfaChallenge, setMfaChallenge] = useState<string | null>(null);
   359→  const [mfaMethod, setMfaMethod] = useState<"TOTP" | "RECOVERY_CODE">("TOTP");
   360→  const [mfaCode, setMfaCode] = useState("");
   361→  // Prompt 27 §53/§58: explicit legal acceptance — UNCHECKED by default, with
   362→  // links to the actual documents BEFORE acceptance. Required by the signup UI.
   363→  const [acceptedLegal, setAcceptedLegal] = useState(false);
   364→  const tLegal = useTranslations("legal.auth");
   365→
   366→  const signInMutation = useMutation({
   367→    mutationFn: async () => {
   368→      const res = await signIn("credentials", { redirect: false, email, password });
   369→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   370→      return res;
   371→    },
   372→    onSuccess: () => {
   373→      setError(null);
   374→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   375→      void queryClient.invalidateQueries({ queryKey: ["me"] });
   376→    },
   377→    onError: (err) => {
   378→      // Prompt 30 §110 — MFA_REQUIRED arrives ONLY after the password was
   379→      // verified inside authorize(); switch to the bounded MFA step.
   380→      if (err.message === "MFA_REQUIRED") {
   381→        void startMfaChallenge();
   382→        return;
   383→      }
   384→      setError(
   385→        err instanceof ApiClientError
   386→          ? mapApiError(err, authErrorMessages)
   387→          : mapSignInError(err.message, authErrorMessages),
   388→      );
   389→    },
   390→  });
   391→
   392→  // Step 2 of the MFA flow (§46): re-verify the password at the dedicated
   393→  // endpoint to obtain the challenge token (rate-limited, hash-only server
   394→  // state, bound to the user's sessionVersion).
   395→  const startMfaChallenge = () => {
   396→    setMfaStage("challenge");
   397→    mfaChallengeMutation.mutate();
   398→  };
   399→
   400→  const mfaChallengeMutation = useMutation({
   401→    mutationFn: async () =>
   402→      api<{ mfaRequired: boolean; challenge: string; expiresAt: string }>(
   403→        "/api/v1/auth/mfa/challenge",
   404→        { method: "POST", body: JSON.stringify({ email, password }), businessId: null },
   405→      ),
   406→    onSuccess: (res) => {
   407→      setMfaChallenge(res.challenge);
   408→      setMfaStage("code");
   409→      setMfaCode("");
   410→      setError(null);
   411→    },
   412→    onError: (err) => {
   413→      setMfaStage("idle");
   414→      if (err instanceof ApiClientError) {
   415→        if (err.code === "MFA_NOT_ENABLED") {
   416→          // The account has no MFA (state changed mid-flow) — retry plain.
   417→          signInMutation.mutate();
   418→          return;
   419→        }
   420→        if (err.code === "RATE_LIMITED") {
   421→          setError(t("mfaRateLimited"));
   422→          return;
   423→        }
   424→        if (err.code === "INVALID_CREDENTIALS") {
   425→          setError(authErrorMessages.invalidCredentials);
   426→          return;
   427→        }
   428→      }
   429→      setError(authErrorMessages.generic);
   430→    },
   431→  });
   432→
   433→  // Step 3 (§51): exchange challenge + code for the one-time assertion,
   434→  // then complete the REAL NextAuth sign-in with it (§52 — the final JWT
   435→  // is still issued by NextAuth only).
   436→  const mfaCompleteMutation = useMutation({
   437→    mutationFn: async () => {
   438→      const assertion = await api<{ assertion: string }>("/api/v1/auth/mfa/complete", {
   439→        method: "POST",
   440→        body: JSON.stringify({ challenge: mfaChallenge, code: mfaCode, method: mfaMethod }),
   441→        businessId: null,
   442→      });
   443→      const res = await signIn("credentials", {
   444→        redirect: false,
   445→        email,
   446→        mfaAssertion: assertion.assertion,
   447→      });
   448→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   449→      return res;
   450→    },
   451→    onSuccess: () => {
   452→      setMfaStage("idle");
   453→      setMfaChallenge(null);
   454→      setMfaCode("");
   455→      setError(null);
   456→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   457→      void queryClient.invalidateQueries({ queryKey: ["me"] });
   458→    },
   459→    onError: (err) => {
   460→      if (err instanceof ApiClientError) {
   461→        if (err.code === "MFA_INVALID_CODE") {
   462→          setError(t("mfaInvalidCode"));
   463→          setMfaCode("");
   464→          return;
   465→        }
   466→        if (err.code === "MFA_CHALLENGE_EXPIRED" || err.code === "MFA_CHALLENGE_INVALID") {
   467→          setError(t("mfaChallengeExpired"));
   468→          return;
   469→        }
   470→        if (err.code === "MFA_RATE_LIMITED" || err.code === "RATE_LIMITED") {
   471→          setError(t("mfaRateLimited"));
   472→          return;
   473→        }
   474→        if (err.code === "MFA_RECOVERY_CODE_INVALID" || err.code === "MFA_RECOVERY_CODE_USED") {
   475→          setError(t("mfaInvalidCode"));
   476→          setMfaCode("");
   477→          return;
   478→        }
   479→      }
   480→      // Assertion replay/expiry inside authorize(): restart the MFA step.
   481→      if (err.message === "MFA_REQUIRED") {
   482→        setError(t("mfaChallengeExpired"));
   483→        setMfaStage("idle");
   484→        setMfaChallenge(null);
   485→        return;
   486→      }
   487→      setError(authErrorMessages.generic);
   488→    },
   489→  });
   490→
   491→  const registerMutation = useMutation({
   492→    mutationFn: async () => {
   493→      await api("/api/v1/auth/register", {
   494→        method: "POST",
   495→        body: JSON.stringify({
   496→          name,
   497→          email,
   498→          password,
   499→          // Server resolves the accepted document versions (Prompt 27 §163).
   500→          acceptedLegal: acceptedLegal ? { terms: true, privacy: true } : undefined,
   501→        }),
   502→      });
   503→      const res = await signIn("credentials", { redirect: false, email, password });
   504→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   505→      return res;
   506→    },
   507→    onSuccess: () => {
   508→      setError(null);
   509→      void queryClient.invalidateQueries();
   510→    },
   511→    onError: (err) => {
   512→      setError(
   513→        err instanceof ApiClientError
   514→          ? mapApiError(err, authErrorMessages)
   515→          : mapSignInError(err.message, authErrorMessages),
   516→      );
   517→    },
   518→  });
   519→
   520→  const busy = signInMutation.isPending || registerMutation.isPending || status === "loading";
   521→
   522→  // Store-driven mode switches (hash updates): password flows always route
   523→  // through navigate() so deep links, back/forward and refresh all work.
   524→  function goBackToSignIn() {
   525→    setMode("signin");
   526→    setError(null);
   527→    navigate({ kind: "dashboard" });
   528→  }
   529→
   530→  function onSubmit(e: React.FormEvent) {
   531→    e.preventDefault();
   532→    setError(null);
   533→    if (mode === "signup") {
   534→      if (password !== confirmPassword) {
   535→        setError(t("errors.passwordMismatch"));
   536→        return;
   537→      }
   538→      if (!acceptedLegal) {
   539→        // Honest gate: the product requires explicit acceptance before an
   540→        // account is created (Prompt 27 §53) — the API records the versions.
   541→        setError(tLegal("acceptRequired"));
   542→        return;
   543→      }
   544→      registerMutation.mutate();
   545→    } else {
   546→      signInMutation.mutate();
   547→    }
   548→  }
   549→
   550→  return (
   551→    <div className="bg-muted/30 flex min-h-svh flex-col lg:flex-row" data-view="auth">
   552→      {/* Brand panel */}
   553→      <div className="from-primary/5 via-primary/10 to-primary/5 relative flex flex-1 flex-col justify-between overflow-hidden p-8 lg:p-14">
   554→        <div
   555→          className="bg-primary/5 pointer-events-none absolute -top-24 -end-24 size-72 rounded-full blur-3xl"
   556→          aria-hidden="true"
   557→        />
   558→        <div
   559→          className="bg-primary/5 pointer-events-none absolute -bottom-24 -start-24 size-72 rounded-full blur-3xl"
   560→          aria-hidden="true"
   561→        />
