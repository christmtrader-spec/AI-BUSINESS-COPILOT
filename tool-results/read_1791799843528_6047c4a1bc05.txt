    82→let passed = 0;
    83→let failed = 0;
    84→const failures: string[] = [];
    85→
    86→function ok(cond: boolean, name: string) {
    87→  if (cond) passed++;
    88→  else {
    89→    failed++;
    90→    failures.push(name);
    91→  }
    92→  console.log(`${cond ? "✓" : "✗"} ${name}`);
    93→}
    94→function eq(a: unknown, b: unknown, name: string) {
    95→  const same = a === b;
    96→  if (same) passed++;
    97→  else {
    98→    failed++;
    99→    failures.push(`${name} (got ${String(a)}, want ${String(b)})`);
   100→  }
   101→  console.log(`${same ? "✓" : "✗"} ${name}`);
   102→}
   103→function throws(fn: () => unknown, name: string, errorName?: string) {
   104→  try {
   105→    fn();
   106→    ok(false, name);
   107→  } catch (e) {
   108→    const matches = errorName
   109→      ? (e as Error).name === errorName || String((e as Error).message).includes(errorName)
   110→      : true;
   111→    ok(matches, name);
   112→  }
   113→}
