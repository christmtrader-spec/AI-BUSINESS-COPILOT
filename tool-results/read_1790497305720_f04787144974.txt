    56→function timezoneOptions(): string[] {
    57→  try {
    58→    const list = Intl.supportedValuesOf("timeZone");
    59→    if (Array.isArray(list) && list.length > 0) return list;
    60→  } catch {
    61→    // older engines
    62→  }
    63→  return FALLBACK_TIMEZONES;
    64→}

   278→  async function onRestoreFileChange(file: File | undefined) {
   279→    setRestoreReport(null);
   280→    setRestoreError(null);
   281→    if (!file) {
   282→      setRestoreFile(null);
   283→      return;
   284→    }

   286→      setRestoreFile(null);
   287→      setRestoreError(t("restoreFileError"));
   288→      return;
   289→    }
   290→    try {
   291→      const text = await file.text();
   292→      const json = JSON.parse(text) as {
   293→        $schema?: string;
   294→        business?: { currencyCode?: string };
   295→      };
   296→      if (json?.$schema !== BACKUP_SCHEMA_ID) {
   297→        setRestoreFile(null);
   298→        setRestoreError(t("restoreInvalidFile"));
   299→        return;
   300→      }
   301→      setRestoreFile({
   302→        name: file.name,
   303→        size: file.size,
   304→        body: json,
   305→        currencyCode: json.business?.currencyCode ?? "?",
   306→      });
   307→    } catch {
   308→      setRestoreFile(null);
   309→      setRestoreError(t("restoreInvalidFile"));
   310→    }
   311→  }

   313→  async function runRestore(mode: "dry_run" | "merge") {
   314→    if (!restoreFile || !activeId) return;

   319→        `/api/v1/businesses/${activeId}/import?mode=${mode}&ids=${restoreIdMode}`,
   320→        {
   321→        method: "POST",
   322→        credentials: "same-origin",
   323→        headers: {
   324→          "content-type": "application/json",
   325→          ...(activeId ? { "x-business-id": activeId } : {}),
   326→        },
   327→        body: JSON.stringify(restoreFile.body),
   328→      });
   329→      const body = (await res.json()) as {
   330→        ok?: boolean;
   331→        data?: { report: ImportReport };
   332→        error?: { code?: string; message?: string };
   333→      };
   334→      if (!res.ok || !body?.ok || !body.data) {
   335→        setRestoreReport(null);
   336→        if (body?.error?.code === "INVALID_BACKUP") setRestoreError(t("restoreInvalidFile"));
   337→        else if (body?.error?.code === "PERMISSION_DENIED") setRestoreError(t("restoreOwnerOnly"));
   338→        else if (body?.error?.code === "RATE_LIMITED") setRestoreError(t("exportRateLimited"));
   339→        else setRestoreError(t("restoreFailed"));
   340→        return;
   341→      }
   342→      setRestoreReport(body.data.report);
   343→      if (mode === "merge") {
   344→        toast({ title: t("restoreDone") });
   345→        void queryClient.invalidateQueries();
   346→        resetRestore();
   347→      }
   348→    } catch {
   349→      setRestoreError(t("restoreFailed"));
   350→    } finally {
   351→      setRestoreBusy(null);
   352→      setConfirmOpen(false);
   353→    }
