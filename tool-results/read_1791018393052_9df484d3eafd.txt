    11→/**
    12→ * NOTIFICATION CATALOG — Prompt 32 §5/§36/§170.
    13→ *
    14→ * THE closed directory of notification types. Every value that can ever be
    15→ * persisted in `Notification.type` is validated against this catalog; there
    16→ * is no dynamic template registration and no client-supplied type (§36/§187).
    17→ *
    18→ * Catalog law (§5): only event types backed by REAL product events are
    19→ * listed. Deliberately ABSENT (documented in docs/product/notification-matrix.md):
    20→ *  - BILLING_TRIAL_ENDING — no scheduler/background worker exists; inventing
    21→ *    a request-time reminder is out of scope (§85).
    22→ *  - BILLING_PAYMENT_REQUIRES_ACTION / RECONCILIATION_REQUIRED — platform-
    23→ *    internal states with no customer-visible product surface (§89).
    24→ *  - AI_COST_UNKNOWN / AI_COST_GOVERNANCE_WARNING — AI COST is platform data;
    25→ *    tenant-visible cost notifications are prohibited (§80).
    26→ *  - SECURITY_SESSION_REVOKED (single other-device revoke) — noise avoidance
    27→ *    is the documented policy (§97); global sign-out is covered by
    28→ *    SECURITY_SESSIONS_SIGNED_OUT.
    29→ */
    30→

    52→  // _REOPENED would fire on every routine task interaction.
    53→  "BUSINESS_CRM_TASK_ASSIGNED",
    54→  "BUSINESS_CRM_TASK_DUE_SOON",
    55→  "BUSINESS_CRM_TASK_OVERDUE",
    56→  // Prompt 38 (§6) — inventory low-stock alerts. Completes the MVP
    57→  // notification scope ("overdue, low stock"). Deliberately ABSENT
    58→  // (noise avoidance): BUSINESS_INVENTORY_RESTOCKED / OUT_OF_STOCK —
    59→  // restock is a success (no alert), and stock-out is already covered by
    60→  // the same low-stock event (qty 0 ≤ any threshold).
    61→  "BUSINESS_INVENTORY_LOW_STOCK",
    62→  "SYSTEM_ANNOUNCEMENT",
    63→] as const;
    64→
    65→export type NotificationType = (typeof NOTIFICATION_TYPES)[number];
    66→
    67→export function isNotificationType(value: unknown): value is NotificationType {
    68→  return (
    69→    typeof value === "string" &&
    70→    (NOTIFICATION_TYPES as readonly string[]).includes(value)
    71→  );
    72→}

    74→/* ------------------------------ data schemas ------------------------------ */
    75→/* Per-type STRICT schemas (§23/§24/§172/§355). notify() validates every data
    76→ * object against its type's schema BEFORE persistence; unknown keys are
    77→ * stripped, oversized payloads are rejected (§231/§346). Values are safe
    78→ * primitives only — never nested arbitrary objects (§24). */
