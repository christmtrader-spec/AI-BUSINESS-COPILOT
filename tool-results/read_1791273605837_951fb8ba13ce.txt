    18→    18→ *  - FINALIZED/CANCELLED are immutable (§50/§51).

   157→   157→export async function listStocktakes(input: {
   158→   158→  businessId: string;
   159→   159→  q?: string;
   160→   160→  status?: string;
   161→   161→  from?: string;
   162→   162→  to?: string;
   163→   163→  page: number;
   164→   164→  pageSize: number;
   165→   165→}) {
   166→   166→  const where: Prisma.InventoryStocktakeWhereInput = { businessId: input.businessId };
   167→   167→  if (input.q) where.reference = { contains: input.q };
   168→   168→  if (input.status && isStocktakeStatus(input.status)) where.status = input.status;
   169→   169→  if (input.from || input.to) {
   170→   170→    where.createdAt = {};
   171→   171→    if (input.from) where.createdAt.gte = new Date(`${input.from}T00:00:00.000Z`);
   172→   172→    if (input.to) where.createdAt.lte = new Date(`${input.to}T23:59:59.999Z`);
   173→   173→  }
   174→   174→
   175→   175→  const [rows, total] = await Promise.all([
   176→   176→    db.inventoryStocktake.findMany({
   177→   177→      where,
   178→   178→      orderBy: { createdAt: "desc" }, // deterministic newest-first list (§134)
   179→   179→      skip: (input.page - 1) * input.pageSize,
   180→   180→      take: input.pageSize,
   181→   181→      // §67 — list select minimization; a bounded line slice feeds the
   182→   182→      // counted/net columns without loading full line tables.
   183→   183→      select: {
   184→   184→        id: true,
   185→   185→        reference: true,
   186→   186→        status: true,
   187→   187→        countType: true,
   188→   188→        createdAt: true,
   189→   189→        createdByUser: { select: { id: true, name: true, email: true } },
   190→   190→        _count: { select: { lines: true } },
   191→   191→        lines: { select: { countedAt: true, varianceQty: true } },
   192→   192→      },
   193→   193→    }),
   194→   194→    db.inventoryStocktake.count({ where }),
   195→   195→  ]);
   196→   196→
   197→   197→  return {
   198→   198→    items: rows.map(stocktakeListItemDto),
   199→   199→    total,
   200→   200→    page: input.page,
   201→   201→    pageSize: input.pageSize,
   202→   202→  };
   203→   203→}

   206→   206→
   207→   207→export async function getStocktake(input: {
   208→   208→  businessId: string;
   209→   209→  id: string;
   210→   210→  page: number;
   211→   211→  pageSize: number;
   212→   212→}) {
   213→   213→  const header = await loadStocktake(db, input.businessId, input.id);
   214→   214→  // §36 — summary over ALL lines (aggregate-light select, §66).

   216→   216→    db.inventoryStocktakeLine.findMany({
   217→   217→      where: { stocktakeId: header.id },
   218→   218→      select: { countedAt: true, varianceQty: true },
   219→   219→    }),
   220→   220→    db.inventoryStocktakeLine.findMany({
   221→   221→      where: { stocktakeId: header.id },
   222→   222→      orderBy: [{ countedAt: "asc" }, { id: "asc" }], // deterministic ordering
   223→   223→      skip: (input.page - 1) * input.pageSize,
   224→   224→      take: input.pageSize,
   225→   225→      select: {
   226→   226→        id: true,
   227→   227→        productId: true,
   228→   228→        productNameAtCount: true,
   229→   229→        skuAtCount: true,
   230→   230→        unitAtCount: true,
   231→   231→        systemQtyAtCount: true,
   232→   232→        countedQty: true,
   233→   233→        varianceQty: true,
   234→   234→        countedAt: true,
   235→   235→        countedByUser: { select: { id: true, name: true, email: true } },
   236→   236→        // Current-product join: display fallback for UNCOUNTED preparation
   237→   237→        // lines (their snapshot is still the empty placeholder). Counted
   238→   238→        // lines always render their historical snapshot (dto decides).
   239→   239→        product: { select: { name: true, sku: true, unit: true, stockQty: true } },
   240→   240→      },
   241→   241→    }),
   242→   242→    db.inventoryStocktakeLine.count({ where: { stocktakeId: header.id } }),
   243→   243→  ]);
   244→   244→
   245→   245→  return stocktakeDetailDto({
   246→   246→    ...header,
   247→   247→    createdByUser: await db.user.findUnique({
   248→   248→      where: { id: header.createdByUserId },
   249→   249→      select: { id: true, name: true, email: true },
   250→   250→    }),
   251→   251→    summary: computeStocktakeSummary(summaryRows),
   252→   252→    lines: lineRows,
   253→   253→    linesPage: { page: input.page, pageSize: input.pageSize, total: lineTotal },
   254→   254→  });
   255→   255→}

   258→   258→
   259→   259→export async function addStocktakeLine(input: {
   260→   260→  businessId: string;
   261→   261→  userId: string;
   262→   262→  stocktakeId: string;
   263→   263→  productId: string;
   264→   264→  ip?: string | null;
   265→   265→  userAgent?: string | null;
   266→   266→}) {
   267→   267→  const cap = stocktakeLineCap(process.env.INVENTORY_STOCKTAKE_MAX_LINES);
   268→   268→  return db.$transaction(async (tx) => {
   269→   269→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   270→   270→    if (stocktake.status !== "DRAFT") {
   271→   271→      // §27 — DRAFT-only line preparation.
   272→   272→      throw new ApiError(
   273→   273→        422,
   274→   274→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   275→   275→        "Products can only be added while the stocktake is a draft.",
   276→   276→      );
   277→   277→    }
   278→   278→    const lineCount = await tx.inventoryStocktakeLine.count({
   279→   279→      where: { stocktakeId: stocktake.id },
   280→   280→    });
   281→   281→    if (lineCount >= cap) {
   282→   282→      // §61 — bounded transaction sizes; surfaced to the UI (lines/max).
   283→   283→      throw new ApiError(
   284→   284→        422,
   285→   285→        STOCKTAKE_ERROR_CODES.LINE_LIMIT,
   286→   286→        `Stocktake line limit reached (${cap}).`,
   287→   287→      );
   288→   288→    }
   289→   289→    // §20/§21/§56/§57 — same business, not deleted, active. Foreign/deleted
   290→   290→    // product → 404 (no existence leakage); inactive → documented 422.
   291→   291→    const product = await tx.product.findFirst({
   292→   292→      where: { id: input.productId, businessId: input.businessId },
   293→   293→      select: { id: true, deletedAt: true, isActive: true },
   294→   294→    });
   295→   295→    if (!product || product.deletedAt) {
   296→   296→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   297→   297→    }
   298→   298→    if (!product.isActive) {
   299→   299→      throw new ApiError(
   300→   300→        422,
   301→   301→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   302→   302→        "Inactive products cannot be added to a stocktake.",
   303→   303→      );
   304→   304→    }
   305→   305→    try {
   306→   306→      const line = await tx.inventoryStocktakeLine.create({
   307→   307→        data: {
   308→   308→          stocktakeId: stocktake.id,
   309→   309→          productId: product.id,
   310→   310→          // Line is PREPARATION data until counted (§29) — zero placeholders.
   311→   311→          systemQtyAtCount: 0,
   312→   312→          countedQty: 0,
   313→   313→          varianceQty: 0,
   314→   314→          countedAt: new Date(0),
   315→   315→          countedByUserId: input.userId,
   316→   316→          productNameAtCount: "",
   317→   317→          skuAtCount: null,
   318→   318→          unitAtCount: null,
   319→   319→        },
   320→   320→        select: { id: true, productId: true },
   321→   321→      });
   322→   322→      await audit(
   323→   323→        auditInput(input.businessId, input.userId, "inventory.stocktake.line_add", stocktake.id, stocktake.reference, {
   324→   324→          lineId: line.id,
   325→   325→          productId: product.id,
   326→   326→        }),
   327→   327→        tx,
   328→   328→      );
   329→   329→      return line;
   330→   330→    } catch (error) {
   331→   331→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   332→   332→        // §28/§48 — DB uniqueness is the final protection.
   333→   333→        throw new ApiError(
   334→   334→          409,
   335→   335→          STOCKTAKE_ERROR_CODES.DUPLICATE_PRODUCT,
   336→   336→          "This product is already in the stocktake.",
   337→   337→        );
   338→   338→      }
   339→   339→      throw error;
   340→   340→    }
   341→   341→  });
   342→   342→}

   380→   380→/* --------------------------------- count ---------------------------------- */
   381→   381→
   382→   382→export async function countStocktakeLine(input: {
   383→   383→  businessId: string;
   384→   384→  userId: string;
   385→   385→  stocktakeId: string;
   386→   386→  lineId: string;
   387→   387→  countedQty: number;
   388→   388→  ip?: string | null;
   389→   389→  userAgent?: string | null;
   390→   390→}) {
   391→   391→  return db.$transaction(async (tx) => {
   392→   392→    // §30 — load stocktake → verify status (COUNTING|REVIEW, §31) → verify
   393→   393→    // line → verify product tenant/active → read current stock → snapshot.
   394→   394→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   395→   395→    if (stocktake.status !== "COUNTING" && stocktake.status !== "REVIEW") {
   396→   396→      throw new ApiError(
   397→   397→        422,
   398→   398→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   399→   399→        "Counting is only possible while the stocktake is counting or in review.",
   400→   400→      );
   401→   401→    }
   402→   402→    const line = await tx.inventoryStocktakeLine.findFirst({
   403→   403→      where: { id: input.lineId, stocktakeId: stocktake.id },
   404→   404→      select: {
   405→   405→        id: true,
   406→   406→        productId: true,
   407→   407→        countedAt: true,
   408→   408→        systemQtyAtCount: true,
   409→   409→        countedQty: true,
   410→   410→        varianceQty: true,
   411→   411→      },
   412→   412→    });
   413→   413→    if (!line) throw new ApiError(404, "NOT_FOUND", "Stocktake line not found.");
   414→   414→
   415→   415→    // §15 — server reads the CURRENT stock; client never supplies it.
   416→   416→    const product = await tx.product.findFirst({
   417→   417→      where: { id: line.productId, businessId: input.businessId },
   418→   418→      select: { id: true, name: true, sku: true, unit: true, stockQty: true, deletedAt: true, isActive: true },
   419→   419→    });
   420→   420→    if (!product || product.deletedAt) {
   421→   421→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   422→   422→    }
   423→   423→    if (!product.isActive) {
   424→   424→      throw new ApiError(
   425→   425→        422,
   426→   426→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   427→   427→        "Inactive products cannot be counted.",
   428→   428→      );
   429→   429→    }
   430→   430→
   431→   431→    const systemQtyAtCount = product.stockQty; // §15 snapshot at count time
   432→   432→    const varianceQty = computeVarianceQty(input.countedQty, systemQtyAtCount); // §16/§18
   433→   433→    const now = new Date();
   434→   434→    const isRecount = line.countedAt !== null && line.countedAt.getTime() !== 0;
   435→   435→
   436→   436→    const updated = await tx.inventoryStocktakeLine.update({
   437→   437→      where: { id: line.id },
   438→   438→      data: {
   439→   439→        systemQtyAtCount,
   440→   440→        countedQty: input.countedQty,
   441→   441→        varianceQty,
   442→   442→        countedAt: now, // §30 step 14
   443→   443→        countedByUserId: input.userId, // last successful count is authoritative (§47)
   444→   444→        productNameAtCount: product.name, // §19 metadata snapshot
   445→   445→        skuAtCount: product.sku,
   446→   446→        unitAtCount: product.unit,
   447→   447→      },
   448→   448→      select: { id: true, productId: true, systemQtyAtCount: true, countedQty: true, varianceQty: true },
   449→   449→    });
   450→   450→
   451→   451→    if (isRecount) {
   452→   452→      // §32/§33 — replace in place; audit the safe before/after metadata.
   453→   453→      await audit(
   454→   454→        auditInput(
   455→   455→          input.businessId,
   456→   456→          input.userId,
   457→   457→          "inventory.stocktake.recount",
   458→   458→          stocktake.id,
   459→   459→          stocktake.reference,
   460→   460→          {
   461→   461→            lineId: line.id,
   462→   462→            productId: line.productId,
   463→   463→            newSystemQty: updated.systemQtyAtCount,
   464→   464→            newCountedQty: updated.countedQty,
   465→   465→            newVariance: updated.varianceQty,
   466→   466→          },
   467→   467→          {
   468→   468→            previousSystemQty: line.systemQtyAtCount,
   469→   469→            previousCountedQty: line.countedQty,
   470→   470→            previousVariance: line.varianceQty,
   471→   471→          },
   472→   472→        ),
   473→   473→        tx,
   474→   474→      );
   475→   475→    } else {
   476→   476→      await audit(
   477→   477→        auditInput(input.businessId, input.userId, "inventory.stocktake.count", stocktake.id, stocktake.reference, {
   478→   478→          lineId: line.id,
   479→   479→          productId: line.productId,
   480→   480→          systemQtyAtCount: updated.systemQtyAtCount,
   481→   481→          countedQty: updated.countedQty,

   588→   588→/* ------------------------------- finalize --------------------------------- */
   589→   589→
   590→   590→export async function finalizeStocktake(input: {
   591→   591→  businessId: string;
   592→   592→  userId: string;
   593→   593→  stocktakeId: string;
   594→   594→  ip?: string | null;
   595→   595→  userAgent?: string | null;
   596→   596→}) {
   597→   597→  return db.$transaction(async (tx) => {
   598→   598→    // §39 — ONE transaction: verify → stale-check → reconcile → finalize.
   599→   599→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   600→   600→    assertTransition(stocktake.status, "FINALIZED"); // §38 — REVIEW only
   601→   601→
   602→   602→    const lines = await tx.inventoryStocktakeLine.findMany({
   603→   603→      where: { stocktakeId: stocktake.id },
   604→   604→      select: {
   605→   605→        id: true,
   606→   606→        productId: true,
   607→   607→        systemQtyAtCount: true,
   608→   608→        countedQty: true,
   609→   609→        varianceQty: true,
   610→   610→        countedAt: true,
   611→   611→        productNameAtCount: true,
   612→   612→      },
   613→   613→    });
   614→   614→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   615→   615→      throw new ApiError(
   616→   616→        422,
   617→   617→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   618→   618→        "Every product must be counted before finalization.",
   619→   619→      );
   620→   620→    }
   621→   621→
   622→   622→    // §40 stale check — every product's CURRENT stockQty must still equal
   623→   623→    // systemQtyAtCount, and the product must still be active + not deleted
   624→   624→    // (§57/§58). No auto-rebase (§41). Collect ALL stale/invalid lines so the
   625→   625→    // client can recount each one (§139 — current-business products only).
   626→   626→    const stale: Array<{
   627→   627→      lineId: string;
   628→   628→      productId: string;
   629→   629→      productName: string;
   630→   630→      systemQtyAtCount: number;
   631→   631→      currentQty: number;
   632→   632→      kind: "STALE" | "INACTIVE" | "DELETED";
   633→   633→    }> = [];
   634→   634→    const products = await tx.product.findMany({
   635→   635→      where: { id: { in: lines.map((l) => l.productId) }, businessId: input.businessId },
   636→   636→      select: { id: true, stockQty: true, deletedAt: true, isActive: true },
   637→   637→    });
   638→   638→    const productById = new Map(products.map((p) => [p.id, p]));
   639→   639→    for (const line of lines) {
   640→   640→      const product = productById.get(line.productId);
   641→   641→      if (!product || product.deletedAt) {
   642→   642→        stale.push({
   643→   643→          lineId: line.id,
   644→   644→          productId: line.productId,
   645→   645→          productName: line.productNameAtCount,
   646→   646→          systemQtyAtCount: line.systemQtyAtCount,
   647→   647→          currentQty: product?.stockQty ?? line.systemQtyAtCount,
   648→   648→          kind: "DELETED",
   649→   649→        });
   650→   650→        continue;
   651→   651→      }
   652→   652→      if (!product.isActive) {
   653→   653→        stale.push({
   654→   654→          lineId: line.id,
   655→   655→          productId: line.productId,
   656→   656→          productName: line.productNameAtCount,
   657→   657→          systemQtyAtCount: line.systemQtyAtCount,
   658→   658→          currentQty: product.stockQty,
   659→   659→          kind: "INACTIVE",
   660→   660→        });
   661→   661→        continue;
   662→   662→      }
   663→   663→      if (product.stockQty !== line.systemQtyAtCount) {
   664→   664→        stale.push({
   665→   665→          lineId: line.id,
   666→   666→          productId: line.productId,
   667→   667→          productName: line.productNameAtCount,
   668→   668→          systemQtyAtCount: line.systemQtyAtCount,
   669→   669→          currentQty: product.stockQty,
   670→   670→          kind: "STALE",
   671→   671→        });
   672→   672→      }
   673→   673→    }
   674→   674→    if (stale.length > 0) {
   675→   675→      const hasInactive = stale.some((s) => s.kind === "INACTIVE");
   676→   676→      const hasDeleted = stale.some((s) => s.kind === "DELETED");
   677→   677→      throw new ApiError(
   678→   678→        409,
   679→   679→        hasInactive
   680→   680→          ? STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE
   681→   681→          : hasDeleted
   682→   682→            ? STOCKTAKE_ERROR_CODES.STALE_COUNT
   683→   683→            : STOCKTAKE_ERROR_CODES.STALE_COUNT,
   684→   684→        "Inventory changed after counting. Recount the affected product(s).",
   685→   685→        { products: stale }, // §139 — affected current-business products only
   686→   686→      );
   687→   687→    }
   688→   688→
   689→   689→    // §43/§44 — reconcile through the EXISTING InventoryAdjustment semantics.
   690→   690→    let adjustedLineCount = 0;
   691→   691→    for (const line of lines) {
   692→   692→      if (line.varianceQty === 0) continue; // §44 — zero variance: no adjustment
   693→   693→      const adjustment = await tx.inventoryAdjustment.create({
   694→   694→        data: {
   695→   695→          businessId: input.businessId,
   696→   696→          productId: line.productId,
   697→   697→          delta: line.varianceQty, // signed — §18
   698→   698→          reason: ADJUSTMENT_REASON(stocktake.reference),
   699→   699→          actorId: input.userId,
   700→   700→        },
   701→   701→        select: { id: true },
   702→   702→      });
   703→   703→      await tx.product.update({
   704→   704→        where: { id: line.productId },
   705→   705→        data: { stockQty: line.countedQty }, // §45 — stock becomes the counted value
   706→   706→      });
   707→   707→      adjustedLineCount += 1;
   708→   708→      // §72/§73 — the InventoryAdjustment row itself carries the quantity
   709→   709→      // history; no duplicate per-line audit event here. The finalize audit
   710→   710→      // below records the workflow-level change once.
   711→   711→      void adjustment;
   712→   712→    }
   713→   713→
   714→   714→    const updated = await tx.inventoryStocktake.update({
   715→   715→      where: { id: stocktake.id },
   716→   716→      data: { status: "FINALIZED", finalizedAt: new Date() },
   717→   717→      select: { id: true, status: true, finalizedAt: true },
   718→   718→    });
   719→   719→
   720→   720→    // §69/§73 — ONE finalize audit for the workflow; InventoryAdjustment rows
   721→   721→    // are the quantity-level history. Safe metadata only (§70/§140).
   722→   722→    await audit(
   723→   723→      auditInput(input.businessId, input.userId, "inventory.stocktake.finalize", stocktake.id, stocktake.reference, {
   724→   724→        status: "FINALIZED",
   725→   725→        lineCount: lines.length,
   726→   726→        adjustedLineCount,
   727→   727→      }),
   728→   728→      tx,
   729→   729→    );
   730→   730→    return { ...updated, adjustedLineCount };
   731→   731→  });
   732→   732→}

   810→   810→    }
   811→   811→  });
