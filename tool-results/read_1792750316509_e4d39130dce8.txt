     3→/**
     4→ * Platform console API layer — Prompt 28.
     5→ *
     6→ * Deliberately separate from the tenant `api()` helper: platform calls
     7→ * never carry the x-business-id header (businessId is NOT a platform
     8→ * authorization key, §11) and never get prefetched for tenant users
     9→ * (§16) — the only call every authenticated user makes is the capability
    10→ * probe (/self), and only once after sign-in.
    11→ *
    12→ * Every mutation body that requires it carries the operator's CURRENT
    13→ * PASSWORD for reauthentication (§56) — the password lives in memory for
    14→ * the duration of the request only and is never stored or logged.
    15→ */
    16→import { ApiClientError } from "@/lib/api-client";
    17→
    18→type Envelope<T> =
    19→  | { ok: true; data: T }
    20→  | { ok: false; error: { code: string; message: string; details?: unknown } };
    21→
    22→export async function platformApi<T>(
    23→  path: string,
    24→  init?: RequestInit,
    25→): Promise<T> {
    26→  let res: Response;
    27→  try {
    28→    res = await fetch(path, {
    29→      ...init,
    30→      headers: { "content-type": "application/json", ...(init?.headers as Record<string, string> | undefined) },
    31→      credentials: "same-origin",
    32→    });
    33→  } catch {
    34→    throw new ApiClientError("NETWORK_ERROR", "Network error. Please try again.", 0);
    35→  }
    36→  let body: unknown;
    37→  try {
    38→    body = await res.json();
    39→  } catch {
    40→    body = null;
    41→  }
    42→  const envelope = body as Envelope<T> | null;
    43→  if (!res.ok || !envelope || envelope.ok === false) {
    44→    const err = envelope && envelope.ok === false ? envelope.error : null;
    45→    throw new ApiClientError(
    46→      err?.code ?? "HTTP_ERROR",
    47→      err?.message ?? "Request failed.",
    48→      res.status,
    49→      err?.details,
    50→    );
    51→  }
    52→  return envelope.data;
    53→}

   866→/**
   867→ * R64 — raw CSV export for the security events feed. Unlike platformApi()
   868→ * (JSON envelope only), a SUCCESSFUL export responds with CSV TEXT plus
   869→ * x-export-truncated / x-export-rows headers, so it needs its own reader.
   870→ * Errors still arrive as the standard JSON error envelope → ApiClientError,
   871→ * so every existing error surface (usePlatformErrorText) just works.
   872→ */
   873→export async function exportSecurityEventsCsv(
   874→  params: string,
   875→): Promise<{ csv: string; truncated: boolean; rows: number }> {
   876→  let res: Response;
   877→  try {
   878→    res = await fetch(`/api/v1/platform/security/events/export${params ? `?${params}` : ""}`, {
   879→      credentials: "same-origin",
   880→    });
   881→  } catch {
   882→    throw new ApiClientError("NETWORK_ERROR", "Network error. Please try again.", 0);
   883→  }
   884→  if (!res.ok) {
   885→    let err: { code: string; message: string; details?: unknown } | null = null;
   886→    try {
   887→      const body = (await res.json()) as Envelope<unknown> | null;
   888→      err = body && body.ok === false ? body.error : null;
   889→    } catch {
   890→      err = null; // non-JSON failure body — fall through to the generic error
   891→    }
   892→    throw new ApiClientError(
   893→      err?.code ?? "HTTP_ERROR",
   894→      err?.message ?? "Request failed.",
   895→      res.status,
   896→      err?.details,
   897→    );
   898→  }
   899→  const csv = await res.text();
   900→  return {
   901→    csv,
   902→    truncated: res.headers.get("x-export-truncated") === "true",
   903→    rows: Number(res.headers.get("x-export-rows") ?? "0"),
   904→  };
   905→}
