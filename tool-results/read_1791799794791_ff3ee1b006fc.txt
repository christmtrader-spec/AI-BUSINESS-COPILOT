     3→/**
     4→ * Prompt 47 — sales-quotation service (server-only).
     5→ *
     6→ * LAWS:
     7→ *  - COMMERCIAL OFFER ONLY (§1): every quotation operation (create, edit,
     8→ *    mark-sent, accept, reject, cancel, expire, convert) mutates NOTHING
     9→ *    but the quotation itself (and, for conversion, the ONE legitimate
    10→ *    linked SalesOrder §13). No stock change, no reservation, no Sale, no
    11→ *    Payment, no Expense, no receivable, no statement, no aging, no
    12→ *    InventoryAdjustment, no notification falsely claiming an invoice or
    13→ *    payment — for ANY lifecycle action.
    14→ *  - CONVERSION LAW (§13): ACCEPTED → exactly ONE CONFIRMED SalesOrder.
    15→ *    ONE atomic transaction: quotation reload + status re-check (inside
    16→ *    the tx), already-converted guard BEFORE the status gate (every loser
    17→ *    gets the ONE stable conflict code), complete-line proof, canonical
    18→ *    totals recomputed from persisted lines, snapshot validation, order +
    19→ *    lines copied from the FROZEN quotation snapshots (never re-priced
    20→ *    from live Products), quotation → CONVERTED + convertedAt, two audit
    21→ *    events, commit together. The DB UNIQUE constraint on
    22→ *    SalesOrder.sourceQuotationId is the final race authority (§7).
    23→ *  - CONVERSION PRODUCES NO FINANCIAL/INVENTORY RECORDS: no Sale invoice,
    24→ *    no Payment, no receivable, no InventoryAdjustment, no stock
    25→ *    reservation, no Product.stockQty change (§13 invariants).
    26→ *  - STATUS LAW (§8): closed vocabulary, canonical transition matrix,
    27→ *    status RE-CHECKED INSIDE the write transaction; corrupted persisted
    28→ *    status is a controlled 500 — never DRAFT.
    29→ *  - VALIDITY LAW (§9): accept/reject require the offer to still be
    30→ *    within validity in the AUTHENTICATED business timezone (server
    31→ *    clock). expire requires the offer to ACTUALLY be past validity.
    32→ *    ACCEPTED never expires.
    33→ *  - TENANT LAW (§2.4): every query is scoped by ctx.businessId; foreign
    34→ *    ids are indistinguishable from unknown ones (404).
    35→ *  - SNAPSHOT LAW (§12): customer/product identity frozen at write time
    36→ *    from verified rows; conversion copies the FROZEN values; corrupted
    37→ *    snapshots surface, never silently substituted.
    38→ *  - NOTES LAW (§10): customerNotes/internalNotes never enter audit
    39→ *    payloads (changed-flag only), never the search index, and
    40→ *    internalNotes never the print view (§16).
    41→ *  - Every mutation writes its audit INSIDE the same transaction.
    42→ *  - READ-CAP LAW (§11): every bounded read fetches cap+1 rows to prove
    43→ *    completeness; over-cap documents fail with the stable invariant code.
    44→ *  - NO quotation delete API exists (§14/§15): history is preserved;
    45→ *    traceability of a converted offer can never be destroyed.
    46→ */

   219→/** Load a PROVEN-COMPLETE, validated line set for one quotation (§11). */
   220→async function loadCompleteLines(
   221→  tx: Tx,
   222→  salesQuotationId: string,
   223→  cap: number,
   224→): Promise<
   225→  Array<{
   226→    id: string;
   227→    productId: string;
   228→    productNameAtQuotation: string;
   229→    skuAtQuotation: string | null;
   230→    unitAtQuotation: string | null;
   231→    qty: number;
   232→    unitPriceMinor: number;
   233→  }>
   234→> {
   235→  const lines = await tx.salesQuotationLine.findMany({
   236→    where: { salesQuotationId },
   237→    orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   238→    take: cap + 1, // cap+1 proves completeness
   239→    select: {
   240→      id: true,
   241→      productId: true,
   242→      productNameAtQuotation: true,
   243→      skuAtQuotation: true,
   244→      unitAtQuotation: true,
   245→      qty: true,
   246→      unitPriceMinor: true,
   247→    },
   248→  });
   249→  const counted = await tx.salesQuotationLine.count({ where: { salesQuotationId } });
   250→  // THREE explicit invariants (§11): fetched ≤ cap, counted ≤ cap,
   251→  // counted === fetched — a mismatch below the cap NEVER passes.
   252→  assertSalesQuotationLineSetComplete(lines.length, counted, cap);
   253→  // Every persisted line must be materially valid BEFORE use (§6/§11/§12).
   254→  for (const line of lines) {
   255→    if (!isValidPersistedQuotationQty(line.qty) || !isValidQuotationPrice(line.unitPriceMinor)) {
   256→      mapPureSalesQuotationDomainError(
   257→        new SalesQuotationIntegrityError(
   258→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: invalid persisted line quantities`,
   259→        ),
   260→      );
   261→    }
   262→    if (typeof line.productNameAtQuotation !== "string" || line.productNameAtQuotation.length === 0) {
   263→      mapPureSalesQuotationDomainError(
   264→        new SalesQuotationIntegrityError(
   265→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: missing product snapshot`,
   266→        ),
   267→      );
   268→    }
   269→  }
   270→  return lines;
   271→}

   273→/** Defensive line-count validation for line writes (§11). */
   274→async function assertLineCap(tx: Tx, salesQuotationId: string, cap: number): Promise<number> {
   275→  const lineCount = await tx.salesQuotationLine.count({ where: { salesQuotationId } });
   276→  if (lineCount > cap) {
   277→    // An ALREADY-corrupted over-cap quotation is database corruption: 500
   278→    // integrity, never disguised as the ordinary 422 user failure.
   279→    throw new ApiError(
   280→      500,
   281→      SALES_QUOTATION_ERROR_CODES.LINE_CAP_INVARIANT,
   282→      "Stored document data exceeds its configured line cap. Please contact support.",
   283→    );
   284→  }
   285→  if (lineCount === cap) {
   286→    // The NORMAL write-time rejection for a legitimately full draft.
   287→    throw new ApiError(
   288→      422,
   289→      SALES_QUOTATION_ERROR_CODES.LINE_LIMIT,
   290→      `A sales quotation can contain at most ${cap} lines.`,
   291→    );
   292→  }
   293→  return lineCount;
   294→}

   298→  const lines = await loadCompleteLines(tx, salesQuotationId, cap);
   299→  try {

   306→/**
   307→ * Input-time subtotal recomputation for LINE WRITES (§11): the client's
   308→ * requested line can push the ACCUMULATED subtotal out of the safe integer
   309→ * range even when the single line total is exact (two max-magnitude lines).
   310→ * The documented contract (§11) maps the INPUT-TIME overflow to the 422
   311→ * SALES_QUOTATION_TOTAL_OVERFLOW — the write rolls back either way, this
   312→ * only fixes the surfaced status/code (persisted-data overflow keeps the
   313→ * integrity 500 through mapPureSalesQuotationDomainError elsewhere).
   314→ */
   315→async function lineWriteSubtotalMinor(tx: Tx, salesQuotationId: string, cap: number): Promise<number> {
   316→  const lines = await loadCompleteLines(tx, salesQuotationId, cap);
   317→  try {
   318→    return computeQuotationSubtotalMinor(lines).subtotalMinor;
   319→  } catch (error) {
   320→    if (error instanceof Error && error.name === "SalesQuotationTotalOverflowError") {
   321→      throw new ApiError(
   322→        422,
   323→        SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW,
   324→        "The requested quantity × price exceeds the supported total range.",
   325→      );
   326→    }
   327→    throw error;
   328→  }

   331→/** Business-local unique reference generation (§10 — never MAX+1). */
   332→async function generateUniqueQuotationReference(
   333→  tx: Tx,
   334→  businessId: string,
   335→): Promise<string> {
   336→  for (let attempt = 0; attempt < 5; attempt += 1) {
   337→    const candidate = generateSalesQuotationReference(new Date(), (max) => randomInt(max));
   338→    const clash = await tx.salesQuotation.findFirst({
   339→      where: { businessId, reference: candidate },
   340→      select: { id: true },
   341→    });
   342→    if (!clash) return candidate;
   343→  }
   344→  // Astronomically unlikely (31^6); fail honestly rather than loop.
   345→  throw new ApiError(
   346→    409,
   347→    SALES_QUOTATION_ERROR_CODES.DUPLICATE_REFERENCE,
   348→    "Could not generate a unique reference. Please try again.",
   349→  );
   350→}

   352→/** Header snapshot validation (§12) — shared by read/convert/export. */
   353→function validateHeaderSnapshots(header: {
   354→  currencyCodeAtQuotation: string;
   355→  currencySymbolAtQuotation: string;
   356→  decimalPrecisionAtQuotation: number;
   357→  customerId: string | null;
   358→  customerNameAtQuotation: string;
   359→}): void {
   360→  if (
   361→    typeof header.currencyCodeAtQuotation !== "string" ||
   362→    header.currencyCodeAtQuotation.length === 0 ||
   363→    typeof header.currencySymbolAtQuotation !== "string" ||
   364→    header.currencySymbolAtQuotation.length === 0 ||
   365→    !Number.isInteger(header.decimalPrecisionAtQuotation)
   366→  ) {
   367→    mapPureSalesQuotationDomainError(
   368→      new SalesQuotationIntegrityError(
   369→        `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: missing currency snapshot`,
   370→      ),
   371→    );
   372→  }
   373→  // A customer-linked quotation must carry its frozen identity snapshot;
   374→  // an anonymous quotation legitimately carries "".
   375→  if (header.customerId !== null && header.customerNameAtQuotation.length === 0) {
   376→    mapPureSalesQuotationDomainError(
   377→      new SalesQuotationIntegrityError(
   378→        `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: missing customer snapshot`,
   379→      ),
   380→    );
   381→  }
   382→}

   384→/** Lifecycle-timestamp consistency law (§8): a persisted timestamp must
   385→ *  never contradict its persisted status — contradictions are corruption. */
   386→function validateLifecycleConsistency(header: {
   387→  status: SalesQuotationStatus;
   388→  sentAt: Date | null;
   389→  acceptedAt: Date | null;
   390→  rejectedAt: Date | null;
   391→  cancelledAt: Date | null;
   392→  convertedAt: Date | null;
   393→}): void {
   394→  const corrupt = (what: string) =>
   395→    mapPureSalesQuotationDomainError(
   396→      new SalesQuotationIntegrityError(
   397→        `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: ${what}`,
   398→      ),
   399→    );
   400→  const needs = (t: Date | null, s: SalesQuotationStatus[]) => {
   401→    const required = s.includes(header.status);
   402→    if (required && !t) corrupt(`status ${header.status} without its transition timestamp`);
   403→    if (!required && t) corrupt(`timestamp present for status ${header.status}`);
   404→  };
   405→  needs(header.sentAt, ["SENT", "ACCEPTED", "REJECTED", "EXPIRED", "CANCELLED", "CONVERTED"]);
   406→  needs(header.acceptedAt, ["ACCEPTED", "CONVERTED"]);
   407→  needs(header.rejectedAt, ["REJECTED"]);
   408→  needs(header.cancelledAt, ["CANCELLED"]);
   409→  needs(header.convertedAt, ["CONVERTED"]);
   410→}

   414→export async function createSalesQuotation(input: {
   415→  businessId: string;
   416→  userId: string;
   417→  timezone: string; // AUTHENTICATED business timezone (server clock law §9)
   418→  customerId?: string;
   419→  reference?: string;
   420→  quotationDate?: string; // calendar date (strict, zod) — default business-local today
   421→  validUntilDate: string;
   422→  expectedDeliveryDate?: string;
   423→  discountMinor?: number;
   424→  customerNotes?: string | null;
   425→  internalNotes?: string | null;
   426→  ip?: string | null;
   427→  userAgent?: string | null;
   428→}) {
   429→  return withWriteTx(async (tx) => {
   430→    // §10 — currency snapshot comes from the BUSINESS configuration
   431→    // (server-derived); the client can never select a different currency.
   432→    const business = await tx.business.findUnique({
   433→      where: { id: input.businessId },
   434→      select: { currencyCode: true, currencySymbol: true, decimalPrecision: true },
   435→    });
   436→    if (!business) throw new ApiError(404, "NOT_FOUND", "Business not found.");
   437→
   438→    // §10 — optional customer: tenant-scoped, not soft-deleted; the name
   439→    // snapshot is frozen at creation (§12).
   440→    let customerNameAtQuotation = "";
   441→    if (input.customerId) {
   442→      const customer = await loadActiveCustomer(tx, input.businessId, input.customerId);
   443→      customerNameAtQuotation = customer.name;
   444→    }
   445→
   446→    // §9 — dates: quotationDate defaults to the SERVER's business-local
   447→    // today (never the browser clock); validUntilDate is required and
   448→    // must not precede quotationDate (inclusive-until law).
   449→    const quotationDate = input.quotationDate ?? businessLocalCalendarDate(new Date(), input.timezone);
   450→    if (!isQuotationDateRangeValid(quotationDate, input.validUntilDate)) {
   451→      throw new ApiError(
   452→        422,
   453→        SALES_QUOTATION_ERROR_CODES.INVALID_DATE_RANGE,
   454→        "The validity date cannot be earlier than the quotation date.",
   455→      );
   456→    }
   457→
   458→    // §11 — a quotation with no lines has subtotal 0: a non-zero creation
   459→    // discount would violate the discount law on day one (mirrors P45).
   460→    if ((input.discountMinor ?? 0) > 0) {
   461→      throw new ApiError(
   462→        422,
   463→        SALES_QUOTATION_ERROR_CODES.INVALID_DISCOUNT,
   464→        "Add lines first — an empty quotation cannot carry a discount.",
   465→      );
   466→    }
   467→
   468→    // Reference: user-supplied (normalized) or server-generated (§10).
   469→    let reference = input.reference;
   470→    if (reference === undefined) {
   471→      reference = await generateUniqueQuotationReference(tx, input.businessId);
   472→    }
   473→
   474→    try {
   475→      const row = await tx.salesQuotation.create({
   476→        data: {
   477→          businessId: input.businessId, // server-derived, never client (§2.4)
   478→          customerId: input.customerId ?? null,
   479→          customerNameAtQuotation, // frozen snapshot ("" when no customer)
   480→          reference,
   481→          status: "DRAFT", // commercial preparation only (§8)
   482→          currencyCodeAtQuotation: business.currencyCode,
   483→          currencySymbolAtQuotation: business.currencySymbol,
   484→          decimalPrecisionAtQuotation: business.decimalPrecision,
   485→          quotationDate: new Date(`${quotationDate}T12:00:00.000Z`),
   486→          validUntilDate: new Date(`${input.validUntilDate}T12:00:00.000Z`),
   487→          expectedDeliveryDate: input.expectedDeliveryDate
   488→            ? new Date(`${input.expectedDeliveryDate}T12:00:00.000Z`)
   489→            : null,
   490→          discountMinor: 0,
   491→          customerNotes: input.customerNotes ?? null, // document data only
   492→          internalNotes: input.internalNotes ?? null, // staff-only document data
   493→          createdByUserId: input.userId, // server-derived
   494→        },
   495→        select: { id: true, reference: true, status: true },
   496→      });
   497→
   498→      await audit(
   499→        auditInput(input.businessId, input.userId, "sales_quotation.create", row.id, row.reference, {
   500→          customerId: input.customerId ?? null,
   501→          customerNameAtQuotation,
   502→          status: row.status,
   503→          quotationDate,
   504→          validUntilDate: input.validUntilDate,
   505→        }),
   506→        tx,
   507→      );
   508→
   509→      return row;
   510→    } catch (error) {
   511→      // Database uniqueness stays authoritative for duplicates (§10).
   512→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   513→        throw new ApiError(
   514→          409,
   515→          SALES_QUOTATION_ERROR_CODES.DUPLICATE_REFERENCE,
   516→          "A sales quotation with this reference already exists.",
   517→        );
   518→      }
   519→      throw error;
   520→    }
   521→  });
   522→}

   556→  pageSize: number;
   557→}> {
   558→  const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
   559→  const now = new Date();
   560→  const where: Prisma.SalesQuotationWhereInput = {
   561→    businessId: input.businessId, // tenant scope
   562→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   563→    ...(input.customerId ? { customerId: input.customerId } : {}),
   564→    ...(input.from || input.to
   565→      ? (() => {
   566→          // from/to are interpreted in the AUTHENTICATED business timezone
   567→          // (inclusive local start of `from`, EXCLUSIVE local start of the
   568→          // day after `to`) — the ONE canonical window law (§types).
   569→          const window = salesQuotationDayWindow(input.from, input.to, input.timezone);
   570→          if (!window) {
   571→            throw new ApiError(422, "VALIDATION_ERROR", "Invalid date filter.");
   572→          }
   573→          return { quotationDate: window };
   574→        })()
   575→      : {}),
   576→    ...(input.q
   577→      ? {
   578→          OR: [
   579→            { reference: { contains: input.q } }, // quotation reference
   580→            { customerNameAtQuotation: { contains: input.q } }, // customer snapshot
   581→            {
   582→              lines: {
   583→                some: {
   584→                  OR: [
   585→                    { productNameAtQuotation: { contains: input.q } }, // product snapshot
   586→                    { skuAtQuotation: { contains: input.q } }, // SKU snapshot
   587→                  ],
   588→                },
   589→              },
   590→            },
   591→          ],
   592→        }
   593→      : {}),
   594→  };
   595→
   596→  const [total, rows] = await Promise.all([
   597→    db.salesQuotation.count({ where }),
   598→    // DB-level pagination; never findMany + slice.
   599→    db.salesQuotation.findMany({
   600→      where,
   601→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   602→      skip: (input.page - 1) * input.pageSize,
   603→      take: input.pageSize,
   604→      select: {
   605→        id: true,
   606→        reference: true,
   607→        status: true,
   608→        customerId: true,
   609→        customerNameAtQuotation: true,
   610→        quotationDate: true,
   611→        validUntilDate: true,
   612→        discountMinor: true,
   613→        createdAt: true,
   614→        convertedSalesOrder: { select: { reference: true } }, // §7 — link presence
   615→        lines: {
   616→          // Bounded per-quotation aggregate relation, not one query per row.
   617→          // take cap+1 PROVES the slice is the whole line set (§11).
   618→          select: { id: true, qty: true, unitPriceMinor: true },
   619→          take: cap + 1,
   620→        },
   621→        _count: { select: { lines: true } },
   622→      },
   623→    }),
   624→  ]);
   625→
   626→  const items = rows.map((row) => {
   627→    try {
   628→      assertSalesQuotationLineSetComplete(row.lines.length, row._count.lines, cap);
   629→      let totalQty = 0;
   630→      let subtotalMinor = 0;
   631→      for (const line of row.lines) {
   632→        if (!isValidPersistedQuotationQty(line.qty) || !isValidQuotationPrice(line.unitPriceMinor)) {
   633→          mapPureSalesQuotationDomainError(
   634→            new SalesQuotationIntegrityError(
   635→              `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: invalid persisted line`,
   636→            ),
   637→          );
   638→        }
   639→        totalQty += line.qty;
   640→        subtotalMinor += computeQuotationLineTotalMinor(line.qty, line.unitPriceMinor);
   641→      }
   642→      if (!Number.isSafeInteger(subtotalMinor)) {
   643→        mapPureSalesQuotationDomainError(
   644→          new SalesQuotationIntegrityError(
   645→            `${SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW}: list subtotal overflow`,
   646→          ),
   647→        );
   648→      }
   649→      const totalMinor = computeQuotationTotalMinor(subtotalMinor, row.discountMinor);
   650→      const persistedStatus = parseDbStatus(row.status);
   651→      // §9 — persisted vs effective are EXPLICIT and distinct.
   652→      const effectiveStatus = effectiveQuotationStatus(
   653→        persistedStatus,
   654→        row.validUntilDate,
   655→        now,
   656→        input.timezone,
   657→      );
   658→      return {
   659→        id: row.id,
   660→        reference: row.reference,
   661→        persistedStatus,
   662→        effectiveStatus,
   663→        customerId: row.customerId,
   664→        customerNameAtQuotation: row.customerNameAtQuotation,
   665→        quotationDate: row.quotationDate.toISOString(),
   666→        validUntilDate: row.validUntilDate.toISOString(),
   667→        lineCount: row._count.lines,
   668→        totalQty,
   669→        subtotalMinor,
   670→        discountMinor: row.discountMinor,
   671→        totalMinor,
   672→        convertedOrderReference: row.convertedSalesOrder?.reference ?? null, // ONLY when the relation exists (§7)
   673→        createdAt: row.createdAt.toISOString(),
   674→      };
   675→    } catch (error) {
   676→      return mapPureSalesQuotationDomainError(error);
   677→    }
   678→  });
   679→  return { items, total, page: input.page, pageSize: input.pageSize };
   680→}

   843→export async function updateSalesQuotation(input: {
   844→  businessId: string;
   845→  userId: string;
   846→  id: string;
   847→  customerId?: string | null;
   848→  reference?: string;
   849→  quotationDate?: string;
   850→  validUntilDate?: string;
   851→  expectedDeliveryDate?: string | null;
   852→  discountMinor?: number;
   853→  customerNotes?: string | null;
   854→  internalNotes?: string | null;
   855→  ip?: string | null;
   856→  userAgent?: string | null;
   857→}) {
   858→  return withWriteTx(async (tx) => {
   859→    const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
   860→    const header = await loadQuotationHeader(tx, input.businessId, input.id);
   861→    if (header.status !== "DRAFT") {
   862→      // §8 — SENT and later are immutable (no silent correction).
   863→      throw new ApiError(
   864→        409,
   865→        SALES_QUOTATION_ERROR_CODES.IMMUTABLE,
   866→        "Only DRAFT sales quotations can be changed.",
   867→      );
   868→    }
   869→
   870→    const data: Prisma.SalesQuotationUpdateInput = {};
   871→    let quotationDate = midnoonDateToCalendarDate(header.quotationDate);
   872→    let validUntilDate = midnoonDateToCalendarDate(header.validUntilDate);
   873→
   874→    if (input.customerId !== undefined) {
   875→      if (input.customerId === null) {
   876→        data.customer = { disconnect: true };
   877→        data.customerNameAtQuotation = "";
   878→      } else {
   879→        const customer = await loadActiveCustomer(tx, input.businessId, input.customerId);
   880→        data.customer = { connect: { id: customer.id } };
   881→        data.customerNameAtQuotation = customer.name; // snapshot refrozen (§12)
   882→      }
   883→    }
   884→    if (input.reference !== undefined) data.reference = input.reference;
   885→    if (input.quotationDate !== undefined) {
   886→      data.quotationDate = new Date(`${input.quotationDate}T12:00:00.000Z`);
   887→      quotationDate = input.quotationDate;
   888→    }
   889→    if (input.validUntilDate !== undefined) {
   890→      data.validUntilDate = new Date(`${input.validUntilDate}T12:00:00.000Z`);
   891→      validUntilDate = input.validUntilDate;
   892→    }
   893→    // §9 — the validity ORDERING holds against the PATCHED pair (a partial
   894→    // patch can never dodge the law).
   895→    if (!isQuotationDateRangeValid(quotationDate, validUntilDate)) {
   896→      throw new ApiError(
   897→        422,
   898→        SALES_QUOTATION_ERROR_CODES.INVALID_DATE_RANGE,
   899→        "The validity date cannot be earlier than the quotation date.",
   900→      );
   901→    }
   902→    if (input.expectedDeliveryDate !== undefined) {
   903→      data.expectedDeliveryDate =
   904→        input.expectedDeliveryDate === null
   905→          ? null
   906→          : new Date(`${input.expectedDeliveryDate}T12:00:00.000Z`);
   907→    }
   908→    if (input.customerNotes !== undefined) data.customerNotes = input.customerNotes;
   909→    if (input.internalNotes !== undefined) data.internalNotes = input.internalNotes;
   910→
   911→    if (input.discountMinor !== undefined) {
   912→      // §11 — discount law against the CURRENT canonical subtotal.
   913→      const subtotalMinor = await currentSubtotalMinor(tx, header.id, cap);
   914→      try {
   915→        validateQuotationDiscount(input.discountMinor, subtotalMinor);
   916→      } catch {
   917→        throw new ApiError(
   918→          422,
   919→          SALES_QUOTATION_ERROR_CODES.INVALID_DISCOUNT,
   920→          "Discount cannot exceed the quotation subtotal.",
   921→        );
   922→      }
   923→      data.discountMinor = input.discountMinor;
   924→    }
   925→
   926→    try {
   927→      const updated = await tx.salesQuotation.update({
   928→        where: { id: header.id },
   929→        data,
   930→        select: { id: true, reference: true, status: true },
   931→      });
   932→
   933→      await audit(
   934→        auditInput(
   935→          input.businessId,
   936→          input.userId,
   937→          "sales_quotation.update",
   938→          header.id,
   939→          updated.reference,
   940→          {
   941→            changed: {
   942→              ...(input.customerId !== undefined ? { customerId: { changed: true } } : {}),
   943→              ...(input.reference !== undefined
   944→                ? { reference: { from: header.reference, to: updated.reference } }
   945→                : {}),
   946→              ...(input.quotationDate !== undefined ? { quotationDate: { changed: true } } : {}),
   947→              ...(input.validUntilDate !== undefined ? { validUntilDate: { changed: true } } : {}),
   948→              ...(input.expectedDeliveryDate !== undefined
   949→                ? { expectedDeliveryDate: { changed: true } }
   950→                : {}),
   951→              ...(input.discountMinor !== undefined
   952→                ? { discountMinor: { from: header.discountMinor, to: input.discountMinor } }
   953→                : {}),
   954→              // Notes: CHANGED-FLAG ONLY — the contents never enter the
   955→              // audit trail (§10/§22).
   956→              ...(input.customerNotes !== undefined ? { customerNotes: { changed: true } } : {}),
   957→              ...(input.internalNotes !== undefined ? { internalNotes: { changed: true } } : {}),
   958→            },
   959→          },
   960→          { reference: header.reference },
   961→        ),
   962→        tx,
   963→      );
   964→
   965→      return updated;
   966→    } catch (error) {
   967→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   968→        throw new ApiError(
   969→          409,
   970→          SALES_QUOTATION_ERROR_CODES.DUPLICATE_REFERENCE,
   971→          "A sales quotation with this reference already exists.",
   972→        );
   973→      }
   974→      throw error;
   975→    }
   976→  });
   977→}

   981→export async function addSalesQuotationLine(input: {
   982→  businessId: string;
   983→  userId: string;
   984→  id: string;
   985→  productId: string;
   986→  qty: number;
   987→  unitPriceMinor?: number;
   988→  ip?: string | null;
   989→  userAgent?: string | null;
   990→}) {
   991→  return withWriteTx(async (tx) => {
   992→    const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
   993→    const header = await loadQuotationHeader(tx, input.businessId, input.id);
   994→    if (header.status !== "DRAFT") {
   995→      throw new ApiError(
   996→        409,
   997→        SALES_QUOTATION_ERROR_CODES.IMMUTABLE,
   998→        "Only DRAFT sales quotations can be changed.",
   999→      );
  1000→    }
  1001→
  1002→    // §6 — eligible products only: same tenant, exists, not soft-deleted,
  1003→    // active. A foreign product is indistinguishable from unknown (404).
  1004→    const product = await tx.product.findFirst({
  1005→      where: { id: input.productId, businessId: input.businessId, deletedAt: null },
  1006→      select: { id: true, name: true, sku: true, unit: true, salePriceMinor: true, isActive: true },
  1007→    });
  1008→    if (!product || !product.isActive) {
  1009→      throw new ApiError(
  1010→        404,
  1011→        SALES_QUOTATION_ERROR_CODES.PRODUCT_NOT_FOUND,
  1012→        "Product not found.",
  1013→      );
  1014→    }
  1015→
  1016→    await assertLineCap(tx, header.id, cap); // defensive count
  1017→
  1018→    // §6/§12 — the offered price: client-supplied or the CURRENT product
  1019→    // sale price as the default; persisted ONCE and never re-derived later
  1020→    // (a later price change never silently reprices the offer §12).
  1021→    const unitPriceMinor = input.unitPriceMinor ?? product.salePriceMinor;
  1022→    if (!isValidQuotationPrice(unitPriceMinor)) {
  1023→      throw new ApiError(
  1024→        422,
  1025→        SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW,
  1026→        "Unit price is outside the allowed range.",
  1027→      );
  1028→    }
  1029→    // §11 — the line total must be exact BEFORE anything persists.
  1030→    try {
  1031→      computeQuotationLineTotalMinor(input.qty, unitPriceMinor);
  1032→    } catch (error) {
  1033→      if (error instanceof Error && error.name === "SalesQuotationTotalOverflowError") {
  1034→        throw new ApiError(
  1035→          422,
  1036→          SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW,
  1037→          "The requested quantity × price exceeds the supported total range.",
  1038→        );
  1039→      }
  1040→      throw error;
  1041→    }
  1042→
  1043→    try {
  1044→      const line = await tx.salesQuotationLine.create({
  1045→        data: {
  1046→          salesQuotationId: header.id,
  1047→          productId: product.id,
  1048→          qty: input.qty,
  1049→          unitPriceMinor,
  1050→          // §6/§12 — snapshots copied from the tenant-scoped Product at
  1051→          // add time, then frozen; the client never authors them.
  1052→          productNameAtQuotation: product.name,
  1053→          skuAtQuotation: product.sku,
  1054→          unitAtQuotation: product.unit,
  1055→        },
  1056→        select: { id: true },
  1057→      });
  1058→
  1059→      // §11 — the stored discount must still satisfy the law against the
  1060→      // NEW subtotal (a line change can shrink the subtotal below it).
  1061→      const subtotalMinor = await lineWriteSubtotalMinor(tx, header.id, cap);
  1062→      if (header.discountMinor > subtotalMinor) {
  1063→        throw new ApiError(
  1064→          422,
  1065→          SALES_QUOTATION_ERROR_CODES.INVALID_DISCOUNT,
  1066→          "Discount cannot exceed the quotation subtotal.",
  1067→        );
  1068→      }
  1069→
  1070→      await audit(
  1071→        auditInput(
  1072→          input.businessId,
  1073→          input.userId,
  1074→          "sales_quotation.line_add",
  1075→          header.id,
  1076→          header.reference,
  1077→          {
  1078→            lineId: line.id,
  1079→            productId: product.id,
  1080→            productNameAtQuotation: product.name,
  1081→            qty: input.qty,
  1082→            unitPriceMinor,
  1083→          },
  1084→        ),
  1085→        tx,
  1086→      );
  1087→
  1088→      return { id: line.id };
  1089→    } catch (error) {
  1090→      // DB uniqueness is authoritative for duplicate lines (§6).
  1091→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
  1092→        throw new ApiError(
  1093→          409,
  1094→          SALES_QUOTATION_ERROR_CODES.DUPLICATE_PRODUCT,
  1095→          "This product is already on the quotation.",
  1096→        );
  1097→      }
  1098→      throw error;
  1099→    }

  1255→/**
  1256→ * The shared in-transaction lifecycle gate (§8): reload + canonical matrix
  1257→ * + optional validity precondition. Every transition endpoint funnels
  1258→ * through here so the law lives in exactly ONE place.
  1259→ */
  1260→async function gateTransition(
  1261→  tx: Tx,
  1262→  input: { businessId: string; id: string; timezone: string },
  1263→  to: SalesQuotationStatus,
  1264→): Promise<{
  1265→  header: Awaited<ReturnType<typeof loadQuotationHeader>>;
  1266→  lines: Awaited<ReturnType<typeof loadCompleteLines>>;
  1267→}> {
  1268→  const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
  1269→  // Reload INSIDE the transaction; never trust an earlier GET (§8).
  1270→  const header = await loadQuotationHeader(tx, input.businessId, input.id);
  1271→  if (!canTransitionSalesQuotation(header.status, to)) {
  1272→    throw new ApiError(
  1273→      409,
  1274→      SALES_QUOTATION_ERROR_CODES.INVALID_TRANSITION,
  1275→      `Cannot move a sales quotation from ${header.status} to ${to}.`,
  1276→    );
  1277→  }
  1278→  const lines = await loadCompleteLines(tx, header.id, cap);
  1279→  return { header, lines };
  1280→}

  1283→ * §10/§D — mark a DRAFT quotation SENT. Requires at least one line (an

  1300→    if (lines.length === 0) {

  1303→        SALES_QUOTATION_ERROR_CODES.EMPTY,
  1304→        "Add at least one line before marking this quotation sent.",
  1305→      );

  1546→/**
  1547→ * §13 — THE conversion: an ACCEPTED quotation becomes exactly ONE
  1548→ * CONFIRMED SalesOrder. ONE atomic transaction:
  1549→ *   1. reload the quotation tenant-scoped;
  1550→ *   2. already-converted guard FIRST (the ONE stable conflict for every
  1551→ *      loser of any race — the P45 D-5 lesson), backed by the DB UNIQUE
  1552→ *      constraint on SalesOrder.sourceQuotationId (§7);
  1553→ *   3. persisted status law (ACCEPTED only — invalid → controlled 500,
  1554→ *      wrong-but-valid status → 409);
  1555→ *   4. complete, materially valid line set (§11);
  1556→ *   5. canonical totals recomputed from authoritative persisted lines;
  1557→ *   6. every snapshot validated (§12);
  1558→ *   7. ONE SalesOrder created CONFIRMED — customer snapshot, currency
  1559→ *      snapshot, expected date, discount copied EXACTLY; orderDate =
  1560→ *      current business-local date (established midnoon rule);
  1561→ *      internalNotes → SalesOrder.notes (staff-visible order notes —
  1562→ *      never the customer print view §16);
  1563→ *   8. ONE SalesOrderLine per quotation line — productId + frozen
  1564→ *      snapshots + exact qty/unitPriceMinor (NEVER re-priced from live
  1565→ *      Products §12/§13);
  1566→ *   9. order totals derived via the EXISTING SalesOrder domain functions;
  1567→ *  10. quotation → CONVERTED + convertedAt;
  1568→ *  11. audit ×2 (sales_order.create + sales_quotation.convert) in-tx;
  1569→ *  12. commit both documents together.
  1570→ *
  1571→ * NO Sale invoice, NO Payment, NO receivable, NO InventoryAdjustment, NO
  1572→ * stock reservation, NO Product.stockQty change (§13 invariants). NO
  1573→ * notification claims an invoice or payment occurred.
  1574→ */
  1575→export async function convertSalesQuotationToOrder(input: {
  1576→  businessId: string;
  1577→  userId: string;
  1578→  id: string;
  1579→  timezone: string;
  1580→  ip?: string | null;
  1581→  userAgent?: string | null;
  1582→}) {
  1583→  return withWriteTx(async (tx) => {
  1584→    const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
  1585→    // 1. tenant-scoped reload INSIDE the transaction (§13 step 1).
  1586→    const header = await loadQuotationHeader(tx, input.businessId, input.id);
  1587→
  1588→    // 2. already-converted guard BEFORE the status gate (§13/D-5 lesson):
  1589→    //    every loser — including a retry after the winner committed —
  1590→    //    receives the ONE documented stable conflict code.
  1591→    const existingOrder = await tx.salesOrder.findFirst({
  1592→      where: { sourceQuotationId: header.id },
  1593→      select: { id: true, reference: true },
  1594→    });
  1595→    if (existingOrder) {
  1596→      throw new ApiError(
  1597→        409,
  1598→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1599→        "This quotation has already been converted to a sales order.",
  1600→      );
  1601→    }
  1602→
  1603→    // 3. persisted status law: an ACCEPTED quotation converts; a
  1604→    //    valid-but-wrong status is an ordinary conflict; a corrupted
  1605→    //    persisted status is the controlled integrity 500 (parseDbStatus).
  1606→    if (header.status !== "ACCEPTED") {
  1607→      if (
  1608→        header.status === "REJECTED" ||
  1609→        header.status === "EXPIRED" ||
  1610→        header.status === "CANCELLED"
  1611→      ) {
  1612→        throw new ApiError(
  1613→          409,
  1614→          SALES_QUOTATION_ERROR_CODES.NOT_ACCEPTED,
  1615→          "Only ACCEPTED sales quotations can be converted.",
  1616→        );
  1617→      }
  1618→      throw new ApiError(
  1619→        409,
  1620→        SALES_QUOTATION_ERROR_CODES.INVALID_STATUS,
  1621→        "Only ACCEPTED sales quotations can be converted.",
  1622→      );
  1623→    }
  1624→    // §9 — ACCEPTED never expires; conversion remains legal after the
  1625→    // original validity date passes because acceptance was already
  1626→    // recorded (documented §9 law).
  1627→
  1628→    // 4–5. complete line set + canonical totals from persisted lines.
  1629→    const lines = await loadCompleteLines(tx, header.id, cap);
  1630→    if (lines.length === 0) {
  1631→      // An ACCEPTED quotation with zero lines is corruption (it could
  1632→      // never have passed mark-sent) — integrity, not a user error.
  1633→      mapPureSalesQuotationDomainError(
  1634→        new SalesQuotationIntegrityError(
  1635→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: accepted quotation without lines`,
  1636→        ),
  1637→      );
  1638→    }
  1639→    // Persisted data: an accumulation overflow is the integrity path.
  1640→    let quotationTotals;
  1641→    try {
  1642→      quotationTotals = computeQuotationSubtotalMinor(lines);
  1643→    } catch (error) {
  1644→      mapPureSalesQuotationDomainError(error);
  1645→    }
  1646→    try {
  1647→      validateQuotationDiscount(header.discountMinor, quotationTotals.subtotalMinor);
  1648→    } catch {
  1649→      mapPureSalesQuotationDomainError(
  1650→        new SalesQuotationIntegrityError(
  1651→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: discount exceeds subtotal`,
  1652→        ),
  1653→      );
  1654→    }
  1655→    const quotationTotalMinor = computeQuotationTotalMinor(
  1656→      quotationTotals.subtotalMinor,
  1657→      header.discountMinor,
  1658→    );
  1659→
  1660→    // 6. every snapshot validated (§12/§13 step 6).
  1661→    validateHeaderSnapshots(header);
  1662→
  1663→    // Order reference: the established SO generator, distinct sequence
  1664→    // domain (§10 — quotation and order references never share a sequence).
  1665→    let orderReference: string | undefined;
  1666→    for (let attempt = 0; attempt < 5; attempt += 1) {
  1667→      const candidate = generateSalesOrderReference(new Date(), (max) => randomInt(max));
  1668→      const clash = await tx.salesOrder.findFirst({
  1669→        where: { businessId: input.businessId, reference: candidate },
  1670→        select: { id: true },
  1671→      });
  1672→      if (!clash) {
  1673→        orderReference = candidate;
  1674→        break;
  1675→      }
  1676→    }
  1677→    if (orderReference === undefined) {
  1678→      throw new ApiError(
  1679→        409,
  1680→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1681→        "Could not generate a unique order reference. Please try again.",
  1682→      );
  1683→    }
  1684→
  1685→    // §13 step 11 — orderDate is the CURRENT business-local date
  1686→    // normalized with the established UTC-midnoon rule (server clock).
  1687→    const orderDate = new Date(
  1688→      `${businessLocalCalendarDate(new Date(), input.timezone)}T12:00:00.000Z`,
  1689→    );
  1690→    const confirmedAt = new Date();
  1691→
  1692→    // 7. ONE SalesOrder — CONFIRMED, source link established (§13 steps
  1693→    //    8–12/19/20). Customer/currency snapshots copied EXACTLY.
  1694→    let order;
  1695→    try {
  1696→      order = await tx.salesOrder.create({
  1697→        data: {
  1698→          businessId: input.businessId, // server-derived
  1699→          customerId: header.customerId,
  1700→          customerNameAtOrder: header.customerNameAtQuotation, // frozen §12
  1701→          reference: orderReference,
  1702→          status: "CONFIRMED", // acceptance = commercial commitment (§13)
  1703→          currencyCodeAtOrder: header.currencyCodeAtQuotation,
  1704→          currencySymbolAtOrder: header.currencySymbolAtQuotation,
  1705→          decimalPrecisionAtOrder: header.decimalPrecisionAtQuotation,
  1706→          orderDate,
  1707→          expectedDate: header.expectedDeliveryDate,
  1708→          discountMinor: header.discountMinor, // copied EXACTLY (§13 step 13)
  1709→          notes: header.internalNotes, // staff notes continuity (§13 step 14)
  1710→          createdByUserId: input.userId,
  1711→          confirmedAt, // §13 step 20 — server-side transaction time
  1712→          sourceQuotationId: header.id, // §13 step 23 — the ONE authority
  1713→        },
  1714→        select: { id: true, reference: true, status: true, confirmedAt: true },
  1715→      });
  1716→    } catch (error) {
  1717→      // The DB unique constraint is the FINAL race authority (§7/§13):
  1718→      // a concurrent winner (or an impossible prior conversion) surfaces
  1719→      // as the ONE stable conflict — never a second order.
  1720→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
  1721→        throw new ApiError(
  1722→          409,
  1723→          SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
  1724→          "This quotation has already been converted to a sales order.",
  1725→        );
  1726→      }
  1727→      throw error;
  1728→    }
  1729→
  1730→    // 8. ONE SalesOrderLine per quotation line — the FROZEN snapshots and
  1731→    //    exact agreed prices (§13 steps 15–17; never re-read/re-priced).
  1732→    //    The order subtotal is re-derived via the EXISTING order domain
  1733→    //    functions so both documents provably agree (§13 step 18).
  1734→    let orderSubtotalMinor = 0;
  1735→    try {
  1736→      for (const line of lines) {
  1737→        await tx.salesOrderLine.create({
  1738→          data: {
  1739→            salesOrderId: order.id,
  1740→            productId: line.productId,
  1741→            orderedQty: line.qty, // exact quantity (§13 step 17)
  1742→            unitPriceMinor: line.unitPriceMinor, // exact agreed price
  1743→            productNameAtOrder: line.productNameAtQuotation,
  1744→            skuAtOrder: line.skuAtQuotation,
  1745→            unitAtOrder: line.unitAtQuotation,
  1746→          },
  1747→          select: { id: true },
  1748→        });
  1749→        orderSubtotalMinor += computeQuotationLineTotalMinor(line.qty, line.unitPriceMinor);
  1750→      }
  1751→      if (!Number.isSafeInteger(orderSubtotalMinor)) {
  1752→        throw new SalesQuotationIntegrityError(
  1753→          `${SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW}: order subtotal overflow during conversion`,
  1754→        );
  1755→      }
  1756→      // §13 step 18 — derive the order totals via the EXISTING domain
  1757→      // functions; quotation and order totals MUST agree exactly.
  1758→      const orderTotals = computeOrderSubtotalMinor(
  1759→        lines.map((l) => ({ orderedQty: l.qty, unitPriceMinor: l.unitPriceMinor })),
  1760→      );
  1761→      if (orderTotals.subtotalMinor !== quotationTotals.subtotalMinor) {
  1762→        mapPureSalesQuotationDomainError(
  1763→          new SalesQuotationIntegrityError(
  1764→            `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: quotation/order subtotal disagreement`,
  1765→          ),
  1766→        );
  1767→      }
  1768→    } catch (error) {
  1769→      // §13/H — a failure here aborts the WHOLE transaction: no partial
  1770→      // order, no partially copied lines, no CONVERTED quotation, no
  1771→      // stray audit (withWriteTx rolls back everything). Domain integrity
  1772→      // failures surface through the ONE stable mapping.
  1773→      mapPureSalesQuotationDomainError(error);
  1774→    }
  1775→
  1776→    // 10. quotation → CONVERTED + convertedAt (§13 steps 21–22).
  1777→    const convertedAt = new Date();
  1778→    await tx.salesQuotation.update({
  1779→      where: { id: header.id },
  1780→      data: { status: "CONVERTED", convertedAt },
  1781→      select: { id: true },
  1782→    });
  1783→
  1784→    // 11. audit ×2, SAME transaction (§13 step 24). Minimal safe payloads;
  1785→    //     never the notes contents.
  1786→    await audit(
  1787→      {
  1788→        businessId: input.businessId,
  1789→        actorId: input.userId,
  1790→        action: "sales_order.create",
  1791→        entityType: "SalesOrder",
  1792→        entityId: order.id,
  1793→        after: {
  1794→          reference: order.reference,
  1795→          sourceQuotationId: header.id,
  1796→          sourceQuotationReference: header.reference,
  1797→          status: order.status,
  1798→          lineCount: lines.length,
  1799→          subtotalMinor: quotationTotals.subtotalMinor,
  1800→          discountMinor: header.discountMinor,
  1801→          totalMinor: quotationTotalMinor,
  1802→        },
  1803→      },
  1804→      tx,
  1805→    );
  1806→    await audit(
  1807→      auditInput(input.businessId, input.userId, "sales_quotation.convert", header.id, header.reference, {
  1808→        salesOrderId: order.id,
  1809→        salesOrderReference: order.reference,
  1810→        lineCount: lines.length,
  1811→        subtotalMinor: quotationTotals.subtotalMinor,
  1812→        discountMinor: header.discountMinor,
  1813→        totalMinor: quotationTotalMinor,
  1814→        previousStatus: header.status,
  1815→        newStatus: "CONVERTED",
  1816→        convertedAt: convertedAt.toISOString(),
  1817→      }),
  1818→      tx,
  1819→    );
  1820→
  1821→    // 12. both documents commit together (§13 step 25).
  1822→    return {
  1823→      quotationId: header.id,
  1824→      quotationReference: header.reference,
  1825→      salesOrder: order,
  1826→    };
  1827→  });
  1828→}

  1833→ * §19 — bounded, read-only quotation history for the Customer 360 (≤10).

  1837→export async function recentQuotationsForCustomer(
  1838→  businessId: string,
  1839→  customerId: string,
  1840→  take = 10,
  1841→): Promise<CustomerRecentSalesQuotationDto[]> {
  1842→  const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
  1843→  const now = new Date();
  1844→  const rows = await db.salesQuotation.findMany({
  1845→    where: { businessId, customerId }, // tenant-scoped (§19)
  1846→    orderBy: [{ createdAt: "desc" }, { id: "desc" }],
  1847→    take,
  1848→    select: {
  1849→      id: true,
  1850→      reference: true,
  1851→      status: true,
  1852→      quotationDate: true,

  1854→      currencyCodeAtQuotation: true,
  1855→      currencySymbolAtQuotation: true,

  1858→      createdAt: true,

  1863→  });

  1868→    const subtotalMinor = computeQuotationSubtotalMinor(row.lines).subtotalMinor;

  1870→    return customerRecentSalesQuotationDto({
  1871→      id: row.id,
  1872→      reference: row.reference,
  1873→      persistedStatus,
  1874→      effectiveStatus,
  1875→      quotationDate: row.quotationDate,
  1876→      currencyCodeAtQuotation: row.currencyCodeAtQuotation,
  1877→      currencySymbolAtQuotation: row.currencySymbolAtQuotation,
  1878→      decimalPrecisionAtQuotation: row.decimalPrecisionAtQuotation,
  1879→      totalMinor,
  1880→      totalQty: row.lines.reduce((acc, l) => acc + l.qty, 0),
  1881→      convertedOrderReference: row.convertedSalesOrder?.reference ?? null,
  1882→    });
  1883→  });
