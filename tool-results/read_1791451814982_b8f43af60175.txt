     3→/**
     4→ * Prompt 42 — goods-receipt service (server-only, §169).
     5→ *
     6→ * LAWS:
     7→ *  - PHYSICAL INVENTORY EVENT ONLY (§3/§4/§5): posting mutates
     8→ *    Product.stockQty and creates one positive InventoryAdjustment per
     9→ *    non-zero received line. It NEVER creates Expense, Payment, Sale,
    10→ *    supplier-statement movement, payable/aging row or any accounting
    11→ *    entry (§5/§160-§164). Draft/cancel mutate NOTHING but the receipt.
    12→ *  - OVER-RECEIPT LAW (§15): SUM(posted received) + new quantity ≤
    13→ *    orderedQty, enforced by the ONE pure helper at line-add (advisory,
    14→ *    §66) AND authoritatively inside the post transaction (§39 step 8).
    15→ *    No tolerance, no bypass, no silent overage (§15).
    16→ *  - Every query is tenant-scoped by ctx.businessId (§89) — never a
    17→ *    client businessId. Foreign PO/PO-line/receipt → 404, no existence
    18→ *    leak (§90/§91).
    19→ *  - Status transitions are server-authoritative through dedicated
    20→ *    operations (§7); the current status is RE-CHECKED INSIDE the write
    21→ *    transaction (§73) so concurrent posts/cancels cannot double-apply.
    22→ *  - Snapshots (supplier name, PO reference, product name/sku/unit) are
    23→ *    frozen at write time from ALREADY-FROZEN PO data (§24/§25/§28/§29)
    24→ *    and NEVER overwritten from live data.
    25→ *  - Quantities are exact integers (§18/§48); stock math is Int32-bounded
    26→ *    (§49); corrupt persisted state is surfaced, never repaired (§46/§47).
    27→ *  - Audit through the single sanctioned helper INSIDE the same
    28→ *    transaction as every mutation (§154) — safe metadata only, never
    29→ *    notes content (§155).
    30→ */

    76→/** §98-analog day-window helpers on createdAt (day-precision filters). */
    77→function dayWindow(from?: string, to?: string): { gte?: Date; lte?: Date } {
    78→  const window: { gte?: Date; lte?: Date } = {};
    79→  if (from) window.gte = new Date(`${from}T00:00:00.000Z`);
    80→  if (to) window.lte = new Date(`${to}T23:59:59.999Z`);
    81→  return window;
    82→}

    86→/** §96 — corrupted persisted status → controlled 500 (never Draft). */
    87→function parseDbStatus(value: string): GoodsReceiptStatus {
    88→  try {
    89→    return parseGoodsReceiptStatus(value);
    90→  } catch (error) {
    91→    if (error instanceof Error && error.name === "GoodsReceiptStatusIntegrityError") {
    92→      throw new ApiError(
    93→        500,
    94→        GOODS_RECEIPT_INVALID_PERSISTED_STATUS,
    95→        "Goods receipt data is in an invalid state. Please contact support.",
    96→      );
    97→    }
    98→    throw error;
    99→  }

   102→/**
   103→ * §100/§283 — map pure canonical-math integrity failures to their
   104→ * controlled HTTP result: corrupted received history is a 500, never a
   105→ * silent clamp to 0. Exported for the PO detail route so corrupted rows
   106→ * surface the SAME stable code there (§283 — receiving summary included).
   107→ */
   108→export function mapPureGoodsReceiptDomainError(error: unknown): never {
   109→  if (error instanceof Error && error.name === "GoodsReceiptStatusIntegrityError") {
   110→    throw new ApiError(
   111→      500,
   112→      GOODS_RECEIPT_INVALID_PERSISTED_STATUS,
   113→      "Goods receipt data is in an invalid state. Please contact support.",
   114→    );
   115→  }
   116→  if (error instanceof GoodsReceiptIntegrityError) {
   117→    throw new ApiError(
   118→      500,
   119→      GOODS_RECEIPT_ERROR_CODES.CORRUPT_RECEIVED,
   120→      "Receiving data is inconsistent. Please contact support.",
   121→    );
   122→  }
   123→  if (error instanceof Error && error.message === "GOODS_RECEIPT_CORRUPT_RECEIVED_HISTORY") {
   124→    throw new ApiError(
   125→      500,
   126→      GOODS_RECEIPT_ERROR_CODES.CORRUPT_RECEIVED,
   127→      "Receiving data is inconsistent. Please contact support.",
   128→    );
   129→  }
   130→  throw error;
   131→}

   154→async function loadReceiptHeader(
   155→  tx: Tx,
   156→  businessId: string,

   159→  id: string;
   160→  reference: string;

   162→  purchaseOrderId: string;
   163→  purchaseOrderReferenceAtReceipt: string;

   165→  notes: string | null;
   166→  createdByUserId: string;
   167→  postedAt: Date | null;
   168→  cancelledAt: Date | null;
   169→  createdAt: Date;
   170→}> {
   171→  const row = await tx.goodsReceipt.findFirst({
   172→    where: { id, businessId }, // §89 — tenant scope from auth context only
   173→    select: {
   174→      id: true,
   175→      reference: true,
   176→      status: true,
   177→      purchaseOrderId: true,
   178→      purchaseOrderReferenceAtReceipt: true,
   179→      supplierNameAtReceipt: true,
   180→      notes: true,
   181→      createdByUserId: true,
   182→      postedAt: true,

   184→      createdAt: true,
   185→    },
   186→  });
   187→  if (!row) throw new ApiError(404, "NOT_FOUND", "Goods receipt not found.");
   188→  // §96 — a corrupted persisted status NEVER silently becomes DRAFT.
   189→  return { ...row, status: parseDbStatus(row.status) };

   229→async function sumPostedQtyForPoLine(tx: Tx, purchaseOrderLineId: string): Promise<number> {
   230→  const rows = await tx.goodsReceiptLine.findMany({
   231→    where: {
   232→      purchaseOrderLineId,
   233→      goodsReceipt: { status: "POSTED" }, // §99 — POSTED only
   234→    },
   235→    select: { receivedQty: true },
   236→  });
   237→  try {
   238→    return computeReceivedQty(rows.map((r) => r.receivedQty));
   239→  } catch (error) {
   240→    return mapPureGoodsReceiptDomainError(error);
   241→  }
   242→}

   274→export async function createGoodsReceipt(input: {
   275→  businessId: string;
   276→  userId: string;
   277→  purchaseOrderId: string;
   278→  reference?: string;
   279→  notes?: string | null;
   280→  ip?: string | null;
   281→  userAgent?: string | null;
   282→}) {
   283→  return withWriteTx(async (tx) => {
   284→    // §11/§23 — the PO must be tenant-scoped and CONFIRMED.
   285→    const po = await tx.purchaseOrder.findFirst({
   286→      where: { id: input.purchaseOrderId, businessId: input.businessId }, // §90 — foreign PO → 404
   287→      select: {
   288→        id: true,
   289→        reference: true,
   290→        status: true,
   291→        supplierNameAtOrder: true, // §25 — copy the ALREADY-FROZEN snapshot
   292→      },
   293→    });
   294→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   295→    assertPurchaseOrderReceivable(po.status); // §11 — DRAFT/SENT/CLOSED/CANCELLED → 422
   296→
   297→    // §59 — reference: user-supplied (normalized) or server-generated.
   298→    let reference = input.reference;
   299→    if (reference === undefined) {
   300→      for (let attempt = 0; attempt < 5; attempt += 1) {
   301→        const candidate = generateGoodsReceiptReference(new Date(), (max) => randomInt(max));
   302→        const clash = await tx.goodsReceipt.findFirst({
   303→          where: { businessId: input.businessId, reference: candidate },
   304→          select: { id: true },
   305→        });
   306→        if (!clash) {
   307→          reference = candidate;
   308→          break;
   309→        }
   310→      }
   311→      if (reference === undefined) {
   312→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   313→        throw new ApiError(
   314→          409,
   315→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   316→          "Could not generate a unique reference. Please try again.",
   317→        );
   318→      }
   319→    }
   320→
   321→    try {
   322→      const row = await tx.goodsReceipt.create({
   323→        data: {
   324→          businessId: input.businessId, // §22 — server-derived, never client
   325→          purchaseOrderId: po.id,
   326→          reference,
   327→          status: "DRAFT", // §58 — every receipt starts as preparation only
   328→          supplierNameAtReceipt: po.supplierNameAtOrder, // §25 — frozen identity
   329→          purchaseOrderReferenceAtReceipt: po.reference, // §24 — frozen identity
   330→          notes: input.notes ?? null, // §62 — document data only
   331→          createdByUserId: input.userId, // §58 — server-derived
   332→        },
   333→        select: { id: true, reference: true, status: true },
   334→      });
   335→
   336→      await audit(
   337→        auditInput(input.businessId, input.userId, "goods_receipt.create", row.id, row.reference, {
   338→          purchaseOrderId: po.id,
   339→          purchaseOrderReferenceAtReceipt: po.reference,
   340→          supplierNameAtReceipt: po.supplierNameAtOrder,
   341→          status: row.status,
   342→        }),
   343→        tx,
   344→      );
   345→
   346→      return row;
   347→    } catch (error) {
   348→      // §61 — database uniqueness stays authoritative for duplicates.
   349→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   350→        throw new ApiError(
   351→          409,
   352→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   353→          "A goods receipt with this reference already exists.",
   354→        );
   355→      }
   356→      throw error;
   357→    }
   358→  });
   359→}

   363→export async function listGoodsReceipts(input: {
   364→  businessId: string;
   365→  q?: string;
   366→  status?: string; // ALL → undefined
   367→  purchaseOrderId?: string;
   368→  from?: string;
   369→  to?: string;
   370→  page: number;
   371→  pageSize: number;
   372→}): Promise<{
   373→  items: GoodsReceiptListItemDto[];
   374→  total: number;
   375→  page: number;
   376→  pageSize: number;
   377→}> {
   378→  const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   379→  const where: Prisma.GoodsReceiptWhereInput = {
   380→    businessId: input.businessId, // §89 — tenant scope
   381→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   382→    ...(input.purchaseOrderId ? { purchaseOrderId: input.purchaseOrderId } : {}),
   383→    ...(input.from || input.to ? { createdAt: dayWindow(input.from, input.to) } : {}),
   384→    ...(input.q
   385→      ? {
   386→          OR: [
   387→            { reference: { contains: input.q } }, // receipt reference (§81)
   388→            { purchaseOrderReferenceAtReceipt: { contains: input.q } }, // PO reference
   389→            { supplierNameAtReceipt: { contains: input.q } }, // supplier snapshot
   390→            {
   391→              lines: {
   392→                some: {
   393→                  OR: [
   394→                    { productNameAtReceipt: { contains: input.q } }, // product snapshot
   395→                    { skuAtReceipt: { contains: input.q } }, // SKU snapshot
   396→                  ],
   397→                },
   398→              },
   399→            },
   400→          ],
   401→        }
   402→      : {}),
   403→  };
   404→
   405→  const [total, rows] = await Promise.all([
   406→    db.goodsReceipt.count({ where }),
   407→    // §175 — DB-level pagination; never findMany + slice.
   408→    db.goodsReceipt.findMany({
   409→      where,
   410→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   411→      skip: (input.page - 1) * input.pageSize,
   412→      take: input.pageSize,
   413→      select: {
   414→        id: true,
   415→        reference: true,
   416→        status: true,
   417→        purchaseOrderId: true,
   418→        purchaseOrderReferenceAtReceipt: true,
   419→        supplierNameAtReceipt: true,
   420→        postedAt: true,
   421→        cancelledAt: true,
   422→        createdAt: true,
   423→        lines: {
   424→          // §180 — bounded aggregate relation, not one query per row.
   425→          select: { receivedQty: true },
   426→          take: cap,
   427→        },
   428→        _count: { select: { lines: true } },
   429→      },
   430→    }),
   431→  ]);
   432→
   433→  const items = rows.map((row) => {
   434→    // §180 — the capped relation slice IS the full line set (a receipt can
   435→    // never exceed its own cap), so the sum below is provably complete.
   436→    const totalReceivedQty = computeReceivedQty(row.lines.map((l) => l.receivedQty));
   437→    return goodsReceiptListItemDto({
   438→      id: row.id,
   439→      reference: row.reference,
   440→      status: row.status,
   441→      purchaseOrderId: row.purchaseOrderId,
   442→      purchaseOrderReferenceAtReceipt: row.purchaseOrderReferenceAtReceipt,
   443→      supplierNameAtReceipt: row.supplierNameAtReceipt,
   444→      lineCount: row._count.lines,
   445→      totalReceivedQty,
   446→      postedAt: row.postedAt,
   447→      cancelledAt: row.cancelledAt,
   448→      createdAt: row.createdAt,
   449→      createdByUserId: "",
   450→      createdByName: null,
   451→      notes: null,
   452→    });
   453→  });
   454→  return { items, total, page: input.page, pageSize: input.pageSize };
   455→}

   459→export async function getGoodsReceipt(input: {
   460→  businessId: string;
   461→  id: string;
   462→}): Promise<GoodsReceiptDetailDto> {
   463→  const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   464→  // §176 — lines are bounded by the document's own cap; one bounded read.
   465→  const row = await db.goodsReceipt.findFirst({
   466→    where: { id: input.id, businessId: input.businessId }, // §89
   467→    select: {
   468→      id: true,
   469→      reference: true,
   470→      status: true,
   471→      notes: true,
   472→      purchaseOrderId: true,
   473→      purchaseOrderReferenceAtReceipt: true,
   474→      supplierNameAtReceipt: true,
   475→      createdByUserId: true,
   476→      postedAt: true,
   477→      cancelledAt: true,
   478→      createdAt: true,
   479→      createdByUser: { select: { name: true } }, // §88 — name only
   480→      lines: {
   481→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   482→        take: cap,
   483→        select: {
   484→          id: true,
   485→          purchaseOrderLineId: true,
   486→          productId: true,
   487→          productNameAtReceipt: true,
   488→          skuAtReceipt: true,
   489→          unitAtReceipt: true,
   490→          receivedQty: true,
   491→          createdAt: true,
   492→          updatedAt: true,
   493→        },
   494→      },
   495→    },
   496→  });
   497→  if (!row) throw new ApiError(404, "NOT_FOUND", "Goods receipt not found.");
   498→
   499→  try {
   500→    return goodsReceiptDetailDto(
   501→      {
   502→        id: row.id,
   503→        reference: row.reference,
   504→        status: row.status,
   505→        notes: row.notes,
   506→        purchaseOrderId: row.purchaseOrderId,
   507→        purchaseOrderReferenceAtReceipt: row.purchaseOrderReferenceAtReceipt,
   508→        supplierNameAtReceipt: row.supplierNameAtReceipt,
   509→        createdByUserId: row.createdByUserId,
   510→        createdByName: row.createdByUser?.name ?? null,
   511→        postedAt: row.postedAt,
   512→        cancelledAt: row.cancelledAt,
   513→        createdAt: row.createdAt,
   514→      },
   515→      row.lines,
   516→    );
   517→  } catch (error) {
   518→    return mapPureGoodsReceiptDomainError(error);
   519→  }
   520→}

   524→export async function updateGoodsReceipt(input: {
   525→  businessId: string;
   526→  userId: string;
   527→  id: string;
   528→  reference?: string;
   529→  notes?: string | null;
   530→  ip?: string | null;
   531→  userAgent?: string | null;
   532→}) {
   533→  return withWriteTx(async (tx) => {
   534→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   535→    requireDraft(header.status); // §107 — DRAFT-only edits
   536→
   537→    const data: Prisma.GoodsReceiptUpdateInput = {};
   538→    if (input.reference !== undefined) data.reference = input.reference;
   539→    if (input.notes !== undefined) data.notes = input.notes;
   540→
   541→    try {
   542→      const updated = await tx.goodsReceipt.update({
   543→        where: { id: header.id },
   544→        data,
   545→        select: { id: true, reference: true, status: true },
   546→      });
   547→
   548→      await audit(
   549→        auditInput(
   550→          input.businessId,
   551→          input.userId,
   552→          "goods_receipt.update",
   553→          header.id,
   554→          updated.reference,
   555→          {
   556→            changed: {
   557→              ...(input.reference !== undefined
   558→                ? { reference: { from: header.reference, to: updated.reference } }
   559→                : {}),
   560→              ...(input.notes !== undefined ? { notes: { changed: true } } : {}),
   561→            },
   562→          },
   563→          { reference: header.reference },
   564→        ),
   565→        tx,
   566→      );
   567→
   568→      return updated;
   569→    } catch (error) {
   570→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   571→        throw new ApiError(
   572→          409,
   573→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   574→          "A goods receipt with this reference already exists.",
   575→        );
   576→      }
   577→      throw error;
   578→    }
   579→  });
   580→}

   584→export async function addGoodsReceiptLine(input: {
   585→  businessId: string;
   586→  userId: string;
   587→  id: string;
   588→  purchaseOrderLineId: string;
   589→  receivedQty: number;
   590→  ip?: string | null;
   591→  userAgent?: string | null;
   592→}) {
   593→  return withWriteTx(async (tx) => {
   594→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   595→    requireDraft(header.status); // §64 — DRAFT-only
   596→
   597→    // §35 — the PO line must belong to THIS receipt's tenant-scoped PO.
   598→    // One tenant-scoped join validates receipt→PO→line ownership; a
   599→    // foreign PO line is indistinguishable from an unknown one (§91 → 404).
   600→    const poLine = await tx.purchaseOrderLine.findFirst({
   601→      where: { id: input.purchaseOrderLineId, purchaseOrderId: header.purchaseOrderId },
   602→      select: {
   603→        id: true,
   604→        productId: true,
   605→        orderedQty: true,
   606→        productNameAtOrder: true, // §29 — snapshots copied from the PO line
   607→        skuAtOrder: true,
   608→        unitAtOrder: true,
   609→        product: { select: { id: true, businessId: true } },
   610→      },
   611→    });
   612→    if (!poLine) throw new ApiError(404, "NOT_FOUND", "Purchase order line not found.");
   613→
   614→    // §35 — product must exist inside the SAME tenant (belt-and-braces:
   615→    // the FK is Restrict so a hard-deleted product is nearly impossible,
   616→    // but §33/§152 demands a safe controlled failure, never fabrication).
   617→    if (poLine.product.businessId !== input.businessId) {
   618→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   619→    }
   620→
   621→    // §65 — the PO must STILL be CONFIRMED at line-add time (a PO that was
   622→    // cancelled/closed after the draft was created stops accepting lines).
   623→    const po = await tx.purchaseOrder.findFirst({
   624→      where: { id: header.purchaseOrderId, businessId: input.businessId },
   625→      select: { status: true },
   626→    });
   627→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   628→    assertPurchaseOrderReceivable(po.status);
   629→
   630→    const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   631→    await assertLineCap(tx, header.id, cap); // §71 — defensive count
   632→
   633→    // §66 — ADVISORY remaining check at line-add time: receivedQty ≤
   634→    // (ordered − POSTED received). Draft lines of OTHER receipts are
   635→    // deliberately NOT subtracted — over-allocation across drafts is the
   636→    // documented §67 stale-draft scenario and only POST is authoritative.
   637→    const alreadyPosted = await sumPostedQtyForPoLine(tx, poLine.id);
   638→    const remaining = computeRemainingQty(poLine.orderedQty, alreadyPosted); // §172
   639→    if (exceedsRemaining(poLine.orderedQty, alreadyPosted, input.receivedQty)) {
   640→      throw new ApiError(
   641→        409,
   642→        GOODS_RECEIPT_ERROR_CODES.EXCEEDS_REMAINING,
   643→        `Only ${remaining} unit(s) remain to receive on this line.`,
   644→      );
   645→    }
   646→
   647→    try {
   648→      const line = await tx.goodsReceiptLine.create({
   649→        data: {
   650→          goodsReceiptId: header.id,
   651→          purchaseOrderLineId: poLine.id,
   652→          productId: poLine.productId, // §36 — derived, never client authority
   653→          receivedQty: input.receivedQty,
   654→          // §28/§29 — snapshots copied from the PO LINE, not live Product.
   655→          productNameAtReceipt: poLine.productNameAtOrder,
   656→          skuAtReceipt: poLine.skuAtOrder,
   657→          unitAtReceipt: poLine.unitAtOrder,
   658→        },
   659→        select: { id: true },
   660→      });
   661→
   662→      await audit(
   663→        auditInput(input.businessId, input.userId, "goods_receipt.line_add", header.id, header.reference, {
   664→          lineId: line.id, // §156 — safe per-line metadata
   665→          purchaseOrderLineId: poLine.id,
   666→          productId: poLine.productId,
   667→          productNameAtReceipt: poLine.productNameAtOrder,
   668→          receivedQty: input.receivedQty,
   669→        }),
   670→        tx,
   671→      );
   672→
   673→      return { id: line.id };
   674→    } catch (error) {
   675→      // §27/§209 — DB uniqueness is authoritative for duplicate lines.
   676→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   677→        throw new ApiError(
   678→          409,
   679→          GOODS_RECEIPT_ERROR_CODES.IMMUTABLE,
   680→          "This purchase order line is already on the receipt.",
   681→        );
   682→      }
   683→      throw error;
   684→    }
   685→  });
   686→}

   690→export async function updateGoodsReceiptLine(input: {
   691→  businessId: string;
   692→  userId: string;
   693→  id: string;
   694→  lineId: string;
   695→  receivedQty: number;
   696→  ip?: string | null;
   697→  userAgent?: string | null;

   700→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   701→    requireDraft(header.status); // §68 — DRAFT-only
   702→
   703→    const line = await tx.goodsReceiptLine.findFirst({
   704→      where: { id: input.lineId, goodsReceiptId: header.id }, // §89 — scoped via parent
   705→      select: {
   706→        id: true,
   707→        purchaseOrderLineId: true,
   708→        productId: true,
   709→        productNameAtReceipt: true,
   710→        receivedQty: true,
   711→      },
   712→    });
   713→    if (!line) throw new ApiError(404, "NOT_FOUND", "Receipt line not found.");
   714→
   715→    // §68 — the receivedQty still respects the advisory remaining check
   716→    // (POSTED sums only — see the §66 note in addGoodsReceiptLine).
   717→    const poLine = await tx.purchaseOrderLine.findFirst({
   718→      where: { id: line.purchaseOrderLineId },
   719→      select: { orderedQty: true },
   720→    });
   721→    if (!poLine) throw new ApiError(404, "NOT_FOUND", "Purchase order line not found.");
   722→    const alreadyPosted = await sumPostedQtyForPoLine(tx, line.purchaseOrderLineId);
   723→    const remaining = computeRemainingQty(poLine.orderedQty, alreadyPosted);

   729→      );
   730→    }
   731→
   732→    // §68-analog — re-read the persisted row after mutation: the response
   733→    // carries the REAL stored snapshot fields and timestamps.
   734→    const updated = await tx.goodsReceiptLine.update({
   735→      where: { id: line.id },
   736→      data: { receivedQty: input.receivedQty },
   737→      select: {
   738→        id: true,
   739→        purchaseOrderLineId: true,
   740→        productId: true,
   741→        productNameAtReceipt: true,
   742→        skuAtReceipt: true,
   743→        unitAtReceipt: true,
   744→        receivedQty: true,
   745→        createdAt: true,
   746→        updatedAt: true,
   747→      },
   748→    });
   749→
   750→    await audit(
   751→      auditInput(input.businessId, input.userId, "goods_receipt.line_update", header.id, header.reference, {
   752→        lineId: line.id, // §156 — before/after quantities only
   753→        receivedQty: { from: line.receivedQty, to: updated.receivedQty },
   754→      }),
   755→      tx,
   756→    );
   757→
   758→    return goodsReceiptLineDto(updated);
   759→  });

   800→/* ---------------------------------- post ----------------------------------- */
   801→
   802→/**
   803→ * §39 — THE atomic stock posting transaction. Steps 1-17 in ONE
   804→ * database transaction (§72); a failure at ANY step rolls back EVERYTHING
   805→ * (§40/§214) — no stock change, no adjustment, no status, no audit.
   806→ */
   807→export async function postGoodsReceipt(input: {
   808→  businessId: string;
   809→  userId: string;
   810→  id: string;
   811→  ip?: string | null;
   812→  userAgent?: string | null;
   813→}) {
   814→  return withWriteTx(async (tx) => {
   815→    // §39 steps 1-2 / §73 — reload receipt INSIDE the transaction; never
   816→    // trust an earlier GET. Corrupted status is surfaced, never Draft.
   817→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   818→    assertTransition(header.status, "POSTED"); // §7 — only DRAFT → POSTED
   819→
   820→    // §39 step 3 — PO must STILL be CONFIRMED (§149/§150: a PO cancelled
   821→    // or closed after the draft was created can never be posted).
   822→    const po = await tx.purchaseOrder.findFirst({
   823→      where: { id: header.purchaseOrderId, businessId: input.businessId },
   824→      select: { id: true, reference: true, status: true, supplierNameAtOrder: true },
   825→    });
   826→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   827→    assertPurchaseOrderReceivable(po.status);
   828→
   829→    // §39 step 4 — load the receipt's own lines (bounded by the document
   830→    // cap, §177 — a receipt can never exceed GOODS_RECEIPT_MAX_LINES).
   831→    const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   832→    const lines = await tx.goodsReceiptLine.findMany({
   833→      where: { goodsReceiptId: header.id },
   834→      orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   835→      take: cap + 1, // cap+1 proves completeness (same law as P41 export)
   836→      select: {
   837→        id: true,
   838→        purchaseOrderLineId: true,
   839→        productId: true,
   840→        receivedQty: true,
   841→        purchaseOrderLine: {
   842→          select: {
   843→            id: true,
   844→            orderedQty: true,
   845→            purchaseOrderId: true,
   846→          },
   847→        },
   848→      },
   849→    });
   850→    if (lines.length > cap) {
   851→      // Impossible by §20/§71 — if it ever happens, fail honestly.
   852→      throw new ApiError(
   853→        422,
   854→        GOODS_RECEIPT_ERROR_CODES.LINE_LIMIT,
   855→        "Goods receipt line limit exceeded.",
   856→      );
   857→    }
   858→
   859→    // §70 — a receipt with zero lines can never post.
   860→    if (lines.length === 0) {
   861→      throw new ApiError(
   862→        422,
   863→        GOODS_RECEIPT_ERROR_CODES.EMPTY,
   864→        "Add at least one line before posting this receipt.",
   865→      );
   866→    }
   867→
   868→    const adjustmentIds: string[] = [];
   869→    const stockChanges: { productId: string; from: number; to: number }[] = [];
   870→
   871→    for (const line of lines) {
   872→      // §39 step 4 — the PO line must belong to THIS purchase order
   873→      // (defensive re-verification of the document chain).
   874→      if (line.purchaseOrderLine.purchaseOrderId !== po.id) {
   875→        throw new ApiError(404, "NOT_FOUND", "Purchase order line not found.");
   876→      }
   877→
   878→      // §39 steps 5-8 — product existence + authoritative remaining check.
   879→      const product = await tx.product.findFirst({
   880→        where: { id: line.productId, businessId: input.businessId }, // §30/§35
   881→        select: { id: true, stockQty: true },
   882→      });
   883→      if (!product) {
   884→        // §33/§152 — a hard-deleted product fails the post safely.
   885→        throw new ApiError(
   886→          422,
   887→          GOODS_RECEIPT_ERROR_CODES.PRODUCT_NOT_FOUND,
   888→          "A product on this receipt no longer exists.",
   889→        );
   890→      }
   891→
   892→      const alreadyPosted = await sumPostedQtyForPoLine(tx, line.purchaseOrderLineId);
   893→      const remaining = computeRemainingQty(
   894→        line.purchaseOrderLine.orderedQty,
   895→        alreadyPosted,
   896→      ); // §100 — throws on corrupted over-received history (§283)
   897→      if (exceedsRemaining(line.purchaseOrderLine.orderedQty, alreadyPosted, line.receivedQty)) {
   898→        // §15/§16/§67 — the HARD over-receipt law, final authority.
   899→        throw new ApiError(
   900→          409,
   901→          GOODS_RECEIPT_ERROR_CODES.EXCEEDS_REMAINING,
   902→          `Only ${remaining} unit(s) remain to receive on this line.`,
   903→        );
   904→      }
   905→
   906→      // §46/§47 — corrupt persisted stock is surfaced, never repaired.
   907→      assertPersistedStockValid(product.stockQty);
   908→
   909→      // §49 — Int32 storage bound; overflow → 422, no partial writes.
   910→      if (stockOverflowsInt32(product.stockQty, line.receivedQty)) {
   911→        throw new ApiError(
   912→          422,
   913→          GOODS_RECEIPT_ERROR_CODES.STOCK_OVERFLOW,
   914→          "Receiving this quantity would exceed the stock storage limit.",
   915→        );
   916→      }
   917→
   918→      // §39 steps 9-11 — exact integer arithmetic, positive delta only.
   919→      const newStock = product.stockQty + line.receivedQty;
   920→      await tx.product.update({
   921→        where: { id: product.id },
   922→        data: { stockQty: newStock },
   923→        select: { id: true },
   924→      });
   925→      stockChanges.push({ productId: product.id, from: product.stockQty, to: newStock });
   926→
   927→      // §39 step 12 / §41-§45 — one positive InventoryAdjustment per line.
   928→      const adjustment = await tx.inventoryAdjustment.create({
   929→        data: {
   930→          businessId: input.businessId, // §43 — server-derived
   931→          productId: product.id, // §44 — derived from the PO line chain
   932→          delta: line.receivedQty, // §45 — always positive
   933→          reason: RECEIPT_ADJUSTMENT_REASON(header.reference), // §41 — stable identifier
   934→          actorId: input.userId, // §42 — server-derived
   935→        },
   936→        select: { id: true },
   937→      });
   938→      adjustmentIds.push(adjustment.id);
   939→    }
   940→
   941→    // §39 steps 14-15 — complete the posting (server timestamps only §74).
   942→    const postedAt = new Date();
   943→    const updated = await tx.goodsReceipt.update({
   944→      where: { id: header.id },
   945→      data: { status: "POSTED", postedAt },
   946→      select: { id: true, reference: true, status: true, postedAt: true },
   947→    });
   948→
   949→    // §39 step 16 / §153/§154 — ONE audit record for the posting, written
   950→    // in the SAME transaction; if it fails, everything above rolls back.
   951→    const totalReceivedQty = computeReceivedQty(lines.map((l) => l.receivedQty));
   952→    await audit(
   953→      auditInput(input.businessId, input.userId, "goods_receipt.post", header.id, header.reference, {
   954→        purchaseOrderId: po.id,
   955→        purchaseOrderReferenceAtReceipt: header.purchaseOrderReferenceAtReceipt,
   956→        lineCount: lines.length,
   957→        totalReceivedQty, // §155 — quantity facts only; never notes
   958→        adjustmentIds,
   959→        previousStatus: header.status,
   960→        newStatus: updated.status,
   961→        postedAt: updated.postedAt?.toISOString() ?? null,
   962→      }),
   963→      tx,
   964→    );
   965→
   966→    return {
   967→      id: updated.id,
   968→      reference: updated.reference,
   969→      status: updated.status,
   970→      postedAt: updated.postedAt,
   971→      lineCount: lines.length,
   972→      totalReceivedQty,
   973→      stockChanges,
   974→    };
   975→  });
   976→}

  1018→/* ------------------- P41 integration: receiving summary --------------------- */
  1019→
  1020→/**
  1021→ * §98/§181 — the PO receiving summary in ONE consistent read (§296): a
  1022→ * transactional read of the PO lines + the POSTED receipt-line aggregates
  1023→ * (grouped, never one query per line). Corrupted over-received history
  1024→ * surfaces as the stable integrity code (§283 — never clamped).
  1025→ */
  1026→export async function getPurchaseOrderReceivingSummary(input: {
  1027→  businessId: string;
  1028→  purchaseOrderId: string;
  1029→}): Promise<{
  1030→  orderedQty: number;
  1031→  receivedQty: number;
  1032→  remainingQty: number;
  1033→  receiptCount: number;
  1034→  fullyReceived: boolean;
  1035→  lines: { lineId: string; orderedQty: number; receivedQty: number; remainingQty: number }[];
  1036→}> {
  1037→  const po = await db.purchaseOrder.findFirst({
  1038→    where: { id: input.purchaseOrderId, businessId: input.businessId }, // §89
  1039→    select: { id: true },
  1040→  });
  1041→  if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
  1042→
  1043→  // §296 — one transactional read of lines + posted aggregates.
  1044→  const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
  1045→  return db.$transaction(async (tx) => {
  1046→    const poLines = await tx.purchaseOrderLine.findMany({
  1047→      where: { purchaseOrderId: po.id },
  1048→      select: { id: true, orderedQty: true },
  1049→      take: cap, // bounded by the PO's own line cap (P41 law)
  1050→    });
  1051→
  1052→    // §178 — sum POSTED receipt lines per PO line, grouped in one query.
  1053→    const grouped = await tx.goodsReceiptLine.groupBy({
  1054→      by: ["purchaseOrderLineId"],
  1055→      where: {
  1056→        purchaseOrderLineId: { in: poLines.map((l) => l.id) },
  1057→        goodsReceipt: { status: "POSTED" }, // §99 — POSTED only
  1058→      },
  1059→      _sum: { receivedQty: true },
  1060→    });
  1061→    const receivedByLine = new Map(
  1062→      grouped.map((g) => [g.purchaseOrderLineId, g._sum.receivedQty ?? 0]),
  1063→    );
  1064→
  1065→    // §104-analog — receipt COUNT (POSTED receipts touching this PO).
  1066→    const receiptCount = await tx.goodsReceipt.count({
  1067→      where: { purchaseOrderId: po.id, status: "POSTED" },
  1068→    });
  1069→
  1070→    let orderedQty = 0;
  1071→    let receivedQty = 0;
  1072→    const lines = poLines.map((l) => {
  1073→      const lineReceived = receivedByLine.get(l.id) ?? 0;
  1074→      const lineRemaining = computeRemainingQty(l.orderedQty, lineReceived); // §100
  1075→      orderedQty += l.orderedQty;
  1076→      receivedQty += lineReceived;
  1077→      return {
  1078→        lineId: l.id,
  1079→        orderedQty: l.orderedQty,
  1080→        receivedQty: lineReceived,
  1081→        remainingQty: lineRemaining,
  1082→      };
  1083→    });
  1084→    const remainingQty = computeRemainingQty(orderedQty, receivedQty); // §100
  1085→
  1086→    return {
  1087→      orderedQty,
  1088→      receivedQty,
  1089→      remainingQty,
  1090→      receiptCount,
  1091→      fullyReceived: remainingQty === 0, // §101 — derived, never stored
  1092→      lines,
  1093→    };
  1094→  }).catch((error) => mapPureGoodsReceiptDomainError(error));
  1095→}
