     4→     4→ * Prompt 40 — stocktake service (server-only, §118).
     5→     5→ *
     6→     6→ * LAWS:
     7→     7→ *  - Every query is tenant-scoped by ctx.businessId (§54) — never a client
     8→     8→ *    businessId. Foreign stocktake/product → 404, no existence leakage (§55).
     9→     9→ *  - Status transitions are server-authoritative through dedicated
    10→    10→ *    operations (§9); invalid transitions → 422 INVALID_STOCKTAKE_TRANSITION.
    11→    11→ *  - systemQtyAtCount is ALWAYS the server-read Product.stockQty at count
    12→    12→ *    time (§15); the client supplies ONLY countedQty (§16); variance is
    13→    13→ *    server-derived (§16/§18).
    14→    14→ *  - Finalization is ONE atomic transaction (§39): stale-count check against
    15→    15→ *    CURRENT stockQty (§40/§41 — never auto-rebased), InventoryAdjustment
    16→    16→ *    rows for non-zero variances (§43), zero-variance lines skip the
    17→    17→ *    adjustment (§44), no partial finalization (§156).
    18→    18→ *  - FINALIZED/CANCELLED are immutable (§50/§51).
    19→    19→ *  - Audit through the single sanctioned helper inside the same transaction
    20→    20→ *    (§71); NO audit on reads (§69); safe metadata only (§70/§140).
    21→    21→ */

    58→    58→  tx: Tx,

    72→    72→  createdAt: Date;
    73→    73→}> {
    74→    74→  const row = await tx.inventoryStocktake.findFirst({

    76→    76→    select: {
    77→    77→      id: true,
    78→    78→      reference: true,
    79→    79→      status: true,
    80→    80→      countType: true,

    89→    89→  });
    90→    90→  if (!row) throw new ApiError(404, "NOT_FOUND", "Stocktake not found.");

    92→    92→}

   156→   156→
   157→   157→export async function listStocktakes(input: {
   158→   158→  businessId: string;
   159→   159→  q?: string;
   160→   160→  status?: string;
   161→   161→  from?: string;
   162→   162→  to?: string;
   163→   163→  page: number;
   164→   164→  pageSize: number;
   165→   165→}) {
   166→   166→  const where: Prisma.InventoryStocktakeWhereInput = { businessId: input.businessId };
   167→   167→  if (input.q) where.reference = { contains: input.q };
   168→   168→  if (input.status && isStocktakeStatus(input.status)) where.status = input.status;
   169→   169→  if (input.from || input.to) {
   170→   170→    where.createdAt = {};
   171→   171→    if (input.from) where.createdAt.gte = new Date(`${input.from}T00:00:00.000Z`);
   172→   172→    if (input.to) where.createdAt.lte = new Date(`${input.to}T23:59:59.999Z`);
   173→   173→  }
   174→   174→
   175→   175→  const [rows, total] = await Promise.all([
   176→   176→    db.inventoryStocktake.findMany({
   177→   177→      where,
   178→   178→      orderBy: { createdAt: "desc" }, // deterministic newest-first list (§134)
   179→   179→      skip: (input.page - 1) * input.pageSize,
   180→   180→      take: input.pageSize,
   181→   181→      // §67 — list select minimization; a bounded line slice feeds the
   182→   182→      // counted/net columns without loading full line tables.
   183→   183→      select: {
   184→   184→        id: true,
   185→   185→        reference: true,
   186→   186→        status: true,
   187→   187→        countType: true,
   188→   188→        createdAt: true,
   189→   189→        createdByUser: { select: { id: true, name: true, email: true } },
   190→   190→        _count: { select: { lines: true } },
   191→   191→        lines: { select: { countedAt: true, varianceQty: true } },
   192→   192→      },
   193→   193→    }),
   194→   194→    db.inventoryStocktake.count({ where }),
   195→   195→  ]);
   196→   196→
   197→   197→  return {
   198→   198→    items: rows.map(stocktakeListItemDto),
   199→   199→    total,
   200→   200→    page: input.page,
   201→   201→    pageSize: input.pageSize,
   202→   202→  };
   203→   203→}

   206→   206→
   207→   207→export async function getStocktake(input: {
   208→   208→  businessId: string;
   209→   209→  id: string;
   210→   210→  page: number;
   211→   211→  pageSize: number;
   212→   212→}) {
   213→   213→  const header = await loadStocktake(db, input.businessId, input.id);
   214→   214→  // §36 — summary over ALL lines (aggregate-light select, §66).
   215→   215→  const [summaryRows, lineRows, lineTotal] = await Promise.all([
   216→   216→    db.inventoryStocktakeLine.findMany({
   217→   217→      where: { stocktakeId: header.id },
   218→   218→      select: { countedAt: true, varianceQty: true },
   219→   219→    }),
   220→   220→    db.inventoryStocktakeLine.findMany({
   221→   221→      where: { stocktakeId: header.id },
   222→   222→      orderBy: [{ countedAt: "asc" }, { id: "asc" }], // deterministic ordering
   223→   223→      skip: (input.page - 1) * input.pageSize,
   224→   224→      take: input.pageSize,
   225→   225→      select: {
   226→   226→        id: true,
   227→   227→        productId: true,
   228→   228→        productNameAtCount: true,
   229→   229→        skuAtCount: true,
   230→   230→        unitAtCount: true,
   231→   231→        systemQtyAtCount: true,
   232→   232→        countedQty: true,
   233→   233→        varianceQty: true,
   234→   234→        countedAt: true,
   235→   235→        countedByUser: { select: { id: true, name: true, email: true } },
   236→   236→      },
   237→   237→    }),
   238→   238→    db.inventoryStocktakeLine.count({ where: { stocktakeId: header.id } }),
   239→   239→  ]);
   240→   240→
   241→   241→  return stocktakeDetailDto({
   242→   242→    ...header,
   243→   243→    createdByUser: await db.user.findUnique({
   244→   244→      where: { id: header.createdByUserId },
   245→   245→      select: { id: true, name: true, email: true },
   246→   246→    }),
   247→   247→    summary: computeStocktakeSummary(summaryRows),
   248→   248→    lines: lineRows,
   249→   249→    linesPage: { page: input.page, pageSize: input.pageSize, total: lineTotal },
   250→   250→  });
   251→   251→}

   253→   253→/* ------------------------------- add line --------------------------------- */
   254→   254→
   255→   255→export async function addStocktakeLine(input: {
   256→   256→  businessId: string;
   257→   257→  userId: string;
   258→   258→  stocktakeId: string;
   259→   259→  productId: string;
   260→   260→  ip?: string | null;
   261→   261→  userAgent?: string | null;
   262→   262→}) {
   263→   263→  const cap = stocktakeLineCap(process.env.INVENTORY_STOCKTAKE_MAX_LINES);
   264→   264→  return db.$transaction(async (tx) => {
   265→   265→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   266→   266→    if (stocktake.status !== "DRAFT") {
   267→   267→      // §27 — DRAFT-only line preparation.
   268→   268→      throw new ApiError(
   269→   269→        422,
   270→   270→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   271→   271→        "Products can only be added while the stocktake is a draft.",
   272→   272→      );
   273→   273→    }
   274→   274→    const lineCount = await tx.inventoryStocktakeLine.count({
   275→   275→      where: { stocktakeId: stocktake.id },
   276→   276→    });
   277→   277→    if (lineCount >= cap) {
   278→   278→      // §61 — bounded transaction sizes; surfaced to the UI (lines/max).
   279→   279→      throw new ApiError(
   280→   280→        422,
   281→   281→        STOCKTAKE_ERROR_CODES.LINE_LIMIT,
   282→   282→        `Stocktake line limit reached (${cap}).`,
   283→   283→      );
   284→   284→    }
   285→   285→    // §20/§21/§56/§57 — same business, not deleted, active. Foreign/deleted
   286→   286→    // product → 404 (no existence leakage); inactive → documented 422.
   287→   287→    const product = await tx.product.findFirst({
   288→   288→      where: { id: input.productId, businessId: input.businessId },
   289→   289→      select: { id: true, deletedAt: true, isActive: true },
   290→   290→    });
   291→   291→    if (!product || product.deletedAt) {
   292→   292→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   293→   293→    }
   294→   294→    if (!product.isActive) {
   295→   295→      throw new ApiError(
   296→   296→        422,
   297→   297→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   298→   298→        "Inactive products cannot be added to a stocktake.",
   299→   299→      );
   300→   300→    }
   301→   301→    try {
   302→   302→      const line = await tx.inventoryStocktakeLine.create({
   303→   303→        data: {
   304→   304→          stocktakeId: stocktake.id,
   305→   305→          productId: product.id,
   306→   306→          // Line is PREPARATION data until counted (§29) — zero placeholders.
   307→   307→          systemQtyAtCount: 0,
   308→   308→          countedQty: 0,
   309→   309→          varianceQty: 0,
   310→   310→          countedAt: new Date(0),
   311→   311→          countedByUserId: input.userId,
   312→   312→          productNameAtCount: "",
   313→   313→          skuAtCount: null,
   314→   314→          unitAtCount: null,
   315→   315→        },
   316→   316→        select: { id: true, productId: true },
   317→   317→      });
   318→   318→      await audit(
   319→   319→        auditInput(input.businessId, input.userId, "inventory.stocktake.line_add", stocktake.id, stocktake.reference, {
   320→   320→          lineId: line.id,
   321→   321→          productId: product.id,
   322→   322→        }),
   323→   323→        tx,
   324→   324→      );
   325→   325→      return line;
   326→   326→    } catch (error) {
   327→   327→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   328→   328→        // §28/§48 — DB uniqueness is the final protection.
   329→   329→        throw new ApiError(
   330→   330→          409,
   331→   331→          STOCKTAKE_ERROR_CODES.DUPLICATE_PRODUCT,
   332→   332→          "This product is already in the stocktake.",
   333→   333→        );
   334→   334→      }
   335→   335→      throw error;
   336→   336→    }
   337→   337→  });
   338→   338→}

   376→   376→/* --------------------------------- count ---------------------------------- */
   377→   377→
   378→   378→export async function countStocktakeLine(input: {
   379→   379→  businessId: string;
   380→   380→  userId: string;
   381→   381→  stocktakeId: string;
   382→   382→  lineId: string;
   383→   383→  countedQty: number;
   384→   384→  ip?: string | null;
   385→   385→  userAgent?: string | null;
   386→   386→}) {
   387→   387→  return db.$transaction(async (tx) => {
   388→   388→    // §30 — load stocktake → verify status (COUNTING|REVIEW, §31) → verify
   389→   389→    // line → verify product tenant/active → read current stock → snapshot.
   390→   390→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   391→   391→    if (stocktake.status !== "COUNTING" && stocktake.status !== "REVIEW") {
   392→   392→      throw new ApiError(
   393→   393→        422,
   394→   394→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   395→   395→        "Counting is only possible while the stocktake is counting or in review.",
   396→   396→      );
   397→   397→    }
   398→   398→    const line = await tx.inventoryStocktakeLine.findFirst({
   399→   399→      where: { id: input.lineId, stocktakeId: stocktake.id },
   400→   400→      select: {
   401→   401→        id: true,
   402→   402→        productId: true,
   403→   403→        countedAt: true,
   404→   404→        systemQtyAtCount: true,
   405→   405→        countedQty: true,
   406→   406→        varianceQty: true,
   407→   407→      },
   408→   408→    });
   409→   409→    if (!line) throw new ApiError(404, "NOT_FOUND", "Stocktake line not found.");
   410→   410→
   411→   411→    // §15 — server reads the CURRENT stock; client never supplies it.
   412→   412→    const product = await tx.product.findFirst({
   413→   413→      where: { id: line.productId, businessId: input.businessId },
   414→   414→      select: { id: true, name: true, sku: true, unit: true, stockQty: true, deletedAt: true, isActive: true },
   415→   415→    });
   416→   416→    if (!product || product.deletedAt) {
   417→   417→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   418→   418→    }
   419→   419→    if (!product.isActive) {
   420→   420→      throw new ApiError(
   421→   421→        422,
   422→   422→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   423→   423→        "Inactive products cannot be counted.",
   424→   424→      );
   425→   425→    }
   426→   426→
   427→   427→    const systemQtyAtCount = product.stockQty; // §15 snapshot at count time
   428→   428→    const varianceQty = computeVarianceQty(input.countedQty, systemQtyAtCount); // §16/§18
   429→   429→    const now = new Date();
   430→   430→    const isRecount = line.countedAt !== null && line.countedAt.getTime() !== 0;
   431→   431→
   432→   432→    const updated = await tx.inventoryStocktakeLine.update({
   433→   433→      where: { id: line.id },
   434→   434→      data: {
   435→   435→        systemQtyAtCount,
   436→   436→        countedQty: input.countedQty,
   437→   437→        varianceQty,
   438→   438→        countedAt: now, // §30 step 14
   439→   439→        countedByUserId: input.userId, // last successful count is authoritative (§47)
   440→   440→        productNameAtCount: product.name, // §19 metadata snapshot
   441→   441→        skuAtCount: product.sku,
   442→   442→        unitAtCount: product.unit,
   443→   443→      },
   444→   444→      select: { id: true, productId: true, systemQtyAtCount: true, countedQty: true, varianceQty: true },
   445→   445→    });
   446→   446→
   447→   447→    if (isRecount) {
   448→   448→      // §32/§33 — replace in place; audit the safe before/after metadata.
   449→   449→      await audit(
   450→   450→        auditInput(
   451→   451→          input.businessId,
   452→   452→          input.userId,
   453→   453→          "inventory.stocktake.recount",
   454→   454→          stocktake.id,
   455→   455→          stocktake.reference,
   456→   456→          {
   457→   457→            lineId: line.id,
   458→   458→            productId: line.productId,
   459→   459→            newSystemQty: updated.systemQtyAtCount,
   460→   460→            newCountedQty: updated.countedQty,
   461→   461→            newVariance: updated.varianceQty,
   462→   462→          },
   463→   463→          {
   464→   464→            previousSystemQty: line.systemQtyAtCount,
   465→   465→            previousCountedQty: line.countedQty,
   466→   466→            previousVariance: line.varianceQty,
   467→   467→          },
   468→   468→        ),
   469→   469→        tx,
   470→   470→      );
   471→   471→    } else {
   472→   472→      await audit(
   473→   473→        auditInput(input.businessId, input.userId, "inventory.stocktake.count", stocktake.id, stocktake.reference, {
   474→   474→          lineId: line.id,
   475→   475→          productId: line.productId,
   476→   476→          systemQtyAtCount: updated.systemQtyAtCount,
   477→   477→          countedQty: updated.countedQty,
   478→   478→          variance: updated.varianceQty,
   479→   479→        }),
   480→   480→        tx,
   481→   481→      );
   482→   482→    }
   483→   483→    return updated;
   484→   484→  });

   584→   584→/* ------------------------------- finalize --------------------------------- */
   585→   585→
   586→   586→export async function finalizeStocktake(input: {
   587→   587→  businessId: string;
   588→   588→  userId: string;
   589→   589→  stocktakeId: string;
   590→   590→  ip?: string | null;
   591→   591→  userAgent?: string | null;
   592→   592→}) {
   593→   593→  return db.$transaction(async (tx) => {
   594→   594→    // §39 — ONE transaction: verify → stale-check → reconcile → finalize.
   595→   595→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   596→   596→    assertTransition(stocktake.status, "FINALIZED"); // §38 — REVIEW only
   597→   597→
   598→   598→    const lines = await tx.inventoryStocktakeLine.findMany({
   599→   599→      where: { stocktakeId: stocktake.id },
   600→   600→      select: {
   601→   601→        id: true,
   602→   602→        productId: true,
   603→   603→        systemQtyAtCount: true,
   604→   604→        countedQty: true,
   605→   605→        varianceQty: true,
   606→   606→        countedAt: true,
   607→   607→        productNameAtCount: true,
   608→   608→      },
   609→   609→    });
   610→   610→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   611→   611→      throw new ApiError(
   612→   612→        422,
   613→   613→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   614→   614→        "Every product must be counted before finalization.",
   615→   615→      );
   616→   616→    }
   617→   617→
   618→   618→    // §40 stale check — every product's CURRENT stockQty must still equal
   619→   619→    // systemQtyAtCount, and the product must still be active + not deleted
   620→   620→    // (§57/§58). No auto-rebase (§41). Collect ALL stale/invalid lines so the
   621→   621→    // client can recount each one (§139 — current-business products only).
   622→   622→    const stale: Array<{
   623→   623→      lineId: string;
   624→   624→      productId: string;
   625→   625→      productName: string;
   626→   626→      systemQtyAtCount: number;
   627→   627→      currentQty: number;
   628→   628→      kind: "STALE" | "INACTIVE" | "DELETED";
   629→   629→    }> = [];
   630→   630→    const products = await tx.product.findMany({
   631→   631→      where: { id: { in: lines.map((l) => l.productId) }, businessId: input.businessId },
   632→   632→      select: { id: true, stockQty: true, deletedAt: true, isActive: true },
   633→   633→    });
   634→   634→    const productById = new Map(products.map((p) => [p.id, p]));
   635→   635→    for (const line of lines) {
   636→   636→      const product = productById.get(line.productId);
   637→   637→      if (!product || product.deletedAt) {
   638→   638→        stale.push({
   639→   639→          lineId: line.id,
   640→   640→          productId: line.productId,
   641→   641→          productName: line.productNameAtCount,
   642→   642→          systemQtyAtCount: line.systemQtyAtCount,
   643→   643→          currentQty: product?.stockQty ?? line.systemQtyAtCount,
   644→   644→          kind: "DELETED",
   645→   645→        });
   646→   646→        continue;
   647→   647→      }
   648→   648→      if (!product.isActive) {
   649→   649→        stale.push({
   650→   650→          lineId: line.id,
   651→   651→          productId: line.productId,
   652→   652→          productName: line.productNameAtCount,
   653→   653→          systemQtyAtCount: line.systemQtyAtCount,
   654→   654→          currentQty: product.stockQty,
   655→   655→          kind: "INACTIVE",
   656→   656→        });
   657→   657→        continue;
   658→   658→      }
   659→   659→      if (product.stockQty !== line.systemQtyAtCount) {
   660→   660→        stale.push({
   661→   661→          lineId: line.id,
   662→   662→          productId: line.productId,
   663→   663→          productName: line.productNameAtCount,
   664→   664→          systemQtyAtCount: line.systemQtyAtCount,
   665→   665→          currentQty: product.stockQty,
   666→   666→          kind: "STALE",
   667→   667→        });
   668→   668→      }
   669→   669→    }
   670→   670→    if (stale.length > 0) {
   671→   671→      const hasInactive = stale.some((s) => s.kind === "INACTIVE");
   672→   672→      const hasDeleted = stale.some((s) => s.kind === "DELETED");
   673→   673→      throw new ApiError(
   674→   674→        409,
   675→   675→        hasInactive
   676→   676→          ? STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE
   677→   677→          : hasDeleted
   678→   678→            ? STOCKTAKE_ERROR_CODES.STALE_COUNT
   679→   679→            : STOCKTAKE_ERROR_CODES.STALE_COUNT,
   680→   680→        "Inventory changed after counting. Recount the affected product(s).",
   681→   681→        { products: stale }, // §139 — affected current-business products only
   682→   682→      );
   683→   683→    }
   684→   684→
   685→   685→    // §43/§44 — reconcile through the EXISTING InventoryAdjustment semantics.
   686→   686→    let adjustedLineCount = 0;
   687→   687→    for (const line of lines) {
   688→   688→      if (line.varianceQty === 0) continue; // §44 — zero variance: no adjustment
   689→   689→      const adjustment = await tx.inventoryAdjustment.create({
   690→   690→        data: {
   691→   691→          businessId: input.businessId,
   692→   692→          productId: line.productId,
   693→   693→          delta: line.varianceQty, // signed — §18
   694→   694→          reason: ADJUSTMENT_REASON(stocktake.reference),
   695→   695→          actorId: input.userId,
   696→   696→        },
   697→   697→        select: { id: true },
   698→   698→      });
   699→   699→      await tx.product.update({
   700→   700→        where: { id: line.productId },
   701→   701→        data: { stockQty: line.countedQty }, // §45 — stock becomes the counted value
   702→   702→      });
   703→   703→      adjustedLineCount += 1;
   704→   704→      // §72/§73 — the InventoryAdjustment row itself carries the quantity
   705→   705→      // history; no duplicate per-line audit event here. The finalize audit
   706→   706→      // below records the workflow-level change once.
   707→   707→      void adjustment;
   708→   708→    }
   709→   709→
   710→   710→    const updated = await tx.inventoryStocktake.update({
   711→   711→      where: { id: stocktake.id },
   712→   712→      data: { status: "FINALIZED", finalizedAt: new Date() },
   713→   713→      select: { id: true, status: true, finalizedAt: true },
   714→   714→    });
   715→   715→
   716→   716→    // §69/§73 — ONE finalize audit for the workflow; InventoryAdjustment rows
   717→   717→    // are the quantity-level history. Safe metadata only (§70/§140).
   718→   718→    await audit(
   719→   719→      auditInput(input.businessId, input.userId, "inventory.stocktake.finalize", stocktake.id, stocktake.reference, {
   720→   720→        status: "FINALIZED",
   721→   721→        lineCount: lines.length,
   722→   722→        adjustedLineCount,
   723→   723→      }),
   724→   724→      tx,
   725→   725→    );
   726→   726→    return { ...updated, adjustedLineCount };
   727→   727→  });
   728→   728→}

   763→   763→  stocktakeId: string;

   784→   784→          ...(input.notes !== undefined ? { notes: input.notes } : {}),
   785→   785→        },

   788→   788→      await audit(
   789→   789→        auditInput(
   790→   790→          input.businessId,
   791→   791→          input.userId,
   792→   792→          "inventory.stocktake.edit",
   793→   793→          stocktake.id,
   794→   794→          updated.reference,
   795→   795→          { status: stocktake.status },
   796→   796→          { reference: stocktake.reference }, // §70 — never the notes content

   800→   800→      return updated;
   801→   801→    } catch (error) {

   803→   803→        throw new ApiError(409, "DUPLICATE_REFERENCE", "A stocktake with this reference already exists.");
   804→   804→      }
   805→   805→      throw error;
   806→   806→    }
   807→   807→  });
