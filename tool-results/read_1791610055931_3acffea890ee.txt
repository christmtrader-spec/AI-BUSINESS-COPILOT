   143→
   144→export type ExportRow = {
   145→  /** JSON row — IDs + relationships preserved, money as minor integers (§54/§226). */
   146→  json: Record<string, unknown>;
   147→  /** CSV row — exact decimal strings for money, ISO-8601 UTC dates (§227/§231). */
   148→  csv: Array<string | number | null>;
   149→};
   150→
   151→export type DatasetFetchResult = {
   152→  rows: ExportRow[];
   153→  /** True when the dataset hit the export record cap (job must FAIL, §26). */
   154→  truncated: boolean;
   155→};
   156→
   157→export type DatasetFetchCtx = {
   158→  businessId: string;
   159→  /** Business decimal precision for money → CSV decimal strings. */
   160→  decimalPrecision: number;
   161→  /** Hard cap applied per fetch — caller passes EXPORT_MAX_RECORDS headroom. */
   162→  maxRecords: number;
   163→};

   165→function iso(value: Date | null | undefined): string | null {
   166→  return value ? value.toISOString() : null;
   167→}

   173→type Customer = Prisma.CustomerGetPayload<object>;

   175→function customerRow(c: Customer, precision: number): ExportRow {
   176→  void precision; // customers carry no money fields
   177→  return {
   178→    json: {
   179→      id: c.id,
   180→      name: c.name,
   181→      phone: c.phone,
   182→      email: c.email,
   183→      address: c.address,
   184→      note: c.note,
   185→      deletedAt: iso(c.deletedAt),
   186→      createdAt: c.createdAt.toISOString(),
   187→      updatedAt: c.updatedAt.toISOString(),
   188→    },
   189→    csv: [c.id, c.name, c.phone, c.email, c.address, c.note, iso(c.deletedAt), c.createdAt.toISOString(), c.updatedAt.toISOString()],
   190→  };
   191→}

   211→function productRow(p: Prisma.ProductGetPayload<object>, precision: number): ExportRow {
   212→  return {
   213→    json: {
   214→      id: p.id,
   215→      name: p.name,
   216→      sku: p.sku,
   217→      unit: p.unit,
   218→      salePriceMinor: p.salePriceMinor,
   219→      costPriceMinor: p.costPriceMinor,
   220→      stockQty: p.stockQty,
   221→      lowStockAt: p.lowStockAt,
   222→      isActive: p.isActive,
   223→      deletedAt: iso(p.deletedAt),
   224→      createdAt: p.createdAt.toISOString(),
   225→      updatedAt: p.updatedAt.toISOString(),
   226→    },
   227→    csv: [
   228→      p.id,
   229→      p.name,
   230→      p.sku,
   231→      p.unit,
   232→      formatMinorToDecimal(p.salePriceMinor, precision),
   233→      formatMinorToDecimal(p.costPriceMinor, precision),
   234→      p.stockQty,
   235→      p.lowStockAt,
   236→      p.isActive ? "true" : "false",
   237→      iso(p.deletedAt),
   238→      p.createdAt.toISOString(),
   239→      p.updatedAt.toISOString(),
   240→    ],
   241→  };
   242→}

   244→async function fetchWithCap<T>(
   245→  cap: number,
   246→  query: (take: number) => Promise<T[]>,
   247→): Promise<{ items: T[]; truncated: boolean }> {
   248→  // Fetch one row beyond the cap so truncation is DETECTABLE, never silent.
   249→  const items = await query(cap + 1);
   250→  return { items: items.slice(0, cap), truncated: items.length > cap };
   251→}
