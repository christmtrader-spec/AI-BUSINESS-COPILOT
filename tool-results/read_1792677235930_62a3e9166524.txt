   297→   297→function priorityClasses(priority: string): string {
   298→   298→  switch (priority) {
   299→   299→    case "URGENT":
   300→   300→      return "border-red-500/40 bg-red-500/10 text-red-700 dark:text-red-400";
   301→   301→    case "HIGH":
   302→   302→      return "border-amber-500/40 bg-amber-500/10 text-amber-700 dark:text-amber-400";
   303→   303→    case "LOW":
   304→   304→      return "border-muted-foreground/30 bg-muted/40 text-muted-foreground";
   305→   305→    default:
   306→   306→      return "";
   307→   307→  }

   352→   352→function TicketDetailBody({
   353→   353→  ticket,
   354→   354→  diagnostics,
   355→   355→  viewer,
   356→   356→  onChanged,
   357→   357→  onClose,
   358→   358→}: {
   359→   359→  ticket: PlatformSupportTicketDetail;
   360→   360→  diagnostics: PlatformTicketDiagnostics;
   361→   361→  viewer: PlatformTicketViewer;
   362→   362→  onChanged: () => void;
   363→   363→  onClose: () => void;
   364→   364→}) {
   365→   365→  const t = useTranslations("platform");
   366→   366→  const tError = usePlatformErrorText();
   367→   367→
   368→   368→  const [reply, setReply] = React.useState("");
   369→   369→  const [visibility, setVisibility] = React.useState<"CUSTOMER" | "INTERNAL">("CUSTOMER");
   370→   370→  const [busy, setBusy] = React.useState(false);
   371→   371→  const [error, setError] = React.useState<string | null>(null);
   372→   372→  const [resolveOpen, setResolveOpen] = React.useState(false);
   373→   373→
   374→   374→  const act = async (fn: () => Promise<unknown>, successKey: string) => {
   375→   375→    setBusy(true);
   376→   376→    setError(null);
   377→   377→    try {
   378→   378→      await fn();
   379→   379→      toast.success(t(successKey));
   380→   380→      onChanged();
   381→   381→    } catch (e) {
   382→   382→      setError(tError(e));
   383→   383→    } finally {
   384→   384→      setBusy(false);
   385→   385→    }
   386→   386→  };
   387→   387→
   388→   388→  const sendReply = () => {
   389→   389→    if (!reply.trim()) return;
   390→   390→    void act(async () => {
   391→   391→      await platformMutations.supportMessage(ticket.id, { body: reply, visibility });
   392→   392→      setReply("");
   393→   393→    }, "support.toastReplySent");
   394→   394→  };
   395→   395→
   396→   396→  return (
   397→   397→    <div className="flex min-h-0 flex-col gap-4">
   398→   398→      <DialogHeader>
   399→   399→        <DialogTitle className="flex flex-wrap items-center gap-2" dir="auto">
   400→   400→          <span className="font-mono text-sm text-muted-foreground">{ticket.reference}</span>
   401→   401→          {ticket.subject}
   402→   402→        </DialogTitle>
   403→   403→        <DialogDescription className="flex flex-wrap items-center gap-2">
   404→   404→          <StatusBadge status={ticket.status} label={t(`support.status.${ticket.status}` as Parameters<typeof t>[0])} />
   405→   405→          <Badge variant="outline" className={priorityClasses(ticket.priority)}>
   406→   406→            {t(`support.priority.${ticket.priority}` as Parameters<typeof t>[0])}
   407→   407→          </Badge>
   408→   408→          <Badge variant="outline">{t(`support.category.${ticket.category}` as Parameters<typeof t>[0])}</Badge>
   409→   409→          <span dir="ltr" className="text-xs">{ticket.requesterEmail}</span>
   410→   410→          {ticket.businessName ? <span className="text-xs" dir="auto">· {ticket.businessName}</span> : null}
   411→   411→        </DialogDescription>
   412→   412→      </DialogHeader>
   413→   413→
   414→   414→      <div className="min-h-0 flex-1 space-y-3 overflow-y-auto pe-1" role="log" aria-label={t("support.threadLabel")}>
   415→   415→        <div className="rounded-md border bg-muted/30 p-3">
   416→   416→          <p className="text-muted-foreground mb-1 text-xs font-medium">{t("support.originalRequest")}</p>
   417→   417→          <p className="whitespace-pre-wrap text-sm" dir="auto">{ticket.description}</p>
   418→   418→        </div>
   419→   419→
   420→   420→        {ticket.messages.map((m) => (
   421→   421→          <div
   422→   422→            key={m.id}
   423→   423→            className={
   424→   424→              m.visibility === "INTERNAL"
   425→   425→                ? "rounded-md border border-amber-500/40 bg-amber-500/10 p-3"
   426→   426→                : "rounded-md border p-3"
   427→   427→            }
   428→   428→          >
   429→   429→            <div className="mb-1 flex flex-wrap items-center gap-2">
   430→   430→              <Badge variant={m.authorKind === "OPERATOR" ? "default" : "outline"} className="text-[10px]">
   431→   431→                {m.authorKind === "OPERATOR" ? t("support.authorOperator") : t("support.authorRequester")}
   432→   432→              </Badge>
   433→   433→              {m.visibility === "INTERNAL" ? (
   434→   434→                <Badge variant="outline" className="border-amber-500/40 bg-amber-500/10 text-[10px] text-amber-700 dark:text-amber-400">
   435→   435→                  {t("support.internalNote")}

   712→   712→function CreateTicketDialog({
   713→   713→  open,

   721→   721→  const t = useTranslations("platform");
   722→   722→  const tError = usePlatformErrorText();
   723→   723→
   724→   724→  const [email, setEmail] = React.useState("");
   725→   725→  const [matches, setMatches] = React.useState<Array<{ id: string; email: string; name: string | null }>>([]);
   726→   726→  const [requesterId, setRequesterId] = React.useState("");
   727→   727→  const [businessId, setBusinessId] = React.useState("");
   728→   728→  const [businessQuery, setBusinessQuery] = React.useState("");
   729→   729→  const [businessMatches, setBusinessMatches] = React.useState<Array<{ id: string; name: string }>>([]);

   731→   731→  const [category, setCategory] = React.useState("TECHNICAL");
   732→   732→  const [priority, setPriority] = React.useState("NORMAL");
   733→   733→  const [description, setDescription] = React.useState("");
   734→   734→  const [busy, setBusy] = React.useState(false);
   735→   735→  const [error, setError] = React.useState<string | null>(null);
   736→   736→
   737→   737→  // Requester lookup — reuses the EXISTING platform users search (§21).

   740→   740→    const emailTrim = email.trim();
   741→   741→    if (emailTrim.length < 3) {

   748→   748→        .then((d) => setMatches(d.users.map((u) => ({ id: u.id, email: u.email, name: u.name }))))

   775→   775→      setEmail("");
   776→   776→      setMatches([]);
   777→   777→      setRequesterId("");
   778→   778→      setBusinessId("");
   779→   779→      setBusinessQuery("");

   806→   806→          priority,
   807→   807→        }),

   810→   810→        | { ok: true; data: { ticket: { id: string } } }
   811→   811→        | { ok: false; error: { code: string; message: string } };

   814→   814→          body.ok === false ? body.error.message : "Request failed.";
   815→   815→        setError(message || "Request failed.");
   816→   816→        return;
   817→   817→      }
   818→   818→      onCreated(body.data.ticket.id);
   819→   819→    } catch (e) {
   820→   820→      setError(tError(e));

   832→   832→        </DialogHeader>
   833→   833→        <div className="space-y-3">

   839→   839→              dir="ltr"

   841→   841→              onChange={(e) => {
   842→   842→                setEmail(e.target.value);
   843→   843→                setRequesterId("");
   844→   844→              }}
   845→   845→              placeholder="user@example.com"

   847→   847→            {matches.length > 0 ? (
   848→   848→              <div className="max-h-32 space-y-1 overflow-y-auto rounded-md border p-1">

   850→   850→                  <button

   862→   862→                ))}
   863→   863→              </div>
   864→   864→            ) : null}
   865→   865→          </div>
   866→   866→
   867→   867→          <div className="space-y-1.5">
