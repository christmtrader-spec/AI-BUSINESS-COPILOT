   177→/** Percent change between two minor-unit totals; null when undefined (÷0). */
   178→function pctChange(cur: number, prev: number): number | null {
   179→  if (prev === 0) return null;
   180→  return Math.round(((cur - prev) / prev) * 100);
   181→}

   195→function MoneyTooltip({
   196→  active,
   197→  payload,
   198→  label,
   199→  fmt,
   200→  labelFmt,
   201→}: {
   202→  active?: boolean;
   203→  payload?: TooltipEntry[];
   204→  label?: string | number;
   205→  fmt: (v: number) => string;
   206→  labelFmt?: (v: string) => string;
   207→}) {
   208→  if (!active || !payload || payload.length === 0) return null;
   209→  const labelText =
   210→    labelFmt && typeof label === "string" ? labelFmt(label) : label != null ? String(label) : "";
   211→  return (
   212→    <div className="bg-popover text-popover-foreground rounded-lg border px-3 py-2 text-xs shadow-md">
   213→      <p className="mb-1 font-medium" dir="auto">{labelText}</p>
   214→      {payload.map((entry) => (
   215→        <p key={String(entry.dataKey)} className="flex items-center gap-1.5 tabular-nums">
   216→          <span
   217→            className="size-2 shrink-0 rounded-[2px]"
   218→            style={{ background: entry.color }}
   219→            aria-hidden="true"
   220→          />
   221→          {entry.name}: <span dir="ltr">{fmt(entry.value ?? 0)}</span>
   222→        </p>
   223→      ))}
   224→    </div>
   225→  );
   226→}
