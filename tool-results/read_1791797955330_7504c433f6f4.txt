   364→   364→  acceptedAt: Date | null;
   365→   365→  rejectedAt: Date | null;

   373→   373→      ),

   375→   375→  const needs = (t: Date | null, s: SalesQuotationStatus[]) => {
   376→   376→    const required = s.includes(header.status);

   382→   382→  needs(header.rejectedAt, ["REJECTED"]);
   383→   383→  needs(header.cancelledAt, ["CANCELLED"]);
   384→   384→  needs(header.convertedAt, ["CONVERTED"]);
   385→   385→}

   389→   389→export async function createSalesQuotation(input: {
   390→   390→  businessId: string;
   391→   391→  userId: string;
   392→   392→  timezone: string; // AUTHENTICATED business timezone (server clock law §9)
   393→   393→  customerId?: string;
   394→   394→  reference?: string;
   395→   395→  quotationDate?: string; // calendar date (strict, zod) — default business-local today
   396→   396→  validUntilDate: string;
   397→   397→  expectedDeliveryDate?: string;
   398→   398→  discountMinor?: number;
   399→   399→  customerNotes?: string | null;
   400→   400→  internalNotes?: string | null;
   401→   401→  ip?: string | null;
   402→   402→  userAgent?: string | null;
   403→   403→}) {
   404→   404→  return withWriteTx(async (tx) => {
   405→   405→    // §10 — currency snapshot comes from the BUSINESS configuration
   406→   406→    // (server-derived); the client can never select a different currency.
   407→   407→    const business = await tx.business.findUnique({
   408→   408→      where: { id: input.businessId },
   409→   409→      select: { currencyCode: true, currencySymbol: true, decimalPrecision: true },
   410→   410→    });
   411→   411→    if (!business) throw new ApiError(404, "NOT_FOUND", "Business not found.");
   412→   412→
   413→   413→    // §10 — optional customer: tenant-scoped, not soft-deleted; the name
   414→   414→    // snapshot is frozen at creation (§12).
   415→   415→    let customerNameAtQuotation = "";
   416→   416→    if (input.customerId) {
   417→   417→      const customer = await loadActiveCustomer(tx, input.businessId, input.customerId);
   418→   418→      customerNameAtQuotation = customer.name;
   419→   419→    }
   420→   420→
   421→   421→    // §9 — dates: quotationDate defaults to the SERVER's business-local
   422→   422→    // today (never the browser clock); validUntilDate is required and
   423→   423→    // must not precede quotationDate (inclusive-until law).
   424→   424→    const quotationDate = input.quotationDate ?? businessLocalCalendarDate(new Date(), input.timezone);
   425→   425→    if (!isQuotationDateRangeValid(quotationDate, input.validUntilDate)) {
   426→   426→      throw new ApiError(
   427→   427→        422,
   428→   428→        SALES_QUOTATION_ERROR_CODES.INVALID_DATE_RANGE,
   429→   429→        "The validity date cannot be earlier than the quotation date.",
   430→   430→      );
   431→   431→    }
   432→   432→
   433→   433→    // §11 — a quotation with no lines has subtotal 0: a non-zero creation
   434→   434→    // discount would violate the discount law on day one (mirrors P45).
   435→   435→    if ((input.discountMinor ?? 0) > 0) {
   436→   436→      throw new ApiError(
   437→   437→        422,
   438→   438→        SALES_QUOTATION_ERROR_CODES.INVALID_DISCOUNT,
   439→   439→        "Add lines first — an empty quotation cannot carry a discount.",
   440→   440→      );
   441→   441→    }
   442→   442→
   443→   443→    // Reference: user-supplied (normalized) or server-generated (§10).
   444→   444→    let reference = input.reference;
   445→   445→    if (reference === undefined) {
   446→   446→      reference = await generateUniqueQuotationReference(tx, input.businessId);
   447→   447→    }
   448→   448→
   449→   449→    try {
   450→   450→      const row = await tx.salesQuotation.create({
   451→   451→        data: {
   452→   452→          businessId: input.businessId, // server-derived, never client (§2.4)
   453→   453→          customerId: input.customerId ?? null,
   454→   454→          customerNameAtQuotation, // frozen snapshot ("" when no customer)
   455→   455→          reference,
   456→   456→          status: "DRAFT", // commercial preparation only (§8)
   457→   457→          currencyCodeAtQuotation: business.currencyCode,
   458→   458→          currencySymbolAtQuotation: business.currencySymbol,
   459→   459→          decimalPrecisionAtQuotation: business.decimalPrecision,
   460→   460→          quotationDate: new Date(`${quotationDate}T12:00:00.000Z`),
   461→   461→          validUntilDate: new Date(`${input.validUntilDate}T12:00:00.000Z`),
   462→   462→          expectedDeliveryDate: input.expectedDeliveryDate
   463→   463→            ? new Date(`${input.expectedDeliveryDate}T12:00:00.000Z`)
   464→   464→            : null,
   465→   465→          discountMinor: 0,
   466→   466→          customerNotes: input.customerNotes ?? null, // document data only
   467→   467→          internalNotes: input.internalNotes ?? null, // staff-only document data
   468→   468→          createdByUserId: input.userId, // server-derived
   469→   469→        },
   470→   470→        select: { id: true, reference: true, status: true },
   471→   471→      });
   472→   472→
   473→   473→      await audit(
   474→   474→        auditInput(input.businessId, input.userId, "sales_quotation.create", row.id, row.reference, {
   475→   475→          customerId: input.customerId ?? null,
   476→   476→          customerNameAtQuotation,
   477→   477→          status: row.status,
   478→   478→          quotationDate,
   479→   479→          validUntilDate: input.validUntilDate,
   480→   480→        }),
   481→   481→        tx,
   482→   482→      );
   483→   483→
   484→   484→      return row;
   485→   485→    } catch (error) {
   486→   486→      // Database uniqueness stays authoritative for duplicates (§10).
   487→   487→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   488→   488→        throw new ApiError(
   489→   489→          409,
   490→   490→          SALES_QUOTATION_ERROR_CODES.DUPLICATE_REFERENCE,
   491→   491→          "A sales quotation with this reference already exists.",
