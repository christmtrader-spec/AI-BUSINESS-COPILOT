     4→ * Prompt 48 — supplier-bill service (server-only).
     5→ *
     6→ * LAWS:
     7→ *  - NO-DUPLICATE-PAYABLE (P48 §2): posting creates EXACTLY ONE linked
     8→ *    Expense inside the SAME transaction (§11). A DRAFT/CANCELLED bill has
     9→ *    NO Expense. The @unique expenseId constraint is the database-level

    11→ *  - THREE-WAY QUANTITY MATCHING (§7): billed ≤ received(POSTED receipts)
    12→ *    − returned(POSTED returns) − previouslyBilled(POSTED bills), computed

    14→ *    surfaced (500), never clamped. The posting transaction is the FINAL
    15→ *    authority; draft-time checks are advisory only (§7).

    18→ *    The ONLY new financial row is the linked Expense payable.

    20→ *    phase. CANCELLED is terminal. No generic status PATCH.
    21→ *  - Every query is tenant-scoped by ctx.businessId — never client input.
    22→ *    Foreign PO/PO-line/bill → 404, no existence leak.
    23→ *  - Snapshots are frozen at write time from ALREADY-FROZEN PO data and
    24→ *    NEVER overwritten from live rows (§5/§6).

    26→ *    ALWAYS server-derived; client totals are structurally rejected.
    27→ *  - internalNotes NEVER enter audit bodies, search, or exports (§16).
    28→ *  - Every mutation writes its audit INSIDE the same transaction (P48 §21);

    82→/**
    83→ * THE ONE mapper from pure domain failures to controlled HTTP results:
    84→ *  - corrupted persisted status → 500 INVALID_PERSISTED_STATUS (never Draft)
    85→ *  - inconsistent matching history → 500 CORRUPT_HISTORY (never clamped)
    86→ *  - exact-math overflow → 422 PRICE/TOTAL_OVERFLOW (never silent rounding)
    87→ * Exported for the PO detail route so corrupted aggregates surface the SAME
    88→ * stable codes there.
    89→ */
    90→export function mapPureSupplierBillDomainError(error: unknown): never {
    91→  if (error instanceof Error && error.name === "SupplierBillStatusIntegrityError") {
    92→    throw new ApiError(
    93→      500,
    94→      CODES.INVALID_PERSISTED_STATUS,
    95→      "Supplier bill data is in an invalid state. Please contact support.",
    96→    );
    97→  }
    98→  if (error instanceof Error && error.name === "SupplierBillMatchingIntegrityError") {
    99→    throw new ApiError(
   100→      500,
   101→      CODES.CORRUPT_HISTORY,
   102→      "Procurement or billing history is inconsistent. Please contact support.",
   103→    );
   104→  }
   105→  if (error instanceof Error) {
   106→    if (error.message === "SUPPLIER_BILL_PRICE_OVERFLOW") {
   107→      throw new ApiError(422, CODES.PRICE_OVERFLOW, "Line amount exceeds the safe integer range.");
   108→    }
   109→    if (error.message === "SUPPLIER_BILL_TOTAL_OVERFLOW") {
   110→      throw new ApiError(422, CODES.TOTAL_OVERFLOW, "Bill total exceeds the safe integer range.");
   111→    }
   112→    if (error.message === "SUPPLIER_BILL_INVALID_DISCOUNT") {
   113→      throw new ApiError(
   114→        422,
   115→        CODES.INVALID_DISCOUNT,
   116→        "Discount must be a non-negative amount not greater than the subtotal.",
   117→      );
   118→    }
   119→  }
   120→  throw error;
   121→}

   141→function assertTransition(from: SupplierBillStatus, to: SupplierBillStatus): void {
   142→  const allowed = from === "DRAFT" && (to === "POSTED" || to === "CANCELLED");
   143→  if (!allowed) {
   144→    if (from === "POSTED" && to === "POSTED") {
   145→      // §14-I — a repeated post after success is the ONE documented
   146→      // stable conflict, not a generic transition error.
   147→      throw new ApiError(409, CODES.ALREADY_POSTED, "This supplier bill has already been posted.");
   148→    }
   149→    throw new ApiError(
   150→      409,
   151→      CODES.INVALID_TRANSITION,
   152→      `Cannot move a supplier bill from ${from} to ${to}.`,
   153→    );
   154→  }

   186→/**
   187→ * Map a Prisma unique-violation to its ONE stable controlled code. Order
   188→ * matters: the composite invoice-reference index name CONTAINS the plain
   189→ * "reference" substring, so it is tested first.
   190→ */
   191→function mapUniqueViolation(error: unknown): ApiError | null {
   192→  if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   193→    const meta = error.meta as { target?: unknown } | undefined;
   194→    const target = Array.isArray(meta?.target)
   195→      ? (meta.target as string[]).join(",")
   196→      : String(meta?.target ?? "");
   197→    if (target.includes("supplierInvoiceReference")) {
   198→      return new ApiError(
   199→        409,
   200→        CODES.DUPLICATE_INVOICE_REFERENCE,
   201→        "This supplier already has a bill with this invoice reference.",
   202→      );
   203→    }
   204→    if (target.includes("purchaseOrderLineId")) {
   205→      return new ApiError(
   206→        409,
   207→        CODES.DUPLICATE_PO_LINE,
   208→        "Each purchase order line can appear at most once on a bill.",
   209→      );
   210→    }
   211→    if (target.includes("expenseId")) {
   212→      return new ApiError(
   213→        409,
   214→        CODES.ALREADY_LINKED,
   215→        "This payable is already linked to a supplier bill.",
   216→      );
   217→    }
   218→    if (target.includes("reference")) {
   219→      return new ApiError(409, CODES.DUPLICATE_REFERENCE, "This bill reference is already in use.");
   220→    }
   221→  }
   222→  return null;
   223→}

   304→/** received = SUM(POSTED GoodsReceiptLine.receivedQty) for the PO line. */
   305→async function sumPostedReceivedQty(tx: AggregateClient, purchaseOrderLineId: string): Promise<number> {
   306→  const agg = await tx.goodsReceiptLine.aggregate({
   307→    where: { purchaseOrderLineId, goodsReceipt: { status: "POSTED" } },
   308→    _sum: { receivedQty: true },
   309→  });
   310→  return agg._sum.receivedQty ?? 0;
   311→}
   312→
   313→/** returned = SUM(POSTED GoodsReturnLine.returnedQty) for the PO line. */
   314→async function sumPostedReturnedQty(tx: AggregateClient, purchaseOrderLineId: string): Promise<number> {
   315→  const agg = await tx.goodsReturnLine.aggregate({
   316→    where: {
   317→      goodsReceiptLine: { purchaseOrderLineId },
   318→      goodsReturn: { status: "POSTED" },
   319→    },
   320→    _sum: { returnedQty: true },
   321→  });
   322→  return agg._sum.returnedQty ?? 0;
   323→}

   325→/** billed = SUM(POSTED SupplierBillLine.billedQty), optionally excluding one bill. */
   326→async function sumPostedBilledQty(
   327→  tx: AggregateClient,
   328→  purchaseOrderLineId: string,
   329→  excludeBillId?: string,
   330→): Promise<number> {
   331→  const agg = await tx.supplierBillLine.aggregate({
   332→    where: {
   333→      purchaseOrderLineId,
   334→      supplierBill: {
   335→        status: "POSTED",
   336→        ...(excludeBillId ? { id: { not: excludeBillId } } : {}),
   337→      },
   338→    },
   339→    _sum: { billedQty: true },
   340→  });
   341→  return agg._sum.billedQty ?? 0;
   342→}

   344→/**
   345→ * THE ONE canonical billable-quantity read for a PO line: DB aggregates +
   346→ * the pure law. Returns the full matching breakdown for display AND the
   347→ * authoritative availability check.
   348→ */
   349→async function computeBillableForPoLine(
   350→  tx: AggregateClient,
   351→  poLine: { id: string; orderedQty: number },
   352→  excludeBillId?: string,
   353→): Promise<{
   354→  orderedQty: number;
   355→  receivedQty: number;
   356→  returnedQty: number;
   357→  previouslyBilledQty: number;
   358→  availableQty: number;
   359→}> {
   360→  const [receivedQty, returnedQty, previouslyBilledQty] = await Promise.all([
   361→    sumPostedReceivedQty(tx, poLine.id),
   362→    sumPostedReturnedQty(tx, poLine.id),
   363→    sumPostedBilledQty(tx, poLine.id, excludeBillId),
   364→  ]);
   365→  let availableQty: number;
   366→  try {
   367→    availableQty = computeAvailableBillableQty({
   368→      orderedQty: poLine.orderedQty,
   369→      receivedQty,
   370→      returnedQty,
   371→      previouslyBilledQty,
   372→    });
   373→  } catch (error) {
   374→    mapPureSupplierBillDomainError(error); // stable CORRUPT_HISTORY 500, never clamped
   375→  }
   376→  return { orderedQty: poLine.orderedQty, receivedQty, returnedQty, previouslyBilledQty, availableQty };
   377→}

   379→/** §7 precision: over-received vs over-returned vs plain over-bill. */
   380→function assertQtyWithinAvailability(
   381→  billedQty: number,
   382→  match: { receivedQty: number; returnedQty: number; previouslyBilledQty: number; availableQty: number },
   383→): void {
   384→  if (billedQty > match.receivedQty) {
   385→    throw new ApiError(
   386→      422,
   387→      CODES.QUANTITY_EXCEEDS_RECEIVED,
   388→      `Only ${match.receivedQty} unit(s) of this line have been received.`,
   389→      { availableQty: match.availableQty },
   390→    );
   391→  }
   392→  if (billedQty > match.availableQty) {
   393→    if (match.previouslyBilledQty === 0 && match.returnedQty > 0) {
   394→      throw new ApiError(
   395→        422,
   396→        CODES.RETURNED_QUANTITY,
   397→        `${match.returnedQty} unit(s) of this line were returned to the supplier and cannot be invoiced.`,
   398→        { availableQty: match.availableQty },
   399→      );
   400→    }
   401→    throw new ApiError(
   402→      422,
   403→      CODES.QUANTITY_EXCEEDS_AVAILABLE,
   404→      `Only ${match.availableQty} unit(s) remain available to invoice on this line.`,
   405→      {
   406→        availableQty: match.availableQty,
   407→        receivedQty: match.receivedQty,
   408→        returnedQty: match.returnedQty,
   409→        previouslyBilledQty: match.previouslyBilledQty,
   410→      },
   411→    );
   412→  }
   413→}

   417→/** Recompute + persist the bill's canonical totals from its current lines. */
   418→async function recomputeBillTotals(
   419→  tx: Tx,
   420→  billId: string,
   421→  discountMinor: number,
   422→): Promise<{ subtotalMinor: number; discountMinor: number; totalMinor: number; lineCount: number }> {
   423→  const lines = await tx.supplierBillLine.findMany({

   427→  let totals: { subtotalMinor: number; discountMinor: number; totalMinor: number; lineCount: number };
   428→  try {

   430→  } catch (error) {
   431→    mapPureSupplierBillDomainError(error);
   432→  }
   433→  await tx.supplierBill.update({
   434→    where: { id: billId },
   435→    data: {
   436→      subtotalMinor: totals.subtotalMinor,
   437→      discountMinor: totals.discountMinor,
   438→      totalMinor: totals.totalMinor,
   439→    },
   440→    select: { id: true },
   441→  });
   442→  return totals;
   443→}

   447→export async function createSupplierBill(input: {
   448→  businessId: string;
   449→  userId: string;
   450→  purchaseOrderId: string;
   451→  reference?: string;
   452→  supplierInvoiceReference?: string | null;
   453→  billDate: string; // YYYY-MM-DD (schema-validated calendar date)
   454→  dueDate?: string;
   455→  discountMinor?: number;
   456→  internalNotes?: string | null;
   457→  ip?: string | null;
   458→  userAgent?: string | null;
   459→}): Promise<SupplierBillDetailDto> {
   460→  const discountMinor = input.discountMinor ?? 0;
   461→  const billDate = calendarDateToUtcMidnoon(input.billDate);
   462→  const dueDate = input.dueDate ? calendarDateToUtcMidnoon(input.dueDate) : null;
   463→
   464→  // §5 — the PO is the procurement authority: tenant-scoped, billable
   465→  // status, and its FROZEN supplier/currency snapshots feed the bill.
   466→  const po = await db.purchaseOrder.findFirst({
   467→    where: { id: input.purchaseOrderId, businessId: input.businessId },
   468→    select: {
   469→      id: true,
   470→      reference: true,
   471→      status: true,
   472→      supplierId: true,
   473→      supplierNameAtOrder: true,
   474→      currencyCodeAtOrder: true,
   475→      currencySymbolAtOrder: true,
   476→      decimalPrecisionAtOrder: true,
   477→    },
   478→  });
   479→  if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   480→  assertPurchaseOrderBillable(po.status);
   481→
   482→  // §5 — early invoice-reference duplicate detection (the composite DB
   483→  // constraint stays the concurrency authority).
   484→  if (input.supplierInvoiceReference) {
   485→    const dup = await db.supplierBill.findFirst({
   486→      where: {
   487→        businessId: input.businessId,
   488→        supplierId: po.supplierId,
   489→        supplierInvoiceReference: input.supplierInvoiceReference,
   490→      },
   491→      select: { id: true, reference: true },
   492→    });
   493→    if (dup) {
   494→      throw new ApiError(
   495→        409,
   496→        CODES.DUPLICATE_INVOICE_REFERENCE,
   497→        "This supplier already has a bill with this invoice reference.",
   498→      );
   499→    }
   500→  }
   501→
   502→  const billId = await withWriteTx(async (tx) => {
   503→    const now = new Date();
   504→    let reference = input.reference ?? generateSupplierBillReference(now, (max) => randomInt(max));
   505→
   506→    for (let attempt = 0; attempt < 5; attempt += 1) {
   507→      try {
   508→        const bill = await tx.supplierBill.create({
   509→          data: {
   510→            businessId: input.businessId, // server-derived tenant scope
   511→            supplierId: po.supplierId, // derived from the PO (§5)
   512→            purchaseOrderId: po.id,
   513→            reference,
   514→            supplierInvoiceReference: input.supplierInvoiceReference ?? null,
   515→            // Snapshots frozen from the PO's ALREADY-FROZEN columns (§5).
   516→            supplierNameAtBill: po.supplierNameAtOrder,
   517→            purchaseOrderReferenceAtBill: po.reference,
   518→            currencyCodeAtBill: po.currencyCodeAtOrder,
   519→            currencySymbolAtBill: po.currencySymbolAtOrder,
   520→            decimalPrecisionAtBill: po.decimalPrecisionAtOrder,
   521→            status: "DRAFT",
   522→            billDate,
   523→            dueDate,
   524→            discountMinor,
   525→            subtotalMinor: 0,
   526→            totalMinor: 0,
   527→            internalNotes: input.internalNotes ?? null,
   528→            createdByUserId: input.userId, // server-derived creator
   529→          },
   530→          select: { id: true },
   531→        });
   532→        await audit(
   533→          auditInput(input.businessId, input.userId, "supplier_bill.create", bill.id, reference, {
   534→            purchaseOrderId: po.id,
   535→            purchaseOrderReferenceAtBill: po.reference,
   536→            supplierInvoiceReference: input.supplierInvoiceReference ?? null,
   537→            billDate: input.billDate,
   538→            dueDate: input.dueDate ?? null,
   539→            discountMinor,
   540→          }),
   541→          tx,
   542→        );
   543→        return bill.id;
   544→      } catch (error) {
   545→        const mapped = mapUniqueViolation(error);
   546→        if (mapped?.code === CODES.DUPLICATE_REFERENCE && !input.reference) {
   547→          // Generated-reference collision (astronomically unlikely) —
   548→          // regenerate; a USER-provided duplicate is a hard conflict.
   549→          reference = generateSupplierBillReference(new Date(), (max) => randomInt(max));
   550→          continue;
   551→        }
   552→        if (mapped) throw mapped;
   553→        throw error;
   554→      }
   555→    }
   556→    throw new ApiError(409, CODES.DUPLICATE_REFERENCE, "This bill reference is already in use.");
   557→  });
   558→
   559→  return getSupplierBill({ businessId: input.businessId, id: billId });
   560→}

   763→export async function updateSupplierBill(input: {
   764→  businessId: string;
   765→  userId: string;

   768→  billDate?: string;

   771→  internalNotes?: string | null;
   772→  ip?: string | null;
   773→  userAgent?: string | null;
   774→}): Promise<SupplierBillDetailDto> {
   775→  await withWriteTx(async (tx) => {
   776→    const header = await loadBillHeader(tx, input.businessId, input.id);
   777→    requireDraft(header.status); // §10 — DRAFT-only edit
   778→
   779→    const patch: Prisma.SupplierBillUncheckedUpdateInput = {};
   780→    if (input.supplierInvoiceReference !== undefined) {
   781→      const next = input.supplierInvoiceReference;
   782→      if (next) {

   784→        const dup = await tx.supplierBill.findFirst({
   785→          where: {
   786→            businessId: input.businessId,

   798→          );
   799→        }
   800→      }
   801→      patch.supplierInvoiceReference = next;
   802→    }
   803→    if (input.billDate !== undefined) patch.billDate = calendarDateToUtcMidnoon(input.billDate);
   804→    if (input.dueDate !== undefined) {
   805→      patch.dueDate = input.dueDate === null ? null : calendarDateToUtcMidnoon(input.dueDate);
   806→    }

   808→    if (input.internalNotes !== undefined) patch.internalNotes = input.internalNotes;
   809→
   810→    await tx.supplierBill.update({ where: { id: header.id }, data: patch });
   811→
   812→    // Discount changes recompute the canonical totals from current lines.
   813→    const discountMinor = input.discountMinor !== undefined ? input.discountMinor : header.discountMinor;
   814→    try {
   815→      await recomputeBillTotals(tx, header.id, discountMinor);
   816→    } catch (error) {
   817→      mapPureSupplierBillDomainError(error);
   818→    }
   819→
   820→    await audit(
   821→      auditInput(input.businessId, input.userId, "supplier_bill.update", header.id, header.reference, {
   822→        supplierInvoiceReference: input.supplierInvoiceReference ?? undefined,
   823→        billDate: input.billDate ?? undefined,
   824→        dueDate: input.dueDate === undefined ? undefined : input.dueDate,
   825→        discountMinor: input.discountMinor ?? undefined,
   826→      }),
   827→      tx,
   828→    );
   829→  });
   830→
   831→  return getSupplierBill({ businessId: input.businessId, id: input.id });
   832→}

  1620→    pricesByLine.set(row.purchaseOrderLineId, list);
  1621→  }
  1622→  const orderedByLine = new Map(poLineRows.map((l) => [l.id, l.orderedQty]));
  1623→  // Received/returned per line via the SAME batch aggregates the PO
  1624→  // receiving summary uses (§7 — POSTED receipts/returns only).
  1625→  const receivedGrouped = await db.goodsReceiptLine.groupBy({
  1626→    by: ["purchaseOrderLineId"],
  1627→    where: {

  1630→    },
  1631→    _sum: { receivedQty: true },
  1632→  });

  1634→    receivedGrouped.map((g) => [g.purchaseOrderLineId, g._sum.receivedQty ?? 0]),
  1635→  );
  1636→  // POSTED returns join their PO line THROUGH the source receipt line
  1637→  // (§7 — the authoritative relation); fetched as a bounded batch and
  1638→  // summed per PO line (never a per-line query).
  1639→  const returnRows = await db.goodsReturnLine.findMany({

  1642→      goodsReceiptLine: { purchaseOrderLineId: { in: input.purchaseOrderLineIds } },
  1643→    },
  1644→    select: {
  1645→      returnedQty: true,
  1646→      goodsReceiptLine: { select: { purchaseOrderLineId: true } },
  1647→    },
  1648→    take: 2000,
  1649→    orderBy: [{ createdAt: "asc" }, { id: "asc" }],
  1650→  });
  1651→  const returnedByLine = new Map<string, number>();
  1652→  for (const row of returnRows) {
  1653→    const poLineId = row.goodsReceiptLine.purchaseOrderLineId;
  1654→    returnedByLine.set(poLineId, (returnedByLine.get(poLineId) ?? 0) + row.returnedQty);
  1655→  }
  1656→  return input.purchaseOrderLineIds.map((lineId) => {
  1657→    const previouslyBilledQty = billedByLine.get(lineId) ?? 0;
  1658→    let availableBillableQty = 0;
  1659→    try {
  1660→      availableBillableQty = computeAvailableBillableQty({
  1661→        orderedQty: orderedByLine.get(lineId) ?? 0,
  1662→        receivedQty: receivedByLine.get(lineId) ?? 0,
  1663→        returnedQty: returnedByLine.get(lineId) ?? 0,
  1664→        previouslyBilledQty,
  1665→      });
  1666→    } catch (error) {
  1667→      mapPureSupplierBillDomainError(error); // stable CORRUPT_HISTORY 500
  1668→    }
  1669→    return {
  1670→      lineId,
  1671→      previouslyBilledQty,
  1672→      availableBillableQty,
  1673→      postedInvoicedUnitPricesMinor: pricesByLine.get(lineId) ?? [],
  1674→    };
  1675→  });
