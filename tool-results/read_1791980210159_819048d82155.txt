   197→   197→/** Percent change between two minor-unit totals; null when undefined (÷0). */
   198→   198→function pctChange(cur: number, prev: number): number | null {

   215→   215→function MoneyTooltip({
   216→   216→  active,
   217→   217→  payload,
   218→   218→  label,
   219→   219→  fmt,
   220→   220→  labelFmt,
   221→   221→}: {
   222→   222→  active?: boolean;
   223→   223→  payload?: TooltipEntry[];
   224→   224→  label?: string | number;
   225→   225→  fmt: (v: number) => string;
   226→   226→  labelFmt?: (v: string) => string;
   227→   227→}) {
   228→   228→  if (!active || !payload || payload.length === 0) return null;
   229→   229→  const labelText =
   230→   230→    labelFmt && typeof label === "string" ? labelFmt(label) : label != null ? String(label) : "";
   231→   231→  return (
   232→   232→    <div className="bg-popover text-popover-foreground rounded-lg border px-3 py-2 text-xs shadow-md">
   233→   233→      <p className="mb-1 font-medium" dir="auto">{labelText}</p>
   234→   234→      {payload.map((entry) => (
   235→   235→        <p key={String(entry.dataKey)} className="flex items-center gap-1.5 tabular-nums">
   236→   236→          <span
   237→   237→            className="size-2 shrink-0 rounded-[2px]"
   238→   238→            style={{ background: entry.color }}
   239→   239→            aria-hidden="true"
   240→   240→          />
   241→   241→          {entry.name}: <span dir="ltr">{fmt(entry.value ?? 0)}</span>
   242→   242→        </p>
   243→   243→      ))}
   244→   244→    </div>
   245→   245→  );
   246→   246→}
