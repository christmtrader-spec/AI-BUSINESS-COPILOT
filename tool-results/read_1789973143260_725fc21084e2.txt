   135→   135→function inline(text: string, keyPrefix: string): ReactNode[] {
   136→   136→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
   137→   137→  return parts.map((part, i) => {
   138→   138→    const key = `${keyPrefix}-${i}`;
   139→   139→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   140→   140→      return (
   141→   141→        <strong key={key} className="font-semibold">
   142→   142→          {part.slice(2, -2)}
   143→   143→        </strong>
   144→   144→      );
   145→   145→    }
   146→   146→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   147→   147→      return (
   148→   148→        <code
   149→   149→          key={key}
   150→   150→          dir="ltr"
   151→   151→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   152→   152→        >
   153→   153→          {part.slice(1, -1)}
   154→   154→        </code>
   155→   155→      );
   156→   156→    }
   157→   157→    return <span key={key}>{part}</span>;
   158→   158→  });
   159→   159→}

   161→   161→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   162→   162→  const nodes: ReactNode[] = [];
   163→   163→  let bullets: string[] = [];
   164→   164→  let para: string[] = [];
   165→   165→
   166→   166→  const flushBullets = () => {
   167→   167→    if (bullets.length === 0) return;
   168→   168→    nodes.push(
   169→   169→      <ul
   170→   170→        key={`ul-${nodes.length}`}
   171→   171→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   172→   172→      >
   173→   173→        {bullets.map((b, i) => (
   174→   174→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   175→   175→        ))}
   176→   176→      </ul>,
   177→   177→    );
   178→   178→    bullets = [];
   179→   179→  };
   180→   180→  const flushPara = () => {
   181→   181→    if (para.length === 0) return;
   182→   182→    nodes.push(
   183→   183→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   184→   184→        {inline(para.join("\n"), `p${nodes.length}`)}
   185→   185→      </p>,
   186→   186→    );
   187→   187→    para = [];
   188→   188→  };
   189→   189→
   190→   190→  for (const raw of text.split("\n")) {
   191→   191→    const line = raw.trimEnd();
   192→   192→    if (/^\s*[-*•]\s+/.test(line)) {
   193→   193→      flushPara();
   194→   194→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   195→   195→    } else if (line === "") {
   196→   196→      flushBullets();
   197→   197→      flushPara();
   198→   198→    } else {
   199→   199→      flushBullets();
   200→   200→      para.push(line);
   201→   201→    }
   202→   202→  }
   203→   203→  flushBullets();
   204→   204→  flushPara();
   205→   205→
   206→   206→  return (
   207→   207→    <div className="space-y-1.5 text-sm">
   208→   208→      {nodes}
   209→   209→      {streaming ? (
   210→   210→        <span
   211→   211→          aria-hidden="true"
   212→   212→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   213→   213→        />
   214→   214→      ) : null}
   215→   215→    </div>
   216→   216→  );
   217→   217→}

   328→   328→type FollowUpKey = (typeof FOLLOW_UP_KEYS)[number];
   329→   329→
   330→   330→const FOLLOW_UP_MATCHERS: Array<{ key: FollowUpKey; re: RegExp }> = [
   331→   331→  { key: "followUpOutstanding", re: /owe|outstanding|receivable|بدهکار|بدهی|طلب/i },
   332→   332→  { key: "followUpLowStock", re: /low.?stock|restock|inventory|موجودی|انبار|شارژ/i },
   333→   333→  { key: "followUpExpenses", re: /expense|spend|cost|هزینه|خرج/i },
   334→   334→  { key: "followUpSales", re: /sale|revenue|invoice|فروش|درآمد|فاکتور/i },
   335→   335→  { key: "followUpCash", re: /cash|payment|collected|نقد|پرداخت|دریافتی/i },
   336→   336→  { key: "followUpProducts", re: /product|best.?seller|top|محصول|پرفروش/i },
   337→   337→];
   338→   338→
   339→   339→/** Live server counts (same deterministic aggregates as the dashboard)
   340→   340→ *  used to RANK suggestions — a chip is only proposed when its subject
   341→   341→ *  actually exists in the data (r10: "only propose restock when there is
   342→   342→ *  something to restock"). */
   343→   343→type SummaryLite = {
   344→   344→  kpis?: { salesCount?: number; expensesCount?: number };
   345→   345→  outstanding?: { receivableCount?: number };
   346→   346→  inventory?: { lowStockCount?: number };
   347→   347→  topProducts?: unknown[];
   348→   348→};
   349→   349→
   350→   350→function followUpRelevant(
   351→   351→  key: FollowUpKey,
   352→   352→  counts: {
   353→   353→    receivableCount: number;
   354→   354→    lowStockCount: number;
   355→   355→    salesCount: number;
   356→   356→    expensesCount: number;
   357→   357→    topProducts: number;
   358→   358→  } | null,
   359→   359→): boolean {
   360→   360→  if (!counts) return true; // counts not loaded yet — keep the neutral default
   361→   361→  switch (key) {
   362→   362→    case "followUpOutstanding":
   363→   363→      return counts.receivableCount > 0;
   364→   364→    case "followUpLowStock":
   365→   365→      return counts.lowStockCount > 0;
   366→   366→    case "followUpExpenses":
   367→   367→      return counts.expensesCount > 0;
   368→   368→    case "followUpProducts":
   369→   369→      return counts.topProducts > 0;
   370→   370→    case "followUpSales":
   371→   371→      return true;
   372→   372→    case "followUpCash":
   373→   373→    case "followUpPayments":
   374→   374→    case "followUpSummary":
   375→   375→      return true;
   376→   376→  }
   377→   377→}
   378→   378→
   379→   379→function pickFollowUps(
   380→   380→  answer: string,
   381→   381→  conversationLength: number,
   382→   382→  counts: Parameters<typeof followUpRelevant>[1] = null,
   383→   383→): FollowUpKey[] {
   384→   384→  const lead = FOLLOW_UP_MATCHERS.find((m) => m.re.test(answer))?.key;
   385→   385→  const pool = FOLLOW_UP_KEYS.filter(
   386→   386→    (k) => k !== lead && followUpRelevant(k, counts),
   387→   387→  );
   388→   388→  const rotate = conversationLength % Math.max(1, pool.length);
   389→   389→  const rotated = [...pool.slice(rotate), ...pool.slice(0, rotate)];
   390→   390→  const picked = rotated.slice(0, lead ? 2 : 3);
   391→   391→  return lead ? [lead, ...picked] : picked;
   392→   392→}
