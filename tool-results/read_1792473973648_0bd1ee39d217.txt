     4→ * Onboarding — Prompt 25 evolution (incremental; the authoritative business
     5→ * creation POST /api/v1/businesses is UNCHANGED).
     6→ *
     7→ * Steps: Business (mandatory, never skippable) → Trial (OPTIONAL, only when
     8→ * the server reports eligibility) → Done (server-derived checklist + honest

    10→ * - A trial is NEVER created before the business exists (funnel order).

    12→ * - The checklist reads REAL server data — no client-asserted progress.
    13→ * - ONBOARDING_COMPLETED is once-per-business (server idempotent).
    14→ * - The signup intent (plan code only, allowlisted) may preselect the trial
    15→ *   offer — amounts/book ids are never carried across the funnel.

    49→const FALLBACK_TIMEZONES = ["UTC", "Asia/Tehran", "Asia/Dubai", "Europe/London", "Europe/Berlin", "America/New_York", "America/Los_Angeles", "Asia/Tokyo"];
    50→
    51→function timezoneOptions(): string[] {
    52→  try {
    53→    const list = Intl.supportedValuesOf("timeZone");
    54→    if (Array.isArray(list) && list.length > 0) return list;
    55→  } catch {
    56→    // older engines
    57→  }
    58→  return FALLBACK_TIMEZONES;
    59→}

    61→type PricingIntent = {
    62→  plan: string | null;
    63→  billingPeriod: "MONTHLY" | "YEARLY" | null;
    64→  currency: string | null;
    65→};
    66→
    67→/** Allowlisted signup intent (plan code ONLY — never amounts or book ids). */
    68→function readPricingIntent(): PricingIntent {
    69→  try {
    70→    const raw = window.sessionStorage.getItem("abc:pricingIntent");
    71→    if (!raw) return { plan: null, billingPeriod: null, currency: null };
    72→    const parsed = JSON.parse(raw) as Partial<PricingIntent>;
    73→    const planCodes = ["FREE", "STARTER", "BUSINESS", "PRO", "ENTERPRISE"];
    74→    return {
    75→      plan: typeof parsed.plan === "string" && planCodes.includes(parsed.plan) ? parsed.plan : null,
    76→      billingPeriod:
    77→        parsed.billingPeriod === "MONTHLY" || parsed.billingPeriod === "YEARLY"
    78→          ? parsed.billingPeriod
    79→          : null,
    80→      currency:
    81→        typeof parsed.currency === "string" && /^[A-Z]{3}$/.test(parsed.currency)
    82→          ? parsed.currency
    83→          : null,
    84→    };
    85→  } catch {
    86→    return { plan: null, billingPeriod: null, currency: null };
    87→  }
    88→}

   174→    }
   175→  }
   176→

   180→        method: "POST",
   181→        businessId: null,

   274→      method: "POST",

   297→    }

   554→                </li>
   555→                <li className="flex items-center gap-2">
   556→                  <CheckCircle2 className="size-4 text-emerald-600" aria-hidden="true" />
   557→                  {t("trialPoint2")}
   558→                </li>
   559→                <li className="flex items-center gap-2">
   560→                  <CheckCircle2 className="size-4 text-emerald-600" aria-hidden="true" />
   561→                  {t("trialPoint3")}
   562→                </li>
   563→              </ul>
   564→              <div className="flex flex-col-reverse gap-3 sm:flex-row sm:justify-end">
   565→                <Button variant="ghost" onClick={skipTrial} disabled={startTrialMutation.isPending}>
   566→                  {t("trialSkip")}
   567→                </Button>
   568→                <Button onClick={() => startTrialMutation.mutate()} disabled={startTrialMutation.isPending}>
   569→                  {startTrialMutation.isPending ? (
   570→                    <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   571→                  ) : (
   572→                    <Sparkles className="size-4" aria-hidden="true" />
   573→                  )}
   574→                  {t("trialStart", { days: eligibility.trialDays ?? 0 })}
   575→                </Button>
   576→              </div>
   577→            </CardContent>
   578→          </Card>
   579→        ) : null}
   580→
   581→        {step === "done" && created ? (
   582→          <Card className="shadow-sm">
   583→            <CardContent className="space-y-5 pt-6">
   584→              <div className="flex items-start gap-3">
   585→                <div className="bg-primary/10 flex size-10 shrink-0 items-center justify-center rounded-xl">
   586→                  <PartyPopper className="text-primary size-5" aria-hidden="true" />
   587→                </div>
   588→                <div>
   589→                  <h2 className="text-lg font-semibold">{t("doneTitle")}</h2>
   590→                  <p className="text-muted-foreground mt-1 text-sm">
   591→                    {t("doneSubtitle", { name: created.business.name })}
   592→                  </p>
   593→                </div>
   594→              </div>
   595→
   596→              {/* Server-derived checklist (counts only — honest state). */}
   597→              <ul className="grid gap-2 sm:grid-cols-2">
   598→                {status?.checklist ? (
   599→                  <>
   600→                    <ChecklistItem done={status.checklist.customersCreated} label={t("checklist.customers")} />
   601→                    <ChecklistItem done={status.checklist.productsCreated} label={t("checklist.products")} />
   602→                    <ChecklistItem done={status.checklist.salesCreated} label={t("checklist.sales")} />
   603→                    <ChecklistItem done={status.checklist.teamInvited} label={t("checklist.team")} />
   604→                    <ChecklistItem done={status.checklist.copilotUsed} label={t("checklist.copilot")} />
   605→                    <ChecklistItem
   606→                      done={!!status.trial?.active}
   607→                      label={t("checklist.trial")}
   608→                      optional
   609→                    />
   610→                  </>
   611→                ) : null}
   612→              </ul>
   613→              <p className="text-muted-foreground text-xs">{t("checklistNote")}</p>
   614→
   615→              <div className="flex justify-end">
   616→                <Button onClick={finish} className="w-full sm:w-auto">
   617→                  <ArrowRight className="size-4 rtl:rotate-180" aria-hidden="true" />
   618→                  {t("doneGo")}
   619→                </Button>
   620→              </div>
   621→            </CardContent>
   622→          </Card>
   623→        ) : null}
   624→      </div>
   625→    </div>
   626→  );
   627→}
