    68→// ---------------------------------------------------------------------------
    69→// Prompt 30 — email ownership verification (§4). ONLY the SHA-256 HASH of a
    70→// one-time verification token is stored; the raw token exists exclusively
    71→// inside the emailed link. Single active token per user is enforced in the
    72→// service layer (outstanding tokens are invalidated when a new one is issued,
    73→// §7). Purpose-separated from PasswordResetToken (§314/§315): a verification
    74→// token can never reset a password and vice versa.
    75→// ---------------------------------------------------------------------------
    76→model EmailVerificationToken {

    82→  usedAt      DateTime?

    86→
    87→  @@index([userId, createdAt])
    88→  @@index([expiresAt])

    91→// ---------------------------------------------------------------------------
    92→// Prompt 30 — TOTP MFA credential (§32). The TOTP secret is stored ENCRYPTED
    93→// (AES-256-GCM secret-box; key lives ONLY in the MFA_ENCRYPTION_KEY env var,
    94→// never in the database §33/§34). enabledAt NULL = pending setup (never

    96→// most one credential per user (§213).

    98→model MfaCredential {
    99→  id               String    @id @default(cuid())
   100→  userId           String    @unique
   101→  user             User      @relation(fields: [userId], references: [id], onDelete: Cascade)
   102→  encryptedSecret  String

   104→  secretVersion    Int       @default(1)

   106→  // Replay protection state (§45): the last TOTP time-step accepted at login.
   107→  // A code from the same (or older) step can never be accepted twice. NULL
   108→  // until the first successful login after enrollment.
   109→  lastUsedTotpStep Int?
   110→  createdAt        DateTime  @default(now())
   111→  updatedAt        DateTime  @updatedAt
   112→
   113→  @@index([userId, enabledAt])

   132→// ---------------------------------------------------------------------------
   133→// Prompt 30 — short-lived pre-auth MFA login challenge (§47). Password
   134→// success alone NEVER creates a session when MFA is enabled (§54): the
   135→// client receives only this bounded challenge. Hash-only storage; single
   136→// use claimed atomically (§50); bound to the user's sessionVersion so a
   137→// password change mid-challenge kills the challenge (§437/§438). The one-
   138→// time assertion hash (§53) is stored here after successful code
   139→// verification and consumed inside NextAuth authorize().
   140→// ---------------------------------------------------------------------------

   147→  usedAt                    DateTime?
   148→  attemptCount              Int       @default(0)
   149→  assertionHash             String?   @unique
   150→  assertionUsedAt           DateTime?
   151→  sessionVersionAtChallenge Int
   152→  createdAt                 DateTime  @default(now())
   153→
   154→  @@index([userId, createdAt])
   155→  @@index([expiresAt])

   526→// Conversations follow the user ACROSS DEVICES: scoped per (business, user) —
   527→// a user's questions are their own, never shared across members. The client
   528→// localStorage store remains the offline/fallback cache; the server copy is
   529→// NOT financial authority (answers are recomputed on every ask — A11-A13).
   530→// Session ids are client-generated (validated format) so local↔server
   531→// reconciliation is identity-based and idempotent.

   534→model ChatSession {

   541→  // User renamed via PATCH — after this the auto-title never overwrites it.
   542→  titleEdited Boolean       @default(false)
   543→  // Pinned conversations sort first in the history list (UI-only affordance).
   544→  pinned      Boolean       @default(false)
   545→  createdAt   DateTime      @default(now())
   546→  updatedAt   DateTime      @updatedAt
   547→  messages    ChatMessage[]

   549→  @@unique([businessId, userId, id]) // defense in depth: id alone is never enough
