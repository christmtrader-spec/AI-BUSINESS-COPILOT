    89→    POSTED: "bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/30",
    90→    CANCELLED: "bg-rose-500/10 text-rose-700 dark:text-rose-300 border-rose-500/30",
    91→  };
    92→  return (
    93→    <Badge variant="outline" className={cn("text-xs font-medium", styles[status])}>
    94→      {label}
    95→    </Badge>
    96→  );

   279→  /** §125-analog — download the CSV through the export endpoint. */
   280→  async function exportCsv() {
   281→    try {
   282→      const res = await fetch(`/api/v1/inventory/goods-receipts/${id}/export`, {
   283→        headers: { "x-business-id": businessId },
   284→      });
   285→      if (res.status === 429) {
   286→        toast({ title: t("exportRateLimited"), variant: "destructive" });
   287→        return;
   288→      }
   289→      if (res.status === 403) {
   290→        toast({ title: t("exportForbidden"), variant: "destructive" });
   291→        return;
   292→      }
   293→      if (!res.ok) {
   294→        toast({ title: t("exportFailed"), variant: "destructive" });
   295→        return;
   296→      }
   297→      const blob = await res.blob();
   298→      const disposition = res.headers.get("content-disposition") ?? "";
   299→      const match = /filename="([^"]+)"/.exec(disposition);
   300→      const url = URL.createObjectURL(blob);
   301→      const a = document.createElement("a");
   302→      a.href = url;
   303→      a.download = match?.[1] ?? "goods-receipt.csv";
   304→      a.click();
   305→      URL.revokeObjectURL(url);
   306→      toast({ description: t("exportToast") });
   307→    } catch {
   308→      toast({ title: t("exportFailed"), variant: "destructive" });
   309→    }
   310→  }

   499→            </div>
   500→          </div>
   501→
   502→          {data.lines.length === 0 ? (

   516→                    </span>
   517→                  ),
   518→                },
   519→                {
   520→                  id: "unit",
   521→                  header: t("unit"),
   522→                  hideBelow: "md",
   523→                  cell: (r: GoodsReceiptLineDto) => (

   525→                  ),
   526→                },
   527→                {
   528→                  id: "qty",
   529→                  header: t("receivedQty"),
   530→                  align: "end",
   531→                  cell: (r: GoodsReceiptLineDto) => (

   539→                  id: "returned",
   540→                  header: rt("returnedQty"),
   541→                  hideBelow: "md",
   542→                  align: "end",
   543→                  cell: (r: GoodsReceiptLineDto) => (

   545→                      {returnFactsById.get(r.id)?.returnedQty ?? 0}

   547→                  ),
   548→                },
   549→                {
   550→                  id: "returnable",
   551→                  header: rt("returnableQty"),

   554→                  cell: (r: GoodsReceiptLineDto) => (

   560→                ...(canEdit
   561→                  ? [
   562→                      {
   563→                        id: "actions",
   564→                        header: "",
   565→                        align: "end" as const,
   566→                        cell: (r: GoodsReceiptLineDto) => (

   578→                              type="button"
   579→                              variant="ghost"
   580→                              size="icon"

   582→                              onClick={() => setRemoveTarget(r)}
   583→                            >

   586→                          </div>
   587→                        ),
   588→                      },
   589→                    ]
   590→                  : []),
   591→              ]}
   592→              rows={data.lines}
   593→              getRowId={(r) => r.id}

   630→
   631→      {pickerOpen ? (

   643→      {addTarget ? (
   644→        <GoodsReceiptLineDialog
   645→          line={addTarget}
   646→          title={t("addLineAction")}

   650→          orderedLabel={t("ordered")}
   651→          receivedLabel={t("received")}
   652→          remainingLabel={t("remaining")}
   653→          invalidQtyMessage={t("invalidQty")}
   654→          exceedsMessage={t("exceedsRemaining")}

   657→              purchaseOrderLineId: addTarget.id,
   658→              receivedQty: value.receivedQty,

   672→          id: editTarget.line.purchaseOrderLineId,

   676→          unitAtOrder: editTarget.line.unitAtReceipt,

   678→          receivedQty: 0,
   679→          remainingQty: 0, // fully consumed elsewhere → only keep/reduce

   683→            line={dialogLine}

   687→            cancelLabel={tCommon("cancel")}
   688→            qtyLabel={t("receivedQty")}
   689→            orderedLabel={t("ordered")}
   690→            receivedLabel={t("received")}
   691→            remainingLabel={t("remaining")}
   692→            invalidQtyMessage={t("invalidQty")}
   693→            exceedsMessage={t("exceedsRemaining")}
   694→            onSubmit={(value) =>
   695→              updateLine.mutate({ lineId: editTarget.line.id, receivedQty: value.receivedQty })
   696→            }
   697→            onClose={() => setEditTarget(null)}
   698→          />
   699→        );
   700→      })() : null}
   701→
   702→      {removeTarget ? (
   703→        <Dialog open onOpenChange={(open) => !open && setRemoveTarget(null)}>
   704→          <DialogContent className="sm:max-w-md">
   705→            <DialogHeader>
   706→              <DialogTitle>{t("removeLineTitle")}</DialogTitle>
   707→              <DialogDescription dir="auto">
   708→                {t("removeLineDesc", { product: removeTarget.productNameAtReceipt })}
   709→              </DialogDescription>
   710→            </DialogHeader>
   711→            <DialogFooter className="gap-2">
   712→              <Button type="button" variant="outline" onClick={() => setRemoveTarget(null)}>
   713→                {tCommon("cancel")}

   715→              <Button
   716→                type="button"
   717→                variant="destructive"
   718→                onClick={() => removeLine.mutate(removeTarget.id)}
   719→                disabled={removeLine.isPending}
   720→              >
   721→                {t("removeAction")}

   836→      {/* Prompt 43 (§87) — create a return draft from THIS receipt. */}
   837→      {returnCreateOpen && data ? (

   845→          onClose={() => setReturnCreateOpen(false)}
   846→          onCreated={(returnId) => {
   847→            setReturnCreateOpen(false);
   848→            navigate({ kind: "inventory-goods-return", id: returnId });

   852→    </div>
