     4→ * PricingView — Prompt 25 §1/§4-§29/§56-§83/§111-§119/§126-§141/§149-§153.
     5→ *
     6→ * The PUBLIC pricing experience, rendered from ONE catalog endpoint
     7→ * (GET /api/v1/public/pricing — the PricingService underneath, §136). Works
     8→ * in two modes through the SAME component:
     9→ *  - public  (unauthenticated): standalone page with its own header/footer;
    10→ *  - in-app  (authenticated): rendered inside the shell; CTAs navigate to
    11→ *    the existing billing flow — this view NEVER mutates a subscription
    12→ *    (§120/§25) and never shows private usage (§13/§141).
    13→ *
    14→ * Honesty invariants:
    15→ * - EVERY amount comes from the payload (amountMinor + currency metadata,
    16→ *   formatted through formatMinor — §17). No price literal exists here.
    17→ * - Yearly is the STORED yearly row — never monthly × anything (§15/§137).
    18→ * - Savings are only shown when deterministically computable from the two
    19→ *   stored prices (§76) and labeled with the computed percent, never
    20→ *   "2 months free".
    21→ * - A currency without an active book shows "not available" — never zero,
    22→ *   never a converted estimate (§19/§23/§78).
    23→ * - CTA states reflect availability: FREE → signup, paid → signup (public)
    24→ *   or billing navigation (authenticated), ENTERPRISE → contact sales,
    25→ *   payments-disabled → contact us (§22/§24-§27).
    26→ */

    56→// Payload types (mirror of the server DTO — no internal ids exist in it)

    59→type PublicPlan = {
    60→  code: string;
    61→  name: string;
    62→  description: string;
    63→  sortOrder: number;
    64→  selfService: boolean;
    65→  availability: "AVAILABLE" | "NOT_PURCHASABLE" | "NOT_SELF_SERVICE" | "NOT_PRICED";
    66→  pricing: {
    67→    monthlyMinor: number;
    68→    yearlyMinor: number;
    69→    currency: string;
    70→    priceVersion: string;
    71→  } | null;
    72→  entitlements: Record<string, number | null | boolean>;
    73→};
    74→
    75→type PublicPricingPayload = {
    76→  plans: PublicPlan[];
    77→  definitions: Array<{ code: string; type: "LIMIT" | "BOOLEAN"; unit: string | null; group: string }>;
    78→  settlementCurrency: string;
    79→  currencies: Array<{
    80→    code: string;
    81→    displayName: string;
    82→    symbol: string;
    83→    minorUnitExponent: number;
    84→    available: boolean;
    85→  }>;
    86→  paymentsEnabled: boolean;
    87→  trialsEnabled: boolean;
    88→  trialDays: number | null;
    89→  featuredPlan: string | null;
    90→};
    91→
    92→type BillingPeriod = "MONTHLY" | "YEARLY";
    93→
    94→type Translator = ReturnType<typeof useTranslations>;

    96→const PENDING_PLAN_KEY = "abc:pendingPlan";
    97→
    98→/** Card-top feature codes (the full matrix lives in the comparison table). */
    99→const CARD_FEATURES = ["CUSTOMERS_MAX", "PRODUCTS_MAX", "TEAM_MEMBERS_MAX", "AI_REQUESTS_MAX"] as const;
   100→
   101→export function PricingView() {
   102→  const t = useTranslations("pricing");
   103→  const tBilling = useTranslations("billing");
   104→  const tCommon = useTranslations("common");
   105→  // Prompt 27 — legal page links in the footer.
   106→  const tLegal = useTranslations("legal");
   107→  const { status } = useSession();
   108→  const authenticated = status === "authenticated";
   109→  const locale = useAppStore((s) => s.locale);
   110→  const setAuthMode = useAppStore((s) => s.setAuthMode);
   111→  const navigate = useAppStore((s) => s.navigate);
   112→
   113→  const [period, setPeriod] = useState<BillingPeriod>("MONTHLY");
   114→  /** null = use the payload default; a code = an explicit visitor choice. */
   115→  const [currency, setCurrency] = useState<string | null>(null);
   116→  const [contactOpen, setContactOpen] = useState(false);
   117→
   118→  const { data, isLoading, isError, refetch, isFetching } = useQuery<PublicPricingPayload>({
   119→    queryKey: ["public-pricing", currency ?? ""],
   120→    queryFn: () =>
   121→      api<PublicPricingPayload>(`/api/v1/public/pricing${currency ? `?currency=${encodeURIComponent(currency)}` : ""}`, {
   122→        businessId: null,
   123→      }),
   124→    staleTime: 30_000,
   125→  });
   126→
   127→  // §13/§141 — the authenticated mode needs the CURRENT plan for CTA
   128→  // wording ("Current plan" / "Upgrade"). Server-verified like everywhere.
   129→  const { data: subscriptionPayload } = useSubscription(authenticated);
   130→  const currentPlanCode = authenticated ? (subscriptionPayload?.subscription.plan.code ?? null) : null;
   131→
   132→  // §49 — PRICING_VIEWED, once per currency per session (the server also
   133→  // dedupes per day; the ref guards re-render storms client-side).
   134→  const viewedFor = useRef<string | null>(null);
   135→  useEffect(() => {
   136→    if (!data) return;
   137→    const key = data.settlementCurrency;
   138→    if (viewedFor.current === key) return;
   139→    viewedFor.current = key;
   140→    trackClientEvent("PRICING_VIEWED", { page: "pricing", currency: key, locale });
   141→  }, [data, locale]);
   142→
   143→  // Document title for the public page (in-app, app-shell owns the title).
   144→  useEffect(() => {
   145→    if (authenticated) return;
   146→    document.title = t("seo.title");
   147→  }, [authenticated, t]);
   148→
   149→  const plans = useMemo(() => data?.plans ?? [], [data]);
   150→
   151→  /** Deterministic yearly savings from the two STORED prices (§76). */
   152→  function savingsPercent(plan: PublicPlan): number | null {
   153→    if (!plan.pricing) return null;
   154→    const { monthlyMinor, yearlyMinor } = plan.pricing;
   155→    const fullYear = monthlyMinor * 12;
   156→    if (fullYear <= 0 || yearlyMinor >= fullYear) return null;
   157→    const percent = Math.round((1 - yearlyMinor / fullYear) * 100);
   158→    return percent > 0 ? percent : null;
   159→  }
   160→
   161→  // §120/§25 — selection NEVER mutates a subscription: it records the funnel
   162→  // step and routes into the appropriate flow (signup or billing).
   163→  function selectPlan(plan: PublicPlan) {
   164→    trackClientEvent("PLAN_SELECTED", {
   165→      planCode: plan.code,
   166→      billingPeriod: period,
   167→      currency: data?.settlementCurrency,
   168→      page: "pricing",
   169→      locale,
   170→    });
   171→  }
   172→
   173→  function startSignup(planCode: string | null) {
   174→    try {
   175→      if (planCode) {
   176→        window.localStorage.setItem(PENDING_PLAN_KEY, JSON.stringify({ planCode, billingPeriod: period }));
   177→      }
   178→    } catch {
   179→      // storage unavailable — signup still works, the preselect is a nicety
   180→    }
   181→    setAuthMode("signup");
   182→    navigate({ kind: "dashboard" });
   183→  }
   184→
   185→  /** CTA model for one plan card (§24-§27). */
   186→  function ctaFor(plan: PublicPlan): {
   187→    label: string;
   188→    onClick?: () => void;
   189→    disabled?: boolean;
   190→    variant: "default" | "outline" | "secondary";
   191→  } {
   192→    const isCurrent = currentPlanCode === plan.code;
   193→
   194→    if (plan.code === "ENTERPRISE") {
   195→      return { label: t("availability.contactSales"), onClick: () => setContactOpen(true), variant: "outline" };
   196→    }
   197→
   198→    if (plan.availability === "NOT_PURCHASABLE") {
   199→      // FREE — signup when public; honest billing navigation when logged in.
   200→      if (!authenticated) {
   201→        return { label: t("availability.startFree"), onClick: () => { selectPlan(plan); startSignup("FREE"); }, variant: "default" };
   202→      }
   203→      return isCurrent
   204→        ? { label: t("availability.currentPlan"), disabled: true, variant: "secondary" }
   205→        : { label: t("availability.changePlan"), onClick: () => navigate({ kind: "settings:billing" }), variant: "outline" };
   206→    }
   207→
   208→    if (plan.availability !== "AVAILABLE" || !plan.pricing) {
   209→      // NOT_PRICED in the selected currency (e.g. IRR today) — §23/§78.
   210→      return { label: t("availability.notAvailable"), disabled: true, variant: "outline" };
   211→    }
   212→
   213→    if (!data?.paymentsEnabled) {
   214→      // §22 — never fake a checkout; the honest CTA is contact.
   215→      return { label: t("availability.contactUs"), onClick: () => setContactOpen(true), variant: "outline" };
   216→    }
   217→
   218→    if (!authenticated) {
   219→      return {
   220→        label: t("availability.choose", { plan: plan.code === "FREE" ? t("plans.FREE.name") : plan.name }),
   221→        onClick: () => { selectPlan(plan); startSignup(plan.code); },
   222→        variant: "default",
   223→      };
   224→    }
   225→
   226→    if (isCurrent) {
   227→      return { label: t("availability.currentPlan"), disabled: true, variant: "secondary" };
   228→    }
   229→
   230→    // §25 — navigation only; the billing flow owns every mutation.
   231→    const higher = plan.sortOrder > (plans.find((p) => p.code === currentPlanCode)?.sortOrder ?? 0);
   232→    return {
   233→      label: higher ? t("availability.upgrade") : t("availability.changePlan"),
   234→      onClick: () => navigate({ kind: "settings:billing" }),
   235→      variant: higher ? "default" : "outline",
   236→    };
   237→  }
   238→
   239→  /** One card's price line (§17 — formatMinor only, no manual math). */
   240→  function priceLine(plan: PublicPlan): React.ReactNode {
   241→    if (plan.code === "FREE") return <span className="text-3xl font-semibold">{t("plans.freePrice")}</span>;
   242→    if (plan.code === "ENTERPRISE") return <span className="text-3xl font-semibold">{t("plans.customPrice")}</span>;
   243→    if (!plan.pricing) {
   244→      return <span className="text-muted-foreground text-sm font-medium">{t("currency.unavailableShort")}</span>;
   245→    }
   246→    const amountMinor = period === "MONTHLY" ? plan.pricing.monthlyMinor : plan.pricing.yearlyMinor;
   247→    const suffix = period === "MONTHLY" ? t("toggle.perMonth") : t("toggle.perYear");
   248→    return (
   249→      <span className="flex items-baseline gap-1">
   250→        <span className="text-3xl font-semibold tabular-nums" dir="ltr">
   251→          {formatMinor(amountMinor, plan.pricing.currency, locale)}
   252→        </span>
   253→        <span className="text-muted-foreground text-sm">{suffix}</span>
   254→      </span>
   255→    );
   256→  }
   257→
   258→  // §126/§128 — catalog failed → honest error state, NEVER fallback prices.
   259→  if (isError) {
   260→    return (
   261→      <div className="bg-background flex min-h-svh flex-col" data-view="pricing-error">
   262→        <PublicHeader authenticated={authenticated} onContact={() => setContactOpen(true)} />
   263→        <main className="flex flex-1 items-center justify-center p-6">
   264→          <div className="mx-auto max-w-md space-y-4 text-center">
   265→            <div className="bg-destructive/10 text-destructive mx-auto flex size-12 items-center justify-center rounded-full" aria-hidden="true">
   266→              <Building2 className="size-6" />
   267→            </div>
   268→            <h1 className="text-xl font-semibold">{t("error.title")}</h1>
   269→            <p className="text-muted-foreground text-sm">{t("error.subtitle")}</p>
   270→            <Button onClick={() => void refetch()} variant="outline">
   271→              <RefreshCw className="size-4" aria-hidden="true" />
   272→              {t("error.retry")}
   273→            </Button>
   274→          </div>
   275→        </main>
   276→        <ContactSalesDialog open={contactOpen} onOpenChange={setContactOpen} />
   277→      </div>
   278→    );
   279→  }
   280→
   281→  const hasUnavailableCurrency = (data?.currencies ?? []).some((c) => !c.available);
   282→
   283→  return (
   284→    <div className="bg-background flex min-h-svh flex-col" data-view="pricing">
   285→      <PublicHeader authenticated={authenticated} onContact={() => setContactOpen(true)} />
   286→
   287→      <main className="flex-1">
   288→        {/* Hero */}
   289→        <section className="from-primary/5 via-primary/10 to-primary/5 border-b bg-gradient-to-b">
   290→          <div className="mx-auto w-full max-w-6xl space-y-6 px-4 py-12 text-center sm:px-6 lg:py-16">
   291→            <h1 className="text-3xl font-semibold tracking-tight gradient-text sm:text-4xl">
   292→              {t("hero.title")}
   293→            </h1>
   294→            <p className="text-muted-foreground mx-auto max-w-2xl text-base">{t("hero.subtitle")}</p>
   295→
   296→            {isLoading || !data ? (
   297→              <div className="text-muted-foreground flex items-center justify-center gap-2 py-6 text-sm" role="status">
   298→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   299→                {tCommon("loading")}
   300→              </div>
   301→            ) : (
   302→              <>
   303→                {/* Billing period toggle (§15) */}
   304→                <div
   305→                  className="bg-background mx-auto flex w-fit items-center gap-1 rounded-full border p-1 shadow-sm"
   306→                  role="tablist"
   307→                  aria-label={t("toggle.monthly")}
   308→                >
   309→                  {(["MONTHLY", "YEARLY"] as const).map((p) => (
   310→                    <button
   311→                      key={p}
   312→                      type="button"
   313→                      role="tab"
   314→                      aria-selected={period === p}
   315→                      onClick={() => setPeriod(p)}
   316→                      className={cn(
   317→                        "rounded-full px-5 py-2 text-sm font-medium transition-colors",
   318→                        period === p
   319→                          ? "bg-primary text-primary-foreground shadow-sm"
   320→                          : "text-muted-foreground hover:text-foreground",
   321→                      )}
   322→                    >
   323→                      {p === "MONTHLY" ? t("toggle.monthly") : t("toggle.yearly")}
   324→                    </button>
   325→                  ))}
   326→                </div>
   327→
   328→                {/* Currency selector (§20/§78) — ONLY server-declared currencies */}
   329→                <div className="mx-auto flex flex-col items-center gap-2">
   330→                  <span className="text-muted-foreground text-xs font-medium uppercase tracking-wide">
   331→                    {t("currency.label")}
   332→                  </span>
   333→                  <div className="flex flex-wrap items-center justify-center gap-2" role="group" aria-label={t("currency.label")}>
   334→                    {(data.currencies ?? []).map((c) => {
   335→                      const active = data.settlementCurrency === c.code;
   336→                      return (
   337→                        <button
   338→                          key={c.code}
   339→                          type="button"
   340→                          disabled={!c.available}
   341→                          aria-pressed={active}
   342→                          title={c.available ? c.displayName : t("currency.unavailable")}
   343→                          onClick={() => setCurrency(c.code)}
   344→                          className={cn(
   345→                            "rounded-full border px-4 py-1.5 text-sm font-medium transition-colors",
   346→                            active
   347→                              ? "border-primary bg-primary/10 text-primary"
   348→                              : "text-muted-foreground hover:border-foreground/20 hover:text-foreground",
   349→                            !c.available && "cursor-not-allowed opacity-50 line-through hover:text-muted-foreground",
   350→                          )}
   351→                        >
   352→                          {c.code}
   353→                          {!c.available ? <span className="ms-1.5 text-xs">({t("availability.notAvailable")})</span> : null}
   354→                        </button>
   355→                      );
   356→                    })}
   357→                  </div>
   358→                  {hasUnavailableCurrency ? (
   359→                    <p className="text-muted-foreground max-w-md text-xs">{t("currency.unavailable")}</p>
   360→                  ) : null}
   361→                </div>
   362→              </>
   363→            )}
   364→          </div>
   365→        </section>
   366→        {/* Plan cards (§69/§152 — semantic, price dominant, honest CTAs) */}
   367→        <section className="mx-auto w-full max-w-6xl px-4 py-10 sm:px-6" aria-label={t("compare.title")}>
   368→          {isLoading || !data ? null : (
   369→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5">
   370→              {plans.map((plan) => {
   371→                const cta = ctaFor(plan);
   372→                const save = period === "YEARLY" ? savingsPercent(plan) : null;
   373→                const featured = data.featuredPlan === plan.code;
   374→                return (
   375→                  <Card
   376→                    key={plan.code}
   377→                    className={cn(
   378→                      "relative flex flex-col transition-shadow duration-200 hover:shadow-md",
   379→                      featured && "border-primary/40 shadow-sm ring-1 ring-primary/20",
   380→                    )}
   381→                    data-plan={plan.code}
   382→                  >
   383→                    <CardContent className="flex flex-1 flex-col gap-4 p-5">
   384→                      <div className="space-y-1.5">
   385→                        <div className="flex items-center justify-between gap-2">
   386→                          <h2 className="text-base font-semibold">{planDisplayName(plan.code, t)}</h2>
   387→                          {featured ? (
   388→                            <Badge className="bg-primary/10 text-primary border-primary/20" variant="outline">
   389→                              <Sparkles className="size-3" aria-hidden="true" />
   390→                              {t("featured")}
   391→                            </Badge>
   392→                          ) : null}
   393→                        </div>
   394→                        <p className="text-muted-foreground min-h-[2.5rem] text-xs leading-relaxed">
   395→                          {planTagline(plan.code, t)}
   396→                        </p>
   397→                      </div>
   398→
   399→                      <div className="space-y-1">
   400→                        <div className="min-h-[2.75rem]">{priceLine(plan)}</div>
   401→                        {period === "YEARLY" && plan.pricing ? (
   402→                          <p className="text-muted-foreground text-xs">
   403→                            {t("toggle.billedYearly")}
   404→                            {save !== null ? (
   405→                              <span className="text-emerald-600 dark:text-emerald-400"> · {t("toggle.savePercent", { percent: save })}</span>
   406→                            ) : null}
   407→                          </p>
   408→                        ) : null}
   409→                      </div>
   410→
   411→                      <Button
   412→                        className="w-full"
   413→                        variant={cta.variant}
   414→                        disabled={cta.disabled}
   415→                        onClick={cta.onClick}
   416→                      >
   417→                        {cta.label}
   418→                      </Button>
   419→
   420→                      <ul className="space-y-2 border-t pt-3" aria-label={t("features.title")}>
   421→                        {CARD_FEATURES.map((code) => {
   422→                          const value = plan.entitlements[code];
   423→                          return (
   424→                            <li key={code} className="text-muted-foreground flex items-center justify-between gap-2 text-xs">
   425→                              <span>{tBilling(`entitlement.${code}` as "entitlement.CUSTOMERS_MAX")}</span>
   426→                              <span className="text-foreground font-medium">
   427→                                {limitLabel(value, plan.code, t)}
   428→                              </span>
   429→                            </li>
   430→                          );
   431→                        })}
   432→                        <li className="text-muted-foreground flex items-center justify-between gap-2 text-xs">
   433→                          <span>{tBilling("entitlement.AI_COPILOT" as "entitlement.AI_COPILOT")}</span>
   434→                          {plan.entitlements.AI_COPILOT ? (
   435→                            <BadgeCheck className="text-emerald-600 size-4 dark:text-emerald-400" aria-label={t("features.included")} />
   436→                          ) : (
   437→                            <Minus className="size-4" aria-label={t("features.notIncluded")} />
   438→                          )}
   439→                        </li>
   440→                      </ul>
   441→                    </CardContent>
   442→                  </Card>
   443→                );
   444→              })}
   445→            </div>
   446→          )}
   447→        </section>
   448→
   449→        {/* Comparison (§10/§14/§70/§138/§153) — desktop table, scrollable on mobile */}
   450→        {data && data.plans.length > 0 ? (
   451→          <section className="mx-auto w-full max-w-6xl px-4 pb-12 sm:px-6" aria-label={t("compare.title")}>
   452→            <div className="mb-4 space-y-1">
   453→              <h2 className="text-xl font-semibold tracking-tight">{t("compare.title")}</h2>
   454→              <p className="text-muted-foreground text-sm">{t("compare.subtitle")}</p>
   455→            </div>
   456→            <ComparisonTable payload={data} locale={locale} />
   457→          </section>
   458→        ) : null}
   459→
   460→        {/* FAQ (§111-§115) — the trial item ONLY when trials are enabled (§114) */}
   461→        {data ? (
   462→          <section className="border-t" aria-label={t("faq.title")}>
   463→            <div className="mx-auto w-full max-w-3xl space-y-4 px-4 py-12 sm:px-6">
   464→              <h2 className="text-xl font-semibold tracking-tight">{t("faq.title")}</h2>
   465→              <Accordion type="single" collapsible className="w-full">
   466→                {faqItems(data.trialsEnabled).map((key) => (
   467→                  <AccordionItem key={key} value={key}>
   468→                    <AccordionTrigger className="text-start text-sm font-medium">
   469→                      {/* Flat keys: faq.q1 / faq.a1 (BUGFIX #17 — the raw
   470→                          key path rendered because faq.q1.q never existed). */}
   471→                      {t(`faq.${key}` as "faq.q1")}
   472→                    </AccordionTrigger>
   473→                    <AccordionContent className="text-muted-foreground text-sm leading-relaxed">
   474→                      {t(`faq.a${key.slice(1)}` as "faq.a1")}
   475→                    </AccordionContent>
   476→                  </AccordionItem>
   477→                ))}
   478→              </Accordion>
   479→            </div>
   480→          </section>
   481→        ) : null}
   482→      </main>
   483→

   579→/**
   580→ * Comparison matrix. Desktop: semantic table, plans as columns (§14).
   581→ * Small screens: horizontally scrollable container with a sticky feature
   582→ * column — no shrunken text, no clipped columns (§153).
   583→ */
   584→function ComparisonTable({ payload, locale }: { payload: PublicPricingPayload; locale: string }) {
   585→  const t = useTranslations("pricing");
   586→  const tBilling = useTranslations("billing");
   587→
   588→  const groups = useMemo(() => {
   589→    const byGroup = new Map<string, typeof payload.definitions>();
   590→    for (const def of payload.definitions) {
   591→      const list = byGroup.get(def.group) ?? [];
   592→      list.push(def);
   593→      byGroup.set(def.group, list);
   594→    }
   595→    return [...byGroup.entries()];
   596→  }, [payload.definitions]);
   597→
   598→  return (
   599→    <div className="bg-background overflow-hidden rounded-xl border shadow-sm">
   600→      <div className="overflow-x-auto" dir="ltr">
   601→        <table className="w-full min-w-[640px] border-collapse text-sm" dir={locale === "fa" ? "rtl" : "ltr"}>
   602→          <caption className="sr-only">{t("compare.title")}</caption>
   603→          <thead>
   604→            <tr className="bg-muted/40 border-b">
   605→              <th scope="col" className="sticky start-0 bg-muted/40 z-10 w-2/5 p-3 text-start font-medium">
   606→                {t("compare.title")}
   607→              </th>
   608→              {payload.plans.map((p) => (
   609→                <th key={p.code} scope="col" className="p-3 text-center font-semibold">
   610→                  {planDisplayName(p.code, t)}
   611→                  <span className="text-muted-foreground block text-xs font-normal tabular-nums">
   612→                    {p.pricing
   613→                      ? formatMinor(
   614→                          payload.plans && p.pricing ? p.pricing.monthlyMinor : 0,
   615→                          p.pricing.currency,
   616→                          locale,
   617→                        )
   618→                      : p.code === "FREE"
   619→                        ? t("plans.freePrice")
   620→                        : t("plans.customPrice")}
   621→                  </span>
   622→                </th>
   623→              ))}
   624→            </tr>
   625→          </thead>
   626→          <tbody>
   627→            {groups.map(([group, defs]) => (
   628→              <Fragment key={`${group}-header`}>
   629→                <tr className="bg-muted/20">
   630→                  <td
   631→                    colSpan={payload.plans.length + 1}
   632→                    className="text-muted-foreground sticky start-0 p-2 text-xs font-semibold uppercase tracking-wide"
   633→                  >
   634→                    {t(`compare.groups.${group}` as "compare.groups.business")}

   638→                  <tr key={def.code} className="border-t">
   639→                    <th scope="row" className="bg-background sticky start-0 z-10 p-3 text-start font-normal">
   640→                      {tBilling(`entitlement.${def.code}` as "entitlement.CUSTOMERS_MAX")}
   641→                    </th>
   642→                    {payload.plans.map((p) => (
   643→                      <td key={p.code} className="text-center p-3 tabular-nums">
   644→                        <Cell plan={p} code={def.code} type={def.type} />
   645→                      </td>
   646→                    ))}
   647→                  </tr>
   648→                ))}
   649→              </Fragment>
   650→            ))}
   651→          </tbody>
   652→        </table>
   653→      </div>
   654→    </div>

   658→function Cell({
   659→  plan,
   660→  code,
   661→  type,
   662→}: {
   663→  plan: PublicPlan;
   664→  code: string;
   665→  type: "LIMIT" | "BOOLEAN";
   666→}) {
   667→  const t = useTranslations("pricing");

   681→
   682→  return <LimitValue value={value} planCode={plan.code} />;

   685→function LimitValue({ value, planCode }: { value: number | null | boolean; planCode: string }) {
   686→  const t = useTranslations("pricing");
   687→  if (typeof value === "boolean" || value === undefined) return <span>—</span>;
   688→  if (value === null) {
   689→    // §11 — "Unlimited" only when the catalog truly defines unlimited;
   690→    // ENTERPRISE nulls are negotiated, displayed as Custom (§83).
   691→    if (planCode === "ENTERPRISE") return <span>{t("features.custom")}</span>;
   692→    return <span>{t("features.unlimited")}</span>;
   693→  }
   694→  return (
   695→    <span>
   696→      {new Intl.NumberFormat().format(value)}
   697→    </span>
   698→  );
   699→}

   701→function limitLabel(
   702→  value: number | null | boolean,
   703→  planCode: string,
   704→  t: Translator,
   705→): string {
   706→  if (typeof value === "boolean" || value === undefined) return "—";
   707→  if (value === null) return planCode === "ENTERPRISE" ? t("features.custom") : t("features.unlimited");
   708→  return t("features.upTo", { count: new Intl.NumberFormat().format(value) });
   709→}

   740→/**
   741→ * §58/§59/§135 — structured data ONLY from the live catalog: one Product
   742→ * with Offers exclusively for plans that are genuinely AVAILABLE in the
   743→ * payload's currency. No IRR offers, no stale prices, no fabricated data.
   744→ */
   745→function JsonLd({ payload, locale }: { payload: PublicPricingPayload | null; locale: string }) {
   746→  const json = useMemo(() => {
   747→    if (!payload) return null;
   748→    const offers = payload.plans
   749→      .filter((p) => p.availability === "AVAILABLE" && p.pricing)
   750→      .map((p) => {
   751→        const digits = payload.currencies.find((c) => c.code === p.pricing!.currency)?.minorUnitExponent ?? 2;
   752→        return {
   753→          "@type": "Offer",
   754→          name: p.name,
   755→          price: (p.pricing!.monthlyMinor / 10 ** digits).toFixed(digits),
   756→          priceCurrency: p.pricing!.currency,
   757→          availability: "https://schema.org/InStock",
   758→        };
   759→      });
   760→    if (offers.length === 0) return null;
   761→    return {
   762→      "@context": "https://schema.org",
   763→      "@type": "Product",
   764→      name: "AI Business Copilot",
   765→      description: locale === "fa" ? "مدیریت کسب‌وکار با دستیار هوشمند" : "Business management with an AI copilot",
   766→      offers,
   767→    };
   768→  }, [payload, locale]);
   769→
   770→  if (!json) return null;
   771→  return (
   772→    <script
   773→      type="application/ld+json"
   774→      // Data comes exclusively from the typed catalog above — no user input.
   775→      dangerouslySetInnerHTML={{ __html: JSON.stringify(json) }}
   776→    />
   777→  );
   778→}
