     3→/**
     4→ * Prompt 40 — stocktake service (server-only, §118).
     5→ *
     6→ * LAWS:
     7→ *  - Every query is tenant-scoped by ctx.businessId (§54) — never a client
     8→ *    businessId. Foreign stocktake/product → 404, no existence leakage (§55).
     9→ *  - Status transitions are server-authoritative through dedicated
    10→ *    operations (§9); invalid transitions → 422 INVALID_STOCKTAKE_TRANSITION.
    11→ *  - systemQtyAtCount is ALWAYS the server-read Product.stockQty at count
    12→ *    time (§15); the client supplies ONLY countedQty (§16); variance is
    13→ *    server-derived (§16/§18).
    14→ *  - Finalization is ONE atomic transaction (§39): stale-count check against
    15→ *    CURRENT stockQty (§40/§41 — never auto-rebased), InventoryAdjustment
    16→ *    rows for non-zero variances (§43), zero-variance lines skip the
    17→ *    adjustment (§44), no partial finalization (§156).
    18→ *  - FINALIZED/CANCELLED are immutable (§50/§51).
    19→ *  - Audit through the single sanctioned helper inside the same transaction
    20→ *    (§71); NO audit on reads (§69); safe metadata only (§70/§140).
    21→ */

    45→/* ---------------------------- shared internals ----------------------------- */
    46→
    47→function assertTransition(from: StocktakeStatus, to: StocktakeStatus): void {
    48→  if (!canTransitionStocktake(from, to)) {
    49→    throw new ApiError(
    50→      422,
    51→      STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
    52→      `Cannot move a stocktake from ${from} to ${to}.`,
    53→    );
    54→  }
    55→}

    57→async function loadStocktake(
    58→  tx: Tx,
    59→  businessId: string,
    60→  id: string,
    61→): Promise<{
    62→  id: string;
    63→  reference: string;
    64→  status: StocktakeStatus;
    65→  countType: string;
    66→  notes: string | null;
    67→  createdByUserId: string;
    68→  startedAt: Date | null;
    69→  reviewedAt: Date | null;
    70→  finalizedAt: Date | null;
    71→  canceledAt: Date | null;
    72→  createdAt: Date;
    73→}> {
    74→  const row = await tx.inventoryStocktake.findFirst({
    75→    where: { id, businessId }, // §54 — tenant scope from auth context only
    76→    select: {
    77→      id: true,
    78→      reference: true,
    79→      status: true,
    80→      countType: true,
    81→      notes: true,
    82→      createdByUserId: true,
    83→      startedAt: true,
    84→      reviewedAt: true,
    85→      finalizedAt: true,
    86→      canceledAt: true,
    87→      createdAt: true,
    88→    },
    89→  });
    90→  if (!row) throw new ApiError(404, "NOT_FOUND", "Stocktake not found.");
    91→  return { ...row, status: isStocktakeStatus(row.status) ? row.status : "DRAFT" };
    92→}

    94→function auditInput(
    95→  businessId: string,
    96→  actorId: string,
    97→  action: string,
    98→  stocktakeId: string,
    99→  reference: string,
   100→  after: Record<string, unknown>,
   101→  before?: Record<string, unknown>,
   102→) {
   103→  return {
   104→    businessId,
   105→    actorId,
   106→    action,
   107→    entityType: "InventoryStocktake",
   108→    entityId: stocktakeId,
   109→    ...(before ? { before } : {}),
   110→    after: { reference, ...after }, // §70 — safe metadata; never notes content
   111→  };
   112→}

   116→export async function createStocktake(input: {
   117→  businessId: string;
   118→  userId: string;
   119→  reference: string;
   120→  notes: string | null;
   121→  countType?: string;
   122→  ip?: string | null;
   123→  userAgent?: string | null;
   124→}) {
   125→  return db.$transaction(async (tx) => {
   126→    try {
   127→      const row = await tx.inventoryStocktake.create({
   128→        data: {
   129→          businessId: input.businessId,
   130→          reference: input.reference,
   131→          notes: input.notes,
   132→          countType: "SELECTED", // §24 — only SELECTED exists
   133→          status: "DRAFT",
   134→          createdByUserId: input.userId,
   135→        },
   136→        select: { id: true, reference: true, status: true },
   137→      });
   138→      await audit(
   139→        auditInput(input.businessId, input.userId, "inventory.stocktake.create", row.id, row.reference, {
   140→          status: row.status,
   141→          countType: "SELECTED",
   142→        }),
   143→        tx,
   144→      );
   145→      return row;
   146→    } catch (error) {
   147→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   148→        throw new ApiError(409, "DUPLICATE_REFERENCE", "A stocktake with this reference already exists.");
   149→      }
   150→      throw error;
   151→    }
   152→  });
   153→}

   155→/* --------------------------------- list ----------------------------------- */
   156→
   157→export async function listStocktakes(input: {
   158→  businessId: string;
   159→  q?: string;
   160→  status?: string;
   161→  from?: string;
   162→  to?: string;
   163→  page: number;
   164→  pageSize: number;
   165→}) {
   166→  const where: Prisma.InventoryStocktakeWhereInput = { businessId: input.businessId };
   167→  if (input.q) where.reference = { contains: input.q };
   168→  if (input.status && isStocktakeStatus(input.status)) where.status = input.status;
   169→  if (input.from || input.to) {
   170→    where.createdAt = {};
   171→    if (input.from) where.createdAt.gte = new Date(`${input.from}T00:00:00.000Z`);
   172→    if (input.to) where.createdAt.lte = new Date(`${input.to}T23:59:59.999Z`);
   173→  }
   174→
   175→  const [rows, total] = await Promise.all([
   176→    db.inventoryStocktake.findMany({
   177→      where,
   178→      orderBy: { createdAt: "desc" }, // deterministic newest-first list (§134)
   179→      skip: (input.page - 1) * input.pageSize,
   180→      take: input.pageSize,
   181→      // §67 — list select minimization; a bounded line slice feeds the
   182→      // counted/net columns without loading full line tables.
   183→      select: {
   184→        id: true,
   185→        reference: true,
   186→        status: true,
   187→        countType: true,
   188→        createdAt: true,
   189→        createdByUser: { select: { id: true, name: true, email: true } },
   190→        _count: { select: { lines: true } },
   191→        lines: { select: { countedAt: true, varianceQty: true } },
   192→      },
   193→    }),
   194→    db.inventoryStocktake.count({ where }),
   195→  ]);
   196→
   197→  return {
   198→    items: rows.map(stocktakeListItemDto),
   199→    total,
   200→    page: input.page,
   201→    pageSize: input.pageSize,
   202→  };
   203→}

   205→/* -------------------------------- detail ---------------------------------- */
   206→
   207→export async function getStocktake(input: {
   208→  businessId: string;
   209→  id: string;
   210→  page: number;
   211→  pageSize: number;
   212→}) {
   213→  const header = await loadStocktake(db, input.businessId, input.id);
   214→  // §36 — summary over ALL lines (aggregate-light select, §66).
   215→  const [summaryRows, lineRows, lineTotal] = await Promise.all([
   216→    db.inventoryStocktakeLine.findMany({
   217→      where: { stocktakeId: header.id },
   218→      select: { countedAt: true, varianceQty: true },
   219→    }),
   220→    db.inventoryStocktakeLine.findMany({
   221→      where: { stocktakeId: header.id },
   222→      orderBy: [{ countedAt: "asc" }, { id: "asc" }], // deterministic ordering
   223→      skip: (input.page - 1) * input.pageSize,
   224→      take: input.pageSize,
   225→      select: {
   226→        id: true,
   227→        productId: true,
   228→        productNameAtCount: true,
   229→        skuAtCount: true,
   230→        unitAtCount: true,
   231→        systemQtyAtCount: true,
   232→        countedQty: true,
   233→        varianceQty: true,
   234→        countedAt: true,
   235→        countedByUser: { select: { id: true, name: true, email: true } },
   236→      },
   237→    }),
   238→    db.inventoryStocktakeLine.count({ where: { stocktakeId: header.id } }),
   239→  ]);
   240→
   241→  return stocktakeDetailDto({
   242→    ...header,
   243→    createdByUser: await db.user.findUnique({
   244→      where: { id: header.createdByUserId },
   245→      select: { id: true, name: true, email: true },
   246→    }),
   247→    summary: computeStocktakeSummary(summaryRows),
   248→    lines: lineRows,
   249→    linesPage: { page: input.page, pageSize: input.pageSize, total: lineTotal },
   250→  });
   251→}

   253→/* ------------------------------- add line --------------------------------- */
   254→
   255→export async function addStocktakeLine(input: {
   256→  businessId: string;
   257→  userId: string;
   258→  stocktakeId: string;
   259→  productId: string;
   260→  ip?: string | null;
   261→  userAgent?: string | null;
   262→}) {
   263→  const cap = stocktakeLineCap(process.env.INVENTORY_STOCKTAKE_MAX_LINES);
   264→  return db.$transaction(async (tx) => {
   265→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   266→    if (stocktake.status !== "DRAFT") {
   267→      // §27 — DRAFT-only line preparation.
   268→      throw new ApiError(
   269→        422,
   270→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   271→        "Products can only be added while the stocktake is a draft.",
   272→      );
   273→    }
   274→    const lineCount = await tx.inventoryStocktakeLine.count({
   275→      where: { stocktakeId: stocktake.id },
   276→    });
   277→    if (lineCount >= cap) {
   278→      // §61 — bounded transaction sizes; surfaced to the UI (lines/max).
   279→      throw new ApiError(
   280→        422,
   281→        STOCKTAKE_ERROR_CODES.LINE_LIMIT,
   282→        `Stocktake line limit reached (${cap}).`,
   283→      );
   284→    }
   285→    // §20/§21/§56/§57 — same business, not deleted, active. Foreign/deleted
   286→    // product → 404 (no existence leakage); inactive → documented 422.
   287→    const product = await tx.product.findFirst({
   288→      where: { id: input.productId, businessId: input.businessId },
   289→      select: { id: true, deletedAt: true, isActive: true },
   290→    });
   291→    if (!product || product.deletedAt) {
   292→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   293→    }
   294→    if (!product.isActive) {
   295→      throw new ApiError(
   296→        422,
   297→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   298→        "Inactive products cannot be added to a stocktake.",
   299→      );
   300→    }
   301→    try {
   302→      const line = await tx.inventoryStocktakeLine.create({
   303→        data: {
   304→          stocktakeId: stocktake.id,
   305→          productId: product.id,
   306→          // Line is PREPARATION data until counted (§29) — zero placeholders.
   307→          systemQtyAtCount: 0,
   308→          countedQty: 0,
   309→          varianceQty: 0,
   310→          countedAt: new Date(0),
   311→          countedByUserId: input.userId,
   312→          productNameAtCount: "",
   313→          skuAtCount: null,
   314→          unitAtCount: null,
   315→        },
   316→        select: { id: true, productId: true },
   317→      });
   318→      await audit(
   319→        auditInput(input.businessId, input.userId, "inventory.stocktake.line_add", stocktake.id, stocktake.reference, {
   320→          lineId: line.id,
   321→          productId: product.id,
   322→        }),
   323→        tx,
   324→      );
   325→      return line;
   326→    } catch (error) {
   327→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   328→        // §28/§48 — DB uniqueness is the final protection.
   329→        throw new ApiError(
   330→          409,
   331→          STOCKTAKE_ERROR_CODES.DUPLICATE_PRODUCT,
   332→          "This product is already in the stocktake.",
   333→        );
   334→      }
   335→      throw error;
   336→    }
   337→  });
   338→}

   376→/* --------------------------------- count ---------------------------------- */
   377→
   378→export async function countStocktakeLine(input: {
   379→  businessId: string;
   380→  userId: string;
   381→  stocktakeId: string;
   382→  lineId: string;
   383→  countedQty: number;
   384→  ip?: string | null;
   385→  userAgent?: string | null;
   386→}) {
   387→  return db.$transaction(async (tx) => {
   388→    // §30 — load stocktake → verify status (COUNTING|REVIEW, §31) → verify
   389→    // line → verify product tenant/active → read current stock → snapshot.
   390→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   391→    if (stocktake.status !== "COUNTING" && stocktake.status !== "REVIEW") {
   392→      throw new ApiError(
   393→        422,
   394→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   395→        "Counting is only possible while the stocktake is counting or in review.",
   396→      );
   397→    }
   398→    const line = await tx.inventoryStocktakeLine.findFirst({
   399→      where: { id: input.lineId, stocktakeId: stocktake.id },
   400→      select: {
   401→        id: true,
   402→        productId: true,
   403→        countedAt: true,
   404→        systemQtyAtCount: true,
   405→        countedQty: true,
   406→        varianceQty: true,
   407→      },
   408→    });
   409→    if (!line) throw new ApiError(404, "NOT_FOUND", "Stocktake line not found.");
   410→
   411→    // §15 — server reads the CURRENT stock; client never supplies it.
   412→    const product = await tx.product.findFirst({
   413→      where: { id: line.productId, businessId: input.businessId },
   414→      select: { id: true, name: true, sku: true, unit: true, stockQty: true, deletedAt: true, isActive: true },
   415→    });
   416→    if (!product || product.deletedAt) {
   417→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   418→    }
   419→    if (!product.isActive) {
   420→      throw new ApiError(
   421→        422,
   422→        STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE,
   423→        "Inactive products cannot be counted.",
   424→      );
   425→    }
   426→
   427→    const systemQtyAtCount = product.stockQty; // §15 snapshot at count time
   428→    const varianceQty = computeVarianceQty(input.countedQty, systemQtyAtCount); // §16/§18
   429→    const now = new Date();
   430→    const isRecount = line.countedAt !== null && line.countedAt.getTime() !== 0;
   431→
   432→    const updated = await tx.inventoryStocktakeLine.update({
   433→      where: { id: line.id },
   434→      data: {
   435→        systemQtyAtCount,
   436→        countedQty: input.countedQty,
   437→        varianceQty,
   438→        countedAt: now, // §30 step 14
   439→        countedByUserId: input.userId, // last successful count is authoritative (§47)
   440→        productNameAtCount: product.name, // §19 metadata snapshot
   441→        skuAtCount: product.sku,
   442→        unitAtCount: product.unit,
   443→      },
   444→      select: { id: true, productId: true, systemQtyAtCount: true, countedQty: true, varianceQty: true },
   445→    });
   446→
   447→    if (isRecount) {
   448→      // §32/§33 — replace in place; audit the safe before/after metadata.
   449→      await audit(
   450→        auditInput(
   451→          input.businessId,
   452→          input.userId,
   453→          "inventory.stocktake.recount",
   454→          stocktake.id,
   455→          stocktake.reference,
   456→          {
   457→            lineId: line.id,
   458→            productId: line.productId,
   459→            newSystemQty: updated.systemQtyAtCount,
   460→            newCountedQty: updated.countedQty,
   461→            newVariance: updated.varianceQty,
   462→          },
   463→          {
   464→            previousSystemQty: line.systemQtyAtCount,
   465→            previousCountedQty: line.countedQty,
   466→            previousVariance: line.varianceQty,
   467→          },
   468→        ),
   469→        tx,
   470→      );
   471→    } else {
   472→      await audit(
   473→        auditInput(input.businessId, input.userId, "inventory.stocktake.count", stocktake.id, stocktake.reference, {
   474→          lineId: line.id,
   475→          productId: line.productId,
   476→          systemQtyAtCount: updated.systemQtyAtCount,
   477→          countedQty: updated.countedQty,
   478→          variance: updated.varianceQty,
   479→        }),
   480→        tx,
   481→      );
   482→    }
   483→    return updated;
   484→  });
   485→}

   489→export async function startStocktake(input: {
   490→  businessId: string;
   491→  userId: string;
   492→  stocktakeId: string;
   493→  ip?: string | null;
   494→  userAgent?: string | null;
   495→}) {
   496→  return db.$transaction(async (tx) => {
   497→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   498→    assertTransition(stocktake.status, "COUNTING"); // §8/§34 — DRAFT only
   499→    const lineCount = await tx.inventoryStocktakeLine.count({
   500→      where: { stocktakeId: stocktake.id },
   501→    });
   502→    if (lineCount < 1) {
   503→      throw new ApiError(422, STOCKTAKE_ERROR_CODES.INVALID_TRANSITION, "Add at least one product before starting.");
   504→    }
   505→    const updated = await tx.inventoryStocktake.update({
   506→      where: { id: stocktake.id },
   507→      data: { status: "COUNTING", startedAt: new Date() },
   508→      select: { id: true, status: true, startedAt: true },
   509→    });
   510→    await audit(
   511→      auditInput(input.businessId, input.userId, "inventory.stocktake.start", stocktake.id, stocktake.reference, {
   512→        status: "COUNTING",
   513→        lineCount,
   514→      }),
   515→      tx,
   516→    );
   517→    return updated;
   518→  });

   546→export async function reviewStocktake(input: {
   547→  businessId: string;
   548→  userId: string;
   549→  stocktakeId: string;
   550→  ip?: string | null;
   551→  userAgent?: string | null;

   554→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   555→    assertTransition(stocktake.status, "REVIEW"); // §35 — COUNTING only
   556→    const lines = await tx.inventoryStocktakeLine.findMany({
   557→      where: { stocktakeId: stocktake.id },
   558→      select: { countedAt: true },
   559→    });
   560→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   561→      // §35 — every line must be counted before review.
   562→      throw new ApiError(
   563→        422,
   564→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   565→        "Count every product before sending the stocktake to review.",
   566→      );
   567→    }
   568→    const updated = await tx.inventoryStocktake.update({
   569→      where: { id: stocktake.id },
   570→      data: { status: "REVIEW", reviewedAt: new Date() },
   571→      select: { id: true, status: true, reviewedAt: true },
   572→    });
   573→    await audit(
   574→      auditInput(input.businessId, input.userId, "inventory.stocktake.review", stocktake.id, stocktake.reference, {
   575→        status: "REVIEW",
   576→        lineCount: lines.length,
   577→      }),
   578→      tx,
   579→    );
   580→    return updated;
   581→  });
   582→}

   586→export async function finalizeStocktake(input: {
   587→  businessId: string;
   588→  userId: string;
   589→  stocktakeId: string;
   590→  ip?: string | null;
   591→  userAgent?: string | null;
   592→}) {
   593→  return db.$transaction(async (tx) => {
   594→    // §39 — ONE transaction: verify → stale-check → reconcile → finalize.
   595→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   596→    assertTransition(stocktake.status, "FINALIZED"); // §38 — REVIEW only
   597→
   598→    const lines = await tx.inventoryStocktakeLine.findMany({
   599→      where: { stocktakeId: stocktake.id },
   600→      select: {
   601→        id: true,
   602→        productId: true,
   603→        systemQtyAtCount: true,
   604→        countedQty: true,
   605→        varianceQty: true,
   606→        countedAt: true,
   607→        productNameAtCount: true,
   608→      },
   609→    });
   610→    if (lines.length === 0 || lines.some((l) => l.countedAt === null || l.countedAt.getTime() === 0)) {
   611→      throw new ApiError(
   612→        422,
   613→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   614→        "Every product must be counted before finalization.",
   615→      );
   616→    }
   617→
   618→    // §40 stale check — every product's CURRENT stockQty must still equal
   619→    // systemQtyAtCount, and the product must still be active + not deleted
   620→    // (§57/§58). No auto-rebase (§41). Collect ALL stale/invalid lines so the
   621→    // client can recount each one (§139 — current-business products only).
   622→    const stale: Array<{
   623→      lineId: string;
   624→      productId: string;
   625→      productName: string;
   626→      systemQtyAtCount: number;
   627→      currentQty: number;
   628→      kind: "STALE" | "INACTIVE" | "DELETED";
   629→    }> = [];
   630→    const products = await tx.product.findMany({
   631→      where: { id: { in: lines.map((l) => l.productId) }, businessId: input.businessId },
   632→      select: { id: true, stockQty: true, deletedAt: true, isActive: true },
   633→    });
   634→    const productById = new Map(products.map((p) => [p.id, p]));
   635→    for (const line of lines) {
   636→      const product = productById.get(line.productId);
   637→      if (!product || product.deletedAt) {
   638→        stale.push({
   639→          lineId: line.id,
   640→          productId: line.productId,
   641→          productName: line.productNameAtCount,
   642→          systemQtyAtCount: line.systemQtyAtCount,
   643→          currentQty: product?.stockQty ?? line.systemQtyAtCount,
   644→          kind: "DELETED",
   645→        });
   646→        continue;
   647→      }
   648→      if (!product.isActive) {
   649→        stale.push({
   650→          lineId: line.id,
   651→          productId: line.productId,
   652→          productName: line.productNameAtCount,
   653→          systemQtyAtCount: line.systemQtyAtCount,
   654→          currentQty: product.stockQty,
   655→          kind: "INACTIVE",
   656→        });
   657→        continue;
   658→      }
   659→      if (product.stockQty !== line.systemQtyAtCount) {
   660→        stale.push({
   661→          lineId: line.id,
   662→          productId: line.productId,
   663→          productName: line.productNameAtCount,
   664→          systemQtyAtCount: line.systemQtyAtCount,
   665→          currentQty: product.stockQty,
   666→          kind: "STALE",
   667→        });
   668→      }
   669→    }
   670→    if (stale.length > 0) {
   671→      const hasInactive = stale.some((s) => s.kind === "INACTIVE");
   672→      const hasDeleted = stale.some((s) => s.kind === "DELETED");
   673→      throw new ApiError(
   674→        409,
   675→        hasInactive
   676→          ? STOCKTAKE_ERROR_CODES.PRODUCT_INACTIVE
   677→          : hasDeleted
   678→            ? STOCKTAKE_ERROR_CODES.STALE_COUNT
   679→            : STOCKTAKE_ERROR_CODES.STALE_COUNT,
   680→        "Inventory changed after counting. Recount the affected product(s).",
   681→        { products: stale }, // §139 — affected current-business products only
   682→      );
   683→    }
   684→
   685→    // §43/§44 — reconcile through the EXISTING InventoryAdjustment semantics.
   686→    let adjustedLineCount = 0;
   687→    for (const line of lines) {
   688→      if (line.varianceQty === 0) continue; // §44 — zero variance: no adjustment
   689→      const adjustment = await tx.inventoryAdjustment.create({
   690→        data: {
   691→          businessId: input.businessId,
   692→          productId: line.productId,
   693→          delta: line.varianceQty, // signed — §18
   694→          reason: ADJUSTMENT_REASON(stocktake.reference),
   695→          actorId: input.userId,
   696→        },
   697→        select: { id: true },
   698→      });
   699→      await tx.product.update({
   700→        where: { id: line.productId },
   701→        data: { stockQty: line.countedQty }, // §45 — stock becomes the counted value
   702→      });
   703→      adjustedLineCount += 1;
   704→      // §72/§73 — the InventoryAdjustment row itself carries the quantity
   705→      // history; no duplicate per-line audit event here. The finalize audit
   706→      // below records the workflow-level change once.
   707→      void adjustment;
   708→    }
   709→
   710→    const updated = await tx.inventoryStocktake.update({
   711→      where: { id: stocktake.id },
   712→      data: { status: "FINALIZED", finalizedAt: new Date() },
   713→      select: { id: true, status: true, finalizedAt: true },
   714→    });
   715→
   716→    // §69/§73 — ONE finalize audit for the workflow; InventoryAdjustment rows
   717→    // are the quantity-level history. Safe metadata only (§70/§140).
   718→    await audit(
   719→      auditInput(input.businessId, input.userId, "inventory.stocktake.finalize", stocktake.id, stocktake.reference, {
   720→        status: "FINALIZED",
   721→        lineCount: lines.length,
   722→        adjustedLineCount,
   723→      }),
   724→      tx,
   725→    );
   726→    return { ...updated, adjustedLineCount };
   727→  });
   728→}

   730→/* --------------------------------- cancel --------------------------------- */
   731→
   732→export async function cancelStocktake(input: {
   733→  businessId: string;
   734→  userId: string;
   735→  stocktakeId: string;
   736→  ip?: string | null;
   737→  userAgent?: string | null;
   738→}) {
   739→  return db.$transaction(async (tx) => {
   740→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   741→    assertTransition(stocktake.status, "CANCELLED"); // §49 — DRAFT|COUNTING|REVIEW
   742→    const updated = await tx.inventoryStocktake.update({
   743→      where: { id: stocktake.id },
   744→      data: { status: "CANCELLED", canceledAt: new Date() },
   745→      select: { id: true, status: true, canceledAt: true },
   746→    });
   747→    await audit(
   748→      auditInput(input.businessId, input.userId, "inventory.stocktake.cancel", stocktake.id, stocktake.reference, {
   749→        status: "CANCELLED",
   750→        previousStatus: stocktake.status,
   751→      }),
   752→      tx,
   753→    );
   754→    return updated;
   755→  });
   756→}

   758→/* ---------------------------- DRAFT-only edit ----------------------------- */
   759→
   760→export async function updateStocktake(input: {
   761→  businessId: string;
   762→  userId: string;
   763→  stocktakeId: string;
   764→  reference?: string;
   765→  notes?: string | null;
   766→  ip?: string | null;
   767→  userAgent?: string | null;
   768→}) {
   769→  return db.$transaction(async (tx) => {
   770→    const stocktake = await loadStocktake(tx, input.businessId, input.stocktakeId);
   771→    // §50/§52 — reference/notes are editable ONLY while DRAFT.
   772→    if (stocktake.status !== "DRAFT") {
   773→      throw new ApiError(
   774→        422,
   775→        STOCKTAKE_ERROR_CODES.INVALID_TRANSITION,
   776→        "Only draft stocktakes can be edited.",
   777→      );
   778→    }
   779→    try {
   780→      const updated = await tx.inventoryStocktake.update({
   781→        where: { id: stocktake.id },
   782→        data: {
   783→          ...(input.reference !== undefined ? { reference: input.reference } : {}),
   784→          ...(input.notes !== undefined ? { notes: input.notes } : {}),
   785→        },
   786→        select: { id: true, reference: true, notes: true },
   787→      });
   788→      await audit(
   789→        auditInput(
   790→          input.businessId,
   791→          input.userId,
   792→          "inventory.stocktake.edit",
   793→          stocktake.id,
   794→          updated.reference,
   795→          { status: stocktake.status },
   796→          { reference: stocktake.reference }, // §70 — never the notes content
   797→        ),
   798→        tx,
   799→      );
   800→      return updated;
   801→    } catch (error) {
   802→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   803→        throw new ApiError(409, "DUPLICATE_REFERENCE", "A stocktake with this reference already exists.");
   804→      }
   805→      throw error;
   806→    }
   807→  });
   808→}
