     1→/**
     2→ * Usage metering service — Prompt 21 (§34/§63/§66/§67/§68).
     3→ *
     4→ * Server-side, authoritative consumption metering for metered SaaS metrics.
     5→ * The client is NEVER the source of truth (§3): quantities are decided by
     6→ * the server, counters live in the database, and every enforcement path
     7→ * runs inside a Serializable transaction.
     8→ *
     9→ * Separation of concepts (§4):
    10→ * - RESOURCE limits (customers/products/team/businesses) stay DERIVED from
    11→ *   authoritative tables in service.ts (Prompt 20) — never duplicated here
    12→ *   (§5/§33).
    13→ * - CONSUMPTION metrics accumulate per USAGE PERIOD aligned with the
    14→ *   subscription billing period (§7/§37). No cron resets (§10/§43): when a
    15→ *   billing period rolls over, the next metered operation simply resolves
    16→ *   to a NEW UsagePeriod row (lazy + race-safe, §76) while historical
    17→ *   periods are retained (§101).
    18→ *
    19→ * AI metering semantics (§12/§13/§45/§46/§47/§48):
    20→ * - ONE user Copilot message that enters the pipeline = 1 × AI_MESSAGES and
    21→ *   1 × AI_REQUESTS. Internal snapshot/tool computations are NOT provider
    22→ *   calls and never meter; stream chunks never meter; loading chat history
    23→ *   never meters.
    24→ * - Strategy = count-before-provider-call + release-on-failure (§20/§21,
    25→ *   adapted to this codebase's single-call provider architecture):
    26→ *     meter (atomic, quota-checked) → call provider → success: charge stands
    27→ *     (tokens recorded when the provider reports them) / failure: released.
    28→ * - IDEMPOTENCY (§17/§18): every logical operation carries a requestId
    29→ *   (server-generated UUID, or client-supplied UUID validated server-side).
    30→ *   UsageEvent uniqueness on (businessId, metric, idempotencyKey) makes a
    31→ *   network retry NEVER double-charge: a replayed requestId whose prior
    32→ *   event is COMMITTED proceeds unmetered; one whose prior event was
    33→ *   RELEASED re-meters (the replacement execution actually runs). Abuse is
    34→ *   bounded by the existing per-user rate limits.
    35→ * - FAIL-CLOSED (§24/§68): if usage state cannot be read/written, metered
    36→ *   premium AI operations fail with BILLING_UNAVAILABLE — they never
    37→ *   degrade into unlimited access. Core (non-metered) business features do
    38→ *   not touch this service and keep working.
    39→ * - NO MONEY (§62/§104): usage is never priced here; quantities are stored
    40→ *   so a future payment phase can interpret them without changing meaning
    41→ *   (§105).
    42→ */

    56→// ---------------------------------------------------------------------------
    57→// Per-business metering queue (single-node SQLite concurrency boundary)
    58→// ---------------------------------------------------------------------------
    59→
    60→/**
    61→ * SQLite allows exactly ONE writer at a time, and this deployment is a
    62→ * single Next.js node over one SQLite file (documented boundary, Prompt 20
    63→ * §112). Ten simultaneous interactive transactions all racing for the write
    64→ * lock degrade catastrophically (engine-level lock queueing measured at
    65→ * ~55s under burst), so metering SERIALIZES per business at the application
    66→ * level instead: every transaction stays short, the burst resolves in
    67→ * milliseconds, and the §82 invariant (never exceed the quota) holds
    68→ * deterministically. Cross-process access (CLI tooling) still relies on the
    69→ * database's own write lock — both paths use guarded atomic updates.
    70→ * Multi-node deployments would replace SQLite with a real RDBMS — out of
    71→ * scope by design.
    72→ */
    73→const meterQueues = new Map<string, Promise<unknown>>();
    74→
    75→async function withBusinessLock<T>(businessId: string, task: () => Promise<T>): Promise<T> {
    76→  const prev = meterQueues.get(businessId) ?? Promise.resolve();
    77→  const run = prev.then(
    78→    () => task(),
    79→    () => task(),
    80→  );
    81→  const tail = run.then(
    82→    () => undefined,
    83→    () => undefined,
    84→  );
    85→  meterQueues.set(businessId, tail);
    86→  try {
    87→    return await run;
    88→  } finally {
    89→    if (meterQueues.get(businessId) === tail) meterQueues.delete(businessId);
    90→  }
    91→}

   120→export type UsageLevel = "ok" | "warning" | "critical" | "reached";

   122→export function usageLevel(percentageUsed: number | null): UsageLevel {
   123→  if (percentageUsed === null) return "ok"; // unlimited — never "reached" (§64)
   124→  if (percentageUsed >= 100) return "reached";
   125→  if (percentageUsed >= USAGE_CRITICAL_PERCENT) return "critical";
   126→  if (percentageUsed >= USAGE_WARNING_PERCENT) return "warning";
   127→  return "ok";

   130→// ---------------------------------------------------------------------------
   131→// Usage period resolution (§7/§8/§37/§42)
   132→// ---------------------------------------------------------------------------
   133→
   134→export type UsagePeriodBounds = { start: Date; end: Date | null };
   135→
   136→/**
   137→ * Derive the CURRENT usage-period boundaries from a subscription DTO.
   138→ * Pure function (unit-testable, `now` injectable):
   139→ * - period still running → exactly the subscription's billing period;
   140→ * - period over (expired / grace passed → FREE fallback) → the billing
   141→ *   cadence rolls forward deterministically (UTC, month-end clamped via
   142→ *   periodEnd) until the window contains `now`. No cron, no timezone
   143→ *   dependence (§37/§43).
   144→ */
   145→export function usagePeriodBounds(
   146→  subscription: Pick<SubscriptionDto, "currentPeriodStart" | "currentPeriodEnd" | "billingPeriod">,
   147→  now: Date = new Date(),
   148→): UsagePeriodBounds {
   149→  const start = subscription.currentPeriodStart ? new Date(subscription.currentPeriodStart) : now;
   150→  const end = subscription.currentPeriodEnd ? new Date(subscription.currentPeriodEnd) : null;
   151→  if (end && end.getTime() > now.getTime()) return { start, end };
   152→
   153→  // Period over: continue the cadence from where it stopped. Cycle guard
   154→  // bounds the loop (240 months = 20 years) against pathological data.
   155→  const period: BillingPeriod = subscription.billingPeriod;
   156→  let cursor = end ?? start;
   157→  let guard = 0;
   158→  while (guard++ < 240) {
   159→    const candidateEnd = periodEnd(cursor, period);
   160→    if (candidateEnd.getTime() > now.getTime()) return { start: cursor, end: candidateEnd };
   161→    cursor = candidateEnd;
   162→  }
   163→  return { start: cursor, end: periodEnd(cursor, period) };
   164→}
   165→
   166→function boundsKey(businessId: string, start: Date) {
   167→  return { businessId_start: { businessId, start } } as const;
   168→}
   169→
   170→/**
   171→ * Find-or-create the UsagePeriod for the given bounds INSIDE the caller's
   172→ * transaction. Race-safe: two concurrent creators collide on the unique
   173→ * (businessId, start) index; the loser refetches (§76/§72).
   174→ */
   175→async function ensureUsagePeriodTx(
   176→  tx: Tx,
   177→  businessId: string,
   178→  bounds: UsagePeriodBounds,
   179→  subscriptionId: string | null,
   180→) {
   181→  const existing = await tx.usagePeriod.findUnique({ where: boundsKey(businessId, bounds.start) });
   182→  if (existing) return existing;
   183→  try {
   184→    return await tx.usagePeriod.create({
   185→      data: {
   186→        businessId,
   187→        subscriptionId,
   188→        start: bounds.start,
   189→        end: bounds.end,
   190→      },
   191→    });
   192→  } catch (error) {
   193→    if (!isUniqueViolation(error)) throw error;
   194→    const raced = await tx.usagePeriod.findUnique({ where: boundsKey(businessId, bounds.start) });
   195→    if (!raced) throw error;
   196→    return raced;
   197→  }
   198→}
   199→
   200→function isUniqueViolation(error: unknown): boolean {
   201→  return (
   202→    typeof error === "object" &&
   203→    error !== null &&
   204→    "code" in error &&
   205→    (error as { code?: string }).code === "P2002"
   206→  );
   207→}

   209→// ---------------------------------------------------------------------------
   210→// Counter primitives — guarded atomic moves (§28/§68)
   211→// ---------------------------------------------------------------------------
   212→
   213→/**
   214→ * Increment a counter by `quantity`, respecting `limit` (null = unlimited).
   215→ * The guarded UPDATE (used < limit ⇒ used+quantity ≤ limit for quantity 1)
   216→ * is a single atomic SQL statement; the create path covers the first use
   217→ * and races on the unique (period, metric) index. Throws the structured
   218→ * USAGE_LIMIT_REACHED error when the quota would be exceeded.
   219→ */
   220→async function bumpCounterTx(
   221→  tx: Tx,
   222→  usagePeriodId: string,
   223→  metric: UsageMetric,
   224→  quantity: number,
   225→  limit: number | null,
   226→  periodEndIso: string | null,
   227→): Promise<void> {
   228→  if (limit === null) {
   229→    const bumped = await tx.usageCounter.updateMany({
   230→      where: { usagePeriodId, metric },
   231→      data: { used: { increment: quantity } },
   232→    });
   233→    if (bumped.count === 0) {
   234→      try {
   235→        await tx.usageCounter.create({ data: { usagePeriodId, metric, used: quantity } });
   236→        return;
   237→      } catch (error) {
   238→        if (!isUniqueViolation(error)) throw error;
   239→        const raced = await tx.usageCounter.updateMany({
   240→          where: { usagePeriodId, metric },
   241→          data: { used: { increment: quantity } },
   242→        });
   243→        if (raced.count === 0) throw new BillingError("BILLING_UNAVAILABLE", "Usage counter is unavailable.");
   244→      }
   245→    }
   246→    return;
   247→  }
   248→
   249→  const bumped = await tx.usageCounter.updateMany({
   250→    where: { usagePeriodId, metric, used: { lt: limit } },
   251→    data: { used: { increment: quantity } },
   252→  });
   253→  if (bumped.count > 0) return;
   254→
   255→  // Either the counter is missing (first use) or the quota is exhausted.
   256→  const existing = await tx.usageCounter.findUnique({
   257→    where: { usagePeriodId_metric: { usagePeriodId, metric } },
   258→  });
   259→  if (!existing) {
   260→    if (quantity > limit) {
   261→      throw usageLimitReachedError(metric, limit, 0, periodEndIso);
   262→    }
   263→    try {
   264→      await tx.usageCounter.create({ data: { usagePeriodId, metric, used: quantity } });
   265→      return;
   266→    } catch (error) {
   267→      if (!isUniqueViolation(error)) throw error;
   268→      // Lost a create race — fall through to the guarded re-check below.
   269→    }
   270→  }
   271→  const current = existing?.used ?? 0;
   272→  const after = await tx.usageCounter.updateMany({
   273→    where: { usagePeriodId, metric, used: { lte: limit - quantity } },
   274→    data: { used: { increment: quantity } },
   275→  });
   276→  if (after.count === 0) {
   277→    throw usageLimitReachedError(metric, limit, Math.max(current, limit), periodEndIso);
   278→  }
   279→}
   280→
   281→/**
   282→ * Decrement a counter (release path) — guarded so it can never go negative
   283→ * (§28). Returns the post-release value, or null when no counter row exists.
   284→ */
   285→async function releaseCounterTx(
   286→  tx: Tx,
   287→  usagePeriodId: string,
   288→  metric: UsageMetric,
   289→  quantity: number,
   290→): Promise<number | null> {
   291→  const existing = await tx.usageCounter.findUnique({
   292→    where: { usagePeriodId_metric: { usagePeriodId, metric } },
   293→  });
   294→  if (!existing) return null;
   295→  const next = Math.max(0, existing.used - quantity);
   296→  await tx.usageCounter.update({ where: { id: existing.id }, data: { used: next } });
   297→  return next;
   298→}

   300→// ---------------------------------------------------------------------------
   301→// AI operation metering (§12/§13/§17/§18/§20/§21/§49)
   302→// ---------------------------------------------------------------------------
   303→
   304→export type AiMeteringResult = {
   305→  requestId: string;
   306→  /** true when at least one metric was newly charged by THIS call. */
   307→  charged: boolean;
   308→  /** true when the requestId was already metered (idempotent replay). */
   309→  replay: boolean;
   310→  period: { start: string; end: string | null };
   311→  /** Post-metering state per enforced metric (for structured responses). */
   312→  metrics: Record<
   313→    string,
   314→    { used: number; limit: number | null; remaining: number | null }
   315→  >;
   316→};
   317→
   318→/**
   319→ * Meter ONE logical AI operation (1 × AI_MESSAGES + 1 × AI_REQUESTS) for
   320→ * the business's current usage period. Runs in ONE Serializable
   321→ * transaction (SQLite serializes writers — the concurrency boundary, §112):
   322→ * entitlements → period → idempotency gate → guarded counter bump.
   323→ *
   324→ * Throws:
   325→ * - 422 USAGE_LIMIT_REACHED (structured §23 body) when an enforced quota
   326→ *   is exhausted — NO usage is consumed by the rejected attempt (§49).
   327→ * - 503 BILLING_UNAVAILABLE when usage state cannot be resolved (fail
   328→ *   closed, §24/§68).
   329→ */
   330→export async function meterAiOperation(
   331→  businessId: string,
   332→  requestId: string,
   333→  opts: { tx?: Tx; meta?: Record<string, unknown> } = {},
   334→): Promise<AiMeteringResult> {
   335→  const run = async (tx: Tx): Promise<AiMeteringResult> => {
   336→    // Entitlement context (lazy subscription ensure inside; effective plan
   337→    // resolution handles expiry/trial/cancellation fallbacks — §17/§40/§41).
   338→    let ctx: EntitlementContext;
   339→    try {
   340→      ctx = await getEntitlementContextTx(tx, businessId);
   341→    } catch (error) {
   342→      if (error instanceof BillingError) throw error;
   343→      console.error("[usage] entitlement resolution failed:", error);
   344→      throw new BillingError("BILLING_UNAVAILABLE", "Usage state is temporarily unavailable. Please try again.");
   345→    }
   346→
   347→    const bounds = usagePeriodBounds(ctx.subscription);
   348→    const period = await ensureUsagePeriodTx(tx, businessId, bounds, ctx.subscription.id);
   349→    const periodEndIso = bounds.end ? bounds.end.toISOString() : null;
   350→
   351→    const limits: Record<string, number | null> = {};
   352→    const metrics: AiMeteringResult["metrics"] = {};
   353→    let charged = false;
   354→    let replay = true;
   355→
   356→    for (const metric of AI_OPERATION_METRICS) {
   357→      const code = METRIC_ENTITLEMENT[metric];
   358→      const limit = code ? getLimitIn(ctx, code) : 0; // unknown → fail closed (0)
   359→      limits[metric] = limit;
   360→
   361→      // Idempotency gate FIRST (§17/§18): the unique event row decides
   362→      // whether this logical operation has already been metered.
   363→      const existing = await tx.usageEvent.findUnique({
   364→        where: {
   365→          businessId_metric_idempotencyKey: {
   366→            businessId,
   367→            metric,
   368→            idempotencyKey: `${requestId}:${metric}`,
   369→          },
   370→        },
   371→      });
   372→
   373→      if (!existing) {
   374→        await tx.usageEvent.create({
   375→          data: {
   376→            businessId,
   377→            usagePeriodId: period.id,
   378→            metric,
   379→            quantity: 1,
   380→            status: "COMMITTED",
   381→            requestId,
   382→            idempotencyKey: `${requestId}:${metric}`,
   383→            ...(opts.meta ? { metaJson: JSON.stringify(opts.meta) } : {}),
   384→          },
   385→        });
   386→        await bumpCounterTx(tx, period.id, metric, 1, limit, periodEndIso);
   387→        charged = true;
   388→        replay = false;
   389→      } else if (existing.status === "RELEASED") {
   390→        // A previous attempt failed before provider acceptance and was
   391→        // released. This retry REPLACES it: re-charge optimistically (and
   392→        // release again if the replacement also fails). Never double-charges
   393→        // the successful original (§17/§49).
   394→        await tx.usageEvent.update({
   395→          where: { id: existing.id },
   396→          data: { status: "COMMITTED", usagePeriodId: period.id },
   397→        });
   398→        await bumpCounterTx(tx, period.id, metric, 1, limit, periodEndIso);
   399→        charged = true;
   400→        replay = false;
   401→      }
   402→      // else: prior event COMMITTED → skip (duplicate submission continues
   403→      // unmetered — the original operation already paid).
   404→
   405→      const counter = await tx.usageCounter.findUnique({
   406→        where: { usagePeriodId_metric: { usagePeriodId: period.id, metric } },
   407→      });
   408→      const used = counter?.used ?? 0;
   409→      metrics[metric] = {
   410→        used,
   411→        limit,
   412→        remaining: limit === null ? null : Math.max(0, limit - used),
   413→      };
   414→    }
   415→
   416→    return {
   417→      requestId,
   418→      charged,
   419→      replay,
   420→      period: { start: bounds.start.toISOString(), end: periodEndIso },
   421→      metrics,
   422→    };
   423→  };
   424→
   425→  if (opts.tx) return run(opts.tx);
   426→  // Serialize per business (see withBusinessLock) — burst-safe quota
   427→  // enforcement on single-node SQLite (§19/§82/§112).
   428→  try {
   429→    return await withBusinessLock(businessId, () =>
   430→      db.$transaction(run, {
   431→        isolationLevel: "Serializable",
   432→        timeout: 30_000,
   433→        maxWait: 5_000,
   434→      }),
   435→    );
   436→  } catch (error) {
   437→    if (error instanceof BillingError) throw error;
   438→    console.error("[usage] metering failed:", error);
   439→    // Fail closed (§24/§68): metered premium AI never degrades into
   440→    // unlimited access on infrastructure failure.
   441→    throw new BillingError("BILLING_UNAVAILABLE", "Usage state is temporarily unavailable. Please try again.");
   442→  }
   443→}
   444→
   445→/**
   446→ * Release a previously metered AI operation (provider never accepted it —
   447→ * §20/§49): decrements the counters and marks the events RELEASED.
   448→ * Idempotent (§48): releasing twice is a no-op; guarded decrements keep
   449→ * counters non-negative (§28). Best-effort by callers — a release failure
   450→ * must never mask the original error; it is logged and audited instead.
   451→ */
   452→export async function releaseAiOperation(
   453→  businessId: string,
   454→  requestId: string,
   455→  opts: { reason?: string } = {},
   456→): Promise<boolean> {
   457→  try {
   458→    return await db.$transaction(
   459→      async (tx) => {
   460→        const events = await tx.usageEvent.findMany({
   461→          where: { businessId, requestId, status: "COMMITTED", metric: { in: [...AI_OPERATION_METRICS] } },
   462→        });
   463→        if (events.length === 0) return false;
   464→        for (const event of events) {
   465→          const metric = event.metric as UsageMetric;
   466→          const next = await releaseCounterTx(tx, event.usagePeriodId, metric, event.quantity);
   467→          if (next === null) {
   468→            // Counter missing while a COMMITTED event exists — an integrity
   469→            // anomaly; report loudly, never silently "fix" (§50/§103).
   470→            console.error(`[usage] integrity: counter missing for ${businessId}/${metric} during release`);
   471→          }
   472→          await tx.usageEvent.update({
   473→            where: { id: event.id },
   474→            data: {
   475→              status: "RELEASED",
   476→              ...(opts.reason ? { metaJson: JSON.stringify({ reason: opts.reason }) } : {}),
   477→            },
   478→          });
   479→        }
   480→        await audit({
   481→          businessId,
   482→          action: "usage.ai_released",
   483→          entityType: "UsageEvent",
   484→          entityId: requestId,
   485→          after: { requestId, reason: opts.reason ?? "provider-not-accepted", metrics: events.map((e) => e.metric) },
   486→        }, tx);
   487→        return true;
   488→      },
   489→      { isolationLevel: "Serializable", timeout: 15_000, maxWait: 5_000 },
   490→    );
   491→  } catch (error) {
   492→    // Release is a reconciliation best-effort: log loudly, never throw into
   493→    // the user-facing error path (the original failure takes precedence).
   494→    console.error("[usage] release failed (counter may be over-stated):", error);
   495→    return false;
   496→  }
   497→}

   499→// ---------------------------------------------------------------------------
   500→// Token recording (§14/§15/§30/§60) — only authoritative provider values
   501→// ---------------------------------------------------------------------------
   502→
   503→export type AiProviderUsage = {
   504→  inputTokens?: number | null;
   505→  outputTokens?: number | null;
   506→  totalTokens?: number | null;
   507→};
   508→
   509→const MAX_TOKEN_QUANTITY = 10_000_000; // sane upper bound (§58)
   510→
   511→function sanitizeTokenCount(value: unknown): number | null {
   512→  if (typeof value !== "number" || !Number.isFinite(value) || value < 0) return null;
   513→  const int = Math.round(value);
   514→  if (int > MAX_TOKEN_QUANTITY) return null; // implausible → ignore, never fabricate
   515→  return int;
   516→}
   517→
   518→/**
   519→ * Record provider-reported token usage for an already-metered AI operation.
   520→ * ONLY genuine provider values are stored (§14): if the provider exposes
   521→ * nothing usable, this is a no-op and usage stays unknown (never estimated,
   522→ * never fabricated). AI_TOKENS is recorded, not enforced (§66) — it carries
   523→ * no quota and exists so future cost computation can interpret quantities
   524→ * without changing their meaning (§105).
   525→ */
   526→export async function recordAiTokenUsage(
   527→  businessId: string,
   528→  requestId: string,
   529→  usage: AiProviderUsage | null | undefined,
   530→  providerMeta: { provider?: string; model?: string } = {},
   531→): Promise<boolean> {
   532→  if (!usage) return false;
   533→  const inputTokens = sanitizeTokenCount(usage.inputTokens);
   534→  const outputTokens = sanitizeTokenCount(usage.outputTokens);
   535→  const totalTokens =
   536→    sanitizeTokenCount(usage.totalTokens) ??
   537→    (inputTokens !== null && outputTokens !== null ? inputTokens + outputTokens : null);
   538→  if (totalTokens === null || totalTokens <= 0) return false;
   539→
   540→  try {
   541→    const created = await db.$transaction(
   542→      async (tx) => {
   543→        const ctx = await getEntitlementContextTx(tx, businessId);
   544→        const bounds = usagePeriodBounds(ctx.subscription);
   545→        const period = await ensureUsagePeriodTx(tx, businessId, bounds, ctx.subscription.id);
   546→
   547→        const meta: Record<string, unknown> = { ...providerMeta };
   548→        if (inputTokens !== null) meta.inputTokens = inputTokens;
   549→        if (outputTokens !== null) meta.outputTokens = outputTokens;
   550→        meta.totalTokens = totalTokens;
   551→
   552→        // Idempotency gate — a replayed requestId never double-counts tokens.
   553→        const existing = await tx.usageEvent.findUnique({
   554→          where: {
   555→            businessId_metric_idempotencyKey: {
   556→              businessId,
   557→              metric: "AI_TOKENS",
   558→              idempotencyKey: `${requestId}:AI_TOKENS`,
   559→            },
   560→          },
   561→        });
   562→        if (existing) return false;
   563→        await tx.usageEvent.create({
   564→          data: {
   565→            businessId,
   566→            usagePeriodId: period.id,
   567→            metric: "AI_TOKENS",
   568→            quantity: totalTokens,
   569→            status: "COMMITTED",
   570→            requestId,
   571→            idempotencyKey: `${requestId}:AI_TOKENS`,
   572→            metaJson: JSON.stringify(meta),
   573→          },
   574→        });
   575→        await bumpCounterTx(tx, period.id, "AI_TOKENS", totalTokens, null, null);
   576→        return true;
   577→      },
   578→      { isolationLevel: "Serializable", timeout: 30_000, maxWait: 5_000 },
   579→    );
   580→    // Propagate the transaction's verdict — a duplicate must report false.
   581→    return created === true;
   582→  } catch (error) {
   583→    // Token recording is observational — never break the AI response path.
   584→    console.error("[usage] token recording failed:", error);
   585→    return false;
   586→  }
   587→}

   590→// Usage summary (§63/§64/§94) — powers the Settings → Subscription → Usage UI
   591→// ---------------------------------------------------------------------------
   592→
   593→export type UsageMetricSummary = {
   594→  metric: UsageMetric;
   595→  used: number;
   596→  /** null = unlimited / not limited (never a fake giant number, §64). */
   597→  limit: number | null;
   598→  remaining: number | null;
   599→  percentageUsed: number | null;
   600→  level: UsageLevel;
   601→};
   602→
   603→export type UsageSummary = {
   604→  planCode: string;
   605→  entitled: boolean;
   606→  status: string;
   607→  period: { start: string; end: string | null };
   608→  metrics: Record<UsageMetric, UsageMetricSummary>;
   609→};
   610→
   611→/**
   612→ * Read-only usage summary for the CURRENT usage period. Uses the 15s
   613→ * entitlement cache (UI read only, §67) — enforcement always re-checks
   614→ * server-side. Creates nothing: zero usage before the first metered
   615→ * operation is reported from computed bounds (§76 lazy initialization).
   616→ */
   617→export async function getUsageSummary(businessId: string): Promise<UsageSummary> {
   618→  const ctx = await getEntitlementContext(businessId);
   619→  const bounds = usagePeriodBounds(ctx.subscription);
   620→
   621→  const period = await db.usagePeriod.findUnique({
   622→    where: boundsKey(businessId, bounds.start),
   623→    include: { counters: true },
   624→  });
   625→  const counterByMetric = new Map(period?.counters.map((c) => [c.metric, c.used]) ?? []);
   626→
   627→  const summarize = (metric: UsageMetric, limit: number | null): UsageMetricSummary => {
   628→    const used = counterByMetric.get(metric) ?? 0;
   629→    const remaining = limit === null ? null : Math.max(0, limit - used);
   630→    // Unlimited → null percentage (§64); a 0-limit plan shows 100% reached.
   631→    const percentageUsed =
   632→      limit === null ? null : limit <= 0 ? (used > 0 ? 100 : null) : Math.min(100, Math.round((used / limit) * 100));
   633→    return { metric, used, limit, remaining, percentageUsed, level: usageLevel(percentageUsed) };
   634→  };
   635→
   636→  return {
   637→    planCode: ctx.subscription.effectivePlanCode,
   638→    entitled: ctx.subscription.entitled,
   639→    status: ctx.subscription.status,
   640→    period: { start: bounds.start.toISOString(), end: bounds.end ? bounds.end.toISOString() : null },
   641→    metrics: {
   642→      AI_REQUESTS: summarize("AI_REQUESTS", getLimitIn(ctx, "AI_REQUESTS_MAX")),
   643→      AI_MESSAGES: summarize("AI_MESSAGES", getLimitIn(ctx, "AI_MESSAGES_MAX")),
   644→      AI_TOKENS: summarize("AI_TOKENS", null),
   645→    },
   646→  };
   647→}
