    63→/** Sidebar label key for the current view (used for <title> too). */
    64→function labelKeyOf(view: AppView): string {
    65→  switch (view.kind) {
    66→    case "dashboard":
    67→      return "dashboard";
    68→    case "audit":
    69→      return "audit";
    70→    case "settings:business":
    71→      return "settings";
    72→    case "settings:members":
    73→      return "teamMembers";
    74→    case "settings:billing":
    75→      return "billing";
    76→    case "settings:security":
    77→      return "security";
    78→    case "settings:data":
    79→      return "dataPortability";
    80→    case "notifications":
    81→      return "notifications";
    82→    case "settings:notifications":
    83→      return "notifications";
    84→    // Prompt 33 — Search Results Center document title.
    85→    case "search":
    86→      return "search";
    87→    // Prompt 35 (§36) — CRM module document title.
    88→    case "crm":
    89→      return "crm";
    90→    // Prompt 34 — entity detail titles reuse the owning module's nav label.
    91→    case "entity":
    92→      return ENTITY_ROUTES[view.entity].labelKey;
    93→    // Prompt 39 — party statement document title (§57).
    94→    case "statements":
    95→      return "statements";
    96→    // Prompt 41 — purchase-order document titles (§71).
    97→    case "purchase-orders":
    98→      return "purchaseOrders";
    99→    case "purchase-order":
   100→      return "purchaseOrders";
   101→    // Prompt 42 — goods-receipt document titles (§117).
   102→    case "inventory-goods-receipts":
   103→      return "goodsReceipts";
   104→    case "inventory-goods-receipt":
   105→      return "goodsReceipts";
   106→    // Prompt 43 — goods-return document titles (§91).
   107→    case "inventory-goods-returns":
   108→      return "goodsReturns";
   109→    case "inventory-goods-return":
   110→      return "goodsReturns";
   111→    // Prompt 44 — goods-dispatch document titles.
   112→    case "inventory-goods-dispatches":
   113→      return "goodsDispatches";
   114→    case "inventory-goods-dispatch":
   115→      return "goodsDispatches";
   116→    // Prompt 45 — sales-order document titles.
   117→    case "sales-orders":
   118→      return "salesOrders";
   119→    case "sales-order":
   120→      return "salesOrders";
   121→    // Prompt 47 — sales-quotation document titles.
   122→    case "sales-quotations":
   123→      return "salesQuotations";
   124→    case "sales-quotation":
   125→      return "salesQuotations";
   126→    // Prompt 48 — supplier-bill document titles (§4.1).
   127→    case "supplier-bills":
   128→      return "supplierBills";
   129→    case "supplier-bill":
   130→      return "supplierBills";
   131→    // Prompt 49 — detailed-aging document titles (the owning balances module).
   132→    case "aging-details":
   133→      return view.side === "payables" ? "payables" : "receivables";
   134→    case "profile":
   135→      return "profile";
   136→    case "module": {
   137→      const map: Record<string, string> = {
   138→        customers: "customers",
   139→        suppliers: "suppliers",
   140→        products: "products",
   141→        inventory: "inventory",
   142→        sales: "sales",
   143→        expenses: "expenses",
   144→        payments: "payments",
   145→        receivables: "receivables",
   146→        payables: "payables",
   147→        reports: "reports",
   148→        copilot: "copilot",
   149→      };
   150→      return map[view.module] ?? "dashboard";
   151→    }
   152→    default:
   153→      return "dashboard";
   154→  }
   155→}

   171→
   172→  const membership =
   173→    memberships.find((m) => m.business.id === activeId) ?? memberships[0] ?? null;
   174→  const role: Role = asRole(membership?.role);
   175→
   176→  // Phase 06 — per-module document titles (localized: "Sales · Business · App").
   177→  const moduleLabel = tNav(labelKeyOf(view) as "dashboard");
   178→  useEffect(() => {
   179→    const parts = [
   180→      moduleLabel,
   181→      membership?.business.name,
   182→      tCommon("appName"),
   183→    ].filter(Boolean);
   184→    document.title = parts.join(" · ");
   185→  }, [moduleLabel, membership?.business.name, tCommon]);
   186→
   187→  // Phase 06 — idle prefetch warming: reports/summary + balances are the
   188→  // heaviest deterministic queries; warm them once per business so deep
   189→  // links into Reports/Receivables/Payables render from cache instantly.
   190→  const queryClient = useQueryClient();
   191→  const activeBusinessId = membership?.business.id;
   192→  useEffect(() => {
   193→    if (!activeBusinessId) return;
   194→    const cancel = onIdle(() => warmHeavyQueries(queryClient));
   195→    return cancel;

   198→  const viewKey = viewKeyOf(view);
   199→
   200→  // Prompt 23 §41 — the provider-callback return page. `?payment_return=<id>`

   205→  // Dismiss the return view as soon as the user navigates elsewhere.
   206→  // Navigation is compared by VIEW OBJECT IDENTITY, not viewKey: on a fresh
   207→  // `/?payment_return=<id>` load the underlying view is already the default
   208→  // dashboard, so clicking «Dashboard» produces the SAME viewKey and would
   209→  // never dismiss. navigate()/replaceView() always pass a fresh object.

   211→  useEffect(() => {
   212→    if (prevViewRef.current === null) {
   213→      prevViewRef.current = view;
   214→      return;
   215→    }
   216→    if (prevViewRef.current !== view) {
   217→      prevViewRef.current = view;
   218→      clearPaymentReturn();
   219→    }
   220→  }, [view, clearPaymentReturn]);

   222→  return (
   223→    <SidebarProvider>
   224→      <AppSidebar membership={membership} side={dir === "rtl" ? "right" : "left"} />
   225→      <SidebarInset>
   226→        {/* a11y: keyboard users can jump past the navigation chrome */}
   227→        <a
   228→          href="#main-content"
   229→          className="bg-primary text-primary-foreground sr-only focus:not-sr-only focus:fixed focus:start-3 focus:top-3 focus:z-50 focus:rounded-md focus:px-3 focus:py-2 focus:text-sm focus:font-medium focus:shadow-lg"
   230→        >
   231→          {tSkip("skipToContent")}
   232→        </a>
   233→        <AppHeader />
   234→        <main id="main-content" className="flex-1 p-4 sm:p-6">
   235→          {paymentReturnSessionId ? (
   236→            <div className="mx-auto w-full max-w-6xl">
   237→              <PaymentReturnView
   238→                sessionId={paymentReturnSessionId}
   239→                onDone={clearPaymentReturn}
   240→              />
   241→            </div>
   242→          ) : (
   243→          <div className="mx-auto w-full max-w-6xl">
   244→            <div key={viewKey} className="animate-in fade-in slide-in-from-bottom-1 duration-300">
   245→              {view.kind === "dashboard" ? <DashboardView membership={membership!} /> : null}
   246→              {view.kind === "settings:business" ? (
   247→                <BusinessSettingsView business={membership!.business} />
   248→              ) : null}
   249→              {view.kind === "settings:members" ? <MembersView membership={membership!} /> : null}
   250→              {view.kind === "settings:billing" ? (
   251→                <SubscriptionView business={membership!.business} role={role} />
   252→              ) : null}
   253→              {view.kind === "settings:security" ? <SecurityCenterView /> : null}
   254→              {view.kind === "settings:data" ? <DataPortabilityView role={role} /> : null}
   255→              {view.kind === "notifications" ? <NotificationsView /> : null}
   256→              {view.kind === "settings:notifications" ? <NotificationsPreferencesView /> : null}

   573→export function useMembershipsQuery(enabled: boolean) {
   574→  const queryClient = useQueryClient();
   575→  return useQuery<BusinessListItem[]>({
   576→    queryKey: ["businesses"],
   577→    queryFn: async () => {
   578→      const res = await fetch("/api/v1/businesses", { credentials: "same-origin" });
   579→      let body: unknown = null;
   580→      try { body = await res.json(); } catch { /* non-JSON */ };
   581→      const err = (body as { error?: { code?: string } } | null)?.error;
   582→      if (!res.ok) {
   583→        // Prompt 30 §83/§229 — a globally invalidated session (password
   584→        // change / MFA enable / admin kill switch) must land the user on
   585→        // the sign-in view, never a permanent splash. Clear the stale
   586→        // cookie client-side; NextAuth's session state flips to signed-out.
   587→        if (res.status === 401 || err?.code === "SESSION_INVALIDATED" || err?.code === "SESSION_REVOKED") {
   588→          queryClient.clear();
   589→          void import("next-auth/react").then((m) => m.signOut({ redirect: false }));
   590→        }
   591→        throw new Error(err?.code ?? `HTTP_${res.status}`);
   592→      }
   593→      return (body as { data: BusinessListItem[] }).data;
   594→    },
   595→    enabled,
   596→    staleTime: 60_000,
   597→    retry: false,
   598→  });
   599→}

   601→/**
   602→ * Prompt 23 §41 — reads `?payment_return=<sessionId>` (set ONLY by our own
   603→ * callback API). Value is validated shape-wise (cuid-ish length bound);
   604→ * the session itself is re-authorized server-side, so a forged id simply
   605→ * renders a 404-ish "unknown" state — never another tenant's data.
   606→ * Listens to popstate so browser back/forward and the view's own
   607→ * clear() stay in sync.
   608→ */
   609→function usePaymentReturnSessionId(): [string | null, () => void] {
   610→  const read = () => {
   611→    try {
   612→      const raw = new URLSearchParams(window.location.search).get("payment_return");
   613→      return raw && raw.length >= 20 && raw.length <= 40 && /^[a-zA-Z0-9_-]+$/.test(raw) ? raw : null;
   614→    } catch {
   615→      return null;
   616→    }
   617→  };
   618→  const [sessionId, setSessionId] = useState<string | null>(read);
   619→  useEffect(() => {
   620→    const sync = () => setSessionId(read());
   621→    window.addEventListener("popstate", sync);
   622→    return () => window.removeEventListener("popstate", sync);
   623→  }, []);
   624→  const clear = useCallback(() => {
   625→    // Strip ONLY the ?payment_return query — never touch the hash (the view
   626→    // may be navigating via the app hash-router at the same time).
   627→    window.history.replaceState(null, "", `${window.location.pathname}${window.location.hash}`);
   628→    setSessionId(null);
   629→  }, []);
   630→  return [sessionId, clear];
   631→}
