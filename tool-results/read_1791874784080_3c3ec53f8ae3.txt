   194→export type ExportRow = {
   195→  /** JSON row — IDs + relationships preserved, money as minor integers (§54/§226). */
   196→  json: Record<string, unknown>;
   197→  /** CSV row — exact decimal strings for money, ISO-8601 UTC dates (§227/§231). */
   198→  csv: Array<string | number | null>;
   199→};
   200→
   201→export type DatasetFetchResult = {
   202→  rows: ExportRow[];
   203→  /** True when the dataset hit the export record cap (job must FAIL, §26). */
   204→  truncated: boolean;
   205→};
   206→
   207→export type DatasetFetchCtx = {
   208→  businessId: string;
   209→  /** Business decimal precision for money → CSV decimal strings. */
   210→  decimalPrecision: number;
   211→  /** Hard cap applied per fetch — caller passes EXPORT_MAX_RECORDS headroom. */
   212→  maxRecords: number;
   213→};

   215→function iso(value: Date | null | undefined): string | null {
   216→  return value ? value.toISOString() : null;
   217→}

   223→type Customer = Prisma.CustomerGetPayload<object>;
   224→
   225→function customerRow(c: Customer, precision: number): ExportRow {
   226→  void precision; // customers carry no money fields
   227→  return {
   228→    json: {
   229→      id: c.id,
   230→      name: c.name,
   231→      phone: c.phone,
   232→      email: c.email,
   233→      address: c.address,
   234→      note: c.note,
   235→      deletedAt: iso(c.deletedAt),
   236→      createdAt: c.createdAt.toISOString(),
   237→      updatedAt: c.updatedAt.toISOString(),
   238→    },
   239→    csv: [c.id, c.name, c.phone, c.email, c.address, c.note, iso(c.deletedAt), c.createdAt.toISOString(), c.updatedAt.toISOString()],
   240→  };
   241→}
   242→
   243→function supplierRow(s: Prisma.SupplierGetPayload<object>, precision: number): ExportRow {
   244→  void precision;
   245→  return {
   246→    json: {
   247→      id: s.id,
   248→      name: s.name,
   249→      phone: s.phone,
   250→      email: s.email,
   251→      address: s.address,
   252→      note: s.note,
   253→      deletedAt: iso(s.deletedAt),
   254→      createdAt: s.createdAt.toISOString(),
   255→      updatedAt: s.updatedAt.toISOString(),
   256→    },
   257→    csv: [s.id, s.name, s.phone, s.email, s.address, s.note, iso(s.deletedAt), s.createdAt.toISOString(), s.updatedAt.toISOString()],
   258→  };
   259→}
   260→
   261→function productRow(p: Prisma.ProductGetPayload<object>, precision: number): ExportRow {
   262→  return {
   263→    json: {
   264→      id: p.id,
   265→      name: p.name,
   266→      sku: p.sku,
   267→      unit: p.unit,
   268→      salePriceMinor: p.salePriceMinor,
   269→      costPriceMinor: p.costPriceMinor,
   270→      stockQty: p.stockQty,
   271→      lowStockAt: p.lowStockAt,
   272→      isActive: p.isActive,
   273→      deletedAt: iso(p.deletedAt),
   274→      createdAt: p.createdAt.toISOString(),
   275→      updatedAt: p.updatedAt.toISOString(),
   276→    },
   277→    csv: [
   278→      p.id,
   279→      p.name,
   280→      p.sku,
   281→      p.unit,
   282→      formatMinorToDecimal(p.salePriceMinor, precision),
   283→      formatMinorToDecimal(p.costPriceMinor, precision),
   284→      p.stockQty,
   285→      p.lowStockAt,
   286→      p.isActive ? "true" : "false",
   287→      iso(p.deletedAt),
   288→      p.createdAt.toISOString(),
   289→      p.updatedAt.toISOString(),
   290→    ],
   291→  };
   292→}

   294→async function fetchWithCap<T>(
   295→  cap: number,
   296→  query: (take: number) => Promise<T[]>,
   297→): Promise<{ items: T[]; truncated: boolean }> {
   298→  // Fetch one row beyond the cap so truncation is DETECTABLE, never silent.
   299→  const items = await query(cap + 1);
   300→  return { items: items.slice(0, cap), truncated: items.length > cap };
   301→}
