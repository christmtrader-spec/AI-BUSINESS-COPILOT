   185→/** Stable identity of a view (for equality checks / keys). */
   186→export function viewKeyOf(view: AppView): string {
   187→  return view.kind === "module"
   188→    ? `module:${view.module}`
   189→    // Prompt 35 — the tab is part of the key so tab switches remount the
   190→    // inner CRM view deterministically.
   191→    : view.kind === "crm"
   192→      ? `crm:${view.tab}`
   193→    // Prompt 34 — entity + id both matter: opening a different record
   194→    // remounts the detail view so no stale record can ever be shown (§36).
   195→    : view.kind === "entity"
   196→      ? `entity:${view.entity}:${view.id}`
   197→    // Prompt 39 — type + id both matter: opening another party's statement
   198→    // remounts the view so no stale statement can ever be shown (§159).
   199→    : view.kind === "statements"
   200→      ? `statements:${view.statementType}:${view.id}`
   201→    // Prompt 40 — stocktake views: list + per-id detail keys so opening
   202→    // another stocktake remounts the detail view (no stale record, §36 parity).
   203→    : view.kind === "inventory-stocktakes"
   204→      ? "inventory-stocktakes"
   205→    : view.kind === "inventory-stocktake"
   206→      ? `inventory-stocktake:${view.id}`
   207→    // Prompt 41 — purchase-order views: list + per-id detail keys so opening
   208→    // another order remounts the detail view (no stale record, §36 parity).
   209→    : view.kind === "purchase-orders"
   210→      ? "purchase-orders"
   211→    : view.kind === "purchase-order"
   212→      ? `purchase-order:${view.id}`
   213→    // Prompt 42 — goods-receipt views: list + per-id detail keys so opening
   214→    // another receipt remounts the detail view (no stale record, §36 parity).
   215→    : view.kind === "inventory-goods-receipts"
   216→      ? "inventory-goods-receipts"
   217→    : view.kind === "inventory-goods-receipt"
   218→      ? `inventory-goods-receipt:${view.id}`
   219→    // Prompt 43 — goods-return views: list + per-id detail keys so opening
   220→    // another return remounts the detail view (§36 parity).
   221→    : view.kind === "inventory-goods-returns"
   222→      ? "inventory-goods-returns"
   223→    : view.kind === "inventory-goods-return"
   224→      ? `inventory-goods-return:${view.id}`
   225→    // Prompt 44 — goods-dispatch views: list + per-id detail keys so opening
   226→    // another dispatch remounts the detail view (P43 parity).
   227→    : view.kind === "inventory-goods-dispatches"
   228→      ? "inventory-goods-dispatches"
   229→    : view.kind === "inventory-goods-dispatch"
   230→      ? `inventory-goods-dispatch:${view.id}`
   231→    // Prompt 45 — sales-order views: list + per-id detail keys (§36 parity).
   232→    : view.kind === "sales-orders"
   233→      ? "sales-orders"
   234→    : view.kind === "sales-order"
   235→      ? `sales-order:${view.id}`
   236→      : view.kind === "join"
   237→      ? `join:${view.token}`
   238→      : view.kind === "reset-password"
   239→        ? `reset:${view.token}`
   240→        : view.kind === "verify-email"
   241→          ? `verify:${view.token}`
   242→          : view.kind === "platform"
   243→            ? `platform:${view.section}`
   244→            // Prompt 33 — the Search Center keeps a STABLE view key so the
   245→            // mounted view (and its focused input) survives query updates;
   246→            // the query itself only matters for sameView/back-forward (§13).
   247→            : view.kind;
   248→}

   250→/**
   251→ * Structural view equality (applyHash). Deliberately FINER than viewKeyOf:
   252→ * module sub-state (?month= deep-link param) must re-enter the store when
   253→ * the hash changes, but must NOT remount the module (viewKey stays stable
   254→ * so React keeps the mounted view alive — reports state survives dialog
   255→ * open/close).
   256→ */
   257→export function sameView(a: AppView, b: AppView): boolean {
   258→  if (viewKeyOf(a) !== viewKeyOf(b)) return false;
   259→  if (a.kind === "module" && b.kind === "module") {
   260→    return (a.param ?? null) === (b.param ?? null);
   261→  }
   262→  // Prompt 34 — entity views compare entity + id (viewKey already carries
   263→  // both, so reaching here means they are equal; kept explicit for clarity).
   264→  if (a.kind === "entity" && b.kind === "entity") {
   265→    return a.entity === b.entity && a.id === b.id;
   266→  }
   267→  // Prompt 35 — CRM tab changes re-enter the store (back/forward) without
   268→  // remounting the module boundary shell; the tab drives the inner view.
   269→  if (a.kind === "crm" && b.kind === "crm") {
   270→    return a.tab === b.tab;
   271→  }
   272→  // Prompt 39 — statement views compare type + id (viewKey already carries
   273→  // both; explicit for parity with entity views).
   274→  if (a.kind === "statements" && b.kind === "statements") {
   275→    return a.statementType === b.statementType && a.id === b.id;
   276→  }
   277→  // Prompt 40 — stocktake detail compares id (viewKey already carries it).
   278→  if (a.kind === "inventory-stocktake" && b.kind === "inventory-stocktake") {
   279→    return a.id === b.id;
   280→  }
   281→  // Prompt 41 — purchase-order detail compares id (viewKey already carries it).
   282→  if (a.kind === "purchase-order" && b.kind === "purchase-order") {
   283→    return a.id === b.id;
   284→  }
   285→  // Prompt 42 — goods-receipt detail compares id (viewKey already carries it).
   286→  if (a.kind === "inventory-goods-receipt" && b.kind === "inventory-goods-receipt") {
   287→    return a.id === b.id;
   288→  }
   289→  // Prompt 43 — goods-return detail compares id (viewKey already carries it).
   290→  if (a.kind === "inventory-goods-return" && b.kind === "inventory-goods-return") {
   291→    return a.id === b.id;
   292→  }
   293→  // Prompt 44 — goods-dispatch detail compares id (viewKey already carries it).
   294→  if (a.kind === "inventory-goods-dispatch" && b.kind === "inventory-goods-dispatch") {
   295→    return a.id === b.id;
   296→  }
   297→  // Prompt 33 — hash query changes re-enter the store (back/forward must
   298→  // update the Search Center) without remounting the view (§13/§33).
   299→  if (a.kind === "search" && b.kind === "search") {
   300→    return (a.query ?? null) === (b.query ?? null);
   301→  }
   302→  return true;
   303→}

   305→/** URL hash for a view. `onboarding` is a transient state — no address. */
   306→export function viewToHash(view: AppView): string | null {
   307→  switch (view.kind) {
   308→    case "dashboard":
   309→      return "#/dashboard";
   310→    case "audit":
   311→      return "#/audit";
   312→    case "settings:business":
   313→      return "#/settings";
   314→    case "settings:members":
   315→      return "#/settings/members";
   316→    case "settings:billing":
   317→      return "#/settings/billing";
   318→    case "profile":
   319→      return "#/profile";
   320→    case "settings:security":
   321→      return "#/settings/security";
   322→    case "settings:data":
   323→      return "#/data";
   324→    case "notifications":
   325→      return "#/notifications";
   326→    case "settings:notifications":
   327→      return "#/settings/notifications";
   328→    case "search": {
   329→      // §33 — the query persists in the hash (encodeURIComponent, §29) so
   330→      // refresh and back/forward keep it; never any other param.
   331→      const q = (view.query ?? "").trim();
   332→      return q ? `#/search?q=${encodeURIComponent(q)}` : "#/search";
   333→    }
   334→    case "crm":
   335→      // Prompt 35 (§37) — the tab persists in the hash; the default tab
   336→      // renders as bare "#/crm" so the address stays canonical.
   337→      return view.tab === CRM_DEFAULT_TAB ? "#/crm" : `#/crm?tab=${view.tab}`;
   338→    case "entity":
   339→      // Prompt 34 — centralized encoder (§7/§39): validated entity + id,
   340→      // safe-charset id, encodeURIComponent. Returns null (never a hash)
   341→      // for anything that could not survive parseEntityRoute.
   342→      return entityToHash(view.entity, view.id);
   343→    case "statements":
   344→      // Prompt 39 (§57/§139) — closed type + validated id only; anything
   345→      // else produces no hash at all (never an arbitrary statement URL).
   346→      if (!isStatementTypeToken(view.statementType)) return null;
   347→      if (!isValidEntityId(view.id)) return null;
   348→      return `#/statements?type=${view.statementType}&id=${encodeURIComponent(view.id)}`;
   349→    case "inventory-stocktakes":
   350→      // Prompt 40 (§76) — canonical list hash.
   351→      return "#/inventory/stocktakes";
   352→    case "inventory-stocktake":
   353→      // Prompt 40 (§77/§136) — only charset-safe ids ever become a hash.
   354→      if (!isValidEntityId(view.id)) return null;
   355→      return `#/inventory/stocktakes/${view.id}`;
   356→    case "purchase-orders":
   357→      // Prompt 41 (§70) — canonical list hash.
   358→      return "#/purchase-orders";
   359→    case "purchase-order":
   360→      // Prompt 41 (§70) — only charset-safe ids ever become a hash.
   361→      if (!isValidEntityId(view.id)) return null;
   362→      return `#/purchase-orders/${view.id}`;
   363→    case "inventory-goods-receipts":
   364→      // Prompt 42 (§116) — canonical list hash.
   365→      return "#/inventory/goods-receipts";
   366→    case "inventory-goods-receipt":
   367→      // Prompt 42 (§116) — only charset-safe ids ever become a hash.
   368→      if (!isValidEntityId(view.id)) return null;
   369→      return `#/inventory/goods-receipts/${view.id}`;
   370→    case "inventory-goods-returns":
   371→      // Prompt 43 (§91) — canonical list hash.
   372→      return "#/inventory/goods-returns";
   373→    case "inventory-goods-return":
   374→      // Prompt 43 (§91) — only charset-safe ids ever become a hash.
   375→      if (!isValidEntityId(view.id)) return null;
   376→      return `#/inventory/goods-returns/${view.id}`;
   377→    case "inventory-goods-dispatches":
   378→      // Prompt 44 — canonical list hash.
   379→      return "#/inventory/goods-dispatches";
   380→    case "inventory-goods-dispatch":
   381→      // Prompt 44 — only charset-safe ids ever become a hash.
   382→      if (!isValidEntityId(view.id)) return null;
   383→      return `#/inventory/goods-dispatches/${view.id}`;
   384→    case "sales-orders":
   385→      // Prompt 45 (§88) — canonical list hash.
   386→      return "#/sales-orders";
   387→    case "sales-order":
   388→      // Prompt 45 (§88) — only charset-safe ids ever become a hash.
   389→      if (!isValidEntityId(view.id)) return null;
   390→      return `#/sales-orders/${view.id}`;
   391→    case "platform":
   392→      // #/platform itself resolves to the overview section (§15).
   393→      return view.section === "overview" ? "#/platform" : `#/platform/${view.section}`;
   394→    case "module": {
   395→      if (!isKnownModule(view.module)) return null;
   396→      // Module sub-state deep links (e.g. #/reports?month=YYYY-MM) stay in
   397→      // the address bar — month is the only sub-param today.
   398→      if (view.param && /^\d{4}-\d{2}$/.test(view.param)) {
   399→        return `#/${view.module}?month=${view.param}`;
   400→      }
   401→      return `#/${view.module}`;
   402→    }
   403→    case "join":
   404→      return `#/join/${view.token}`;
   405→    case "forgot-password":
   406→      return "#/forgot-password";
   407→    case "reset-password":
   408→      return `#/reset-password/${view.token}`;
   409→    case "verify-email":
   410→      return `#/verify-email/${view.token}`;
   411→    case "onboarding":
   412→      return null;
   413→  }
   414→}

   416→/** Parse a location hash into a view. Unknown → dashboard (never onboarding). */
   417→export function hashToView(hash: string): AppView {
   418→  const raw = hash.replace(/^#\/?/, "").replace(/\/+$/, "");
   419→  // Query part (e.g. #/reports?month=2026-04) — module-scoped deep-link params.
   420→  const qIndex = raw.indexOf("?");
   421→  const path = (qIndex === -1 ? raw : raw.slice(0, qIndex)).trim();
   422→  const query = new URLSearchParams(qIndex === -1 ? "" : raw.slice(qIndex + 1));
   423→  if (!path || path === "dashboard") return { kind: "dashboard" };
   424→  if (path === "audit") return { kind: "audit" };
   425→  if (path === "settings") return { kind: "settings:business" };
   426→  if (path === "settings/members") return { kind: "settings:members" };
   427→  if (path === "settings/billing") return { kind: "settings:billing" };
   428→  if (path === "settings/security") return { kind: "settings:security" };
   429→  if (path === "data") return { kind: "settings:data" };
   430→  if (path === "notifications") return { kind: "notifications" };
   431→  if (path === "settings/notifications") return { kind: "settings:notifications" };
   432→  // Prompt 33 — Search Results Center deep links (#/search, #/search?q=…).
   433→  if (path === "search") {
   434→    const q = query.get("q") ?? "";
   435→    const clean = sanitizeSearchHashQuery(q);
   436→    return clean ? { kind: "search", query: clean } : { kind: "search" };
   437→  }
   438→  // Prompt 35 (§37) — CRM deep links: #/crm, #/crm?tab=<closed tab>.
   439→  // Unknown tab values fall back to CRM_DEFAULT_TAB (pipeline) — the hash
   440→  // query is never interpreted as arbitrary state.
   441→  if (path === "crm") {
   442→    const tab = query.get("tab") ?? "";
   443→    return { kind: "crm", tab: isCrmTab(tab) ? tab : CRM_DEFAULT_TAB };
   444→  }
   445→  if (path === "profile") return { kind: "profile" };
   446→  // Prompt 28 — platform console deep links (#/platform, #/platform/<section>).
   447→  if (path === "platform") return { kind: "platform", section: "overview" };
   448→  if (path.startsWith("platform/")) {
   449→    const section = path.slice("platform/".length);
   450→    if (isPlatformSection(section)) return { kind: "platform", section };
   451→    return { kind: "platform", section: "overview" };
   452→  }
   453→  if (path.startsWith("join/")) {
   454→    const token = path.slice("join/".length);
   455→    // Opaque invite token: non-empty, bounded length (server validates the rest).
   456→    if (token.length > 0 && token.length <= 128) return { kind: "join", token };
   457→  }
   458→  if (path === "forgot-password") return { kind: "forgot-password" };
   459→  if (path.startsWith("verify-email/")) {
   460→    const token = path.slice("verify-email/".length);
   461→    // Opaque verification token: bounded length; the server validates the rest.
   462→    if (token.length > 0 && token.length <= 200) return { kind: "verify-email", token };
   463→    return { kind: "dashboard" };
   464→  }
   465→  if (path.startsWith("reset-password/")) {
   466→    const token = path.slice("reset-password/".length);

   587→  // Prompt 34 — entity detail deep links (#/customers/<id>, §5/§6) are
   588→  // attempted BEFORE the bare-module match: "customers" alone stays the
   589→  // module list while "customers/<id>" resolves through the centralized
   590→  // parser (allowlisted first segment, bounded charset-safe id). Anything
   591→  // else falls through to the dashboard fallback (§6.6) — never an
   592→  // arbitrary "#/anything/<id>" interpretation.
   593→  const entityRoute = parseEntityRoute(path);
   594→  if (entityRoute) return { kind: "entity", ...entityRoute };
   595→  if (isKnownModule(path)) {
   596→    // Module sub-param deep links: ?month=YYYY-MM opens reports straight
   597→    // into the drill-down, or pre-filters sales/expenses to that month
   598→    // (e.g. "Open in Sales" from the reports month dialog).
   599→    const month = query.get("month");

   601→      return { kind: "module", module: path, param: month };
   602→    }
   603→    return { kind: "module", module: path };
   604→  }
   605→  return { kind: "dashboard" };
   606→}

   608→/** Sync the URL to a view without triggering navigation events. */
   609→function syncUrl(view: AppView, replace: boolean) {
   610→  if (typeof window === "undefined") return;
   611→  const hash = viewToHash(view);
   612→  if (!hash) return;
   613→  const target = `${window.location.pathname}${window.location.search}${hash}`;
   614→  if (`${window.location.pathname}${window.location.search}${window.location.hash}` === target) {
   615→    return;
   616→  }
   617→  try {
   618→    if (replace) window.history.replaceState(null, "", target);
   619→    else window.history.pushState(null, "", target);
   620→  } catch {
   621→    // history might be unavailable in exotic embeds — URL sync is best-effort
   622→  }
   623→}

   625→/**
   626→ * Prompt 33 — hash `?q=` hygiene (§16/§29): the query is rendered strictly
   627→ * as text, so only control characters/null bytes are stripped and the
   628→ * length is capped to the search schema bound. URLSearchParams already
   629→ * performs percent-decoding; no other transformation is applied.
   630→ */
   631→function sanitizeSearchHashQuery(raw: string): string {
   632→  const capped = Array.from(raw)
   633→    .filter((ch) => {
   634→      const code = ch.codePointAt(0) ?? 0;
   635→      return code > 0x1f && code !== 0x7f;
   636→    })
   637→    .slice(0, 64)
   638→    .join("");
   639→  const trimmed = capped.trim();
   640→  return trimmed.length > 0 ? trimmed : "";
   641→}
