     3→/**
     4→ * Prompt 43 — goods-return service (server-only, §136).
     5→ *
     6→ * LAWS:
     7→ *  - PHYSICAL INVENTORY OUTFLOW ONLY (§4/§6/§7): posting DECREASES
     8→ *    Product.stockQty and creates ONE negative InventoryAdjustment per
     9→ *    line (delta = -returnedQty). It NEVER creates Expense, Payment, Sale,
    10→ *    Credit Note, supplier-statement movement, payable/aging row or any
    11→ *    accounting entry (§7). Draft/cancel mutate NOTHING but the return.
    12→ *  - RECEIPT IMMUTABILITY (§8): a return NEVER mutates its source receipt
    13→ *    — status, receivedQty, snapshots and lines stay frozen forever.
    14→ *  - CUMULATIVE RETURN CAP (§35): SUM(POSTED returned) + new ≤ receivedQty,
    15→ *    enforced by the ONE pure helper at line-add (advisory, §62) AND
    16→ *    authoritatively inside the post transaction (§43 step 11). No
    17→ *    tolerance, no bypass, no rounding.
    18→ *  - CURRENT STOCK CHECK (§39/§41): each posted line requires
    19→ *    Product.stockQty ≥ returnedQty; the aggregate stock can NEVER go
    20→ *    negative — no override, no silent zero-clamp (§42 exact integer math).
    21→ *  - Every query is tenant-scoped by ctx.businessId (§67) — never a client
    22→ *    businessId. Foreign receipt/receipt-line/return/return-line → 404, no
    23→ *    existence leak (§68).
    24→ *  - Status transitions are server-authoritative through dedicated
    25→ *    operations (§9); the current status is RE-CHECKED INSIDE the write
    26→ *    transaction (§43 step 3) so concurrent posts/cancels cannot double-apply.
    27→ *  - Snapshots (supplier name, receipt reference, PO reference, product
    28→ *    name/sku/unit) are frozen at write time from ALREADY-FROZEN receipt
    29→ *    data (§16/§17/§18/§30) and NEVER overwritten from live data.
    30→ *  - Quantities are exact integers (§32/§42); corrupt persisted state is
    31→ *    surfaced, never repaired (§38/§41/§65).
    32→ *  - READ-CAP LAW (§76/§140): every bounded read fetches cap+1 rows to
    33→ *    PROVE completeness — a document holding more rows than its own cap
    34→ *    (or a `_count` mismatch) fails the read with the stable
    35→ *    GOODS_RETURN_LINE_CAP_INVARIANT code. Read paths NEVER render a
    36→ *    silently truncated slice or a partial total.
    37→ *  - AGGREGATE LAW (§37/§140): POSTED returned quantities are summed by
    38→ *    the DATABASE (aggregate/groupBy _sum), never by loading return
    39→ *    history into application memory. The aggregate result is re-validated
    40→ *    (integer, ≥ 0); corruption still surfaces, never clamps.
    41→ *  - Audit through the single sanctioned helper INSIDE the same
    42→ *    transaction as every mutation (§48/§133) — safe metadata only, never
    43→ *    notes content (§22/§132).

   109→/** §65 — corrupted persisted status → controlled 500 (never Draft). */
   110→function parseDbStatus(value: string): GoodsReturnStatus {
   111→  try {
   112→    return parseGoodsReturnStatus(value);
   113→  } catch (error) {
   114→    if (error instanceof Error && error.name === "GoodsReturnStatusIntegrityError") {
   115→      throw new ApiError(
   116→        500,
   117→        GOODS_RETURN_INVALID_PERSISTED_STATUS,
   118→        "Goods return data is in an invalid state. Please contact support.",
   119→      );
   120→    }
   121→    throw error;
   122→  }
   123→}

   125→/**
   126→ * §38/§41 — map pure canonical-math integrity failures to their controlled
   127→ * HTTP result: corrupted return history / corrupt stock is a 500, never a
   128→ * silent clamp. Exported for cross-service mapping (the P42 receiving
   129→ * path surfaces return-history corruption under the SAME stable code).
   130→ */
   131→export function mapPureGoodsReturnDomainError(error: unknown): never {
   132→  if (error instanceof Error && error.name === "GoodsReturnStatusIntegrityError") {
   133→    throw new ApiError(
   134→      500,
   135→      GOODS_RETURN_INVALID_PERSISTED_STATUS,
   136→      "Goods return data is in an invalid state. Please contact support.",
   137→    );
   138→  }
   139→  if (error instanceof Error && error.name === "GoodsReceiptStatusIntegrityError") {
   140→    // §66 — corrupted SOURCE receipt status uses P42's own stable code.
   141→    throw new ApiError(
   142→      500,
   143→      GOODS_RECEIPT_INVALID_PERSISTED_STATUS,
   144→      "Goods receipt data is in an invalid state. Please contact support.",
   145→    );
   146→  }
   147→  if (error instanceof Error && error.name === "GoodsReturnIntegrityError") {
   148→    // §15/§16 — the line-set completeness invariant (over-cap document or a
   149→    // counted/fetched mismatch) is ITS OWN stable integrity code — never
   150→    // misreported as generic history corruption.
   151→    if (error.message.includes(GOODS_RETURN_LINE_CAP_INVARIANT)) {
   152→      throw new ApiError(
   153→        500,
   154→        GOODS_RETURN_LINE_CAP_INVARIANT,
   155→        "Stored document data exceeds its configured line cap. Please contact support.",
   156→      );
   157→    }
   158→    const corruptStock = error.message.includes("GOODS_RETURN_INVALID_PERSISTED_STOCK");
   159→    throw new ApiError(
   160→      500,
   161→      corruptStock
   162→        ? GOODS_RETURN_ERROR_CODES.INVALID_PERSISTED_STOCK
   163→        : GOODS_RETURN_ERROR_CODES.CORRUPT_HISTORY,
   164→      corruptStock
   165→        ? "Product stock data is in an invalid state. Please contact support."
   166→        : "Goods return data is inconsistent. Please contact support.",
   167→    );
   168→  }
   169→  throw error;
   170→}

   239→/**
   240→ * §13/§43 step 5/§66 — load the tenant-scoped source receipt and require
   241→ * persisted status POSTED. A corrupted receipt status NEVER becomes
   242→ * POSTED — P42's own parser + stable code are reused (§66).

   244→async function loadPostedSourceReceipt(tx: Tx, businessId: string, goodsReceiptId: string) {
   245→  const receipt = await tx.goodsReceipt.findFirst({
   246→    where: { id: goodsReceiptId, businessId }, // §67/§68 — foreign → 404
   247→    select: {
   248→      id: true,
   249→      reference: true,
   250→      status: true,
   251→      supplierNameAtReceipt: true,
   252→      purchaseOrderReferenceAtReceipt: true,
   253→    },
   254→  });
   255→  if (!receipt) throw new ApiError(404, "NOT_FOUND", "Goods receipt not found.");
   256→  let status: string;
   257→  try {
   258→    status = parseGoodsReceiptStatus(receipt.status);
   259→  } catch (error) {
   260→    throw mapPureGoodsReturnDomainError(error); // §66 — stable P42 code
   261→  }
   262→  if (status !== "POSTED") {
   263→    // §13 — returns may only be created from POSTED receipts.
   264→    throw new ApiError(
   265→      422,
   266→      GOODS_RETURN_ERROR_CODES.SOURCE_NOT_POSTED,
   267→      "Goods can be returned only from a posted goods receipt.",
   268→    );
   269→  }
   270→  return { ...receipt, status };
   271→}

   293→/**
   294→ * §37/§140 — the ONE canonical POSTED-returned-quantity sum for ONE
   295→ * receipt line. DRAFT/CANCELLED returns never count (§9/§10). Database
   296→ * aggregate (_sum), never a findMany of history into memory; the result
   297→ * is re-validated (exact integer, ≥ 0) and corruption surfaces through
   298→ * the ONE mapping, never clamped.
   299→ */
   300→export async function sumPostedReturnedQtyForReceiptLine(
   301→  tx: Tx,
   302→  goodsReceiptLineId: string,
   303→): Promise<number> {
   304→  const agg = await tx.goodsReturnLine.aggregate({
   305→    where: {
   306→      goodsReceiptLineId,
   307→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   308→    },
   309→    _sum: { returnedQty: true },
   310→  });
   311→  return validateReturnedSum(agg._sum.returnedQty ?? 0);
   312→}
   313→
   314→/**
   315→ * §37/§77/§80 — the ONE canonical POSTED-returned-quantity sum for ONE
   316→ * PO line (all return lines whose SOURCE receipt line belongs to that PO
   317→ * line). Database aggregate with a join filter — no receipt-line
   318→ * materialization. Used by the P42 net-receiving guard (line add / line
   319→ * update / receipt POST) and by the PO net receiving summary.
   320→ */
   321→export async function sumPostedReturnedQtyForPoLine(
   322→  tx: Tx,
   323→  purchaseOrderLineId: string,
   324→): Promise<number> {
   325→  const agg = await tx.goodsReturnLine.aggregate({
   326→    where: {
   327→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   328→      goodsReceiptLine: { purchaseOrderLineId },
   329→    },
   330→    _sum: { returnedQty: true },
   331→  });
   332→  return validateReturnedSum(agg._sum.returnedQty ?? 0);
   333→}
   334→
   335→/** §38 — aggregate result re-validation (SAFE integer, ≥ 0, no clamp, §8). */
   336→function validateReturnedSum(sum: number): number {
   337→  if (!Number.isSafeInteger(sum) || sum < 0) {
   338→    throw mapPureGoodsReturnDomainError(
   339→      new Error("GOODS_RETURN_CORRUPT_HISTORY: negative or unsafe aggregate"),
   340→    );
   341→  }
   342→  return sum;
   343→}
   344→
   345→/**
   346→ * §141 — grouped returned-quantity sums for MANY receipt lines (bounded
   347→ * batch, no N+1): one groupBy + (only when needed) one bounded id-in
   348→ * mapping fetch. Used by the receipt-detail return facts.
   349→ */
   350→export async function sumPostedReturnedQtyForReceiptLines(
   351→  db: DbLike,
   352→  goodsReceiptLineIds: string[],
   353→): Promise<Map<string, number>> {
   354→  const result = new Map(goodsReceiptLineIds.map((id) => [id, 0]));
   355→  if (goodsReceiptLineIds.length === 0) return result;
   356→  const grouped = await db.goodsReturnLine.groupBy({
   357→    by: ["goodsReceiptLineId"],
   358→    where: {
   359→      goodsReceiptLineId: { in: goodsReceiptLineIds },
   360→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   361→    },
   362→    _sum: { returnedQty: true },
   363→  });
   364→  for (const g of grouped) {
   365→    result.set(g.goodsReceiptLineId, validateReturnedSum(g._sum.returnedQty ?? 0));
   366→  }
   367→  return result;
   368→}
   369→
   370→/**
   371→ * §77/§141 — grouped returned-quantity sums for MANY PO lines (bounded
   372→ * batch, no N+1): one groupBy (by source receipt line) + one bounded
   373→ * id-in mapping fetch (only receipt lines that actually HAVE returns —
   374→ * never the PO's whole receipt history). Used by the PO net receiving
   375→ * summary and the PO detail aggregate.
   376→ */
   377→export async function sumPostedReturnedQtyForPoLines(
   378→  db: DbLike,
   379→  purchaseOrderLineIds: string[],
   380→): Promise<Map<string, number>> {
   381→  const result = new Map(purchaseOrderLineIds.map((id) => [id, 0]));
   382→  if (purchaseOrderLineIds.length === 0) return result;
   383→  const grouped = await db.goodsReturnLine.groupBy({
   384→    by: ["goodsReceiptLineId"],
   385→    where: {
   386→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   387→      goodsReceiptLine: { purchaseOrderLineId: { in: purchaseOrderLineIds } },
   388→    },
   389→    _sum: { returnedQty: true },
   390→  });
   391→  if (grouped.length === 0) return result;
   392→  // §141 — bounded batch: only the receipt lines that appear in return
   393→  // history, fetched by indexed id (NOT the PO's whole receipt set).
   394→  const receiptLineIds = grouped.map((g) => g.goodsReceiptLineId);
   395→  const mapping = await db.goodsReceiptLine.findMany({
   396→    where: { id: { in: receiptLineIds } },
   397→    select: { id: true, purchaseOrderLineId: true },
   398→  });
   399→  const poLineByReceiptLine = new Map(mapping.map((m) => [m.id, m.purchaseOrderLineId]));
   400→  for (const g of grouped) {
   401→    const poLineId = poLineByReceiptLine.get(g.goodsReceiptLineId);
   402→    if (!poLineId) continue; // defensive — FK guarantees existence
   403→    // §6 — every accumulation step stays a safe non-negative integer; the
   404→    // addition is re-validated, never trusted.
   405→    result.set(
   406→      poLineId,
   407→      validateReturnedSum((result.get(poLineId) ?? 0) + validateReturnedSum(g._sum.returnedQty ?? 0)),
   408→    );
   409→  }
   410→  return result;
   411→}

   413→/** §71/§20 — defensive line-count validation (add line). */
   414→async function assertLineCap(tx: Tx, goodsReturnId: string, cap: number): Promise<number> {
   415→  const lineCount = await tx.goodsReturnLine.count({ where: { goodsReturnId } });
   416→  if (lineCount > cap) {
   417→    // §20 — an ALREADY-corrupted document (more lines than its own cap) is
   418→    // database corruption: 500 integrity, never disguised as the ordinary
   419→    // 422 user validation failure.
   420→    throw new ApiError(
   421→      500,
   422→      GOODS_RETURN_LINE_CAP_INVARIANT,
   423→      "Stored document data exceeds its configured line cap. Please contact support.",
   424→    );
   425→  }
   426→  if (lineCount === cap) {
   427→    // §20 — the NORMAL write-time rejection for a legitimately full draft.
   428→    throw new ApiError(
   429→      422,
   430→      GOODS_RETURN_ERROR_CODES.LINE_LIMIT,
   431→      `A goods return can contain at most ${cap} lines.`,
   432→    );
   433→  }
   434→  return lineCount;
   435→}

   439→export async function createGoodsReturn(input: {
   440→  businessId: string;
   441→  userId: string;
   442→  goodsReceiptId: string;
   443→  reference?: string;
   444→  notes?: string | null;
   445→  ip?: string | null;
   446→  userAgent?: string | null;
   447→}) {
   448→  return withWriteTx(async (tx) => {
   449→    // §13/§16/§17/§18 — the receipt must be tenant-scoped and POSTED; all
   450→    // snapshots derive from ITS frozen columns (never client input, never
   451→    // live Supplier/Product rows).
   452→    const receipt = await loadPostedSourceReceipt(tx, input.businessId, input.goodsReceiptId);
   453→
   454→    // §20 — reference: user-supplied (normalized) or server-generated.
   455→    let reference = input.reference;
   456→    if (reference === undefined) {
   457→      for (let attempt = 0; attempt < 5; attempt += 1) {
   458→        const candidate = generateGoodsReturnReference(new Date(), (max) => randomInt(max));
   459→        const clash = await tx.goodsReturn.findFirst({
   460→          where: { businessId: input.businessId, reference: candidate },
   461→          select: { id: true },
   462→        });
   463→        if (!clash) {
   464→          reference = candidate;
   465→          break;
   466→        }
   467→      }
   468→      if (reference === undefined) {
   469→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   470→        throw new ApiError(
   471→          409,
   472→          GOODS_RETURN_ERROR_CODES.DUPLICATE_REFERENCE,
   473→          "Could not generate a unique reference. Please try again.",
   474→        );
   475→      }
   476→    }
   477→
   478→    try {
   479→      const row = await tx.goodsReturn.create({
   480→        data: {
   481→          businessId: input.businessId, // §25 — server-derived, never client
   482→          goodsReceiptId: receipt.id,
   483→          reference,
   484→          status: "DRAFT", // §10 — preparation only
   485→          supplierNameAtReturn: receipt.supplierNameAtReceipt, // §16 — frozen identity
   486→          goodsReceiptReferenceAtReturn: receipt.reference, // §17 — frozen identity
   487→          purchaseOrderReferenceAtReturn: receipt.purchaseOrderReferenceAtReceipt, // §18 — frozen identity
   488→          notes: input.notes ?? null, // §22 — document data only
   489→          createdByUserId: input.userId, // §56 — server-derived
   490→        },
   491→        select: { id: true, reference: true, status: true },
   492→      });
   493→
   494→      await audit(
   495→        auditInput(input.businessId, input.userId, "goods_return.create", row.id, row.reference, {
   496→          goodsReceiptId: receipt.id,
   497→          goodsReceiptReferenceAtReturn: receipt.reference,
   498→          purchaseOrderReferenceAtReturn: receipt.purchaseOrderReferenceAtReceipt,
   499→          supplierNameAtReturn: receipt.supplierNameAtReceipt,
   500→          status: row.status,
   501→        }),
   502→        tx,
   503→      );
   504→
   505→      return row;
   506→    } catch (error) {
   507→      // §21 — database uniqueness stays authoritative for duplicates.
   508→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   509→        throw new ApiError(
   510→          409,
   511→          GOODS_RETURN_ERROR_CODES.DUPLICATE_REFERENCE,
   512→          "A goods return with this reference already exists.",
   513→        );
   514→      }
   515→      throw error;
   516→    }
   517→  });
   518→}

   522→export async function listGoodsReturns(input: {
   523→  businessId: string;
   524→  q?: string;
   525→  status?: string; // ALL → undefined
   526→  goodsReceiptId?: string;
   527→  from?: string;
   528→  to?: string;
   529→  timezone?: string; // §22-§23 — AUTHENTICATED business timezone (never client authority)
   530→  page: number;
   531→  pageSize: number;
   532→}): Promise<{
   533→  items: GoodsReturnListItemDto[];
   534→  total: number;
   535→  page: number;
   536→  pageSize: number;
   537→}> {
   538→  const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
   539→  const where: Prisma.GoodsReturnWhereInput = {
   540→    businessId: input.businessId, // §67 — tenant scope
   541→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   542→    ...(input.goodsReceiptId ? { goodsReceiptId: input.goodsReceiptId } : {}),
   543→    ...(input.from || input.to
   544→      ? (() => {
   545→          // §22-§24 — the from/to calendar dates are interpreted in the
   546→          // AUTHENTICATED BUSINESS's timezone (inclusive local start of
   547→          // `from`, EXCLUSIVE local start of the day after `to`). A record
   548→          // at 00:15 local belongs to its local date even when its UTC
   549→          // date is the previous day. Unresolvable bounds → controlled 422.
   550→          const window = goodsReturnDayWindow(input.from, input.to, input.timezone);
   551→          if (!window) {
   552→            throw new ApiError(422, "VALIDATION_ERROR", "Invalid date filter.");
   553→          }
   554→          return { createdAt: window };
   555→        })()
   556→      : {}),
   557→    ...(input.q
   558→      ? {
   559→          OR: [
   560→            { reference: { contains: input.q } }, // return reference (§73)
   561→            { goodsReceiptReferenceAtReturn: { contains: input.q } }, // receipt reference
   562→            { purchaseOrderReferenceAtReturn: { contains: input.q } }, // PO reference snapshot
   563→            { supplierNameAtReturn: { contains: input.q } }, // supplier snapshot
   564→            {
   565→              lines: {
   566→                some: {
   567→                  OR: [
   568→                    { productNameAtReturn: { contains: input.q } }, // product snapshot
   569→                    { skuAtReturn: { contains: input.q } }, // SKU snapshot
   570→                  ],
   571→                },
   572→              },
   573→            },
   574→          ],
   575→        }
   576→      : {}),
   577→  };
   578→
   579→  const [total, rows] = await Promise.all([
   580→    db.goodsReturn.count({ where }),
   581→    // §72 — DB-level pagination; never findMany + slice.
   582→    db.goodsReturn.findMany({
   583→      where,
   584→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   585→      skip: (input.page - 1) * input.pageSize,
   586→      take: input.pageSize,
   587→      select: {
   588→        id: true,
   589→        reference: true,
   590→        status: true,
   591→        goodsReceiptId: true,
   592→        goodsReceiptReferenceAtReturn: true,
   593→        purchaseOrderReferenceAtReturn: true,
   594→        supplierNameAtReturn: true,
   595→        postedAt: true,
   596→        cancelledAt: true,
   597→        createdAt: true,
   598→        lines: {
   599→          // §76 — bounded aggregate relation, not one query per row.
   600→          // take cap+1 PROVES the slice is the whole line set: a return
   601→          // over its own cap is corruption and fails the read below —
   602→          // never a silently truncated total.
   603→          select: { returnedQty: true },
   604→          take: cap + 1,
   605→        },
   606→        _count: { select: { lines: true } },
   607→      },
   608→    }),
   609→  ]);
   610→
   611→  const items = rows.map((row) => {
   612→    try {
   613→      // §15/§16 — the THREE explicit completeness invariants (fetched ≤ cap,
   614→      // counted ≤ cap, counted === fetched) are enforced by the ONE
   615→      // canonical pure validator: a count mismatch fails the read EVEN WHEN
   616→      // both numbers are below the cap — a subset total must never
   617→      // masquerade as complete. Then §76 — the proven-complete slice feeds
   618→      // the canonical sum, which re-validates every persisted quantity
   619→      // (safe integer ≥ 1 — a stored zero is corruption, §7).
   620→      assertGoodsReturnLineSetComplete(row.lines.length, row._count.lines, cap);
   621→      const totalReturnedQty = computeReturnedQty(row.lines.map((l) => l.returnedQty));
   622→      return goodsReturnListItemDto({
   623→        id: row.id,
   624→        reference: row.reference,
   625→        status: row.status,
   626→        goodsReceiptId: row.goodsReceiptId,
   627→        goodsReceiptReferenceAtReturn: row.goodsReceiptReferenceAtReturn,
   628→        purchaseOrderReferenceAtReturn: row.purchaseOrderReferenceAtReturn,
   629→        supplierNameAtReturn: row.supplierNameAtReturn,
   630→        lineCount: row._count.lines,
   631→        totalReturnedQty,
   632→        postedAt: row.postedAt,
   633→        cancelledAt: row.cancelledAt,
   634→        createdAt: row.createdAt,
   635→        createdByUserId: "",
   636→        createdByName: null,
   637→        notes: null,
   638→      });
   639→    } catch (error) {
   640→      // §38 — a corrupted quantity inside history surfaces as the stable
   641→      // integrity 500, never a partial row and never a generic 500.
   642→      return mapPureGoodsReturnDomainError(error);
   643→    }
   644→  });
   645→  return { items, total, page: input.page, pageSize: input.pageSize };
   646→}

   785→export async function addGoodsReturnLine(input: {
   786→  businessId: string;
   787→  userId: string;
   788→  id: string;
   789→  goodsReceiptLineId: string;
   790→  returnedQty: number;
   791→  reasonCode: string;
   792→  ip?: string | null;
   793→  userAgent?: string | null;
   794→}) {
   795→  return withWriteTx(async (tx) => {
   796→    const header = await loadReturnHeader(tx, input.businessId, input.id);
   797→    requireDraft(header.status); // §62 — DRAFT-only
   798→
   799→    // §62/§28 — the source receipt line must belong to THIS return's
   800→    // tenant-scoped receipt. One tenant-scoped join validates
   801→    // return→receipt→line ownership; a foreign line is indistinguishable
   802→    // from an unknown one (§68 → 404, no existence leak).
   803→    const receiptLine = await tx.goodsReceiptLine.findFirst({
   804→      where: { id: input.goodsReceiptLineId, goodsReceiptId: header.goodsReceiptId },
   805→      select: {
   806→        id: true,
   807→        productId: true,
   808→        receivedQty: true,
   809→        productNameAtReceipt: true, // §30 — snapshots copied from the receipt line
   810→        skuAtReceipt: true,
   811→        unitAtReceipt: true,
   812→        product: { select: { id: true, businessId: true } },
   813→      },
   814→    });
   815→    if (!receiptLine) {
   816→      throw new ApiError(
   817→        404,
   818→        GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
   819→        "Goods receipt line not found.",
   820→      );
   821→    }
   822→
   823→    // §62 — the product must exist inside the SAME tenant (belt-and-braces:
   824→    // the FK is Restrict so a hard-deleted product is nearly impossible,
   825→    // but the safe controlled failure is demanded, never fabrication).
   826→    if (receiptLine.product.businessId !== input.businessId) {
   827→      throw new ApiError(
   828→        404,
   829→        GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
   830→        "Goods receipt line not found.",
   831→      );
   832→    }
   833→
   834→    // §62 — the source receipt must STILL be POSTED at line-add time (a
   835→    // receipt cannot leave POSTED, but the defensive re-check keeps the
   836→    // invariant local and fail-loud if that ever changed).
   837→    await loadPostedSourceReceipt(tx, input.businessId, header.goodsReceiptId);
   838→
   839→    const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
   840→    await assertLineCap(tx, header.id, cap); // §62 — defensive count
   841→
   842→    // §62/§35 — ADVISORY cumulative check at line-add time: returnedQty ≤
   843→    // (received − POSTED returned). Draft lines of OTHER returns are
   844→    // deliberately NOT subtracted — over-allocation across drafts is the
   845→    // documented §63 stale-draft scenario and only POST is authoritative.
   846→    const alreadyReturned = await sumPostedReturnedQtyForReceiptLine(tx, receiptLine.id);
   847→    const returnable = computeReturnableQty(receiptLine.receivedQty, alreadyReturned); // §36
   848→    if (exceedsReturnable(receiptLine.receivedQty, alreadyReturned, input.returnedQty)) {
   849→      throw new ApiError(
   850→        409,
   851→        GOODS_RETURN_ERROR_CODES.EXCEEDS_RETURNABLE,
   852→        `Only ${returnable} unit(s) remain returnable on this receipt line.`,
   853→      );
   854→    }
   855→
   856→    try {
   857→      const line = await tx.goodsReturnLine.create({
   858→        data: {
   859→          goodsReturnId: header.id,
   860→          goodsReceiptLineId: receiptLine.id,
   861→          productId: receiptLine.productId, // §28/§47 — derived, never client authority
   862→          returnedQty: input.returnedQty,
   863→          reasonCode: input.reasonCode, // §23 — closed vocabulary (zod)
   864→          // §30 — snapshots copied from the RECEIPT LINE, not live Product.
   865→          productNameAtReturn: receiptLine.productNameAtReceipt,
   866→          skuAtReturn: receiptLine.skuAtReceipt,
   867→          unitAtReturn: receiptLine.unitAtReceipt,
   868→        },
   869→        select: { id: true },
   870→      });
   871→
   872→      await audit(
   873→        auditInput(input.businessId, input.userId, "goods_return.line_add", header.id, header.reference, {
   874→          lineId: line.id, // §132 — safe per-line metadata
   875→          goodsReceiptLineId: receiptLine.id,
   876→          productId: receiptLine.productId,
   877→          productNameAtReturn: receiptLine.productNameAtReceipt,
   878→          returnedQty: input.returnedQty,
   879→          reasonCode: input.reasonCode,
   880→        }),
   881→        tx,
   882→      );
   883→
   884→      return { id: line.id };
   885→    } catch (error) {
   886→      // §29/§75 — DB uniqueness is authoritative for duplicate lines.
   887→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   888→        throw new ApiError(
   889→          409,
   890→          GOODS_RETURN_ERROR_CODES.DUPLICATE_LINE,
   891→          "This goods receipt line is already on the return.",
   892→        );
   893→      }
   894→      throw error;
   895→    }
   896→  });
   897→}

  1020→/* ---------------------------------- post ----------------------------------- */
  1021→
  1022→/**
  1023→ * §43 — THE atomic stock posting transaction. Steps 1-21 in ONE
  1024→ * database transaction (§48); a failure at ANY step rolls back EVERYTHING
  1025→ * (§52) — no stock change, no adjustment, no status, no audit.
  1026→ */
  1027→export async function postGoodsReturn(input: {
  1028→  businessId: string;
  1029→  userId: string;
  1030→  id: string;
  1031→  ip?: string | null;
  1032→  userAgent?: string | null;
  1033→}) {
  1034→  return withWriteTx(async (tx) => {
  1035→    // §43 steps 1-3 — reload return INSIDE the transaction; never trust an
  1036→    // earlier GET. Corrupted status is surfaced, never Draft. Only
  1037→    // DRAFT → POSTED (§9 — at most one successful transition §53).
  1038→    const header = await loadReturnHeader(tx, input.businessId, input.id);
  1039→    assertTransition(header.status, "POSTED");
  1040→
  1041→    // §43 steps 5-6 — the source receipt must STILL exist in this tenant
  1042→    // and be POSTED (corrupted receipt status never becomes POSTED, §66).
  1043→    const receipt = await loadPostedSourceReceipt(tx, input.businessId, header.goodsReceiptId);
  1044→
  1045→    // §43 steps 7-8 — load the return's own lines (bounded by the document
  1046→    // cap; cap+1 proves completeness).
  1047→    const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
  1048→    const lines = await tx.goodsReturnLine.findMany({
  1049→      where: { goodsReturnId: header.id },
  1050→      orderBy: [{ createdAt: "asc" }, { id: "asc" }],
  1051→      take: cap + 1, // cap+1 proves completeness
  1052→      select: {
  1053→        id: true,
  1054→        goodsReceiptLineId: true,
  1055→        productId: true,
  1056→        returnedQty: true,
  1057→        productNameAtReturn: true,
  1058→      },
  1059→    });
  1060→    if (lines.length > cap) {
  1061→      // §18/§19 — a PERSISTED document holding more lines than its own
  1062→      // invariant permits is DATA CORRUPTION → 500
  1063→      // GOODS_RETURN_LINE_CAP_INVARIANT. Never the 422 write-time
  1064→      // LINE_LIMIT error (that classification is for a newly submitted
  1065→      // over-limit request). No status change, no stock mutation, no
  1066→      // adjustment, no audit.
  1067→      throw new ApiError(
  1068→        500,
  1069→        GOODS_RETURN_LINE_CAP_INVARIANT,
  1070→        "Stored document data exceeds its configured line cap. Please contact support.",
  1071→      );
  1072→    }
  1073→
  1074→    // §61 — a return with zero lines can never post.
  1075→    if (lines.length === 0) {
  1076→      throw new ApiError(
  1077→        422,
  1078→        GOODS_RETURN_ERROR_CODES.EMPTY,
  1079→        "Add at least one line before posting this return.",
  1080→      );
  1081→    }
  1082→
  1083→    // §13 — validate EVERY persisted line quantity BEFORE any mutation: a
  1084→    // corrupted persisted line (zero, negative, fractional, unsafe, or over
  1085→    // the domain max) must NEVER become a zero-delta or positive-stock
  1086→    // adjustment. Controlled integrity error — no stock mutation, no
  1087→    // InventoryAdjustment, no status mutation, no post audit.
  1088→    for (const line of lines) {
  1089→      if (!isValidPersistedReturnedQty(line.returnedQty)) {
  1090→        throw new ApiError(
  1091→          500,
  1092→          GOODS_RETURN_ERROR_CODES.CORRUPT_HISTORY,
  1093→          "Goods return data is inconsistent. Please contact support.",
  1094→        );
  1095→      }
  1096→    }
  1097→
  1098→    // §43 step 9 — every line's source receipt line must belong to THIS
  1099→    // receipt (defensive re-verification of the document chain), with the
  1100→    // immutable receivedQty as the quantity authority (§34).
  1101→    const receiptLineIds = lines.map((l) => l.goodsReceiptLineId);
  1102→    const receiptLines = await tx.goodsReceiptLine.findMany({
  1103→      where: { id: { in: receiptLineIds } },
  1104→      select: { id: true, goodsReceiptId: true, receivedQty: true },
  1105→    });
  1106→    const receiptLineById = new Map(receiptLines.map((rl) => [rl.id, rl]));
  1107→
  1108→    // §43 step 10 / §37 — ONE grouped aggregate for ALL prior POSTED
  1109→    // returns of these source lines (never per-line queries, never an
  1110→    // unbounded history read).
  1111→    const returnedByLine = await sumPostedReturnedQtyForReceiptLines(tx, receiptLineIds);
  1112→
  1113→    const adjustmentIds: string[] = [];
  1114→    const stockChanges: { productId: string; from: number; to: number }[] = [];
  1115→
  1116→    for (const line of lines) {
  1117→      const sourceLine = receiptLineById.get(line.goodsReceiptLineId);
  1118→      if (!sourceLine || sourceLine.goodsReceiptId !== receipt.id) {
  1119→        // §43 step 9 / §142 — a detached source line is corruption; fail
  1120→        // safely (no partial posting), no existence leak.
  1121→        throw new ApiError(
  1122→          404,
  1123→          GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
  1124→          "Goods receipt line not found.",
  1125→        );
  1126→      }
  1127→
  1128→      // §43 step 11 / §35 — the HARD cumulative return cap, final
  1129→      // authority. Prior POSTED returns + this line ≤ receivedQty.
  1130→      const alreadyReturned = returnedByLine.get(line.goodsReceiptLineId) ?? 0;
  1131→      const returnable = computeReturnableQty(sourceLine.receivedQty, alreadyReturned); // §36
  1132→      if (exceedsReturnable(sourceLine.receivedQty, alreadyReturned, line.returnedQty)) {
  1133→        throw new ApiError(
  1134→          409,
  1135→          GOODS_RETURN_ERROR_CODES.EXCEEDS_RETURNABLE,
  1136→          `Only ${returnable} unit(s) remain returnable on this receipt line.`,
  1137→        );
  1138→      }
  1139→
  1140→      // §43 step 12 — reload the product (tenant-scoped; the product id
  1141→      // derives from the receipt line, §47 — never client authority).
  1142→      const product = await tx.product.findFirst({
  1143→        where: { id: line.productId, businessId: input.businessId }, // §67
  1144→        select: { id: true, stockQty: true },
  1145→      });
  1146→      if (!product) {
  1147→        // A hard-deleted product fails the post safely (never fabrication).
  1148→        throw new ApiError(
  1149→          422,
  1150→          GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
  1151→          "A product on this return no longer exists.",
  1152→        );
  1153→      }
  1154→
  1155→      // §41/§164 — corrupt/negative persisted stock is surfaced as the
  1156→      // controlled integrity failure BEFORE any sufficiency comparison:
  1157→      // a negative stock reading is corruption, not "insufficient stock",
  1158→      // and it is NEVER silently repaired or zeroed.
  1159→      if (!Number.isInteger(product.stockQty) || product.stockQty < 0) {
  1160→        throw new ApiError(
  1161→          500,
  1162→          GOODS_RETURN_ERROR_CODES.INVALID_PERSISTED_STOCK,
  1163→          "Product stock data is in an invalid state. Please contact support.",
  1164→        );
  1165→      }
  1166→
  1167→      // §39 — CURRENT STOCK CHECK: the physical decrement needs enough
  1168→      // aggregate stock NOW (the application has no lot/batch provenance —
  1169→      // documented limitation §40/§197).
  1170→      if (product.stockQty < line.returnedQty) {
  1171→        throw new ApiError(
  1172→          409,
  1173→          GOODS_RETURN_ERROR_CODES.INSUFFICIENT_STOCK,
  1174→          `Insufficient stock for ${line.productNameAtReturn}: ${product.stockQty} on hand, ${line.returnedQty} to return.`,
  1175→        );
  1176→      }
  1177→
  1178→      // §42 — exact integer arithmetic; stockAfterReturn re-validates both
  1179→      // operands and the non-negative result (defense in depth).
  1180→      const newStock = stockAfterReturn(product.stockQty, line.returnedQty);
  1181→
  1182→      // §43 steps 16 — decrement stock.
  1183→      await tx.product.update({
  1184→        where: { id: product.id },
  1185→        data: { stockQty: newStock },
  1186→        select: { id: true },
  1187→      });
  1188→      stockChanges.push({ productId: product.id, from: product.stockQty, to: newStock });
  1189→
  1190→      // §43 step 17 / §44-§47 — ONE negative InventoryAdjustment per line
  1191→      // (delta = -returnedQty), server-derived actor/business/product.
  1192→      const adjustment = await tx.inventoryAdjustment.create({
  1193→        data: {
  1194→          businessId: input.businessId, // §46 — server-derived
  1195→          productId: product.id, // §47 — derived from the receipt line chain
  1196→          delta: -line.returnedQty, // §44 — always negative
  1197→          reason: RETURN_ADJUSTMENT_REASON(header.reference), // §44 — stable identifier
  1198→          actorId: input.userId, // §45 — server-derived
  1199→        },
  1200→        select: { id: true },
  1201→      });
  1202→      adjustmentIds.push(adjustment.id);
  1203→    }
  1204→
  1205→    // §43 steps 18-19 — complete the posting (server timestamps only).
  1206→    const postedAt = new Date();
  1207→    const updated = await tx.goodsReturn.update({
  1208→      where: { id: header.id },
  1209→      data: { status: "POSTED", postedAt },
  1210→      select: { id: true, reference: true, status: true, postedAt: true },
  1211→    });
  1212→
  1213→    // §43 step 20 / §48/§133 — ONE audit record for the posting, written
  1214→    // in the SAME transaction; if it fails, everything above rolls back.
  1215→    const totalReturnedQty = computeReturnedQty(lines.map((l) => l.returnedQty));
  1216→    await audit(
  1217→      auditInput(input.businessId, input.userId, "goods_return.post", header.id, header.reference, {
  1218→        goodsReceiptId: receipt.id,
  1219→        goodsReceiptReferenceAtReturn: header.goodsReceiptReferenceAtReturn,
  1220→        lineCount: lines.length,
  1221→        totalReturnedQty, // §132 — quantity facts only; never notes
  1222→        adjustmentIds,
  1223→        previousStatus: header.status,
  1224→        newStatus: updated.status,
  1225→        postedAt: updated.postedAt?.toISOString() ?? null,
  1226→      }),
  1227→      tx,
  1228→    );
  1229→
  1230→    return {
  1231→      id: updated.id,
  1232→      reference: updated.reference,
  1233→      status: updated.status,
  1234→      postedAt: updated.postedAt,
  1235→      lineCount: lines.length,
  1236→      totalReturnedQty,
  1237→      stockChanges,
  1238→    };
  1239→  });
  1240→}

  1284→/**
  1285→ * §86/§87 — ADDITIVE per-receipt-line return facts for the GoodsReceipt
  1286→ * detail: receivedQty (immutable gross, §34), returnedQty (POSTED only,
  1287→ * §36) and returnableQty = received − returned. One grouped aggregate —
  1288→ * never one query per line (§141). `returnable` reports whether ANY line
  1289→ * still has returnableQty > 0 (the §87 CTA gate; client-advisory only —
  1290→ * the server revalidates at post time).
  1291→ */
  1292→export async function getGoodsReceiptReturnFacts(input: {
  1293→  businessId: string;
  1294→  receiptId: string;
  1295→  receivedLines: ReadonlyArray<{ id: string; receivedQty: number }>;
  1296→}): Promise<{
  1297→  returnable: boolean;
  1298→  lines: ReceiptLineReturnFactsDto[];
  1299→}> {
  1300→  const returnedByLine = await sumPostedReturnedQtyForReceiptLines(
  1301→    db,
  1302→    input.receivedLines.map((l) => l.id),
  1303→  );
  1304→  let returnable = false;
  1305→  const lines = input.receivedLines.map((l) => {
  1306→    const returnedQty = returnedByLine.get(l.id) ?? 0;
  1307→    try {
  1308→      // §36 — ONE canonical checked helper: returned > received (or any
  1309→      // invalid operand) is corruption; surface it, never clamp.
  1310→      const returnableQty = computeReturnableQty(l.receivedQty, returnedQty);
  1311→      if (returnableQty > 0) returnable = true;
  1312→      return receiptLineReturnFactsDto({
  1313→        goodsReceiptLineId: l.id,
  1314→        receivedQty: l.receivedQty,
  1315→        returnedQty,
  1316→      });
  1317→    } catch (error) {
  1318→      return mapPureGoodsReturnDomainError(error);
  1319→    }
  1320→  });
  1321→  return { returnable, lines };
  1322→}
