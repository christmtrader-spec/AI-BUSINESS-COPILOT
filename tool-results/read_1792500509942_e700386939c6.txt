   220→/** Stable identity of a view (for equality checks / keys). */
   221→export function viewKeyOf(view: AppView): string {
   222→  return view.kind === "module"
   223→    ? `module:${view.module}`
   224→    // Prompt 35 — the tab is part of the key so tab switches remount the
   225→    // inner CRM view deterministically.
   226→    : view.kind === "crm"
   227→      ? `crm:${view.tab}`
   228→    // Prompt 34 — entity + id both matter: opening a different record
   229→    // remounts the detail view so no stale record can ever be shown (§36).
   230→    : view.kind === "entity"
   231→      ? `entity:${view.entity}:${view.id}`
   232→    // Prompt 39 — type + id both matter: opening another party's statement
   233→    // remounts the view so no stale statement can ever be shown (§159).
   234→    : view.kind === "statements"
   235→      ? `statements:${view.statementType}:${view.id}`
   236→    // Prompt 40 — stocktake views: list + per-id detail keys so opening
   237→    // another stocktake remounts the detail view (no stale record, §36 parity).
   238→    : view.kind === "inventory-stocktakes"
   239→      ? "inventory-stocktakes"
   240→    : view.kind === "inventory-stocktake"
   241→      ? `inventory-stocktake:${view.id}`
   242→    // Prompt 41 — purchase-order views: list + per-id detail keys so opening
   243→    // another order remounts the detail view (no stale record, §36 parity).
   244→    : view.kind === "purchase-orders"
   245→      ? "purchase-orders"
   246→    : view.kind === "purchase-order"
   247→      ? `purchase-order:${view.id}`
   248→    // Prompt 42 — goods-receipt views: list + per-id detail keys so opening
   249→    // another receipt remounts the detail view (no stale record, §36 parity).
   250→    : view.kind === "inventory-goods-receipts"
   251→      ? "inventory-goods-receipts"
   252→    : view.kind === "inventory-goods-receipt"
   253→      ? `inventory-goods-receipt:${view.id}`
   254→    // Prompt 43 — goods-return views: list + per-id detail keys so opening
   255→    // another return remounts the detail view (§36 parity).
   256→    : view.kind === "inventory-goods-returns"
   257→      ? "inventory-goods-returns"
   258→    : view.kind === "inventory-goods-return"
   259→      ? `inventory-goods-return:${view.id}`
   260→    // Prompt 44 — goods-dispatch views: list + per-id detail keys so opening
   261→    // another dispatch remounts the detail view (P43 parity).
   262→    : view.kind === "inventory-goods-dispatches"
   263→      ? "inventory-goods-dispatches"
   264→    : view.kind === "inventory-goods-dispatch"
   265→      ? `inventory-goods-dispatch:${view.id}`
   266→    // Prompt 45 — sales-order views: list + per-id detail keys (§36 parity).
   267→    : view.kind === "sales-orders"
   268→      ? "sales-orders"
   269→    : view.kind === "sales-order"
   270→      ? `sales-order:${view.id}`
   271→    // Prompt 47 — sales-quotation views: list + per-id detail keys.
   272→    : view.kind === "sales-quotations"
   273→      ? "sales-quotations"
   274→    : view.kind === "sales-quotation"
   275→      ? `sales-quotation:${view.id}`
   276→    // Prompt 48 — supplier-bill views: list + per-id detail keys (§36 parity).
   277→    : view.kind === "supplier-bills"
   278→      ? "supplier-bills"
   279→    : view.kind === "supplier-bill"
   280→      ? `supplier-bill:${view.id}`
   281→    // Prompt 49 — aging-details remounts on ANY filter change so each
   282→    // side/party/bucket view holds its own cursor stack and scroll state.
   283→    : view.kind === "aging-details"
   284→      ? `aging-details:${view.side}:${view.partyId ?? ""}:${view.bucket ?? ""}:${view.pastDue ? 1 : 0}`
   285→      : view.kind === "join"
   286→      ? `join:${view.token}`
   287→      : view.kind === "reset-password"
   288→        ? `reset:${view.token}`
   289→        : view.kind === "verify-email"
   290→          ? `verify:${view.token}`
   291→          : view.kind === "platform"
   292→            ? `platform:${view.section}`
   293→            // Prompt 33 — the Search Center keeps a STABLE view key so the
   294→            // mounted view (and its focused input) survives query updates;
   295→            // the query itself only matters for sameView/back-forward (§13).
   296→            : view.kind;
   297→}

   299→/**
   300→ * Structural view equality (applyHash). Deliberately FINER than viewKeyOf:
   301→ * module sub-state (?month= deep-link param) must re-enter the store when
   302→ * the hash changes, but must NOT remount the module (viewKey stays stable
   303→ * so React keeps the mounted view alive — reports state survives dialog
   304→ * open/close).
   305→ */
   306→export function sameView(a: AppView, b: AppView): boolean {
   307→  if (viewKeyOf(a) !== viewKeyOf(b)) return false;
   308→  if (a.kind === "module" && b.kind === "module") {
   309→    return (a.param ?? null) === (b.param ?? null);
   310→  }
   311→  // Prompt 34 — entity views compare entity + id (viewKey already carries
   312→  // both, so reaching here means they are equal; kept explicit for clarity).
   313→  if (a.kind === "entity" && b.kind === "entity") {
   314→    return a.entity === b.entity && a.id === b.id;
   315→  }
   316→  // Prompt 35 — CRM tab changes re-enter the store (back/forward) without
   317→  // remounting the module boundary shell; the tab drives the inner view.
   318→  if (a.kind === "crm" && b.kind === "crm") {
   319→    return a.tab === b.tab;
   320→  }
   321→  // Prompt 39 — statement views compare type + id (viewKey already carries
   322→  // both; explicit for parity with entity views).
   323→  if (a.kind === "statements" && b.kind === "statements") {
   324→    return a.statementType === b.statementType && a.id === b.id;
   325→  }
   326→  // Prompt 40 — stocktake detail compares id (viewKey already carries it).
   327→  if (a.kind === "inventory-stocktake" && b.kind === "inventory-stocktake") {
   328→    return a.id === b.id;
   329→  }
   330→  // Prompt 41 — purchase-order detail compares id (viewKey already carries it).
   331→  if (a.kind === "purchase-order" && b.kind === "purchase-order") {
   332→    return a.id === b.id;
   333→  }
   334→  // Prompt 42 — goods-receipt detail compares id (viewKey already carries it).
   335→  if (a.kind === "inventory-goods-receipt" && b.kind === "inventory-goods-receipt") {
   336→    return a.id === b.id;
   337→  }
   338→  // Prompt 43 — goods-return detail compares id (viewKey already carries it).
   339→  if (a.kind === "inventory-goods-return" && b.kind === "inventory-goods-return") {
   340→    return a.id === b.id;
   341→  }
   342→  // Prompt 44 — goods-dispatch detail compares id (viewKey already carries it).
   343→  if (a.kind === "inventory-goods-dispatch" && b.kind === "inventory-goods-dispatch") {
   344→    return a.id === b.id;
   345→  }
   346→  // Prompt 48 — supplier-bill detail compares id (viewKey already carries it).
   347→  if (a.kind === "supplier-bill" && b.kind === "supplier-bill") {
   348→    return a.id === b.id;
   349→  }
   350→  // Prompt 49 — aging-details compare every canonical filter: a hash change
   351→  // on any of them re-enters the store (back/forward correctness, §18)
   352→  // while viewKeyOf keeps the mounted view stable per filter combination.
   353→  if (a.kind === "aging-details" && b.kind === "aging-details") {
   354→    return (
   355→      a.side === b.side &&
   356→      (a.partyId ?? null) === (b.partyId ?? null) &&
   357→      (a.bucket ?? null) === (b.bucket ?? null) &&
   358→      Boolean(a.pastDue) === Boolean(b.pastDue)
   359→    );
   360→  }
   361→  // Prompt 33 — hash query changes re-enter the store (back/forward must
   362→  // update the Search Center) without remounting the view (§13/§33).
   363→  if (a.kind === "search" && b.kind === "search") {
   364→    return (a.query ?? null) === (b.query ?? null);
   365→  }
   366→  return true;
   367→}

   369→/** URL hash for a view. `onboarding` is a transient state — no address. */
   370→export function viewToHash(view: AppView): string | null {
   371→  switch (view.kind) {
   372→    case "dashboard":
   373→      return "#/dashboard";
   374→    case "audit":
   375→      return "#/audit";
   376→    case "settings:business":
   377→      return "#/settings";
   378→    case "settings:members":
   379→      return "#/settings/members";
   380→    case "settings:billing":
   381→      return "#/settings/billing";
   382→    case "profile":
   383→      return "#/profile";
   384→    case "settings:security":
   385→      return "#/settings/security";
   386→    case "settings:data":
   387→      return "#/data";
   388→    case "notifications":
   389→      return "#/notifications";
   390→    case "settings:notifications":
   391→      return "#/settings/notifications";
   392→    case "search": {
   393→      // §33 — the query persists in the hash (encodeURIComponent, §29) so
   394→      // refresh and back/forward keep it; never any other param.
   395→      const q = (view.query ?? "").trim();
   396→      return q ? `#/search?q=${encodeURIComponent(q)}` : "#/search";
   397→    }
   398→    case "crm":
   399→      // Prompt 35 (§37) — the tab persists in the hash; the default tab
   400→      // renders as bare "#/crm" so the address stays canonical.
   401→      return view.tab === CRM_DEFAULT_TAB ? "#/crm" : `#/crm?tab=${view.tab}`;
   402→    case "entity":
   403→      // Prompt 34 — centralized encoder (§7/§39): validated entity + id,
   404→      // safe-charset id, encodeURIComponent. Returns null (never a hash)
   405→      // for anything that could not survive parseEntityRoute.
   406→      return entityToHash(view.entity, view.id);
   407→    case "statements":
   408→      // Prompt 39 (§57/§139) — closed type + validated id only; anything
   409→      // else produces no hash at all (never an arbitrary statement URL).
   410→      if (!isStatementTypeToken(view.statementType)) return null;
   411→      if (!isValidEntityId(view.id)) return null;
   412→      return `#/statements?type=${view.statementType}&id=${encodeURIComponent(view.id)}`;
   413→    case "inventory-stocktakes":
   414→      // Prompt 40 (§76) — canonical list hash.
   415→      return "#/inventory/stocktakes";
   416→    case "inventory-stocktake":
   417→      // Prompt 40 (§77/§136) — only charset-safe ids ever become a hash.
   418→      if (!isValidEntityId(view.id)) return null;
   419→      return `#/inventory/stocktakes/${view.id}`;
   420→    case "purchase-orders":
   421→      // Prompt 41 (§70) — canonical list hash.
   422→      return "#/purchase-orders";
   423→    case "purchase-order":
   424→      // Prompt 41 (§70) — only charset-safe ids ever become a hash.
   425→      if (!isValidEntityId(view.id)) return null;
   426→      return `#/purchase-orders/${view.id}`;
   427→    case "inventory-goods-receipts":
   428→      // Prompt 42 (§116) — canonical list hash.
   429→      return "#/inventory/goods-receipts";
   430→    case "inventory-goods-receipt":
   431→      // Prompt 42 (§116) — only charset-safe ids ever become a hash.
   432→      if (!isValidEntityId(view.id)) return null;
   433→      return `#/inventory/goods-receipts/${view.id}`;
   434→    case "inventory-goods-returns":
   435→      // Prompt 43 (§91) — canonical list hash.
   436→      return "#/inventory/goods-returns";
   437→    case "inventory-goods-return":
   438→      // Prompt 43 (§91) — only charset-safe ids ever become a hash.
   439→      if (!isValidEntityId(view.id)) return null;
   440→      return `#/inventory/goods-returns/${view.id}`;
   441→    case "inventory-goods-dispatches":
   442→      // Prompt 44 — canonical list hash.
   443→      return "#/inventory/goods-dispatches";
   444→    case "inventory-goods-dispatch":
   445→      // Prompt 44 — only charset-safe ids ever become a hash.
   446→      if (!isValidEntityId(view.id)) return null;
   447→      return `#/inventory/goods-dispatches/${view.id}`;
   448→    case "sales-orders":
   449→      // Prompt 45 (§88) — canonical list hash.
   450→      return "#/sales-orders";
   451→    case "sales-order":
   452→      // Prompt 45 (§88) — only charset-safe ids ever become a hash.
   453→      if (!isValidEntityId(view.id)) return null;
   454→      return `#/sales-orders/${view.id}`;
   455→    case "sales-quotations":
   456→      // Prompt 47 — canonical list hash.
   457→      return "#/sales-quotations";
   458→    case "sales-quotation":
   459→      // Prompt 47 — only charset-safe ids ever become a hash.
   460→      if (!isValidEntityId(view.id)) return null;
   461→      return `#/sales-quotations/${view.id}`;
   462→    case "supplier-bills":
   463→      // Prompt 48 — canonical list hash.
   464→      return "#/supplier-bills";
   465→    case "supplier-bill":
   466→      // Prompt 48 — only charset-safe ids ever become a hash.
   467→      if (!isValidEntityId(view.id)) return null;
   468→      return `#/supplier-bills/${view.id}`;
   469→    case "aging-details": {
   470→      // Prompt 49 (§18) — canonical filter hash. A partyId that cannot
   471→      // survive parsing produces NO hash (never a fake destination); the
   472→      // bucket rides only when it is a closed bucket key; pastDue only on
   473→      // the payables side (receivables have no authoritative due date).
   474→      if (view.side !== "receivables" && view.side !== "payables") return null;
   475→      if (view.partyId && !isValidEntityId(view.partyId)) return null;
   476→      const params = new URLSearchParams({ side: view.side });
   477→      if (view.partyId) params.set("party", view.partyId);
   478→      if (view.bucket && (AGING_BUCKET_KEYS as readonly string[]).includes(view.bucket)) {
   479→        params.set("bucket", view.bucket);
   480→      }
   481→      if (view.pastDue && view.side === "payables") params.set("pastDue", "1");
   482→      return `#/reports/aging-details?${params.toString()}`;
   483→    }
   484→    case "platform":
   485→      // #/platform itself resolves to the overview section (§15).
   486→      return view.section === "overview" ? "#/platform" : `#/platform/${view.section}`;
   487→    case "module": {
   488→      if (!isKnownModule(view.module)) return null;
   489→      // Module sub-state deep links stay in the address bar. Reports has
   490→      // three: the month drill-down (?month=YYYY-MM), — Prompt 51 — the
   491→      // detailed inventory valuation (?inventory=1) and — Prompt 55 —
   492→      // the reorder suggestions (?reorder=1). Mutually exclusive.
   493→      if (view.module === "reports" && view.param === "inventory") {
   494→        return "#/reports?inventory=1";
   495→      }
   496→      if (view.module === "reports" && view.param === "reorder") {
   497→        return "#/reports?reorder=1";
   498→      }
   499→      if (view.param && /^\d{4}-\d{2}$/.test(view.param)) {
   500→        return `#/${view.module}?month=${view.param}`;
   501→      }
   502→      return `#/${view.module}`;
   503→    }
   504→    case "join":
   505→      return `#/join/${view.token}`;
   506→    case "forgot-password":
   507→      return "#/forgot-password";
   508→    case "reset-password":
   509→      return `#/reset-password/${view.token}`;
   510→    case "verify-email":
   511→      return `#/verify-email/${view.token}`;
   512→    case "onboarding":
   513→      return null;
   514→  }
   515→}

   517→/** Parse a location hash into a view. Unknown → dashboard (never onboarding). */
   518→export function hashToView(hash: string): AppView {
   519→  const raw = hash.replace(/^#\/?/, "").replace(/\/+$/, "");
   520→  // Query part (e.g. #/reports?month=2026-04) — module-scoped deep-link params.
   521→  const qIndex = raw.indexOf("?");
   522→  const path = (qIndex === -1 ? raw : raw.slice(0, qIndex)).trim();
   523→  const query = new URLSearchParams(qIndex === -1 ? "" : raw.slice(qIndex + 1));
   524→  if (!path || path === "dashboard") return { kind: "dashboard" };
   525→  if (path === "audit") return { kind: "audit" };
   526→  if (path === "settings") return { kind: "settings:business" };
   527→  if (path === "settings/members") return { kind: "settings:members" };
   528→  if (path === "settings/billing") return { kind: "settings:billing" };
   529→  if (path === "settings/security") return { kind: "settings:security" };
   530→  if (path === "data") return { kind: "settings:data" };
   531→  if (path === "notifications") return { kind: "notifications" };
   532→  if (path === "settings/notifications") return { kind: "settings:notifications" };
   533→  // Prompt 33 — Search Results Center deep links (#/search, #/search?q=…).
   534→  if (path === "search") {
   535→    const q = query.get("q") ?? "";
   536→    const clean = sanitizeSearchHashQuery(q);
   537→    return clean ? { kind: "search", query: clean } : { kind: "search" };
   538→  }
   539→  // Prompt 35 (§37) — CRM deep links: #/crm, #/crm?tab=<closed tab>.
   540→  // Unknown tab values fall back to CRM_DEFAULT_TAB (pipeline) — the hash
   541→  // query is never interpreted as arbitrary state.
   542→  if (path === "crm") {
   543→    const tab = query.get("tab") ?? "";
   544→    return { kind: "crm", tab: isCrmTab(tab) ? tab : CRM_DEFAULT_TAB };
   545→  }
   546→  if (path === "profile") return { kind: "profile" };
   547→  // Prompt 28 — platform console deep links (#/platform, #/platform/<section>).
   548→  if (path === "platform") return { kind: "platform", section: "overview" };
   549→  if (path.startsWith("platform/")) {
   550→    const section = path.slice("platform/".length);
   551→    if (isPlatformSection(section)) return { kind: "platform", section };
   552→    return { kind: "platform", section: "overview" };
   553→  }
   554→  if (path.startsWith("join/")) {
   555→    const token = path.slice("join/".length);
   556→    // Opaque invite token: non-empty, bounded length (server validates the rest).
   557→    if (token.length > 0 && token.length <= 128) return { kind: "join", token };
   558→  }
   559→  if (path === "forgot-password") return { kind: "forgot-password" };
   560→  if (path.startsWith("verify-email/")) {
   561→    const token = path.slice("verify-email/".length);
   562→    // Opaque verification token: bounded length; the server validates the rest.
   563→    if (token.length > 0 && token.length <= 200) return { kind: "verify-email", token };
   564→    return { kind: "dashboard" };
   565→  }
   566→  if (path.startsWith("reset-password/")) {
   567→    const token = path.slice("reset-password/".length);
   568→    // Opaque reset token: non-empty, bounded length (server validates the rest).
   569→    if (token.length > 0 && token.length <= 200) return { kind: "reset-password", token };
   570→    // Malformed reset link → the forgot form so a fresh link can be requested.
   571→    return { kind: "forgot-password" };
   572→  }
   573→  // Prompt 39 (§57/§58/§139) — statement deep links: #/statements with a
   574→  // closed `type` token and a safe `id`. Unknown type / missing or unsafe
   575→  // id → the existing unknown-hash fallback (dashboard); the query string
   576→  // is never interpreted as arbitrary state.
   577→  if (path === "statements") {
   578→    const type = query.get("type") ?? "";
   579→    const rawId = query.get("id") ?? "";
   580→    if (!isStatementTypeToken(type)) return { kind: "dashboard" };
   581→    let id = rawId;
   582→    try {
   583→      id = decodeURIComponent(rawId);
   584→    } catch {
   585→      return { kind: "dashboard" };
   586→    }
   587→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   588→    return { kind: "statements", statementType: type, id };
   589→  }
   590→  // Prompt 49 (§18) — detailed aging deep links:
   591→  //   #/reports/aging-details?side=…&party=…&bucket=…&pastDue=1
   592→  // EVERY filter falls back safely (never a blank page): an unknown side
   593→  // defaults to receivables, an unknown bucket / unsafe party id is dropped,
   594→  // pastDue is honored only on the payables side. The parsed party id is
   595→  // resolved server-side inside the tenant — a foreign id renders the
   596→  // honest not-found state (404), never another business's data.
   597→  if (path === "reports/aging-details") {
   598→    const side = query.get("side") === "payables" ? "payables" : "receivables";
   599→    const rawBucket = query.get("bucket") ?? "";
   600→    const bucket = (AGING_BUCKET_KEYS as readonly string[]).includes(rawBucket)
   601→      ? (rawBucket as AgingBucketKey)
   602→      : null;
   603→    const rawParty = query.get("party") ?? "";
   604→    let partyId: string | null = null;
   605→    try {
   606→      const decoded = rawParty ? decodeURIComponent(rawParty) : "";
   607→      partyId = decoded && isValidEntityId(decoded) ? decoded : null;
   608→    } catch {
   609→      partyId = null;
   610→    }
   611→    const pastDue = side === "payables" && (query.get("pastDue") === "1" || query.get("pastDue") === "true");
   612→    return { kind: "aging-details", side, bucket, partyId, pastDue };
   613→  }
   614→  // Prompt 40 (§76/§77/§136) — inventory stocktake deep links, matched
   615→  // BEFORE the module/entity fallbacks: #/inventory/stocktakes opens the
   616→  // list; #/inventory/stocktakes/<id> opens the detail only when the id is
   617→  // charset-safe; anything else falls through to the dashboard fallback.
   618→  if (path === "inventory/stocktakes") return { kind: "inventory-stocktakes" };
   619→  if (path.startsWith("inventory/stocktakes/")) {
   620→    const rawId = path.slice("inventory/stocktakes/".length);
   621→    let id = rawId;
   622→    try {
   623→      id = decodeURIComponent(rawId);
   624→    } catch {
   625→      return { kind: "dashboard" };
   626→    }
   627→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   628→    return { kind: "inventory-stocktake", id };
   629→  }
   630→  // Prompt 41 (§69/§70) — purchase-order deep links, matched BEFORE the
   631→  // module/entity fallbacks: #/purchase-orders opens the list;
   632→  // #/purchase-orders/<id> opens the detail only when the id is charset-safe;
   633→  // anything else falls through to the dashboard fallback.
   634→  if (path === "purchase-orders") return { kind: "purchase-orders" };
   635→  if (path.startsWith("purchase-orders/")) {
   636→    const rawId = path.slice("purchase-orders/".length);
   637→    let id = rawId;
   638→    try {
   639→      id = decodeURIComponent(rawId);
   640→    } catch {
   641→      return { kind: "dashboard" };
   642→    }
   643→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   644→    return { kind: "purchase-order", id };
   645→  }
   646→  // Prompt 42 (§116) — goods-receipt deep links, matched BEFORE the
   647→  // module/entity fallbacks: #/inventory/goods-receipts opens the list;
   648→  // #/inventory/goods-receipts/<id> opens the detail only when the id is
   649→  // charset-safe; anything else falls through to the dashboard fallback.
   650→  if (path === "inventory/goods-receipts") return { kind: "inventory-goods-receipts" };
   651→  if (path.startsWith("inventory/goods-receipts/")) {
   652→    const rawId = path.slice("inventory/goods-receipts/".length);
   653→    let id = rawId;
   654→    try {
   655→      id = decodeURIComponent(rawId);
   656→    } catch {
   657→      return { kind: "dashboard" };
   658→    }
   659→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   660→    return { kind: "inventory-goods-receipt", id };
   661→  }
   662→  // Prompt 43 (§91) — goods-return deep links, matched BEFORE the
   663→  // module/entity fallbacks: #/inventory/goods-returns opens the list;
   664→  // #/inventory/goods-returns/<id> opens the detail only when the id is
   665→  // charset-safe; anything else falls through to the dashboard fallback.
   666→  if (path === "inventory/goods-returns") return { kind: "inventory-goods-returns" };
   667→  if (path.startsWith("inventory/goods-returns/")) {
   668→    const rawId = path.slice("inventory/goods-returns/".length);
   669→    let id = rawId;
   670→    try {
   671→      id = decodeURIComponent(rawId);
   672→    } catch {
   673→      return { kind: "dashboard" };
   674→    }
   675→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   676→    return { kind: "inventory-goods-return", id };

   678→  // Prompt 44 — goods-dispatch deep links, matched BEFORE the
   679→  // module/entity fallbacks: #/inventory/goods-dispatches opens the list;
   680→  // #/inventory/goods-dispatches/<id> opens the detail only when the id is
   681→  // charset-safe; anything else falls through to the dashboard fallback —
   682→  // never an arbitrary "#/anything/<id>" interpretation.
   683→  if (path === "inventory/goods-dispatches") return { kind: "inventory-goods-dispatches" };
   684→  if (path.startsWith("inventory/goods-dispatches/")) {
   685→    const rawId = path.slice("inventory/goods-dispatches/".length);
   686→    let id = rawId;
   687→    try {
   688→      id = decodeURIComponent(rawId);
   689→    } catch {
   690→      return { kind: "dashboard" };
   691→    }
   692→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   693→    return { kind: "inventory-goods-dispatch", id };
   694→  }
   695→  // Prompt 45 — sales-order deep links, matched BEFORE the
   696→  // module/entity fallbacks: #/sales-orders opens the list;
   697→  // #/sales-orders/<id> opens the detail only when the id is charset-safe;
   698→  // anything else falls through to the dashboard fallback — never an
   699→  // arbitrary "#/anything/<id>" interpretation.
   700→  if (path === "sales-orders") return { kind: "sales-orders" };
   701→  if (path.startsWith("sales-orders/")) {
   702→    const rawId = path.slice("sales-orders/".length);
   703→    let id = rawId;
   704→    try {
   705→      id = decodeURIComponent(rawId);
   706→    } catch {

   709→    if (!isValidEntityId(id)) return { kind: "dashboard" };

   711→  }
   712→  // Prompt 47 — sales-quotation deep links, matched BEFORE the
   713→  // module/entity fallbacks: #/sales-quotations opens the list;
   714→  // #/sales-quotations/<id> opens the detail only when the id is
   715→  // charset-safe; anything else falls through to the dashboard fallback —
   716→  // never an arbitrary "#/anything/<id>" interpretation.
   717→  if (path === "sales-quotations") return { kind: "sales-quotations" };
   718→  if (path.startsWith("sales-quotations/")) {
   719→    const rawId = path.slice("sales-quotations/".length);
   720→    let id = rawId;
   721→    try {
   722→      id = decodeURIComponent(rawId);
   723→    } catch {
   724→      return { kind: "dashboard" };
   725→    }
   726→    if (!isValidEntityId(id)) return { kind: "dashboard" };
   727→    return { kind: "sales-quotation", id };
   728→  }
   729→  // Prompt 48 — supplier-bill list + detail deep links (#/supplier-bills/<id>).
   730→  if (path === "supplier-bills") return { kind: "supplier-bills" };
   731→  if (path.startsWith("supplier-bills/")) {
   732→    const rawId = path.slice("supplier-bills/".length);
   733→    let id = rawId;
   734→    try {
   735→      id = decodeURIComponent(rawId);
   736→    } catch {
   737→      return { kind: "dashboard" };
   738→    }
   739→    if (!isValidEntityId(id)) return { kind: "dashboard" };

   741→  }
   742→  // Prompt 34 — entity detail deep links (#/customers/<id>, §5/§6) are
   743→  // attempted BEFORE the bare-module match: "customers" alone stays the
   744→  // module list while "customers/<id>" resolves through the centralized
   745→  // parser (allowlisted first segment, bounded charset-safe id). Anything
   746→  // else falls through to the dashboard fallback (§6.6) — never an
   747→  // arbitrary "#/anything/<id>" interpretation.
   748→  const entityRoute = parseEntityRoute(path);
   749→  if (entityRoute) return { kind: "entity", ...entityRoute };
   750→  if (isKnownModule(path)) {
   751→    // Module sub-param deep links: ?month=YYYY-MM opens reports straight
   752→    // into the drill-down, or pre-filters sales/expenses to that month
   753→    // (e.g. "Open in Sales" from the reports month dialog). Prompt 51:
   754→    // ?inventory=1 opens the detailed inventory valuation sub-view.
   755→    const month = query.get("month");
   756→    if (month && /^\d{4}-\d{2}$/.test(month)) {
   757→      return { kind: "module", module: path, param: month };
   758→    }
   759→    if (path === "reports" && query.get("inventory") === "1") {
   760→      return { kind: "module", module: path, param: "inventory" };
   761→    }
   762→    // Prompt 55 (§X2) — ?reorder=1 opens the low-stock reorder suggestions
   763→    // sub-view. Closed token, never arbitrary state.
   764→    if (path === "reports" && query.get("reorder") === "1") {
   765→      return { kind: "module", module: path, param: "reorder" };
   766→    }
   767→    return { kind: "module", module: path };
   768→  }
   769→  return { kind: "dashboard" };
   770→}

   772→/** Sync the URL to a view without triggering navigation events. */
   773→function syncUrl(view: AppView, replace: boolean) {
   774→  if (typeof window === "undefined") return;
   775→  const hash = viewToHash(view);
   776→  if (!hash) return;
   777→  const target = `${window.location.pathname}${window.location.search}${hash}`;
   778→  if (`${window.location.pathname}${window.location.search}${window.location.hash}` === target) {
   779→    return;
   780→  }
   781→  try {
   782→    if (replace) window.history.replaceState(null, "", target);
   783→    else window.history.pushState(null, "", target);
   784→  } catch {
   785→    // history might be unavailable in exotic embeds — URL sync is best-effort
   786→  }
   787→}

   789→/**
   790→ * Prompt 33 — hash `?q=` hygiene (§16/§29): the query is rendered strictly
   791→ * as text, so only control characters/null bytes are stripped and the
   792→ * length is capped to the search schema bound. URLSearchParams already
   793→ * performs percent-decoding; no other transformation is applied.
   794→ */
   795→function sanitizeSearchHashQuery(raw: string): string {
   796→  const capped = Array.from(raw)
   797→    .filter((ch) => {
   798→      const code = ch.codePointAt(0) ?? 0;
   799→      return code > 0x1f && code !== 0x7f;
   800→    })
   801→    .slice(0, 64)
   802→    .join("");
   803→  const trimmed = capped.trim();
   804→  return trimmed.length > 0 ? trimmed : "";
   805→}
