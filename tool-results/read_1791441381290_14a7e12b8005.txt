    50→    50→/**
    51→    51→ * §25/§28 — SQLite is a single-writer database. Prisma interactive
    52→    52→ * transactions issue BEGIN IMMEDIATE, so N concurrent P41 writes contend on
    53→    53→ * the one write lock; losers used to sit until the 5s transaction timeout
    54→    54→ * and die with "Transaction already closed" (a generic 500). Two defenses:
    55→    55→ *
    56→    56→ *  1. In-process FIFO serialization of P41 WRITE transactions — requests
    57→    57→ *     still run concurrently up to the transaction boundary, then take the
    58→    58→ *     write lock one at a time (each transaction is a few ms, so latency
    59→    59→ *     stays bounded; reads are never blocked).
    60→    60→ *  2. A bounded retry for TRANSIENT transaction failures (P2034 write
    61→    61→ *     conflict / P2024 timeout / expired transaction from another process,
    62→    62→ *     e.g. test scripts sharing the dev DB) with deterministic backoff.
    63→    63→ *
    64→    64→ * Domain errors (ApiError) are NEVER retried — they propagate unchanged.
    65→    65→ */
    66→    66→let writeChain: Promise<unknown> = Promise.resolve();
    67→    67→function runSerializedWrite<T>(fn: () => Promise<T>): Promise<T> {
    68→    68→  const result = writeChain.then(() => fn());
    69→    69→  writeChain = result.then(
    70→    70→    () => undefined,
    71→    71→    () => undefined,
    72→    72→  );
    73→    73→  return result;
    74→    74→}

    76→    76→function isTransientTxError(error: unknown): boolean {
    77→    77→  if (error instanceof Prisma.PrismaClientKnownRequestError) {
    78→    78→    return error.code === "P2034" || error.code === "P2024";
    79→    79→  }
    80→    80→  if (error instanceof Prisma.PrismaClientUnknownRequestError) {
    81→    81→    return /Transaction already closed|SQLITE_BUSY|database is locked/i.test(error.message);
    82→    82→  }
    83→    83→  return false;
    84→    84→}

    88→    88→async function withWriteTx<T>(fn: (tx: Tx) => Promise<T>, attempts = 3): Promise<T> {
    89→    89→  let lastError: unknown;
    90→    90→  for (let attempt = 0; attempt < attempts; attempt += 1) {
    91→    91→    try {
    92→    92→      return await runSerializedWrite(() => db.$transaction(fn, TX_OPTIONS));
    93→    93→    } catch (error) {
    94→    94→      lastError = error;
    95→    95→      if (!isTransientTxError(error)) throw error;
    96→    96→      // Deterministic backoff; bounded (§25 — graceful, never a hot spin).
    97→    97→      await new Promise((resolve) => setTimeout(resolve, 30 * (attempt + 1)));
    98→    98→    }
    99→    99→  }
   100→   100→  throw lastError;
   101→   101→}

   287→   287→
   288→   288→export async function createPurchaseOrder(input: {
   289→   289→  businessId: string;
   290→   290→  userId: string;
   291→   291→  supplierId: string;
   292→   292→  reference?: string; // undefined/blank → server-generated (§13)
   293→   293→  orderedAt?: string; // calendar date; default now
   294→   294→  expectedDate?: string | null;
   295→   295→  notes?: string | null;
   296→   296→  ip?: string | null;
   297→   297→  userAgent?: string | null;
   298→   298→}) {
   299→   299→  const cap = purchaseOrderLineCap(process.env.PURCHASE_ORDER_MAX_LINES);
   300→   300→  void cap; // creation starts with zero lines; cap bounds LINES (§33)
   301→   301→  const orderedAt = input.orderedAt ? midnoon(input.orderedAt) : new Date();
   302→   302→  const expectedDate = input.expectedDate ? midnoon(input.expectedDate) : null;
   303→   303→
   304→   304→  return withWriteTx(async (tx) => {
   305→   305→    const supplier = await loadSupplier(tx, input.businessId, input.supplierId);
   306→   306→    const business = await tx.business.findUnique({
   307→   307→      where: { id: input.businessId },
   308→   308→      select: { currencyCode: true, currencySymbol: true, decimalPrecision: true },
   309→   309→    });
   310→   310→    if (!business) throw new ApiError(404, "NOT_FOUND", "Business not found.");
   311→   311→
   312→   312→    const generated = !input.reference;
   313→   313→    const baseReference = generated
   314→   314→      ? generatePurchaseOrderReference(new Date(), (max) => randomInt(max))
   315→   315→      : (input.reference as string);
   316→   316→
   317→   317→    // §13 — bounded retry ONLY for generated references; a user-chosen
   318→   318→    // duplicate fails immediately with the stable 409 (§12/§60).
   319→   319→    let lastCollision = false;
   320→   320→    for (let attempt = 0; attempt < 5; attempt += 1) {
   321→   321→      const reference = generated && attempt > 0
   322→   322→        ? generatePurchaseOrderReference(new Date(), (max) => randomInt(max))
   323→   323→        : baseReference;
   324→   324→      try {
   325→   325→        const row = await tx.purchaseOrder.create({
   326→   326→          data: {
   327→   327→            businessId: input.businessId,
   328→   328→            supplierId: supplier.id,
   329→   329→            reference,
   330→   330→            status: "DRAFT",
   331→   331→            // §22 — currency snapshot from the business, never the client.
   332→   332→            currencyCodeAtOrder: business.currencyCode,
   333→   333→            currencySymbolAtOrder: business.currencySymbol,
   334→   334→            decimalPrecisionAtOrder: business.decimalPrecision,
   335→   335→            orderedAt,
   336→   336→            expectedDate,
   337→   337→            notes: input.notes ?? null,
   338→   338→            supplierNameAtOrder: supplier.name, // §11 — historical identity
   339→   339→            createdByUserId: input.userId, // §209 — always ctx.userId
   340→   340→          },
   341→   341→          select: { id: true, reference: true, status: true },
   342→   342→        });
   343→   343→        await audit(
   344→   344→          auditInput(input.businessId, input.userId, "purchase_order.create", row.id, row.reference, {
   345→   345→            status: row.status,
   346→   346→            supplierId: supplier.id,
   347→   347→            supplierNameAtOrder: supplier.name,
   348→   348→            currencyCodeAtOrder: business.currencyCode,
   349→   349→            lineCount: 0,
   350→   350→          }),
   351→   351→          tx,
   352→   352→        );
   353→   353→        return row;
   354→   354→      } catch (error) {
   355→   355→        if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   356→   356→          lastCollision = true;
   357→   357→          if (!generated) break; // §60 — user-chosen reference collision → 409
   358→   358→          continue; // §13 — regenerate and retry (bounded)
   359→   359→        }
   360→   360→        throw error;
   361→   361→      }
   362→   362→    }
   363→   363→    if (lastCollision) {
   364→   364→      throw new ApiError(
   365→   365→        409,
   366→   366→        PURCHASE_ORDER_ERROR_CODES.DUPLICATE_REFERENCE,
   367→   367→        "A purchase order with this reference already exists.",
   368→   368→      );
   369→   369→    }
   370→   370→    throw new ApiError(500, "INTERNAL_ERROR", "Purchase order could not be created.");
   371→   371→  });
   372→   372→}

   375→   375→
   376→   376→export async function listPurchaseOrders(input: {
   377→   377→  businessId: string;
   378→   378→  q?: string;
   379→   379→  status?: string;
   380→   380→  from?: string;
   381→   381→  to?: string;
   382→   382→  page: number;
   383→   383→  pageSize: number;
   384→   384→}) {
   385→   385→  const cap = purchaseOrderLineCap(process.env.PURCHASE_ORDER_MAX_LINES);
   386→   386→  const where: Prisma.PurchaseOrderWhereInput = { businessId: input.businessId };
   387→   387→  if (input.q) {
   388→   388→    // §53 — reference + supplier snapshot + product snapshot names/SKUs.
   389→   389→    where.OR = [
   390→   390→      { reference: { contains: input.q } },
   391→   391→      { supplierNameAtOrder: { contains: input.q } },
   392→   392→      { supplier: { is: { name: { contains: input.q } } } },
   393→   393→      { lines: { some: { OR: [
   394→   394→        { productNameAtOrder: { contains: input.q } },
   395→   395→        { skuAtOrder: { contains: input.q } },
   396→   396→      ] } } },
   397→   397→    ];
   398→   398→  }
   399→   399→  if (input.status && isPurchaseOrderStatus(input.status)) where.status = input.status;
   400→   400→  if (input.from || input.to) where.orderedAt = dayWindow(input.from, input.to);
   401→   401→
   402→   402→  const [rows, total] = await Promise.all([
   403→   403→    db.purchaseOrder.findMany({
   404→   404→      where,
   405→   405→      // §55 — deterministic: orderedAt DESC, then createdAt DESC, then id DESC.
   406→   406→      orderBy: [{ orderedAt: "desc" }, { createdAt: "desc" }, { id: "desc" }],
   407→   407→      skip: (input.page - 1) * input.pageSize,
   408→   408→      take: input.pageSize,
   409→   409→      select: {
   410→   410→        id: true,
   411→   411→        reference: true,
   412→   412→        status: true,
   413→   413→        supplierId: true,
   414→   414→        supplierNameAtOrder: true,
   415→   415→        currencyCodeAtOrder: true,
   416→   416→        currencySymbolAtOrder: true,
   417→   417→        decimalPrecisionAtOrder: true,
   418→   418→        orderedAt: true,
   419→   419→        expectedDate: true,
   420→   420→        createdAt: true,
   421→   421→        updatedAt: true,
   422→   422→        // §16 — the line slice is EXPLICITLY bounded by the same cap law as
   423→   423→        // the write path (defense against invariant drift/corrupt rows).
   424→   424→        lines: { select: { orderedQty: true, unitPriceMinor: true }, take: cap },
   425→   425→        _count: { select: { lines: true } },
   426→   426→      },
   427→   427→    }),
   428→   428→    db.purchaseOrder.count({ where }),
   429→   429→  ]);
   430→   430→
   431→   431→  let items;
   432→   432→  try {
   433→   433→    items = rows.map((row) => {
   434→   434→      if (row._count.lines > row.lines.length) {
   435→   435→        // §16 — a truncated slice would make every total a LIE. The write
   436→   436→        // path forbids this; treat it as the invariant violation it is.
   437→   437→        throw new Error("PURCHASE_ORDER_LINE_CAP_INVARIANT");
   438→   438→      }
   439→   439→      return purchaseOrderListItemDto({
   440→   440→        ...row,
   441→   441→        supplierName: null,
   442→   442→        createdByName: null,
   443→   443→        createdByUserId: "",
   444→   444→        sentAt: null,
   445→   445→        confirmedAt: null,
   446→   446→        closedAt: null,
   447→   447→        cancelledAt: null,
   448→   448→        notes: null,
   449→   449→        lineCount: row._count.lines,
   450→   450→      });
   451→   451→    });
   452→   452→  } catch (error) {
   453→   453→    if (error instanceof Error && error.message === "PURCHASE_ORDER_LINE_CAP_INVARIANT") {
   454→   454→      throw new ApiError(
   455→   455→        500,
   456→   456→        "PURCHASE_ORDER_LINE_CAP_INVARIANT",
   457→   457→        "Purchase order line cap invariant violated.",
   458→   458→      );
   459→   459→    }
   460→   460→    mapPureDomainError(error); // §17 — overflow/integrity → controlled errors
   461→   461→  }
   462→   462→
   463→   463→  return {
   464→   464→    items,
   465→   465→    total,
   466→   466→    page: input.page,
   467→   467→    pageSize: input.pageSize,
   468→   468→  };
   469→   469→}

   473→   473→export async function getPurchaseOrder(input: { businessId: string; id: string }) {
   474→   474→  const cap = purchaseOrderLineCap(process.env.PURCHASE_ORDER_MAX_LINES);
   475→   475→  // §22 — header, supplier state, creator and lines are read in ONE
   476→   476→  // consistent transaction so the response can never combine a stale header
   477→   477→  // with mismatched line state (production-quality document read).
   478→   478→  const { header, supplier, creator, lineRows } = await db.$transaction(async (tx) => {
   479→   479→    const hdr = await loadOrderHeader(tx, input.businessId, input.id);
   480→   480→    const [sup, usr, lineRws] = await Promise.all([
   481→   481→      // §23 — EXPLICIT tenant scope on the live-supplier read: a foreign
   482→   482→      // tenant's supplier row is unreachable even by internal id.
   483→   483→      tx.supplier.findFirst({
   484→   484→        where: { id: hdr.supplierId, businessId: input.businessId },
   485→   485→        select: { name: true, deletedAt: true },
   486→   486→      }),
   487→   487→      // §24 — creator read exposes ONLY id + name (no hashes/secrets).
   488→   488→      tx.user.findUnique({
   489→   489→        where: { id: hdr.createdByUserId },
   490→   490→        select: { id: true, name: true },
   491→   491→      }),
   492→   492→      // §158 — ALL lines (the order is hard-capped); cap+1 proves completeness.
   493→   493→      tx.purchaseOrderLine.findMany({
   494→   494→        where: { purchaseOrderId: hdr.id },
   495→   495→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   496→   496→        take: cap + 1,
   497→   497→        select: {
   498→   498→          id: true,
   499→   499→          productId: true,
   500→   500→          productNameAtOrder: true,
   501→   501→          skuAtOrder: true,
   502→   502→          unitAtOrder: true,
   503→   503→          orderedQty: true,
   504→   504→          unitPriceMinor: true,
   505→   505→          createdAt: true,
   506→   506→          updatedAt: true,
   507→   507→        },
   508→   508→      }),
   509→   509→    ]);
   510→   510→    return { header: hdr, supplier: sup, creator: usr, lineRows: lineRws };
   511→   511→  });
   512→   512→
   513→   513→  if (lineRows.length > cap) {
   514→   514→    // Defensive: the write path prevents this; never render a partial document.
   515→   515→    throw new ApiError(500, "INTERNAL_ERROR", "Purchase order line cap invariant violated.");
   516→   516→  }
   517→   517→
   518→   518→  try {
   519→   519→    return purchaseOrderDetailDto(
   520→   520→      {
   521→   521→        ...header,
   522→   522→        // §11 — live name ONLY as a navigation aid; display identity is the
   523→   523→        // snapshot. A soft-deleted supplier renders as "no longer listed".
   524→   524→        supplierName: supplier && supplier.deletedAt === null ? supplier.name : null,
   525→   525→        createdByName: creator?.name ?? null,
   526→   526→      },
   527→   527→      lineRows,
   528→   528→    );
   529→   529→  } catch (error) {
   530→   530→    mapPureDomainError(error); // §13/§17 — controlled integrity/overflow
   531→   531→  }
   532→   532→}

   594→   594→export async function addPurchaseOrderLine(input: {
   595→   595→  businessId: string;
   596→   596→  userId: string;
   597→   597→  id: string;
   598→   598→  productId: string;
   599→   599→  orderedQty: number;
   600→   600→  unitPriceMinor: number;
   601→   601→  ip?: string | null;
   602→   602→  userAgent?: string | null;
   603→   603→}) {
   604→   604→  const cap = purchaseOrderLineCap(process.env.PURCHASE_ORDER_MAX_LINES);
   605→   605→  return withWriteTx(async (tx) => {
   606→   606→    // §58 steps 1-2 — order exists in tenant + DRAFT.
   607→   607→    const header = await loadOrderHeader(tx, input.businessId, input.id);
   608→   608→    requireDraft(header.status);
   609→   609→
   610→   610→    // §58 step 3-5 — product exists in the same business, not deleted, and
   611→   611→    // ACTIVE for NEW selections (existing draft lines stay historical §26).
   612→   612→    const product = await tx.product.findFirst({
   613→   613→      where: { id: input.productId, businessId: input.businessId, deletedAt: null },
   614→   614→      select: { id: true, name: true, sku: true, unit: true, isActive: true },
   615→   615→    });
   616→   616→    if (!product) {
   617→   617→      throw new ApiError(404, PURCHASE_ORDER_ERROR_CODES.INVALID_PRODUCT, "Product not found.");
   618→   618→    }
   619→   619→    if (!product.isActive) {
   620→   620→      throw new ApiError(
   621→   621→        422,
   622→   622→        PURCHASE_ORDER_ERROR_CODES.PRODUCT_INACTIVE,
   623→   623→        "Inactive products cannot be added to a purchase order.",
   624→   624→      );
   625→   625→    }
   626→   626→
   627→   627→    // §58 step 6 — line cap.
   628→   628→    const lineCount = await tx.purchaseOrderLine.count({ where: { purchaseOrderId: header.id } });
   629→   629→    if (lineCount >= cap) {
   630→   630→      throw new ApiError(
   631→   631→        422,
   632→   632→        PURCHASE_ORDER_ERROR_CODES.LINE_LIMIT,
   633→   633→        `Purchase order line limit reached (${cap}).`,
   634→   634→      );
   635→   635→    }
   636→   636→
   637→   637→    // §58 step 9 — snapshot product metadata at write time (§27).
   638→   638→    try {
   639→   639→      const line = await tx.purchaseOrderLine.create({
   640→   640→        data: {
   641→   641→          purchaseOrderId: header.id,
   642→   642→          productId: product.id,
   643→   643→          orderedQty: input.orderedQty,
   644→   644→          unitPriceMinor: input.unitPriceMinor,
   645→   645→          productNameAtOrder: product.name,
   646→   646→          skuAtOrder: product.sku,
   647→   647→          unitAtOrder: product.unit,
   648→   648→        },
   649→   649→        // §67 — the FULL persisted row feeds the DTO (real timestamps).
   650→   650→        select: {
   651→   651→          id: true,
   652→   652→          orderedQty: true,
   653→   653→          unitPriceMinor: true,
   654→   654→          createdAt: true,
   655→   655→          updatedAt: true,
   656→   656→        },
   657→   657→      });
   658→   658→
   659→   659→      // §25 — belt-and-braces cap re-check INSIDE the transaction: two
   660→   660→      // racing adds cannot both survive SQLite serialization — if the
   661→   661→      // create pushed the count past the cap, this throw ROLLS BACK the
   662→   662→      // whole transaction so >cap lines can never persist.
   663→   663→      const lineCountAfter = await tx.purchaseOrderLine.count({
   664→   664→        where: { purchaseOrderId: header.id },
   665→   665→      });
   666→   666→      if (lineCountAfter > cap) {
   667→   667→        throw new ApiError(
   668→   668→          422,
   669→   669→          PURCHASE_ORDER_ERROR_CODES.LINE_LIMIT,
   670→   670→          `Purchase order line limit reached (${cap}).`,
   671→   671→        );
   672→   672→      }
   673→   673→
   674→   674→      // §58 step 10 — audit inside the same transaction.
   675→   675→      await audit(
   676→   676→        auditInput(input.businessId, input.userId, "purchase_order.line_add", header.id, header.reference, {
   677→   677→          lineId: line.id,
   678→   678→          productId: product.id,
   679→   679→          productNameAtOrder: product.name,
   680→   680→          orderedQty: line.orderedQty,
   681→   681→          unitPriceMinor: line.unitPriceMinor,
   682→   682→          lineCount: lineCountAfter,
   683→   683→        }),
   684→   684→        tx,
   685→   685→      );
   686→   686→
   687→   687→      // §32 — the resulting document total must remain a safe integer.
   688→   688→      const lines = await tx.purchaseOrderLine.findMany({
   689→   689→        where: { purchaseOrderId: header.id },
   690→   690→        select: { orderedQty: true, unitPriceMinor: true },
   691→   691→      });
   692→   692→      assertSafeTotal(lines);
   693→   693→
   694→   694→      // §67 — DTO built from the PERSISTED row: real snapshots + real
   695→   695→      // timestamps, nothing fabricated.
   696→   696→      return purchaseOrderLineDto({
   697→   697→        ...line,
   698→   698→        productId: product.id,
   699→   699→        productNameAtOrder: product.name,
   700→   700→        skuAtOrder: product.sku,
   701→   701→        unitAtOrder: product.unit,
   702→   702→      });
   703→   703→    } catch (error) {
   704→   704→      // §59 — DB uniqueness is authoritative for duplicate products.
   705→   705→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   706→   706→        throw new ApiError(
   707→   707→          409,
   708→   708→          PURCHASE_ORDER_ERROR_CODES.DUPLICATE_PRODUCT,
   709→   709→          "This product is already on the purchase order.",
   710→   710→        );
   711→   711→      }
   712→   712→      throw error;
   713→   713→    }
   714→   714→  });
   715→   715→}

   838→   838→export async function transitionPurchaseOrder(input: {
   839→   839→  businessId: string;
   840→   840→  userId: string;
   841→   841→  id: string;
   842→   842→  op: "send" | "confirm" | "close" | "cancel";
   843→   843→  ip?: string | null;
   844→   844→  userAgent?: string | null;
   845→   845→}) {
   846→   846→  const target = PURCHASE_ORDER_TRANSITION_TARGET[input.op];
   847→   847→
   848→   848→  return withWriteTx(async (tx) => {
   849→   849→    // §61 — current status re-checked INSIDE the transaction; two concurrent
   850→   850→    // confirms cannot both succeed (the second sees the new status).
   851→   851→    const header = await loadOrderHeader(tx, input.businessId, input.id);
   852→   852→    assertTransition(header.status, target);
   853→   853→
   854→   854→    if (input.op === "send") {
   855→   855→      // §37/§38 — a document needs at least one line to be sent.
   856→   856→      const lineCount = await tx.purchaseOrderLine.count({ where: { purchaseOrderId: header.id } });
   857→   857→      if (lineCount < 1) {
   858→   858→        throw new ApiError(
   859→   859→          422,
   860→   860→          PURCHASE_ORDER_ERROR_CODES.EMPTY_ORDER,
   861→   861→          "Add at least one line before sending.",
   862→   862→        );
   863→   863→      }
   864→   864→      // §38 — supplier must still exist and not be soft-deleted.
   865→   865→      await assertSupplierStillValid(tx, input.businessId, header.supplierId);
   866→   866→    }
   867→   867→
   868→   868→    const now = new Date();
   869→   869→    const data: Prisma.PurchaseOrderUpdateInput = { status: target };
   870→   870→    if (target === "SENT") data.sentAt = now;
   871→   871→    if (target === "CONFIRMED") data.confirmedAt = now;
   872→   872→    if (target === "CLOSED") data.closedAt = now;
   873→   873→    if (target === "CANCELLED") data.cancelledAt = now;
   874→   874→
   875→   875→    const updated = await tx.purchaseOrder.update({
   876→   876→      where: { id: header.id },
   877→   877→      data,
   878→   878→      select: { id: true, reference: true, status: true, sentAt: true, confirmedAt: true, closedAt: true, cancelledAt: true },
   879→   879→    });
   880→   880→
   881→   881→    // §17/§18 — CONFIRMED/CLOSED are pure document-state transitions: NO
   882→   882→    // inventory, expense, payment or statement write happens here (pinned
   883→   883→    // by the §110-§113 negative tests and the §147-§150 static scan).
   884→   884→    await audit(
   885→   885→      auditInput(input.businessId, input.userId, `purchase_order.${input.op}`, header.id, header.reference, {
   886→   886→        previousStatus: header.status,
   887→   887→        newStatus: target,
   888→   888→      }),
   889→   889→      tx,
   890→   890→    );
   891→   891→
   892→   892→    return updated;
   893→   893→  });
