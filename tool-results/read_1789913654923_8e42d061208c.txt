   498→   498→                streamError = payload.error;
   499→   499→              } else if (payload.done) {
   500→   500→                sawDone = true;
   501→   501→              }
   502→   502→            }
   503→   503→          }

   533→   533→            content: localizedError("NETWORK_ERROR"),
