   178→/** Stable identity of a view (for equality checks / keys). */
   179→export function viewKeyOf(view: AppView): string {
   180→  return view.kind === "module"
   181→    ? `module:${view.module}`
   182→    // Prompt 35 — the tab is part of the key so tab switches remount the
   183→    // inner CRM view deterministically.
   184→    : view.kind === "crm"
   185→      ? `crm:${view.tab}`
   186→    // Prompt 34 — entity + id both matter: opening a different record
   187→    // remounts the detail view so no stale record can ever be shown (§36).
   188→    : view.kind === "entity"
   189→      ? `entity:${view.entity}:${view.id}`
   190→    // Prompt 39 — type + id both matter: opening another party's statement
   191→    // remounts the view so no stale statement can ever be shown (§159).
   192→    : view.kind === "statements"
   193→      ? `statements:${view.statementType}:${view.id}`
   194→    // Prompt 40 — stocktake views: list + per-id detail keys so opening
   195→    // another stocktake remounts the detail view (no stale record, §36 parity).
   196→    : view.kind === "inventory-stocktakes"
   197→      ? "inventory-stocktakes"
   198→    : view.kind === "inventory-stocktake"
   199→      ? `inventory-stocktake:${view.id}`
   200→    // Prompt 41 — purchase-order views: list + per-id detail keys so opening
   201→    // another order remounts the detail view (no stale record, §36 parity).
   202→    : view.kind === "purchase-orders"
   203→      ? "purchase-orders"
   204→    : view.kind === "purchase-order"
   205→      ? `purchase-order:${view.id}`
   206→    // Prompt 42 — goods-receipt views: list + per-id detail keys so opening
   207→    // another receipt remounts the detail view (no stale record, §36 parity).
   208→    : view.kind === "inventory-goods-receipts"
   209→      ? "inventory-goods-receipts"
   210→    : view.kind === "inventory-goods-receipt"
   211→      ? `inventory-goods-receipt:${view.id}`
   212→    // Prompt 43 — goods-return views: list + per-id detail keys so opening
   213→    // another return remounts the detail view (§36 parity).
   214→    : view.kind === "inventory-goods-returns"
   215→      ? "inventory-goods-returns"
   216→    : view.kind === "inventory-goods-return"
   217→      ? `inventory-goods-return:${view.id}`
   218→    // Prompt 44 — goods-dispatch views: list + per-id detail keys so opening
   219→    // another dispatch remounts the detail view (P43 parity).
   220→    : view.kind === "inventory-goods-dispatches"
   221→      ? "inventory-goods-dispatches"
   222→    : view.kind === "inventory-goods-dispatch"
   223→      ? `inventory-goods-dispatch:${view.id}`
   224→      : view.kind === "join"
   225→      ? `join:${view.token}`
   226→      : view.kind === "reset-password"
   227→        ? `reset:${view.token}`
   228→        : view.kind === "verify-email"
   229→          ? `verify:${view.token}`
   230→          : view.kind === "platform"
   231→            ? `platform:${view.section}`
   232→            // Prompt 33 — the Search Center keeps a STABLE view key so the
   233→            // mounted view (and its focused input) survives query updates;
   234→            // the query itself only matters for sameView/back-forward (§13).
   235→            : view.kind;
   236→}

   238→/**
   239→ * Structural view equality (applyHash). Deliberately FINER than viewKeyOf:
   240→ * module sub-state (?month= deep-link param) must re-enter the store when
   241→ * the hash changes, but must NOT remount the module (viewKey stays stable
   242→ * so React keeps the mounted view alive — reports state survives dialog
   243→ * open/close).
   244→ */
   245→export function sameView(a: AppView, b: AppView): boolean {
   246→  if (viewKeyOf(a) !== viewKeyOf(b)) return false;
   247→  if (a.kind === "module" && b.kind === "module") {
   248→    return (a.param ?? null) === (b.param ?? null);
   249→  }
   250→  // Prompt 34 — entity views compare entity + id (viewKey already carries
   251→  // both, so reaching here means they are equal; kept explicit for clarity).
   252→  if (a.kind === "entity" && b.kind === "entity") {
   253→    return a.entity === b.entity && a.id === b.id;
   254→  }
   255→  // Prompt 35 — CRM tab changes re-enter the store (back/forward) without
   256→  // remounting the module boundary shell; the tab drives the inner view.
   257→  if (a.kind === "crm" && b.kind === "crm") {
   258→    return a.tab === b.tab;
   259→  }
   260→  // Prompt 39 — statement views compare type + id (viewKey already carries
   261→  // both; explicit for parity with entity views).
   262→  if (a.kind === "statements" && b.kind === "statements") {
   263→    return a.statementType === b.statementType && a.id === b.id;
   264→  }
   265→  // Prompt 40 — stocktake detail compares id (viewKey already carries it).
   266→  if (a.kind === "inventory-stocktake" && b.kind === "inventory-stocktake") {
   267→    return a.id === b.id;
   268→  }
   269→  // Prompt 41 — purchase-order detail compares id (viewKey already carries it).
   270→  if (a.kind === "purchase-order" && b.kind === "purchase-order") {
   271→    return a.id === b.id;
   272→  }
   273→  // Prompt 42 — goods-receipt detail compares id (viewKey already carries it).
   274→  if (a.kind === "inventory-goods-receipt" && b.kind === "inventory-goods-receipt") {
   275→    return a.id === b.id;
   276→  }
   277→  // Prompt 43 — goods-return detail compares id (viewKey already carries it).
   278→  if (a.kind === "inventory-goods-return" && b.kind === "inventory-goods-return") {
   279→    return a.id === b.id;
   280→  }
   281→  // Prompt 44 — goods-dispatch detail compares id (viewKey already carries it).
   282→  if (a.kind === "inventory-goods-dispatch" && b.kind === "inventory-goods-dispatch") {
   283→    return a.id === b.id;
   284→  }
   285→  // Prompt 33 — hash query changes re-enter the store (back/forward must
   286→  // update the Search Center) without remounting the view (§13/§33).
   287→  if (a.kind === "search" && b.kind === "search") {
   288→    return (a.query ?? null) === (b.query ?? null);
   289→  }
   290→  return true;
   291→}

   293→/** URL hash for a view. `onboarding` is a transient state — no address. */
   294→export function viewToHash(view: AppView): string | null {
   295→  switch (view.kind) {
   296→    case "dashboard":
   297→      return "#/dashboard";
   298→    case "audit":
   299→      return "#/audit";
   300→    case "settings:business":
   301→      return "#/settings";
   302→    case "settings:members":
   303→      return "#/settings/members";
   304→    case "settings:billing":
   305→      return "#/settings/billing";
   306→    case "profile":
   307→      return "#/profile";
   308→    case "settings:security":
   309→      return "#/settings/security";
   310→    case "settings:data":
   311→      return "#/data";
   312→    case "notifications":
   313→      return "#/notifications";
   314→    case "settings:notifications":
   315→      return "#/settings/notifications";
   316→    case "search": {
   317→      // §33 — the query persists in the hash (encodeURIComponent, §29) so
   318→      // refresh and back/forward keep it; never any other param.
   319→      const q = (view.query ?? "").trim();
   320→      return q ? `#/search?q=${encodeURIComponent(q)}` : "#/search";
   321→    }
   322→    case "crm":
   323→      // Prompt 35 (§37) — the tab persists in the hash; the default tab
   324→      // renders as bare "#/crm" so the address stays canonical.
   325→      return view.tab === CRM_DEFAULT_TAB ? "#/crm" : `#/crm?tab=${view.tab}`;
   326→    case "entity":
   327→      // Prompt 34 — centralized encoder (§7/§39): validated entity + id,
   328→      // safe-charset id, encodeURIComponent. Returns null (never a hash)
   329→      // for anything that could not survive parseEntityRoute.
   330→      return entityToHash(view.entity, view.id);
   331→    case "statements":
   332→      // Prompt 39 (§57/§139) — closed type + validated id only; anything
   333→      // else produces no hash at all (never an arbitrary statement URL).
   334→      if (!isStatementTypeToken(view.statementType)) return null;
   335→      if (!isValidEntityId(view.id)) return null;
   336→      return `#/statements?type=${view.statementType}&id=${encodeURIComponent(view.id)}`;
   337→    case "inventory-stocktakes":
   338→      // Prompt 40 (§76) — canonical list hash.
   339→      return "#/inventory/stocktakes";
   340→    case "inventory-stocktake":
   341→      // Prompt 40 (§77/§136) — only charset-safe ids ever become a hash.
   342→      if (!isValidEntityId(view.id)) return null;
   343→      return `#/inventory/stocktakes/${view.id}`;
   344→    case "purchase-orders":
   345→      // Prompt 41 (§70) — canonical list hash.
   346→      return "#/purchase-orders";
   347→    case "purchase-order":
   348→      // Prompt 41 (§70) — only charset-safe ids ever become a hash.
   349→      if (!isValidEntityId(view.id)) return null;
   350→      return `#/purchase-orders/${view.id}`;
   351→    case "inventory-goods-receipts":
   352→      // Prompt 42 (§116) — canonical list hash.
   353→      return "#/inventory/goods-receipts";
   354→    case "inventory-goods-receipt":
   355→      // Prompt 42 (§116) — only charset-safe ids ever become a hash.
   356→      if (!isValidEntityId(view.id)) return null;
   357→      return `#/inventory/goods-receipts/${view.id}`;
   358→    case "inventory-goods-returns":
   359→      // Prompt 43 (§91) — canonical list hash.
   360→      return "#/inventory/goods-returns";
   361→    case "inventory-goods-return":
   362→      // Prompt 43 (§91) — only charset-safe ids ever become a hash.
   363→      if (!isValidEntityId(view.id)) return null;
   364→      return `#/inventory/goods-returns/${view.id}`;
   365→    case "inventory-goods-dispatches":
   366→      // Prompt 44 — canonical list hash.
   367→      return "#/inventory/goods-dispatches";
   368→    case "inventory-goods-dispatch":
   369→      // Prompt 44 — only charset-safe ids ever become a hash.
   370→      if (!isValidEntityId(view.id)) return null;
   371→      return `#/inventory/goods-dispatches/${view.id}`;
   372→    case "platform":
   373→      // #/platform itself resolves to the overview section (§15).
   374→      return view.section === "overview" ? "#/platform" : `#/platform/${view.section}`;
   375→    case "module": {
   376→      if (!isKnownModule(view.module)) return null;
   377→      // Module sub-state deep links (e.g. #/reports?month=YYYY-MM) stay in
   378→      // the address bar — month is the only sub-param today.
   379→      if (view.param && /^\d{4}-\d{2}$/.test(view.param)) {
   380→        return `#/${view.module}?month=${view.param}`;
   381→      }
   382→      return `#/${view.module}`;
   383→    }
   384→    case "join":
   385→      return `#/join/${view.token}`;
   386→    case "forgot-password":
   387→      return "#/forgot-password";
   388→    case "reset-password":
   389→      return `#/reset-password/${view.token}`;
   390→    case "verify-email":
   391→      return `#/verify-email/${view.token}`;
   392→    case "onboarding":
   393→      return null;
   394→  }
   395→}

   397→/** Parse a location hash into a view. Unknown → dashboard (never onboarding). */
   398→export function hashToView(hash: string): AppView {
   399→  const raw = hash.replace(/^#\/?/, "").replace(/\/+$/, "");
   400→  // Query part (e.g. #/reports?month=2026-04) — module-scoped deep-link params.
   401→  const qIndex = raw.indexOf("?");
   402→  const path = (qIndex === -1 ? raw : raw.slice(0, qIndex)).trim();
   403→  const query = new URLSearchParams(qIndex === -1 ? "" : raw.slice(qIndex + 1));
   404→  if (!path || path === "dashboard") return { kind: "dashboard" };
   405→  if (path === "audit") return { kind: "audit" };
   406→  if (path === "settings") return { kind: "settings:business" };
   407→  if (path === "settings/members") return { kind: "settings:members" };
   408→  if (path === "settings/billing") return { kind: "settings:billing" };
   409→  if (path === "settings/security") return { kind: "settings:security" };
   410→  if (path === "data") return { kind: "settings:data" };
   411→  if (path === "notifications") return { kind: "notifications" };
   412→  if (path === "settings/notifications") return { kind: "settings:notifications" };
   413→  // Prompt 33 — Search Results Center deep links (#/search, #/search?q=…).
   414→  if (path === "search") {
   415→    const q = query.get("q") ?? "";
   416→    const clean = sanitizeSearchHashQuery(q);
   417→    return clean ? { kind: "search", query: clean } : { kind: "search" };
   418→  }
   419→  // Prompt 35 (§37) — CRM deep links: #/crm, #/crm?tab=<closed tab>.
   420→  // Unknown tab values fall back to CRM_DEFAULT_TAB (pipeline) — the hash
   421→  // query is never interpreted as arbitrary state.
   422→  if (path === "crm") {
   423→    const tab = query.get("tab") ?? "";
   424→    return { kind: "crm", tab: isCrmTab(tab) ? tab : CRM_DEFAULT_TAB };
   425→  }
   426→  if (path === "profile") return { kind: "profile" };
   427→  // Prompt 28 — platform console deep links (#/platform, #/platform/<section>).
   428→  if (path === "platform") return { kind: "platform", section: "overview" };
   429→  if (path.startsWith("platform/")) {
   430→    const section = path.slice("platform/".length);
   431→    if (isPlatformSection(section)) return { kind: "platform", section };
   432→    return { kind: "platform", section: "overview" };
   433→  }
   434→  if (path.startsWith("join/")) {
   435→    const token = path.slice("join/".length);
   436→    // Opaque invite token: non-empty, bounded length (server validates the rest).
   437→    if (token.length > 0 && token.length <= 128) return { kind: "join", token };
   438→  }
   439→  if (path === "forgot-password") return { kind: "forgot-password" };
   440→  if (path.startsWith("verify-email/")) {
   441→    const token = path.slice("verify-email/".length);
   442→    // Opaque verification token: bounded length; the server validates the rest.
   443→    if (token.length > 0 && token.length <= 200) return { kind: "verify-email", token };
   444→    return { kind: "dashboard" };
   445→  }
   446→  if (path.startsWith("reset-password/")) {
   447→    const token = path.slice("reset-password/".length);
   448→    // Opaque reset token: non-empty, bounded length (server validates the rest).
   449→    if (token.length > 0 && token.length <= 200) return { kind: "reset-password", token };
   450→    // Malformed reset link → the forgot form so a fresh link can be requested.
   451→    return { kind: "forgot-password" };
   452→  }
   453→  // Prompt 39 (§57/§58/§139) — statement deep links: #/statements with a
   454→  // closed `type` token and a safe `id`. Unknown type / missing or unsafe
   455→  // id → the existing unknown-hash fallback (dashboard); the query string
   456→  // is never interpreted as arbitrary state.
   457→  if (path === "statements") {
   458→    const type = query.get("type") ?? "";
   459→    const rawId = query.get("id") ?? "";
   460→    if (!isStatementTypeToken(type)) return { kind: "dashboard" };
   461→    let id = rawId;
   462→    try {
   463→      id = decodeURIComponent(rawId);
   464→    } catch {
   465→      return { kind: "dashboard" };
   466→    }

   565→      return { kind: "module", module: path, param: month };
   566→    }
   567→    return { kind: "module", module: path };
   568→  }
   569→  return { kind: "dashboard" };

   572→/** Sync the URL to a view without triggering navigation events. */
   573→function syncUrl(view: AppView, replace: boolean) {
   574→  if (typeof window === "undefined") return;
   575→  const hash = viewToHash(view);
   576→  if (!hash) return;
   577→  const target = `${window.location.pathname}${window.location.search}${hash}`;
   578→  if (`${window.location.pathname}${window.location.search}${window.location.hash}` === target) {
   579→    return;
   580→  }
   581→  try {
   582→    if (replace) window.history.replaceState(null, "", target);
   583→    else window.history.pushState(null, "", target);
   584→  } catch {
   585→    // history might be unavailable in exotic embeds — URL sync is best-effort
   586→  }
   587→}

   589→/**
   590→ * Prompt 33 — hash `?q=` hygiene (§16/§29): the query is rendered strictly
   591→ * as text, so only control characters/null bytes are stripped and the
   592→ * length is capped to the search schema bound. URLSearchParams already
   593→ * performs percent-decoding; no other transformation is applied.
   594→ */
   595→function sanitizeSearchHashQuery(raw: string): string {
   596→  const capped = Array.from(raw)
   597→    .filter((ch) => {
   598→      const code = ch.codePointAt(0) ?? 0;
   599→      return code > 0x1f && code !== 0x7f;
   600→    })
   601→    .slice(0, 64)
   602→    .join("");
   603→  const trimmed = capped.trim();
   604→  return trimmed.length > 0 ? trimmed : "";
   605→}
