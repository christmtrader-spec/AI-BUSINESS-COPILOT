   422→  1521→ *   1. reload the quotation tenant-scoped;
   423→  1522→ *   2. already-converted guard FIRST (the ONE stable conflict for every
   424→  1523→ *      loser of any race — the P45 D-5 lesson), backed by the DB UNIQUE
   425→  1524→ *      constraint on SalesOrder.sourceQuotationId (§7);
   426→  1525→ *   3. persisted status law (ACCEPTED only — invalid → controlled 500,
   427→  1526→ *      wrong-but-valid status → 409);
   428→  1527→ *   4. complete, materially valid line set (§11);
   429→  1528→ *   5. canonical totals recomputed from authoritative persisted lines;
   430→  1529→ *   6. every snapshot validated (§12);
   431→  1530→ *   7. ONE SalesOrder created CONFIRMED — customer snapshot, currency
   432→  1531→ *      snapshot, expected date, discount copied EXACTLY; orderDate =
   433→  1532→ *      current business-local date (established midnoon rule);
   434→  1533→ *      internalNotes → SalesOrder.notes (staff-visible order notes —
   435→  1534→ *      never the customer print view §16);

   437→  1536→ *      snapshots + exact qty/unitPriceMinor (NEVER re-priced from live
   438→  1537→ *      Products §12/§13);
   439→  1538→ *   9. order totals derived via the EXISTING SalesOrder domain functions;
   440→  1539→ *  10. quotation → CONVERTED + convertedAt;
   441→  1540→ *  11. audit ×2 (sales_order.create + sales_quotation.convert) in-tx;
   442→  1541→ *  12. commit both documents together.
   443→  1542→ *
   444→  1543→ * NO Sale invoice, NO Payment, NO receivable, NO InventoryAdjustment, NO
   445→  1544→ * stock reservation, NO Product.stockQty change (§13 invariants). NO
   446→  1545→ * notification claims an invoice or payment occurred.
   447→  1546→ */
   448→  1547→export async function convertSalesQuotationToOrder(input: {
   449→  1548→  businessId: string;
   450→  1549→  userId: string;
   451→  1550→  id: string;
   452→  1551→  timezone: string;
   453→  1552→  ip?: string | null;
   454→  1553→  userAgent?: string | null;
   455→  1554→}) {
   456→  1555→  return withWriteTx(async (tx) => {
   457→  1556→    const cap = salesQuotationLineCap(process.env.SALES_QUOTATION_MAX_LINES);
   458→  1557→    // 1. tenant-scoped reload INSIDE the transaction (§13 step 1).
   459→  1558→    const header = await loadQuotationHeader(tx, input.businessId, input.id);
   460→  1559→
   461→  1560→    // 2. already-converted guard BEFORE the status gate (§13/D-5 lesson):
   462→  1561→    //    every loser — including a retry after the winner committed —
   463→  1562→    //    receives the ONE documented stable conflict code.
   464→  1563→    const existingOrder = await tx.salesOrder.findFirst({
   465→  1564→      where: { sourceQuotationId: header.id },
   466→  1565→      select: { id: true, reference: true },
   467→  1566→    });
   468→  1567→    if (existingOrder) {
   469→  1568→      throw new ApiError(
   470→  1569→        409,
   471→  1570→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
   472→  1571→        "This quotation has already been converted to a sales order.",
   473→  1572→      );
   474→  1573→    }
   475→  1574→
   476→  1575→    // 3. persisted status law: an ACCEPTED quotation converts; a
   477→  1576→    //    valid-but-wrong status is an ordinary conflict; a corrupted
   478→  1577→    //    persisted status is the controlled integrity 500 (parseDbStatus).
   479→  1578→    if (header.status !== "ACCEPTED") {
   480→  1579→      if (
   481→  1580→        header.status === "REJECTED" ||
   482→  1581→        header.status === "EXPIRED" ||
   483→  1582→        header.status === "CANCELLED"
   484→  1583→      ) {
   485→  1584→        throw new ApiError(
   486→  1585→          409,
   487→  1586→          SALES_QUOTATION_ERROR_CODES.NOT_ACCEPTED,
   488→  1587→          "Only ACCEPTED sales quotations can be converted.",
   489→  1588→        );
   490→  1589→      }
   491→  1590→      throw new ApiError(
   492→  1591→        409,
   493→  1592→        SALES_QUOTATION_ERROR_CODES.INVALID_STATUS,
   494→  1593→        "Only ACCEPTED sales quotations can be converted.",
   495→  1594→      );
   496→  1595→    }
   497→  1596→    // §9 — ACCEPTED never expires; conversion remains legal after the
   498→  1597→    // original validity date passes because acceptance was already
   499→  1598→    // recorded (documented §9 law).
   500→  1599→
   501→  1600→    // 4–5. complete line set + canonical totals from persisted lines.

   504→  1603→      // An ACCEPTED quotation with zero lines is corruption (it could
   505→  1604→      // never have passed mark-sent) — integrity, not a user error.
   506→  1605→      mapPureSalesQuotationDomainError(
   507→  1606→        new SalesQuotationIntegrityError(
   508→  1607→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: accepted quotation without lines`,
   509→  1608→        ),
   510→  1609→      );
   511→  1610→    }
   512→  1611→    const quotationTotals = computeQuotationSubtotalMinor(lines);
   513→  1612→    try {
   514→  1613→      validateQuotationDiscount(header.discountMinor, quotationTotals.subtotalMinor);
   515→  1614→    } catch {
   516→  1615→      mapPureSalesQuotationDomainError(
   517→  1616→        new SalesQuotationIntegrityError(
   518→  1617→          `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: discount exceeds subtotal`,
   519→  1618→        ),
   520→  1619→      );
   521→  1620→    }
   522→  1621→    const quotationTotalMinor = computeQuotationTotalMinor(
   523→  1622→      quotationTotals.subtotalMinor,
   524→  1623→      header.discountMinor,
   525→  1624→    );
   526→  1625→
   527→  1626→    // 6. every snapshot validated (§12/§13 step 6).
   528→  1627→    validateHeaderSnapshots(header);
   529→  1628→
   530→  1629→    // Order reference: the established SO generator, distinct sequence
   531→  1630→    // domain (§10 — quotation and order references never share a sequence).
   532→  1631→    let orderReference: string | undefined;
   533→  1632→    for (let attempt = 0; attempt < 5; attempt += 1) {
   534→  1633→      const candidate = generateSalesOrderReference(new Date(), (max) => randomInt(max));
   535→  1634→      const clash = await tx.salesOrder.findFirst({
   536→  1635→        where: { businessId: input.businessId, reference: candidate },
   537→  1636→        select: { id: true },
   538→  1637→      });
   539→  1638→      if (!clash) {
   540→  1639→        orderReference = candidate;
   541→  1640→        break;
   542→  1641→      }
   543→  1642→    }
   544→  1643→    if (orderReference === undefined) {
   545→  1644→      throw new ApiError(
   546→  1645→        409,
   547→  1646→        SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
   548→  1647→        "Could not generate a unique order reference. Please try again.",
   549→  1648→      );
   550→  1649→    }
   551→  1650→
   552→  1651→    // §13 step 11 — orderDate is the CURRENT business-local date
   553→  1652→    // normalized with the established UTC-midnoon rule (server clock).
   554→  1653→    const orderDate = new Date(
   555→  1654→      `${businessLocalCalendarDate(new Date(), input.timezone)}T12:00:00.000Z`,
   556→  1655→    );
   557→  1656→    const confirmedAt = new Date();
   558→  1657→
   559→  1658→    // 7. ONE SalesOrder — CONFIRMED, source link established (§13 steps
   560→  1659→    //    8–12/19/20). Customer/currency snapshots copied EXACTLY.
   561→  1660→    let order;
   562→  1661→    try {
   563→  1662→      order = await tx.salesOrder.create({
   564→  1663→        data: {
   565→  1664→          businessId: input.businessId, // server-derived
   566→  1665→          customerId: header.customerId,
   567→  1666→          customerNameAtOrder: header.customerNameAtQuotation, // frozen §12
   568→  1667→          reference: orderReference,
   569→  1668→          status: "CONFIRMED", // acceptance = commercial commitment (§13)
   570→  1669→          currencyCodeAtOrder: header.currencyCodeAtQuotation,
   571→  1670→          currencySymbolAtOrder: header.currencySymbolAtQuotation,
   572→  1671→          decimalPrecisionAtOrder: header.decimalPrecisionAtQuotation,
   573→  1672→          orderDate,
   574→  1673→          expectedDate: header.expectedDeliveryDate,
   575→  1674→          discountMinor: header.discountMinor, // copied EXACTLY (§13 step 13)
   576→  1675→          notes: header.internalNotes, // staff notes continuity (§13 step 14)
   577→  1676→          createdByUserId: input.userId,
   578→  1677→          confirmedAt, // §13 step 20 — server-side transaction time
   579→  1678→          sourceQuotationId: header.id, // §13 step 23 — the ONE authority
   580→  1679→        },
   581→  1680→        select: { id: true, reference: true, status: true, confirmedAt: true },
   582→  1681→      });
   583→  1682→    } catch (error) {
   584→  1683→      // The DB unique constraint is the FINAL race authority (§7/§13):
   585→  1684→      // a concurrent winner (or an impossible prior conversion) surfaces
   586→  1685→      // as the ONE stable conflict — never a second order.
   587→  1686→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   588→  1687→        throw new ApiError(
   589→  1688→          409,
   590→  1689→          SALES_QUOTATION_ERROR_CODES.ALREADY_CONVERTED,
   591→  1690→          "This quotation has already been converted to a sales order.",
   592→  1691→        );
   593→  1692→      }
   594→  1693→      throw error;
   595→  1694→    }
   596→  1695→
   597→  1696→    // 8. ONE SalesOrderLine per quotation line — the FROZEN snapshots and
   598→  1697→    //    exact agreed prices (§13 steps 15–17; never re-read/re-priced).
   599→  1698→    //    The order subtotal is re-derived via the EXISTING order domain
   600→  1699→    //    functions so both documents provably agree (§13 step 18).
   601→  1700→    let orderSubtotalMinor = 0;
   602→  1701→    try {
   603→  1702→      for (const line of lines) {
   604→  1703→        await tx.salesOrderLine.create({
   605→  1704→          data: {
   606→  1705→            salesOrderId: order.id,
   607→  1706→            productId: line.productId,
   608→  1707→            orderedQty: line.qty, // exact quantity (§13 step 17)
   609→  1708→            unitPriceMinor: line.unitPriceMinor, // exact agreed price
   610→  1709→            productNameAtOrder: line.productNameAtQuotation,
   611→  1710→            skuAtOrder: line.skuAtQuotation,
   612→  1711→            unitAtOrder: line.unitAtQuotation,
   613→  1712→          },
   614→  1713→          select: { id: true },
   615→  1714→        });
   616→  1715→        orderSubtotalMinor += computeQuotationLineTotalMinor(line.qty, line.unitPriceMinor);
   617→  1716→      }
   618→  1717→      if (!Number.isSafeInteger(orderSubtotalMinor)) {
   619→  1718→        throw new SalesQuotationIntegrityError(
   620→  1719→          `${SALES_QUOTATION_ERROR_CODES.TOTAL_OVERFLOW}: order subtotal overflow during conversion`,
   621→  1720→        );
   622→  1721→      }
   623→  1722→      // §13 step 18 — derive the order totals via the EXISTING domain
   624→  1723→      // functions; quotation and order totals MUST agree exactly.
   625→  1724→      const orderTotals = computeOrderSubtotalMinor(
   626→  1725→        lines.map((l) => ({ orderedQty: l.qty, unitPriceMinor: l.unitPriceMinor })),
   627→  1726→      );
   628→  1727→      if (orderTotals.subtotalMinor !== quotationTotals.subtotalMinor) {
   629→  1728→        mapPureSalesQuotationDomainError(
   630→  1729→          new SalesQuotationIntegrityError(
   631→  1730→            `${SALES_QUOTATION_ERROR_CODES.CORRUPT_HISTORY}: quotation/order subtotal disagreement`,
   632→  1731→          ),
   633→  1732→        );
   634→  1733→      }
   635→  1734→    } catch (error) {
   636→  1735→      // §13/H — a failure here aborts the WHOLE transaction: no partial
   637→  1736→      // order, no partially copied lines, no CONVERTED quotation, no
   638→  1737→      // stray audit (withWriteTx rolls back everything).
   639→  1738→      throw error;
   640→  1739→    }
   641→  1740→
   642→  1741→    // 10. quotation → CONVERTED + convertedAt (§13 steps 21–22).
   643→  1742→    const convertedAt = new Date();
   644→  1743→    await tx.salesQuotation.update({
   645→  1744→      where: { id: header.id },
   646→  1745→      data: { status: "CONVERTED", convertedAt },
   647→  1746→      select: { id: true },
   648→  1747→    });
   649→  1748→
   650→  1749→    // 11. audit ×2, SAME transaction (§13 step 24). Minimal safe payloads;
   651→  1750→    //     never the notes contents.
   652→  1751→    await audit(
   653→  1752→      {
   654→  1753→        businessId: input.businessId,
   655→  1754→        actorId: input.userId,
   656→  1755→        action: "sales_order.create",
   657→  1756→        entityType: "SalesOrder",
   658→  1757→        entityId: order.id,
   659→  1758→        after: {
   660→  1759→          reference: order.reference,
   661→  1760→          sourceQuotationId: header.id,
   662→  1761→          sourceQuotationReference: header.reference,
   663→  1762→          status: order.status,
   664→  1763→          lineCount: lines.length,
   665→  1764→          subtotalMinor: quotationTotals.subtotalMinor,
   666→  1765→          discountMinor: header.discountMinor,
   667→  1766→          totalMinor: quotationTotalMinor,
   668→  1767→        },
   669→  1768→      },
   670→  1769→      tx,
   671→  1770→    );
   672→  1771→    await audit(
   673→  1772→      auditInput(input.businessId, input.userId, "sales_quotation.convert", header.id, header.reference, {
   674→  1773→        salesOrderId: order.id,
   675→  1774→        salesOrderReference: order.reference,
   676→  1775→        lineCount: lines.length,
   677→  1776→        subtotalMinor: quotationTotals.subtotalMinor,
   678→  1777→        discountMinor: header.discountMinor,
   679→  1778→        totalMinor: quotationTotalMinor,
   680→  1779→        previousStatus: header.status,
   681→  1780→        newStatus: "CONVERTED",
   682→  1781→        convertedAt: convertedAt.toISOString(),
   683→  1782→      }),
   684→  1783→      tx,
   685→  1784→    );
   686→  1785→
   687→  1786→    // 12. both documents commit together (§13 step 25).
   688→  1787→    return {
   689→  1788→      quotationId: header.id,
   690→  1789→      quotationReference: header.reference,
   691→  1790→      salesOrder: order,
   692→  1791→    };
   693→  1792→  });
   694→  1793→}
