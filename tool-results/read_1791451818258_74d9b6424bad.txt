     6→     6→ * LAWS:
     7→     7→ *  - PHYSICAL INVENTORY EVENT ONLY (§3/§4/§5): posting mutates
     8→     8→ *    Product.stockQty and creates one positive InventoryAdjustment per
     9→     9→ *    non-zero received line. It NEVER creates Expense, Payment, Sale,

    13→    13→ *    orderedQty, enforced by the ONE pure helper at line-add (advisory,

    22→    22→ *  - Snapshots (supplier name, PO reference, product name/sku/unit) are
    23→    23→ *    frozen at write time from ALREADY-FROZEN PO data (§24/§25/§28/§29)

   274→   274→export async function createGoodsReceipt(input: {
   275→   275→  businessId: string;
   276→   276→  userId: string;
   277→   277→  purchaseOrderId: string;
   278→   278→  reference?: string;
   279→   279→  notes?: string | null;
   280→   280→  ip?: string | null;
   281→   281→  userAgent?: string | null;
   282→   282→}) {
   283→   283→  return withWriteTx(async (tx) => {
   284→   284→    // §11/§23 — the PO must be tenant-scoped and CONFIRMED.
   285→   285→    const po = await tx.purchaseOrder.findFirst({
   286→   286→      where: { id: input.purchaseOrderId, businessId: input.businessId }, // §90 — foreign PO → 404
   287→   287→      select: {
   288→   288→        id: true,
   289→   289→        reference: true,
   290→   290→        status: true,
   291→   291→        supplierNameAtOrder: true, // §25 — copy the ALREADY-FROZEN snapshot
   292→   292→      },
   293→   293→    });
   294→   294→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   295→   295→    assertPurchaseOrderReceivable(po.status); // §11 — DRAFT/SENT/CLOSED/CANCELLED → 422
   296→   296→
   297→   297→    // §59 — reference: user-supplied (normalized) or server-generated.
   298→   298→    let reference = input.reference;
   299→   299→    if (reference === undefined) {
   300→   300→      for (let attempt = 0; attempt < 5; attempt += 1) {
   301→   301→        const candidate = generateGoodsReceiptReference(new Date(), (max) => randomInt(max));
   302→   302→        const clash = await tx.goodsReceipt.findFirst({
   303→   303→          where: { businessId: input.businessId, reference: candidate },
   304→   304→          select: { id: true },
   305→   305→        });
   306→   306→        if (!clash) {
   307→   307→          reference = candidate;
   308→   308→          break;
   309→   309→        }
   310→   310→      }
   311→   311→      if (reference === undefined) {
   312→   312→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   313→   313→        throw new ApiError(
   314→   314→          409,
   315→   315→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   316→   316→          "Could not generate a unique reference. Please try again.",
   317→   317→        );
   318→   318→      }
   319→   319→    }
   320→   320→
   321→   321→    try {
   322→   322→      const row = await tx.goodsReceipt.create({
   323→   323→        data: {
   324→   324→          businessId: input.businessId, // §22 — server-derived, never client
   325→   325→          purchaseOrderId: po.id,
   326→   326→          reference,
   327→   327→          status: "DRAFT", // §58 — every receipt starts as preparation only
   328→   328→          supplierNameAtReceipt: po.supplierNameAtOrder, // §25 — frozen identity
   329→   329→          purchaseOrderReferenceAtReceipt: po.reference, // §24 — frozen identity
   330→   330→          notes: input.notes ?? null, // §62 — document data only
   331→   331→          createdByUserId: input.userId, // §58 — server-derived
   332→   332→        },
   333→   333→        select: { id: true, reference: true, status: true },
   334→   334→      });
   335→   335→
   336→   336→      await audit(
   337→   337→        auditInput(input.businessId, input.userId, "goods_receipt.create", row.id, row.reference, {
   338→   338→          purchaseOrderId: po.id,
   339→   339→          purchaseOrderReferenceAtReceipt: po.reference,
   340→   340→          supplierNameAtReceipt: po.supplierNameAtOrder,
   341→   341→          status: row.status,
   342→   342→        }),
   343→   343→        tx,
   344→   344→      );
   345→   345→
   346→   346→      return row;
   347→   347→    } catch (error) {
   348→   348→      // §61 — database uniqueness stays authoritative for duplicates.
   349→   349→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   350→   350→        throw new ApiError(
   351→   351→          409,
   352→   352→          GOODS_RECEIPT_ERROR_CODES.DUPLICATE_REFERENCE,
   353→   353→          "A goods receipt with this reference already exists.",
   354→   354→        );
   355→   355→      }
   356→   356→      throw error;
   357→   357→    }
   358→   358→  });
   359→   359→}

   363→   363→export async function listGoodsReceipts(input: {
   364→   364→  businessId: string;
   365→   365→  q?: string;
   366→   366→  status?: string; // ALL → undefined
   367→   367→  purchaseOrderId?: string;
   368→   368→  from?: string;
   369→   369→  to?: string;
   370→   370→  page: number;
   371→   371→  pageSize: number;
   372→   372→}): Promise<{
   373→   373→  items: GoodsReceiptListItemDto[];
   374→   374→  total: number;
   375→   375→  page: number;
   376→   376→  pageSize: number;
   377→   377→}> {
   378→   378→  const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   379→   379→  const where: Prisma.GoodsReceiptWhereInput = {
   380→   380→    businessId: input.businessId, // §89 — tenant scope
   381→   381→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   382→   382→    ...(input.purchaseOrderId ? { purchaseOrderId: input.purchaseOrderId } : {}),
   383→   383→    ...(input.from || input.to ? { createdAt: dayWindow(input.from, input.to) } : {}),
   384→   384→    ...(input.q
   385→   385→      ? {
   386→   386→          OR: [
   387→   387→            { reference: { contains: input.q } }, // receipt reference (§81)
   388→   388→            { purchaseOrderReferenceAtReceipt: { contains: input.q } }, // PO reference
   389→   389→            { supplierNameAtReceipt: { contains: input.q } }, // supplier snapshot
   390→   390→            {
   391→   391→              lines: {
   392→   392→                some: {
   393→   393→                  OR: [
   394→   394→                    { productNameAtReceipt: { contains: input.q } }, // product snapshot
   395→   395→                    { skuAtReceipt: { contains: input.q } }, // SKU snapshot
   396→   396→                  ],
   397→   397→                },
   398→   398→              },
   399→   399→            },
   400→   400→          ],
   401→   401→        }
   402→   402→      : {}),
   403→   403→  };
   404→   404→
   405→   405→  const [total, rows] = await Promise.all([
   406→   406→    db.goodsReceipt.count({ where }),
   407→   407→    // §175 — DB-level pagination; never findMany + slice.
   408→   408→    db.goodsReceipt.findMany({
   409→   409→      where,
   410→   410→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   411→   411→      skip: (input.page - 1) * input.pageSize,
   412→   412→      take: input.pageSize,
   413→   413→      select: {
   414→   414→        id: true,
   415→   415→        reference: true,
   416→   416→        status: true,
   417→   417→        purchaseOrderId: true,
   418→   418→        purchaseOrderReferenceAtReceipt: true,
   419→   419→        supplierNameAtReceipt: true,
   420→   420→        postedAt: true,
   421→   421→        cancelledAt: true,
   422→   422→        createdAt: true,
   423→   423→        lines: {
   424→   424→          // §180 — bounded aggregate relation, not one query per row.
   425→   425→          select: { receivedQty: true },
   426→   426→          take: cap,
   427→   427→        },
   428→   428→        _count: { select: { lines: true } },
   429→   429→      },
   430→   430→    }),
   431→   431→  ]);
   432→   432→
   433→   433→  const items = rows.map((row) => {
   434→   434→    // §180 — the capped relation slice IS the full line set (a receipt can
   435→   435→    // never exceed its own cap), so the sum below is provably complete.
   436→   436→    const totalReceivedQty = computeReceivedQty(row.lines.map((l) => l.receivedQty));
   437→   437→    return goodsReceiptListItemDto({
   438→   438→      id: row.id,
   439→   439→      reference: row.reference,
   440→   440→      status: row.status,
   441→   441→      purchaseOrderId: row.purchaseOrderId,
   442→   442→      purchaseOrderReferenceAtReceipt: row.purchaseOrderReferenceAtReceipt,
   443→   443→      supplierNameAtReceipt: row.supplierNameAtReceipt,
   444→   444→      lineCount: row._count.lines,
   445→   445→      totalReceivedQty,
   446→   446→      postedAt: row.postedAt,
   447→   447→      cancelledAt: row.cancelledAt,
   448→   448→      createdAt: row.createdAt,
   449→   449→      createdByUserId: "",
   450→   450→      createdByName: null,
   451→   451→      notes: null,
   452→   452→    });
   453→   453→  });
   454→   454→  return { items, total, page: input.page, pageSize: input.pageSize };
   455→   455→}

   584→   584→export async function addGoodsReceiptLine(input: {
   585→   585→  businessId: string;
   586→   586→  userId: string;
   587→   587→  id: string;
   588→   588→  purchaseOrderLineId: string;
   589→   589→  receivedQty: number;
   590→   590→  ip?: string | null;
   591→   591→  userAgent?: string | null;
   592→   592→}) {
   593→   593→  return withWriteTx(async (tx) => {
   594→   594→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   595→   595→    requireDraft(header.status); // §64 — DRAFT-only
   596→   596→
   597→   597→    // §35 — the PO line must belong to THIS receipt's tenant-scoped PO.
   598→   598→    // One tenant-scoped join validates receipt→PO→line ownership; a
   599→   599→    // foreign PO line is indistinguishable from an unknown one (§91 → 404).
   600→   600→    const poLine = await tx.purchaseOrderLine.findFirst({
   601→   601→      where: { id: input.purchaseOrderLineId, purchaseOrderId: header.purchaseOrderId },
   602→   602→      select: {
   603→   603→        id: true,
   604→   604→        productId: true,
   605→   605→        orderedQty: true,
   606→   606→        productNameAtOrder: true, // §29 — snapshots copied from the PO line
   607→   607→        skuAtOrder: true,
   608→   608→        unitAtOrder: true,
   609→   609→        product: { select: { id: true, businessId: true } },
   610→   610→      },
   611→   611→    });
   612→   612→    if (!poLine) throw new ApiError(404, "NOT_FOUND", "Purchase order line not found.");
   613→   613→
   614→   614→    // §35 — product must exist inside the SAME tenant (belt-and-braces:
   615→   615→    // the FK is Restrict so a hard-deleted product is nearly impossible,
   616→   616→    // but §33/§152 demands a safe controlled failure, never fabrication).
   617→   617→    if (poLine.product.businessId !== input.businessId) {
   618→   618→      throw new ApiError(404, "NOT_FOUND", "Product not found.");
   619→   619→    }
   620→   620→
   621→   621→    // §65 — the PO must STILL be CONFIRMED at line-add time (a PO that was
   622→   622→    // cancelled/closed after the draft was created stops accepting lines).
   623→   623→    const po = await tx.purchaseOrder.findFirst({
   624→   624→      where: { id: header.purchaseOrderId, businessId: input.businessId },
   625→   625→      select: { status: true },
   626→   626→    });
   627→   627→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   628→   628→    assertPurchaseOrderReceivable(po.status);
   629→   629→
   630→   630→    const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   631→   631→    await assertLineCap(tx, header.id, cap); // §71 — defensive count
   632→   632→
   633→   633→    // §66 — ADVISORY remaining check at line-add time: receivedQty ≤
   634→   634→    // (ordered − POSTED received). Draft lines of OTHER receipts are
   635→   635→    // deliberately NOT subtracted — over-allocation across drafts is the
   636→   636→    // documented §67 stale-draft scenario and only POST is authoritative.
   637→   637→    const alreadyPosted = await sumPostedQtyForPoLine(tx, poLine.id);
   638→   638→    const remaining = computeRemainingQty(poLine.orderedQty, alreadyPosted); // §172
   639→   639→    if (exceedsRemaining(poLine.orderedQty, alreadyPosted, input.receivedQty)) {
   640→   640→      throw new ApiError(
   641→   641→        409,
   642→   642→        GOODS_RECEIPT_ERROR_CODES.EXCEEDS_REMAINING,
   643→   643→        `Only ${remaining} unit(s) remain to receive on this line.`,
   644→   644→      );
   645→   645→    }
   646→   646→
   647→   647→    try {
   648→   648→      const line = await tx.goodsReceiptLine.create({
   649→   649→        data: {
   650→   650→          goodsReceiptId: header.id,
   651→   651→          purchaseOrderLineId: poLine.id,
   652→   652→          productId: poLine.productId, // §36 — derived, never client authority
   653→   653→          receivedQty: input.receivedQty,
   654→   654→          // §28/§29 — snapshots copied from the PO LINE, not live Product.
   655→   655→          productNameAtReceipt: poLine.productNameAtOrder,
   656→   656→          skuAtReceipt: poLine.skuAtOrder,
   657→   657→          unitAtReceipt: poLine.unitAtOrder,
   658→   658→        },
   659→   659→        select: { id: true },
   660→   660→      });
   661→   661→
   662→   662→      await audit(
   663→   663→        auditInput(input.businessId, input.userId, "goods_receipt.line_add", header.id, header.reference, {
   664→   664→          lineId: line.id, // §156 — safe per-line metadata
   665→   665→          purchaseOrderLineId: poLine.id,
   666→   666→          productId: poLine.productId,
   667→   667→          productNameAtReceipt: poLine.productNameAtOrder,
   668→   668→          receivedQty: input.receivedQty,
   669→   669→        }),
   670→   670→        tx,
   671→   671→      );
   672→   672→
   673→   673→      return { id: line.id };
   674→   674→    } catch (error) {
   675→   675→      // §27/§209 — DB uniqueness is authoritative for duplicate lines.
   676→   676→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   677→   677→        throw new ApiError(
   678→   678→          409,
   679→   679→          GOODS_RECEIPT_ERROR_CODES.IMMUTABLE,
   680→   680→          "This purchase order line is already on the receipt.",
   681→   681→        );
   682→   682→      }
   683→   683→      throw error;
   684→   684→    }
   685→   685→  });
   686→   686→}

   800→   800→/* ---------------------------------- post ----------------------------------- */
   801→   801→
   802→   802→/**
   803→   803→ * §39 — THE atomic stock posting transaction. Steps 1-17 in ONE
   804→   804→ * database transaction (§72); a failure at ANY step rolls back EVERYTHING
   805→   805→ * (§40/§214) — no stock change, no adjustment, no status, no audit.
   806→   806→ */
   807→   807→export async function postGoodsReceipt(input: {
   808→   808→  businessId: string;
   809→   809→  userId: string;
   810→   810→  id: string;
   811→   811→  ip?: string | null;
   812→   812→  userAgent?: string | null;
   813→   813→}) {
   814→   814→  return withWriteTx(async (tx) => {
   815→   815→    // §39 steps 1-2 / §73 — reload receipt INSIDE the transaction; never
   816→   816→    // trust an earlier GET. Corrupted status is surfaced, never Draft.
   817→   817→    const header = await loadReceiptHeader(tx, input.businessId, input.id);
   818→   818→    assertTransition(header.status, "POSTED"); // §7 — only DRAFT → POSTED
   819→   819→
   820→   820→    // §39 step 3 — PO must STILL be CONFIRMED (§149/§150: a PO cancelled
   821→   821→    // or closed after the draft was created can never be posted).
   822→   822→    const po = await tx.purchaseOrder.findFirst({
   823→   823→      where: { id: header.purchaseOrderId, businessId: input.businessId },
   824→   824→      select: { id: true, reference: true, status: true, supplierNameAtOrder: true },
   825→   825→    });
   826→   826→    if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
   827→   827→    assertPurchaseOrderReceivable(po.status);
   828→   828→
   829→   829→    // §39 step 4 — load the receipt's own lines (bounded by the document
   830→   830→    // cap, §177 — a receipt can never exceed GOODS_RECEIPT_MAX_LINES).
   831→   831→    const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
   832→   832→    const lines = await tx.goodsReceiptLine.findMany({
   833→   833→      where: { goodsReceiptId: header.id },
   834→   834→      orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   835→   835→      take: cap + 1, // cap+1 proves completeness (same law as P41 export)
   836→   836→      select: {
   837→   837→        id: true,
   838→   838→        purchaseOrderLineId: true,
   839→   839→        productId: true,
   840→   840→        receivedQty: true,
   841→   841→        purchaseOrderLine: {
   842→   842→          select: {
   843→   843→            id: true,
   844→   844→            orderedQty: true,
   845→   845→            purchaseOrderId: true,
   846→   846→          },
   847→   847→        },
   848→   848→      },
   849→   849→    });
   850→   850→    if (lines.length > cap) {
   851→   851→      // Impossible by §20/§71 — if it ever happens, fail honestly.
   852→   852→      throw new ApiError(
   853→   853→        422,
   854→   854→        GOODS_RECEIPT_ERROR_CODES.LINE_LIMIT,
   855→   855→        "Goods receipt line limit exceeded.",
   856→   856→      );
   857→   857→    }
   858→   858→
   859→   859→    // §70 — a receipt with zero lines can never post.
   860→   860→    if (lines.length === 0) {
   861→   861→      throw new ApiError(
   862→   862→        422,
   863→   863→        GOODS_RECEIPT_ERROR_CODES.EMPTY,
   864→   864→        "Add at least one line before posting this receipt.",
   865→   865→      );
   866→   866→    }
   867→   867→
   868→   868→    const adjustmentIds: string[] = [];
   869→   869→    const stockChanges: { productId: string; from: number; to: number }[] = [];
   870→   870→
   871→   871→    for (const line of lines) {
   872→   872→      // §39 step 4 — the PO line must belong to THIS purchase order
   873→   873→      // (defensive re-verification of the document chain).
   874→   874→      if (line.purchaseOrderLine.purchaseOrderId !== po.id) {
   875→   875→        throw new ApiError(404, "NOT_FOUND", "Purchase order line not found.");
   876→   876→      }
   877→   877→
   878→   878→      // §39 steps 5-8 — product existence + authoritative remaining check.
   879→   879→      const product = await tx.product.findFirst({
   880→   880→        where: { id: line.productId, businessId: input.businessId }, // §30/§35
   881→   881→        select: { id: true, stockQty: true },
   882→   882→      });
   883→   883→      if (!product) {
   884→   884→        // §33/§152 — a hard-deleted product fails the post safely.
   885→   885→        throw new ApiError(
   886→   886→          422,
   887→   887→          GOODS_RECEIPT_ERROR_CODES.PRODUCT_NOT_FOUND,
   888→   888→          "A product on this receipt no longer exists.",
   889→   889→        );
   890→   890→      }
   891→   891→
   892→   892→      const alreadyPosted = await sumPostedQtyForPoLine(tx, line.purchaseOrderLineId);
   893→   893→      const remaining = computeRemainingQty(
   894→   894→        line.purchaseOrderLine.orderedQty,
   895→   895→        alreadyPosted,
   896→   896→      ); // §100 — throws on corrupted over-received history (§283)
   897→   897→      if (exceedsRemaining(line.purchaseOrderLine.orderedQty, alreadyPosted, line.receivedQty)) {
   898→   898→        // §15/§16/§67 — the HARD over-receipt law, final authority.
   899→   899→        throw new ApiError(
   900→   900→          409,
   901→   901→          GOODS_RECEIPT_ERROR_CODES.EXCEEDS_REMAINING,
   902→   902→          `Only ${remaining} unit(s) remain to receive on this line.`,
   903→   903→        );
   904→   904→      }
   905→   905→
   906→   906→      // §46/§47 — corrupt persisted stock is surfaced, never repaired.
   907→   907→      assertPersistedStockValid(product.stockQty);
   908→   908→
   909→   909→      // §49 — Int32 storage bound; overflow → 422, no partial writes.
   910→   910→      if (stockOverflowsInt32(product.stockQty, line.receivedQty)) {
   911→   911→        throw new ApiError(
   912→   912→          422,
   913→   913→          GOODS_RECEIPT_ERROR_CODES.STOCK_OVERFLOW,
   914→   914→          "Receiving this quantity would exceed the stock storage limit.",
   915→   915→        );
   916→   916→      }
   917→   917→
   918→   918→      // §39 steps 9-11 — exact integer arithmetic, positive delta only.
   919→   919→      const newStock = product.stockQty + line.receivedQty;
   920→   920→      await tx.product.update({
   921→   921→        where: { id: product.id },
   922→   922→        data: { stockQty: newStock },
   923→   923→        select: { id: true },
   924→   924→      });
   925→   925→      stockChanges.push({ productId: product.id, from: product.stockQty, to: newStock });
   926→   926→
   927→   927→      // §39 step 12 / §41-§45 — one positive InventoryAdjustment per line.
   928→   928→      const adjustment = await tx.inventoryAdjustment.create({
   929→   929→        data: {
   930→   930→          businessId: input.businessId, // §43 — server-derived
   931→   931→          productId: product.id, // §44 — derived from the PO line chain
   932→   932→          delta: line.receivedQty, // §45 — always positive
   933→   933→          reason: RECEIPT_ADJUSTMENT_REASON(header.reference), // §41 — stable identifier
   934→   934→          actorId: input.userId, // §42 — server-derived
   935→   935→        },
   936→   936→        select: { id: true },
   937→   937→      });
   938→   938→      adjustmentIds.push(adjustment.id);
   939→   939→    }
   940→   940→
   941→   941→    // §39 steps 14-15 — complete the posting (server timestamps only §74).
   942→   942→    const postedAt = new Date();
   943→   943→    const updated = await tx.goodsReceipt.update({
   944→   944→      where: { id: header.id },
   945→   945→      data: { status: "POSTED", postedAt },
   946→   946→      select: { id: true, reference: true, status: true, postedAt: true },
   947→   947→    });
   948→   948→
   949→   949→    // §39 step 16 / §153/§154 — ONE audit record for the posting, written
   950→   950→    // in the SAME transaction; if it fails, everything above rolls back.
   951→   951→    const totalReceivedQty = computeReceivedQty(lines.map((l) => l.receivedQty));
   952→   952→    await audit(
   953→   953→      auditInput(input.businessId, input.userId, "goods_receipt.post", header.id, header.reference, {
   954→   954→        purchaseOrderId: po.id,
   955→   955→        purchaseOrderReferenceAtReceipt: header.purchaseOrderReferenceAtReceipt,
   956→   956→        lineCount: lines.length,
   957→   957→        totalReceivedQty, // §155 — quantity facts only; never notes
   958→   958→        adjustmentIds,
   959→   959→        previousStatus: header.status,
   960→   960→        newStatus: updated.status,
   961→   961→        postedAt: updated.postedAt?.toISOString() ?? null,
   962→   962→      }),
   963→   963→      tx,
   964→   964→    );
   965→   965→
   966→   966→    return {
   967→   967→      id: updated.id,
   968→   968→      reference: updated.reference,
   969→   969→      status: updated.status,
   970→   970→      postedAt: updated.postedAt,
   971→   971→      lineCount: lines.length,
   972→   972→      totalReceivedQty,
   973→   973→      stockChanges,
   974→   974→    };
   975→   975→  });
   976→   976→}

  1026→  1026→export async function getPurchaseOrderReceivingSummary(input: {
  1027→  1027→  businessId: string;
  1028→  1028→  purchaseOrderId: string;
  1029→  1029→}): Promise<{
  1030→  1030→  orderedQty: number;
  1031→  1031→  receivedQty: number;
  1032→  1032→  remainingQty: number;
  1033→  1033→  receiptCount: number;
  1034→  1034→  fullyReceived: boolean;
  1035→  1035→  lines: { lineId: string; orderedQty: number; receivedQty: number; remainingQty: number }[];
  1036→  1036→}> {
  1037→  1037→  const po = await db.purchaseOrder.findFirst({
  1038→  1038→    where: { id: input.purchaseOrderId, businessId: input.businessId }, // §89
  1039→  1039→    select: { id: true },
  1040→  1040→  });
  1041→  1041→  if (!po) throw new ApiError(404, "NOT_FOUND", "Purchase order not found.");
  1042→  1042→
  1043→  1043→  // §296 — one transactional read of lines + posted aggregates.
  1044→  1044→  const cap = goodsReceiptLineCap(process.env.GOODS_RECEIPT_MAX_LINES);
  1045→  1045→  return db.$transaction(async (tx) => {
  1046→  1046→    const poLines = await tx.purchaseOrderLine.findMany({
  1047→  1047→      where: { purchaseOrderId: po.id },
  1048→  1048→      select: { id: true, orderedQty: true },
  1049→  1049→      take: cap, // bounded by the PO's own line cap (P41 law)
  1050→  1050→    });
  1051→  1051→
  1052→  1052→    // §178 — sum POSTED receipt lines per PO line, grouped in one query.
  1053→  1053→    const grouped = await tx.goodsReceiptLine.groupBy({
  1054→  1054→      by: ["purchaseOrderLineId"],
  1055→  1055→      where: {
  1056→  1056→        purchaseOrderLineId: { in: poLines.map((l) => l.id) },
  1057→  1057→        goodsReceipt: { status: "POSTED" }, // §99 — POSTED only
  1058→  1058→      },
  1059→  1059→      _sum: { receivedQty: true },
  1060→  1060→    });
  1061→  1061→    const receivedByLine = new Map(
  1062→  1062→      grouped.map((g) => [g.purchaseOrderLineId, g._sum.receivedQty ?? 0]),
  1063→  1063→    );
  1064→  1064→
  1065→  1065→    // §104-analog — receipt COUNT (POSTED receipts touching this PO).
  1066→  1066→    const receiptCount = await tx.goodsReceipt.count({
  1067→  1067→      where: { purchaseOrderId: po.id, status: "POSTED" },
  1068→  1068→    });
  1069→  1069→
  1070→  1070→    let orderedQty = 0;
  1071→  1071→    let receivedQty = 0;
  1072→  1072→    const lines = poLines.map((l) => {
  1073→  1073→      const lineReceived = receivedByLine.get(l.id) ?? 0;
  1074→  1074→      const lineRemaining = computeRemainingQty(l.orderedQty, lineReceived); // §100
  1075→  1075→      orderedQty += l.orderedQty;
  1076→  1076→      receivedQty += lineReceived;
  1077→  1077→      return {
  1078→  1078→        lineId: l.id,
  1079→  1079→        orderedQty: l.orderedQty,
  1080→  1080→        receivedQty: lineReceived,
  1081→  1081→        remainingQty: lineRemaining,
  1082→  1082→      };
  1083→  1083→    });
  1084→  1084→    const remainingQty = computeRemainingQty(orderedQty, receivedQty); // §100
  1085→  1085→
  1086→  1086→    return {
  1087→  1087→      orderedQty,
  1088→  1088→      receivedQty,
  1089→  1089→      remainingQty,
  1090→  1090→      receiptCount,
  1091→  1091→      fullyReceived: remainingQty === 0, // §101 — derived, never stored
  1092→  1092→      lines,
  1093→  1093→    };
  1094→  1094→  }).catch((error) => mapPureGoodsReceiptDomainError(error));
  1095→  1095→}
