     3→/**
     4→ * Export dataset catalog — Prompt 31 §5/§6/§41-§54.
     5→ *
     6→ * LAWS:
     7→ *  - EXPLICIT allowlist: a dataset key that is not in this catalog can
     8→ *    never be exported (§39/§239 — unknown dataset rejected).
     9→ *  - Every dataset is classified (§6): PUBLIC_BUSINESS / CONFIDENTIAL /
    10→ *    HIGHLY_SENSITIVE. No dataset below contains authentication or
    11→ *    security secrets (§9/§12-§19 — enforced again by the static scanner).
    12→ *  - Tenant-owned only: fetchers filter by businessId unconditionally
    13→ *    (§32); the caller's ctx.businessId is the ONLY business key used.
    14→ *  - Per-row businessId is deliberately EXCLUDED (§10 — the manifest
    15→ *    carries the business identity; per-row copies are unnecessary).
    16→ *  - Financial export is READ-ONLY representation of the same records the
    17→ *    UI already shows (§222/§223) — amounts as exact integer minor units
    18→ *    in JSON, exact decimal strings in CSV (§226/§227).
    19→ *  - NOT exported (explicit Prompt 31 decisions, documented):
    20→ *      · AI chat transcripts (user-personal, out of portability scope §53)
    21→ *      · tenant audit trail (actor identities + history, not in catalog §5)
    22→ *      · receivables/payables (DERIVED views of sales/expenses — the same
    23→ *        facts are already inside those datasets)
    24→ *      · anything platform: subscriptions internals, provider costs, AI
    25→ *        rates, platform audit, operators (§9/§51/§52/§189).
    26→ */

   103→export type ExportRow = {
   104→  /** JSON row — IDs + relationships preserved, money as minor integers (§54/§226). */
   105→  json: Record<string, unknown>;
   106→  /** CSV row — exact decimal strings for money, ISO-8601 UTC dates (§227/§231). */
   107→  csv: Array<string | number | null>;
   108→};
   109→
   110→export type DatasetFetchResult = {
   111→  rows: ExportRow[];
   112→  /** True when the dataset hit the export record cap (job must FAIL, §26). */
   113→  truncated: boolean;
   114→};

   116→export type DatasetFetchCtx = {
   117→  businessId: string;
   118→  /** Business decimal precision for money → CSV decimal strings. */
   119→  decimalPrecision: number;

   121→  maxRecords: number;

   124→function iso(value: Date | null | undefined): string | null {
   125→  return value ? value.toISOString() : null;
   126→}

   132→type Customer = Prisma.CustomerGetPayload<object>;

   134→function customerRow(c: Customer, precision: number): ExportRow {
   135→  void precision; // customers carry no money fields
   136→  return {
   137→    json: {
   138→      id: c.id,
   139→      name: c.name,
   140→      phone: c.phone,
   141→      email: c.email,
   142→      address: c.address,
   143→      note: c.note,
   144→      deletedAt: iso(c.deletedAt),
   145→      createdAt: c.createdAt.toISOString(),
   146→      updatedAt: c.updatedAt.toISOString(),
   147→    },
   148→    csv: [c.id, c.name, c.phone, c.email, c.address, c.note, iso(c.deletedAt), c.createdAt.toISOString(), c.updatedAt.toISOString()],
   149→  };
   150→}

   170→function productRow(p: Prisma.ProductGetPayload<object>, precision: number): ExportRow {
   171→  return {
   172→    json: {
   173→      id: p.id,
   174→      name: p.name,
   175→      sku: p.sku,
   176→      unit: p.unit,
   177→      salePriceMinor: p.salePriceMinor,
   178→      costPriceMinor: p.costPriceMinor,
   179→      stockQty: p.stockQty,
   180→      lowStockAt: p.lowStockAt,
   181→      isActive: p.isActive,
   182→      deletedAt: iso(p.deletedAt),
   183→      createdAt: p.createdAt.toISOString(),
   184→      updatedAt: p.updatedAt.toISOString(),
   185→    },
   186→    csv: [
   187→      p.id,
   188→      p.name,
   189→      p.sku,
   190→      p.unit,
   191→      formatMinorToDecimal(p.salePriceMinor, precision),
   192→      formatMinorToDecimal(p.costPriceMinor, precision),
   193→      p.stockQty,
   194→      p.lowStockAt,
   195→      p.isActive ? "true" : "false",
   196→      iso(p.deletedAt),
   197→      p.createdAt.toISOString(),
   198→      p.updatedAt.toISOString(),
   199→    ],
   200→  };
   201→}
   202→
   203→async function fetchWithCap<T>(
   204→  cap: number,
   205→  query: (take: number) => Promise<T[]>,
   206→): Promise<{ items: T[]; truncated: boolean }> {
   207→  // Fetch one row beyond the cap so truncation is DETECTABLE, never silent.
   208→  const items = await query(cap + 1);
   209→  return { items: items.slice(0, cap), truncated: items.length > cap };
   210→}
