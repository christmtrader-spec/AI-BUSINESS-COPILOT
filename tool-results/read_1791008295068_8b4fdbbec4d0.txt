    69→function jarFetch(jar) {
    70→  return async (path, init = {}) => {
    71→    const cookie = [...jar.entries()].map(([k, v]) => `${k}=${v}`).join("; ");
    72→    const headers = { "x-forwarded-for": randomXff(), ...(init.headers ?? {}) };
    73→    if (init.body && !headers["content-type"]) headers["content-type"] = "application/json";
    74→    const res = await fetch(`${BASE}${path}`, { ...init, headers: cookie ? { ...headers, cookie } : headers });
    75→    for (const c of res.headers.getSetCookie?.() ?? []) {
    76→      const [pair] = c.split(";");
    77→      const idx = pair.indexOf("=");
    78→      if (idx > 0) jar.set(pair.slice(0, idx).trim(), pair.slice(idx + 1));
    79→    }
    80→    return res;
    81→  };
    82→}

   121→/** Runs the REAL operator CLI. Returns parsed {result} or null on failure. */
   122→function runProcessor(flags = []) {
   123→  const proc = Bun.spawnSync({
   124→    cmd: ["bun", "scripts/process-crm-task-reminders.mjs", ...flags],
   125→    cwd: process.cwd(),
   126→    stdout: "pipe",
   127→    stderr: "pipe",
   128→    env: { ...process.env },
   129→  });
   130→  const out = proc.stdout.toString();
   131→  const lines = out.split("\n").filter((l) => l.trim().startsWith("{"));
   132→  let parsed = null;
   133→  for (const line of lines) {
   134→    try {
   135→      const obj = JSON.parse(line);
   136→      if (obj.result) parsed = obj;
   137→    } catch {}
   138→  }
   139→  if (proc.exitCode !== 0) {
   140→    console.log(`  ⚠ CLI exited ${proc.exitCode}: ${proc.stderr.toString().slice(0, 300)}`);
   141→  }
   142→  console.log(`  ↳ CLI ${flags.join(" ")} → ${JSON.stringify(parsed?.result ?? null)}`);
   143→  return { exitCode: proc.exitCode, parsed, out, err: proc.stderr.toString() };
   144→}
