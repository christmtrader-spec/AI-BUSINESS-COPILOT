    96→// ---- Markdown-lite renderer (no deps, no raw HTML — safe by construction) --

    98→function inline(text: string, keyPrefix: string): ReactNode[] {
    99→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
   100→  return parts.map((part, i) => {
   101→    const key = `${keyPrefix}-${i}`;
   102→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   103→      return (
   104→        <strong key={key} className="font-semibold">
   105→          {part.slice(2, -2)}
   106→        </strong>
   107→      );
   108→    }
   109→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   110→      return (
   111→        <code
   112→          key={key}
   113→          dir="ltr"
   114→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   115→        >
   116→          {part.slice(1, -1)}
   117→        </code>
   118→      );
   119→    }
   120→    return <span key={key}>{part}</span>;
   121→  });
   122→}

   124→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   125→  const nodes: ReactNode[] = [];
   126→  let bullets: string[] = [];
   127→  let para: string[] = [];
   128→
   129→  const flushBullets = () => {
   130→    if (bullets.length === 0) return;
   131→    nodes.push(
   132→      <ul
   133→        key={`ul-${nodes.length}`}
   134→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   135→      >
   136→        {bullets.map((b, i) => (
   137→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   138→        ))}
   139→      </ul>,
   140→    );
   141→    bullets = [];
   142→  };
   143→  const flushPara = () => {
   144→    if (para.length === 0) return;
   145→    nodes.push(
   146→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   147→        {inline(para.join("\n"), `p${nodes.length}`)}
   148→      </p>,
   149→    );
   150→    para = [];
   151→  };
   152→
   153→  for (const raw of text.split("\n")) {
   154→    const line = raw.trimEnd();
   155→    if (/^\s*[-*•]\s+/.test(line)) {
   156→      flushPara();
   157→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   158→    } else if (line === "") {
   159→      flushBullets();
   160→      flushPara();
   161→    } else {
   162→      flushBullets();
   163→      para.push(line);
   164→    }
   165→  }
   166→  flushBullets();
   167→  flushPara();
   168→
   169→  return (
   170→    <div className="space-y-1.5 text-sm">
   171→      {nodes}
   172→      {streaming ? (
   173→        <span
   174→          aria-hidden="true"
   175→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   176→        />
   177→      ) : null}
   178→    </div>
   179→  );
   180→}

   272→// ---- Follow-up suggestions ---------------------------------------------------
   273→
   274→/**
   275→ * Context-aware follow-up pool. The chips are UI suggestions only — they
   276→ * never touch data. Deterministic: keyword match on the answer text picks
   277→ * the lead chip; the rest rotate by conversation length so repeated
   278→ * answers surface different prompts.
   279→ */
   280→const FOLLOW_UP_KEYS = [
   281→  "followUpOutstanding",
   282→  "followUpLowStock",
   283→  "followUpExpenses",
   284→  "followUpSales",
   285→  "followUpCash",
   286→  "followUpProducts",
   287→  "followUpPayments",
   288→  "followUpSummary",
   289→] as const;
   290→
   291→type FollowUpKey = (typeof FOLLOW_UP_KEYS)[number];
   292→
   293→const FOLLOW_UP_MATCHERS: Array<{ key: FollowUpKey; re: RegExp }> = [
   294→  { key: "followUpOutstanding", re: /owe|outstanding|receivable|بدهکار|بدهی|طلب/i },
   295→  { key: "followUpLowStock", re: /low.?stock|restock|inventory|موجودی|انبار|شارژ/i },
   296→  { key: "followUpExpenses", re: /expense|spend|cost|هزینه|خرج/i },
   297→  { key: "followUpSales", re: /sale|revenue|invoice|فروش|درآمد|فاکتور/i },
   298→  { key: "followUpCash", re: /cash|payment|collected|نقد|پرداخت|دریافتی/i },
   299→  { key: "followUpProducts", re: /product|best.?seller|top|محصول|پرفروش/i },
   300→];
   301→
   302→function pickFollowUps(answer: string, conversationLength: number): FollowUpKey[] {
   303→  const lead = FOLLOW_UP_MATCHERS.find((m) => m.re.test(answer))?.key;
   304→  const rest = FOLLOW_UP_KEYS.filter((k) => k !== lead);
   305→  const rotate = conversationLength % rest.length;
   306→  const rotated = [...rest.slice(rotate), ...rest.slice(0, rotate)];
   307→  const picked = rotated.slice(0, lead ? 2 : 3);
   308→  return lead ? [lead, ...picked] : picked;
   309→}

   313→export function CopilotView({
   314→  business,
   315→  role,
   316→}: {
   317→  business: BusinessSummary;
   318→  role: Role;
   319→}) {
   320→  const t = useTranslations("copilot");
   321→  const tCommon = useTranslations("common");
   322→  const tErrors = useTranslations("errors");
   323→
   324→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   325→  const [input, setInput] = useState("");
   326→  const [activeStreamId, setActiveStreamId] = useState<string | null>(null);
   327→  const [days, setDays] = useState<7 | 30 | 90>(30);
   328→  const [copiedId, setCopiedId] = useState<string | null>(null);
   329→  const [dataOpen, setDataOpen] = useState(false);
   330→  const [atBottom, setAtBottom] = useState(true);
   331→  const lastQuestionRef = useRef<string | null>(null);
   332→  const abortRef = useRef<AbortController | null>(null);
   333→  const scrollRef = useRef<HTMLDivElement>(null);
   334→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   335→
   336→  const suggestions = useMemo(
   337→    () => [
   338→      t("suggestion1"),
   339→      t("suggestion2"),
   340→      t("suggestion3"),
   341→      t("suggestion4"),
   342→      t("suggestion5"),
   343→      t("suggestion6"),
   344→    ],
   345→    [t],
   346→  );
   347→
   348→  const canUse = can(role, "copilot:use");
   349→  const pending = activeStreamId !== null;
   350→
   351→  // Follow-up chips — only under the LAST completed answer.
   352→  const lastMessage = messages[messages.length - 1];
   353→  const showFollowUps =
   354→    !pending &&
   355→    messages.length >= 2 &&
   356→    lastMessage?.role === "assistant" &&
   357→    lastMessage.failed !== true &&
   358→    lastMessage.content.trim().length > 0;
   359→  const followUps = useMemo(
   360→    () => (showFollowUps ? pickFollowUps(lastMessage.content, messages.length) : []),
   361→    [showFollowUps, lastMessage?.content, messages.length],
   362→  );
   363→
   364→  // Keep the latest message in view (scroll-only effect — no setState).
   365→  useEffect(() => {
   366→    if (atBottom && scrollRef.current) {
   367→      scrollRef.current.scrollTo({
   368→        top: scrollRef.current.scrollHeight,
   369→        behavior: "smooth",
   370→      });
   371→    }
   372→  }, [messages, pending, atBottom]);
   373→
   374→  // Abort any in-flight stream when the module unmounts.
   375→  useEffect(() => {
   376→    return () => abortRef.current?.abort();
   377→  }, []);
   378→
   379→  const patchAssistant = useCallback(
   380→    (id: string, patch: (m: ChatMessage) => ChatMessage) => {
   381→      setMessages((prev) => prev.map((m) => (m.id === id ? patch(m) : m)));
   382→    },
   383→    [],
   384→  );
   385→
   386→  const localizedError = useCallback(
   387→    (code: string, retryAfterMs?: number) => {
   388→      switch (code) {
   389→        case "RATE_LIMITED":
   390→          return t("errorRateLimited", {
   391→            seconds: String(Math.max(1, Math.ceil((retryAfterMs ?? 0) / 1000))),
   392→          });
   393→        case "PERMISSION_DENIED":
   394→          return t("errorForbidden");
   395→        case "AI_UNAVAILABLE":
   396→          return t("errorAiUnavailable");
   397→        default:
   398→          return tErrors("generic");
   399→      }
   400→    },
   401→    [t, tErrors],
   402→  );
   403→
   404→  const send = useCallback(
   405→    async (question: string, history: ChatMessage[]) => {
   406→      const trimmed = question.trim();
   407→      if (!trimmed || trimmed.length > 500 || activeStreamId !== null) return;
   408→      lastQuestionRef.current = trimmed;
   409→      setAtBottom(true);
   410→
   411→      const assistantId = nextId();
   412→      setMessages((prev) => [
   413→        ...prev,
   414→        { id: assistantId, role: "assistant", content: "" },
   415→      ]);
   416→      setActiveStreamId(assistantId);
   417→
   418→      const controller = new AbortController();
   419→      abortRef.current = controller;
   420→
   421→      try {
   422→        const res = await fetch("/api/v1/copilot/stream", {
   423→          method: "POST",
   424→          credentials: "same-origin",
   425→          headers: {
   426→            "content-type": "application/json",
   427→            "x-business-id": business.id,
   428→          },
   429→          body: JSON.stringify({
   430→            question: trimmed,
   431→            history: history
   432→              .filter((m) => !m.failed)
   433→              .slice(-6)
   434→              .map((m) => ({ role: m.role, content: m.content })),
   435→            days,
   436→          }),
   437→          signal: controller.signal,
   438→        });
   439→
   440→        if (!res.ok || !res.body) {
   441→          let code = "INTERNAL_ERROR";
   442→          let retryAfterMs: number | undefined;
   443→          try {
   444→            const body = (await res.json()) as {
   445→              error?: { code?: string; details?: { retryAfterMs?: number } };
   446→            };
   447→            if (body?.error?.code) code = body.error.code;
   448→            retryAfterMs = body?.error?.details?.retryAfterMs;
   449→          } catch {
   450→            // keep defaults
   451→          }
   452→          patchAssistant(assistantId, (m) => ({
   453→            ...m,
   454→            content: localizedError(code, retryAfterMs),
   455→            failed: true,
   456→          }));
   457→          return;
   458→        }
   459→
   460→        // Parse the SSE stream incrementally.
   461→        const reader = res.body.getReader();
   462→        const decoder = new TextDecoder();
   463→        let buffer = "";
   464→        let streamError: { code?: string; message?: string } | null = null;
   465→        let sawDone = false;
   466→
   467→        while (true) {
   468→          const { done, value } = await reader.read();
   469→          if (done) break;
   470→          buffer += decoder.decode(value, { stream: true });
   471→          const events = buffer.split("\n\n");
   472→          buffer = events.pop() ?? "";
   473→          for (const evt of events) {
   474→            for (const line of evt.split("\n")) {
   475→              if (!line.startsWith("data: ")) continue;
   476→              let payload: {
   477→                meta?: CopilotMeta;
   478→                delta?: string;
   479→                done?: boolean;
   480→                error?: { code?: string; message?: string };
   481→              };
   482→              try {
   483→                payload = JSON.parse(line.slice(6));
   484→              } catch {
   485→                continue;
   486→              }
   487→              if (payload.meta) {
   488→                // First event — dataset provenance for this answer.
   489→                const meta = payload.meta;
   490→                patchAssistant(assistantId, (m) => ({ ...m, meta }));
   491→              } else if (typeof payload.delta === "string" && payload.delta.length > 0) {
   492→                const delta = payload.delta;
   493→                patchAssistant(assistantId, (m) => ({
   494→                  ...m,
   495→                  content: m.content + delta,
   496→                }));
   497→              } else if (payload.error) {
   498→                streamError = payload.error;
   499→              } else if (payload.done) {
   500→                sawDone = true;
   501→              }
   502→            }
   503→          }
   504→        }
   505→
   506→        if (streamError) {
   507→          const code = streamError.code ?? "INTERNAL_ERROR";
   508→          if (code === "RATE_LIMITED" || code === "AI_UNAVAILABLE" || !sawDone) {
   509→            patchAssistant(assistantId, (m) =>
   510→              m.content.trim().length === 0
   511→                ? { ...m, content: localizedError(code), failed: true }
   512→                : { ...m, content: `${m.content}\n\n⚠ ${localizedError(code)}` },
   513→            );
   514→          }
   515→        }
   516→      } catch (error) {
   517→        if ((error as Error)?.name === "AbortError") {
   518→          // User pressed Stop — keep the partial answer, drop it if empty.
   519→          setMessages((prev) =>
   520→            prev.flatMap((m) =>
   521→              m.id === assistantId ? (m.content.trim() ? [m] : []) : [m],
   522→            ),
   523→          );
   524→        } else if (error instanceof ApiClientError) {
   525→          patchAssistant(assistantId, (m) => ({
   526→            ...m,
   527→            content: localizedError(error.code),
   528→            failed: true,
   529→          }));
   530→        } else {
   531→          patchAssistant(assistantId, (m) => ({
   532→            ...m,
   533→            content: localizedError("NETWORK_ERROR"),
   534→            failed: true,
   535→          }));
   536→        }
   537→      } finally {
   538→        abortRef.current = null;
   539→        setActiveStreamId((cur) => (cur === assistantId ? null : cur));
   540→      }
   541→    },
   542→    [activeStreamId, business.id, days, localizedError, patchAssistant],
   543→  );
   544→
   545→  const handleSubmit = () => {
   546→    const trimmed = input.trim();
   547→    if (!trimmed || pending) return;
   548→    setInput("");
   549→    if (textareaRef.current) textareaRef.current.style.height = "auto";
   550→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: trimmed }]);
   551→    void send(trimmed, messages);
   552→  };
   553→
   554→  const handleStop = () => {
   555→    abortRef.current?.abort();
   556→  };
   557→
   558→  const handleRetry = () => {
   559→    const q = lastQuestionRef.current;
   560→    if (!q || pending) return;
   561→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: q }]);
   562→    void send(q, messages);
   563→  };
   564→
   565→  const copyAnswer = async (m: ChatMessage) => {
   566→    try {
   567→      await navigator.clipboard.writeText(m.content);
   568→      setCopiedId(m.id);
   569→      window.setTimeout(() => setCopiedId((c) => (c === m.id ? null : c)), 1500);
   570→    } catch {
   571→      /* clipboard unavailable — silently ignore */
   572→    }
   573→  };
   574→
   575→  /** Download the Q&A transcript as a shareable Markdown file. */
   576→  const exportTranscript = () => {
   577→    if (messages.length === 0 || pending) return;
   578→    try {
   579→      const localeTag = business.locale === "fa" ? "fa-IR" : "en-GB";
   580→      const nowLabel = new Intl.DateTimeFormat(localeTag, {
   581→        dateStyle: "long",
   582→        timeStyle: "short",
   583→      }).format(new Date());
   584→      const windowLabel = t(`window_${days}` as "window_7");
   585→
   586→      const lines: string[] = [];
   587→      lines.push(`# ${t("title")} — ${t("exportTranscript")}`);
   588→      lines.push("");
   589→      lines.push(`- **${business.name}**`);
   590→      lines.push(`- ${t("windowLabel")}: ${windowLabel}`);
   591→      lines.push(`- ${t("citationsGenerated", { time: nowLabel })}`);
   592→      lines.push(`- ${t("groundedNote")}`);
   593→      lines.push("");
   594→      lines.push("---");
   595→
   596→      let qIndex = 0;

   626→            }).format(new Date(meta.generatedAt));

   629→            /* timestamp is decorative in the export */

   638→      if (qIndex === 0) return; // nothing answerable to export
   639→
