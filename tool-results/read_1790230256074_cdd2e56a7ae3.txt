    63→function formatDate(iso: string | null, locale: string): string {
    64→  if (!iso) return "—";
    65→  try {
    66→    return new Intl.DateTimeFormat(locale === "fa" ? "fa-IR" : locale, {
    67→      dateStyle: "medium",
    68→      timeStyle: "short",
    69→    }).format(new Date(iso));
    70→  } catch {
    71→    return iso.slice(0, 10);
    72→  }
    73→}

    75→function StatusBadge({ status }: { status: string }) {
    76→  const t = useTranslations("billing");
    77→  const tone =
    78→    status === "ACTIVE"
    79→      ? "border-emerald-500/30 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400"
    80→      : status === "TRIALING"
    81→        ? "border-amber-500/30 bg-amber-500/15 text-amber-700 dark:text-amber-400"
    82→        : status === "CANCELED"
    83→          ? "border-orange-500/30 bg-orange-500/15 text-orange-700 dark:text-orange-400"
    84→          : status === "PAST_DUE" || status === "PAUSED"
    85→            ? "border-yellow-500/30 bg-yellow-500/15 text-yellow-700 dark:text-yellow-400"
    86→            : "text-muted-foreground border-border bg-muted"; // EXPIRED
    87→  return (
    88→    <Badge variant="outline" className={cn("font-medium", tone)} data-testid="billing-status">
    89→      {t(`status_${status}` as "status_ACTIVE")}
    90→    </Badge>
    91→  );
    92→}

    94→function usagePercent(used: number, limit: number | null): number | null {
    95→  if (limit === null || limit <= 0) return null;
    96→  return Math.min(100, Math.round((used / limit) * 100));
    97→}

   104→function UsageMetricRow({
   105→  metric,
   106→  m,
   107→  periodEnd,
   108→  locale,
   109→}: {
   110→  metric: string;
   111→  m: UsageMetricPayload;
   112→  periodEnd: string | null;
   113→  locale: string;
   114→}) {
   115→  const tUsage = useTranslations("billing.usage");
   116→  const pct = m.percentageUsed;
   117→  const tone =
   118→    m.level === "reached"
   119→      ? "border-destructive/40 bg-destructive/10 text-destructive"
   120→      : m.level === "critical"
   121→        ? "border-orange-500/40 bg-orange-500/10 text-orange-700 dark:text-orange-400"
   122→        : m.level === "warning"
   123→          ? "border-amber-500/40 bg-amber-500/10 text-amber-700 dark:text-amber-400"
   124→          : null;
   125→  const barTone =
   126→    m.level === "reached"
   127→      ? "[&>div]:bg-destructive"
   128→      : m.level === "critical"
   129→        ? "[&>div]:bg-orange-500"
   130→        : m.level === "warning"
   131→          ? "[&>div]:bg-amber-500"
   132→          : "";
   133→  const badge =
   134→    m.level === "reached"
   135→      ? tUsage("reachedBadge")
   136→      : m.level === "critical"
   137→        ? tUsage("criticalBadge")
   138→        : m.level === "warning"
   139→          ? tUsage("warningBadge")
   140→          : null;
   141→  return (
   142→    <div className="space-y-1.5" data-testid={`usage-${metric}`}>
   143→      <div className="flex flex-wrap items-center justify-between gap-2 text-sm">
   144→        <span className="flex items-center gap-2">
   145→          {tUsage(`metric_${metric}` as "metric_AI_REQUESTS")}
   146→          {badge ? (
   147→            <Badge variant="outline" className={cn("font-medium", tone)}>
   148→              {badge}
   149→            </Badge>
   150→          ) : null}
   151→        </span>
   152→        {m.limit === null ? (
   153→          <span className="text-muted-foreground font-medium">
   154→            {tUsage("usedOnly", { used: m.used })} · {tUsage("unlimitedMetric")}
   155→          </span>
   156→        ) : (
   157→          <span className="text-muted-foreground font-medium">
   158→            {tUsage("usedOf", { used: m.used, limit: m.limit })} ·{" "}
   159→            <span>
   160→              {m.remaining === 0
   161→                ? tUsage("remainingNone")
   162→                : tUsage("remaining", { count: m.remaining ?? 0 })}
   163→            </span>
   164→          </span>
   165→        )}
   166→      </div>
   167→      {pct !== null ? (
   168→        <Progress
   169→          value={pct}
   170→          aria-label={tUsage(`metric_${metric}` as "metric_AI_REQUESTS")}
   171→          className={cn("h-1.5", barTone)}
   172→        />
   173→      ) : null}
   174→      {m.level === "reached" ? (
   175→        <p className="text-destructive flex items-start gap-1.5 text-xs">
   176→          <Info className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   177→          {periodEnd
   178→            ? tUsage("reachedNotice", { date: formatDate(periodEnd, locale) })
   179→            : tUsage("reachedNoticeNoDate")}
   180→        </p>
   181→      ) : null}
   182→    </div>
   183→  );
   184→}
