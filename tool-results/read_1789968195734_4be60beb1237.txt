   117→   117→function inline(text: string, keyPrefix: string): ReactNode[] {
   118→   118→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
   119→   119→  return parts.map((part, i) => {
   120→   120→    const key = `${keyPrefix}-${i}`;
   121→   121→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   122→   122→      return (
   123→   123→        <strong key={key} className="font-semibold">
   124→   124→          {part.slice(2, -2)}
   125→   125→        </strong>
   126→   126→      );
   127→   127→    }
   128→   128→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   129→   129→      return (
   130→   130→        <code
   131→   131→          key={key}
   132→   132→          dir="ltr"
   133→   133→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   134→   134→        >
   135→   135→          {part.slice(1, -1)}
   136→   136→        </code>
   137→   137→      );
   138→   138→    }
   139→   139→    return <span key={key}>{part}</span>;
   140→   140→  });
   141→   141→}

   143→   143→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   144→   144→  const nodes: ReactNode[] = [];
   145→   145→  let bullets: string[] = [];
   146→   146→  let para: string[] = [];
   147→   147→
   148→   148→  const flushBullets = () => {
   149→   149→    if (bullets.length === 0) return;
   150→   150→    nodes.push(
   151→   151→      <ul
   152→   152→        key={`ul-${nodes.length}`}
   153→   153→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   154→   154→      >
   155→   155→        {bullets.map((b, i) => (
   156→   156→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   157→   157→        ))}
   158→   158→      </ul>,
   159→   159→    );
   160→   160→    bullets = [];
   161→   161→  };
   162→   162→  const flushPara = () => {
   163→   163→    if (para.length === 0) return;
   164→   164→    nodes.push(
   165→   165→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   166→   166→        {inline(para.join("\n"), `p${nodes.length}`)}
   167→   167→      </p>,
   168→   168→    );
   169→   169→    para = [];
   170→   170→  };
   171→   171→
   172→   172→  for (const raw of text.split("\n")) {
   173→   173→    const line = raw.trimEnd();
   174→   174→    if (/^\s*[-*•]\s+/.test(line)) {
   175→   175→      flushPara();
   176→   176→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   177→   177→    } else if (line === "") {
   178→   178→      flushBullets();
   179→   179→      flushPara();
   180→   180→    } else {
   181→   181→      flushBullets();
   182→   182→      para.push(line);
   183→   183→    }
   184→   184→  }
   185→   185→  flushBullets();
   186→   186→  flushPara();
   187→   187→
   188→   188→  return (
   189→   189→    <div className="space-y-1.5 text-sm">
   190→   190→      {nodes}
   191→   191→      {streaming ? (
   192→   192→        <span
   193→   193→          aria-hidden="true"
   194→   194→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   195→   195→        />
   196→   196→      ) : null}
   197→   197→    </div>
   198→   198→  );
   199→   199→}

   293→   293→/**

   311→   311→
   312→   312→const FOLLOW_UP_MATCHERS: Array<{ key: FollowUpKey; re: RegExp }> = [
   313→   313→  { key: "followUpOutstanding", re: /owe|outstanding|receivable|بدهکار|بدهی|طلب/i },
   314→   314→  { key: "followUpLowStock", re: /low.?stock|restock|inventory|موجودی|انبار|شارژ/i },
   315→   315→  { key: "followUpExpenses", re: /expense|spend|cost|هزینه|خرج/i },
   316→   316→  { key: "followUpSales", re: /sale|revenue|invoice|فروش|درآمد|فاکتور/i },
   317→   317→  { key: "followUpCash", re: /cash|payment|collected|نقد|پرداخت|دریافتی/i },
   318→   318→  { key: "followUpProducts", re: /product|best.?seller|top|محصول|پرفروش/i },
   319→   319→];
   320→   320→
   321→   321→/** Live server counts (same deterministic aggregates as the dashboard)
   322→   322→ *  used to RANK suggestions — a chip is only proposed when its subject
   323→   323→ *  actually exists in the data (r10: "only propose restock when there is
   324→   324→ *  something to restock"). */
   325→   325→type SummaryLite = {
   326→   326→  kpis?: { salesCount?: number; expensesCount?: number };
   327→   327→  outstanding?: { receivableCount?: number };
   328→   328→  inventory?: { lowStockCount?: number };
   329→   329→  topProducts?: unknown[];
   330→   330→};
   331→   331→
   332→   332→function followUpRelevant(
   333→   333→  key: FollowUpKey,
   334→   334→  counts: {
   335→   335→    receivableCount: number;
   336→   336→    lowStockCount: number;
   337→   337→    salesCount: number;
   338→   338→    expensesCount: number;
   339→   339→    topProducts: number;
   340→   340→  } | null,
   341→   341→): boolean {
   342→   342→  if (!counts) return true; // counts not loaded yet — keep the neutral default
   343→   343→  switch (key) {
   344→   344→    case "followUpOutstanding":
   345→   345→      return counts.receivableCount > 0;
   346→   346→    case "followUpLowStock":
   347→   347→      return counts.lowStockCount > 0;
   348→   348→    case "followUpExpenses":
   349→   349→      return counts.expensesCount > 0;
   350→   350→    case "followUpProducts":
   351→   351→      return counts.topProducts > 0;
   352→   352→    case "followUpSales":
   353→   353→      return true;
   354→   354→    case "followUpCash":
   355→   355→    case "followUpPayments":
   356→   356→    case "followUpSummary":
   357→   357→      return true;
   358→   358→  }
   359→   359→}
   360→   360→
   361→   361→function pickFollowUps(
   362→   362→  answer: string,
   363→   363→  conversationLength: number,
   364→   364→  counts: Parameters<typeof followUpRelevant>[1] = null,
   365→   365→): FollowUpKey[] {
   366→   366→  const lead = FOLLOW_UP_MATCHERS.find((m) => m.re.test(answer))?.key;
   367→   367→  const pool = FOLLOW_UP_KEYS.filter(
   368→   368→    (k) => k !== lead && followUpRelevant(k, counts),
   369→   369→  );
   370→   370→  const rotate = conversationLength % Math.max(1, pool.length);
   371→   371→  const rotated = [...pool.slice(rotate), ...pool.slice(0, rotate)];
   372→   372→  const picked = rotated.slice(0, lead ? 2 : 3);
   373→   373→  return lead ? [lead, ...picked] : picked;
   374→   374→}
