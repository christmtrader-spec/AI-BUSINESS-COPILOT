   122→function errText(err: unknown, fallback: string): string {
   123→  if (err instanceof ApiClientError) {
   124→    const code = err.code;
   125→    if (code === "INVALID_CURRENT_PASSWORD") return "errors.invalidPassword";
   126→    if (code === "MFA_INVALID_CODE") return "errors.invalidCode";
   127→    if (code === "MFA_CODE_REQUIRED") return "errors.mfaCodeRequired";
   128→    if (code === "MFA_REAUTH_REQUIRED" || code === "MFA_DISABLE_REAUTH_REQUIRED") return "errors.reauthRequired";
   129→    if (code === "RATE_LIMITED" || code === "MFA_RATE_LIMITED" || code === "EMAIL_VERIFICATION_RATE_LIMITED")
   130→      return "errors.rateLimited";
   131→    if (code === "SESSION_NOT_FOUND") return "errors.sessionNotFound";
   132→    if (code === "SESSION_CURRENT") return "errors.sessionCurrent";
   133→    if (code === "EMAIL_VERIFICATION_REQUIRED") return "errors.emailVerificationRequired";
   134→  }
   135→  return fallback;
   136→}

   138→function formatDateTime(iso: string, locale: string): string {
   139→  try {
   140→    return new Intl.DateTimeFormat(locale === "fa" ? "fa-IR" : "en-US", {
   141→      dateStyle: "medium",
   142→      timeStyle: "short",
   143→    }).format(new Date(iso));
   144→  } catch {
   145→    return iso;
   146→  }
   147→}

   481→function EmailSection({
   482→  summary,
   483→  onChanged,
   484→}: {
   485→  summary: SecuritySummary;
   486→  onChanged: () => void;
   487→}) {
   488→  const t = useTranslations("security");
   489→  const [pending, setPending] = useState(false);
   490→
   491→  const resend = useMutation({
   492→    mutationFn: async () =>
   493→      api<{ sent: boolean; alreadyVerified: boolean }>("/api/v1/auth/email-verification/resend", {
   494→        method: "POST",
   495→        businessId: null,
   496→      }),
   497→    onMutate: () => setPending(true),
   498→    onSettled: () => setPending(false),
   499→    onSuccess: (res) => {
   500→      if (res.alreadyVerified) {
   501→        toast({ title: t("emailAlreadyVerified") });
   502→      } else if (res.sent) {
   503→        toast({ title: t("emailResendSent") });
   504→        onChanged();
   505→      } else {
   506→        toast({ title: t("emailResendFailed"), variant: "destructive" });
   507→      }
   508→    },
   509→    onError: (err) => {
   510→      toast({
   511→        title: t(errText(err, "emailResendFailed")),
   512→        variant: "destructive",
   513→      });
   514→    },
   515→  });
   516→
   517→  const modeOff = summary.emailVerificationMode === "OFF";
   518→
   519→  return (
   520→    <CardContent className="space-y-3">
   521→      <div className="flex flex-wrap items-center gap-2">
   522→        {summary.emailVerified ? (
   523→          <Badge className="gap-1" variant="default">
   524→            <BadgeCheck className="h-3.5 w-3.5" aria-hidden />
   525→            {t("emailVerified")}
   526→          </Badge>
   527→        ) : (
   528→          <Badge className="gap-1" variant="outline">
   529→            <CircleAlert className="h-3.5 w-3.5" aria-hidden />
   530→            {t("emailNotVerified")}
   531→          </Badge>
   532→        )}
   533→        {!summary.emailVerified && !modeOff ? (
   534→          <Button
   535→            variant="outline"
   536→            size="sm"
   537→            disabled={pending}
   538→            onClick={() => resend.mutate()}
   539→          >
   540→            {pending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   541→            {t("emailResendAction")}
   542→          </Button>
   543→        ) : null}
   544→      </div>
   545→      {modeOff && !summary.emailVerified ? (
   546→        <p className="text-xs text-muted-foreground">{t("emailModeOff")}</p>
   547→      ) : null}
   548→    </CardContent>
   549→  );
   550→}

   554→function ReauthDialog({
   555→  open,
   556→  onOpenChange,
   557→  mfaEnabled,
   558→  onSuccess,
   559→}: {
   560→  open: boolean;
   561→  onOpenChange: (open: boolean) => void;
   562→  mfaEnabled: boolean;
   563→  onSuccess: () => void;
   564→}) {
   565→  const t = useTranslations("security");
   566→  const [password, setPassword] = useState("");
   567→  const [mfaCode, setMfaCode] = useState("");
   568→  const [error, setError] = useState<string | null>(null);
   569→
   570→  const reauth = useMutation({
   571→    mutationFn: async () =>
   572→      api<{ passwordVerified: boolean; mfaVerified: boolean; windowSeconds: number }>(
   573→        "/api/v1/me/security/reauth",
   574→        { method: "POST", body: JSON.stringify({ password, ...(mfaCode ? { mfaCode } : {}) }), businessId: null },
   575→      ),
   576→    onSuccess: () => {
