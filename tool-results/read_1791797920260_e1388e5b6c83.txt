    73→let passed = 0;
    74→let failed = 0;
    75→const failures: string[] = [];
    76→
    77→function ok(cond: boolean, name: string) {
    78→  if (cond) passed++;
    79→  else {
    80→    failed++;
    81→    failures.push(name);
    82→  }
    83→  console.log(`${cond ? "✓" : "✗"} ${name}`);
    84→}
    85→function eq(a: unknown, b: unknown, name: string) {
    86→  const same = a === b;
    87→  if (same) passed++;
    88→  else {
    89→    failed++;
    90→    failures.push(`${name} (got ${String(a)}, want ${String(b)})`);
    91→  }
    92→  console.log(`${same ? "✓" : "✗"} ${name}`);
    93→}
    94→function throws(fn: () => unknown, name: string, errorName?: string) {
    95→  try {
    96→    fn();
    97→    ok(false, name);
    98→  } catch (e) {
    99→    const matches = errorName
   100→      ? (e as Error).name === errorName || String((e as Error).message).includes(errorName)
   101→      : true;
   102→    ok(matches, name);
   103→  }
   104→}
