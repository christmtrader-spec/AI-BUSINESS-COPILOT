    17→    17→type AuthErrors = {
    18→    18→  invalidCredentials: string;
    19→    19→  rateLimited: string;
    20→    20→  emailTaken: string;
    21→    21→  passwordMismatch: string;

    28→    28→/**
    29→    29→ * Server 422 envelopes carry the zod issues in `details`. When the failed
    30→    30→ * field is the password we can give the user the REAL reason instead of a
    31→    31→ * generic "try again" (BUGFIX P51-R1: signup showed a useless generic error
    32→    32→ * when the password policy rejected the submitted password).
    33→    33→ */
    34→    34→function hasValidationPath(details: unknown, field: string): boolean {
    35→    35→  if (!Array.isArray(details)) return false;
    36→    36→  return details.some(
    37→    37→    (issue) =>
    38→    38→      issue !== null &&
    39→    39→      typeof issue === "object" &&
    40→    40→      Array.isArray((issue as { path?: unknown }).path) &&
    41→    41→      (issue as { path: unknown[] }).path[0] === field,
    42→    42→  );
    43→    43→}
    44→    44→
    45→    45→function mapSignInError(error: string | null | undefined, t: AuthErrors): string {
    46→    46→  if (error === "CredentialsSignin") return t.invalidCredentials;
    47→    47→  return t.generic;
    48→    48→}
    49→    49→
    50→    50→function mapApiError(err: unknown, t: AuthErrors): string {
    51→    51→  if (err instanceof ApiClientError) {
    52→    52→    if (err.status === 0) return t.network;
    53→    53→    if (err.code === "EMAIL_TAKEN") return t.emailTaken;
    54→    54→    if (err.code === "RATE_LIMITED") return t.rateLimited;
    55→    55→    if (err.code === "VALIDATION_ERROR") {
    56→    56→      if (hasValidationPath(err.details, "password")) return t.passwordPolicy;
    57→    57→      return t.invalidData;
    58→    58→    }
    59→    59→  }
    60→    60→  return t.generic;
    61→    61→}

   189→   189→/**
   190→   190→ * "Reset password" flow — consumes a single-use token from the deep link
   191→   191→ * (#/reset-password/<token>). On success every previous session is dead:
   192→   192→ * any stale local session is cleared immediately and the user is sent to
   193→   193→ * sign-in via the success card button.

   195→   195→function ResetPasswordCard({
   196→   196→  token,
   197→   197→  onGoToSignIn,
   198→   198→  onRequestNewLink,
   199→   199→}: {
   200→   200→  token: string;
   201→   201→  onGoToSignIn: () => void;
   202→   202→  onRequestNewLink: () => void;
   203→   203→}) {
   204→   204→  const t = useTranslations("auth");
   205→   205→  const queryClient = useQueryClient();
   206→   206→  const [newPassword, setNewPassword] = useState("");
   207→   207→  const [confirmPassword, setConfirmPassword] = useState("");
   208→   208→  const [error, setError] = useState<string | null>(null);
   209→   209→  const [tokenInvalid, setTokenInvalid] = useState(false);
   210→   210→  const [doneMessage, setDoneMessage] = useState<string | null>(null);
   211→   211→
   212→   212→  const resetMutation = useMutation({
   213→   213→    mutationFn: async () =>
   214→   214→      api<{ message?: string }>("/api/v1/auth/password-reset/confirm", {
   215→   215→        method: "POST",
   216→   216→        body: JSON.stringify({ token, newPassword, confirmPassword }),
   217→   217→      }),
   218→   218→    onSuccess: () => {
   219→   219→      setError(null);
   220→   220→      setTokenInvalid(false);
   221→   221→      setDoneMessage(t("resetSuccessBody"));

   223→   223→      // local one right away (the success card below routes to sign-in).

   229→   229→        if (err.code === "INVALID_RESET_TOKEN") {
   230→   230→          setTokenInvalid(true);
   231→   231→          setError(null);
   232→   232→        } else if (err.status === 0) {
   233→   233→          setTokenInvalid(false);
   234→   234→          setError(t("errors.network"));
   235→   235→        } else if (err.code === "RATE_LIMITED") {
   236→   236→          setTokenInvalid(false);
   237→   237→          setError(t("errors.rateLimited"));
   238→   238→        } else {
   239→   239→          setTokenInvalid(false);
   240→   240→          setError(t("errors.generic"));
   241→   241→        }
   242→   242→      } else {
   243→   243→        setTokenInvalid(false);
   244→   244→        setError(t("errors.generic"));
   245→   245→      }
   246→   246→    },
   247→   247→  });

   251→   251→      <CardHeader className="space-y-1.5">
   252→   252→        <CardTitle className="text-2xl">
   253→   253→          {doneMessage ? t("resetSuccessTitle") : t("resetTitle")}
   254→   254→        </CardTitle>
   255→   255→        {doneMessage ? null : <CardDescription>{t("resetSubtitle")}</CardDescription>}
   256→   256→      </CardHeader>
   257→   257→      <CardContent>
   258→   258→        {doneMessage ? (
   259→   259→          <div className="space-y-5 py-2 text-center">
   260→   260→            <div
   261→   261→              className="bg-primary/10 mx-auto flex size-12 items-center justify-center rounded-full"
   262→   262→              aria-hidden="true"
   263→   263→            >
   264→   264→              <KeyRound className="text-primary size-6" />

   266→   266→            <p className="text-foreground text-sm" role="status">
   267→   267→              {doneMessage}
   268→   268→            </p>
   269→   269→            <Button type="button" className="w-full" onClick={onGoToSignIn}>
   270→   270→              <LogIn className="size-4" aria-hidden="true" />
   271→   271→              {t("goToSignIn")}
   272→   272→            </Button>
   273→   273→          </div>
   274→   274→        ) : (
   275→   275→          <form
   276→   276→            onSubmit={(e) => {
   277→   277→              e.preventDefault();
   278→   278→              setError(null);
   279→   279→              setTokenInvalid(false);
   280→   280→              if (newPassword !== confirmPassword) {
   281→   281→                setError(t("errors.passwordMismatch"));
   282→   282→                return;
   283→   283→              }
   284→   284→              resetMutation.mutate();
   285→   285→            }}
   286→   286→            className="space-y-4"
   287→   287→            noValidate
   288→   288→          >
   289→   289→            <div className="space-y-2">
   290→   290→              <Label htmlFor="reset-password">{t("newPassword")}</Label>
   291→   291→              <Input
   292→   292→                id="reset-password"
   293→   293→                type="password"
   294→   294→                autoComplete="new-password"
   295→   295→                dir="ltr"
   296→   296→                placeholder={t("passwordPlaceholder")}
   297→   297→                value={newPassword}
   298→   298→                onChange={(e) => setNewPassword(e.target.value)}
   299→   299→                required
   300→   300→                minLength={8}
   301→   301→              />
   302→   302→              <p className="text-muted-foreground text-xs">{t("passwordHint")}</p>

   314→   314→                required

   318→   318→            {tokenInvalid ? (
   319→   319→              <div className={errorBoxClass} role="alert">
   320→   320→                <p>{t("errors.invalidResetToken")}</p>
   321→   321→                <button
   322→   322→                  type="button"

   324→   324→                  onClick={onRequestNewLink}
   325→   325→                >
   326→   326→                  {t("sendResetLink")}
   327→   327→                </button>
   328→   328→              </div>
   329→   329→            ) : error ? (
   330→   330→              <p className={errorBoxClass} role="alert">
   331→   331→                {error}

   342→   342→            </Button>
   343→   343→          </form>
   344→   344→        )}
   345→   345→      </CardContent>

   350→   350→export function AuthView() {
   351→   351→  const t = useTranslations("auth");
   352→   352→  const tCommon = useTranslations("common");
   353→   353→  const view = useAppStore((s) => s.view);
   354→   354→  const navigate = useAppStore((s) => s.navigate);
   355→   355→  // next-intl's t is a function — nested property access (t.errors) is
   356→   356→  // undefined at runtime, which crashed the onError handlers. Build the
   357→   357→  // message map with explicit t() calls instead (BUGFIX r8).
   358→   358→  const authErrorMessages: AuthErrors = {
   359→   359→    invalidCredentials: t("errors.invalidCredentials"),
   360→   360→    rateLimited: t("errors.rateLimited"),
   361→   361→    emailTaken: t("errors.emailTaken"),
   362→   362→    passwordMismatch: t("errors.passwordMismatch"),
   363→   363→    passwordPolicy: t("errors.passwordPolicy"),
   364→   364→    invalidData: t("errors.invalidData"),
   365→   365→    network: t("errors.network"),
   366→   366→    generic: t("errors.generic"),
   367→   367→  };
   368→   368→  const { status } = useSession();
   369→   369→  const queryClient = useQueryClient();
   370→   370→  const [mode, setMode] = useState<"signin" | "signup">("signin");
   371→   371→  const [name, setName] = useState("");
   372→   372→  const [email, setEmail] = useState("");
   373→   373→  const [password, setPassword] = useState("");
   374→   374→  const [confirmPassword, setConfirmPassword] = useState("");
   375→   375→  const [error, setError] = useState<string | null>(null);
   376→   376→
   377→   377→  // Prompt 30 §46/§109 — the MFA login step machine. Password success with
   378→   378→  // MFA enabled NEVER creates a session (authorize throws the fixed
   379→   379→  // MFA_REQUIRED code); the client then obtains a SHORT-LIVED challenge,
   380→   380→  // collects the TOTP/recovery code, and exchanges it for a ONE-TIME
   381→   381→  // assertion that completes the real NextAuth login.
   382→   382→  const [mfaStage, setMfaStage] = useState<"idle" | "challenge" | "code">("idle");
   383→   383→  const [mfaChallenge, setMfaChallenge] = useState<string | null>(null);
   384→   384→  const [mfaMethod, setMfaMethod] = useState<"TOTP" | "RECOVERY_CODE">("TOTP");
   385→   385→  const [mfaCode, setMfaCode] = useState("");
   386→   386→  // Prompt 27 §53/§58: explicit legal acceptance — UNCHECKED by default, with
   387→   387→  // links to the actual documents BEFORE acceptance. Required by the signup UI.
   388→   388→  const [acceptedLegal, setAcceptedLegal] = useState(false);
   389→   389→  const tLegal = useTranslations("legal.auth");
   390→   390→
   391→   391→  const signInMutation = useMutation({
   392→   392→    mutationFn: async () => {
   393→   393→      const res = await signIn("credentials", { redirect: false, email, password });
   394→   394→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   395→   395→      return res;
   396→   396→    },
   397→   397→    onSuccess: () => {
   398→   398→      setError(null);
   399→   399→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   400→   400→      void queryClient.invalidateQueries({ queryKey: ["me"] });
   401→   401→    },
   402→   402→    onError: (err) => {
   403→   403→      // Prompt 30 §110 — MFA_REQUIRED arrives ONLY after the password was
   404→   404→      // verified inside authorize(); switch to the bounded MFA step.
   405→   405→      if (err.message === "MFA_REQUIRED") {
   406→   406→        void startMfaChallenge();
   407→   407→        return;
   408→   408→      }
   409→   409→      setError(
   410→   410→        err instanceof ApiClientError
   411→   411→          ? mapApiError(err, authErrorMessages)
   412→   412→          : mapSignInError(err.message, authErrorMessages),
   413→   413→      );
   414→   414→    },
   415→   415→  });
   416→   416→
   417→   417→  // Step 2 of the MFA flow (§46): re-verify the password at the dedicated
   418→   418→  // endpoint to obtain the challenge token (rate-limited, hash-only server
   419→   419→  // state, bound to the user's sessionVersion).
   420→   420→  const startMfaChallenge = () => {
   421→   421→    setMfaStage("challenge");
   422→   422→    mfaChallengeMutation.mutate();
   423→   423→  };
   424→   424→
   425→   425→  const mfaChallengeMutation = useMutation({
   426→   426→    mutationFn: async () =>
   427→   427→      api<{ mfaRequired: boolean; challenge: string; expiresAt: string }>(
   428→   428→        "/api/v1/auth/mfa/challenge",
   429→   429→        { method: "POST", body: JSON.stringify({ email, password }), businessId: null },
   430→   430→      ),
   431→   431→    onSuccess: (res) => {
   432→   432→      setMfaChallenge(res.challenge);
   433→   433→      setMfaStage("code");
   434→   434→      setMfaCode("");
   435→   435→      setError(null);
   436→   436→    },
   437→   437→    onError: (err) => {
   438→   438→      setMfaStage("idle");
   439→   439→      if (err instanceof ApiClientError) {
   440→   440→        if (err.code === "MFA_NOT_ENABLED") {
   441→   441→          // The account has no MFA (state changed mid-flow) — retry plain.
   442→   442→          signInMutation.mutate();
   443→   443→          return;
   444→   444→        }
   445→   445→        if (err.code === "RATE_LIMITED") {
   446→   446→          setError(t("mfaRateLimited"));
   447→   447→          return;
   448→   448→        }
   449→   449→        if (err.code === "INVALID_CREDENTIALS") {
   450→   450→          setError(authErrorMessages.invalidCredentials);
   451→   451→          return;
   452→   452→        }
   453→   453→      }
   454→   454→      setError(authErrorMessages.generic);
   455→   455→    },
   456→   456→  });
   457→   457→
   458→   458→  // Step 3 (§51): exchange challenge + code for the one-time assertion,
   459→   459→  // then complete the REAL NextAuth sign-in with it (§52 — the final JWT
   460→   460→  // is still issued by NextAuth only).
   461→   461→  const mfaCompleteMutation = useMutation({
   462→   462→    mutationFn: async () => {
   463→   463→      const assertion = await api<{ assertion: string }>("/api/v1/auth/mfa/complete", {
   464→   464→        method: "POST",
   465→   465→        body: JSON.stringify({ challenge: mfaChallenge, code: mfaCode, method: mfaMethod }),
