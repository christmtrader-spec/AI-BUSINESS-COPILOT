     3→/**
     4→ * Prompt 44 — goods-dispatch service (server-only).
     5→ *
     6→ * LAWS:
     7→ *  - PHYSICAL INVENTORY OUTFLOW ONLY: posting DECREASES Product.stockQty
     8→ *    and creates ONE negative InventoryAdjustment per line (delta = -qty).
     9→ *    It NEVER creates Sale, Expense, Payment, receivable, credit note,
    10→ *    statement movement or any accounting entry. Draft/cancel mutate
    11→ *    NOTHING but the dispatch document.
    12→ *  - PRODUCT IS THE SOURCE: an outbound dispatch is not chained to a
    13→ *    purchase-order document. The tenant-scoped Product row and its
    14→ *    CURRENT aggregate stock are the quantity authority at POST time (no
    15→ *    lot/batch provenance — the same documented aggregate-stock trade-off
    16→ *    as P40/P42/P43).
    17→ *  - CURRENT STOCK CHECK: each posted line requires Product.stockQty ≥
    18→ *    qty; the aggregate stock can NEVER go negative — no override, no
    19→ *    silent zero-clamp (exact integer math via the ONE pure helper).
    20→ *  - Every query is tenant-scoped by ctx.businessId — never a client
    21→ *    businessId. Foreign product/customer/dispatch/dispatch-line → 404,
    22→ *    no existence leak.
    23→ *  - Status transitions are server-authoritative through dedicated
    24→ *    operations; the current status is RE-CHECKED INSIDE the write
    25→ *    transaction so concurrent posts/cancels cannot double-apply.
    26→ *  - Snapshots (customer name, product name/sku/unit) are frozen at write
    27→ *    time and NEVER overwritten from live data.
    28→ *  - Quantities are exact integers; corrupt persisted state is surfaced,
    29→ *    never repaired (safe-integer laws inherited from P43).
    30→ *  - READ-CAP LAW: every bounded read fetches cap+1 rows to PROVE
    31→ *    completeness — a document holding more rows than its own cap (or a
    32→ *    `_count` mismatch) fails the read with the stable
    33→ *    GOODS_DISPATCH_LINE_CAP_INVARIANT code. Read paths NEVER render a
    34→ *    silently truncated slice or a partial total.
    35→ *  - AGGREGATE LAW: POSTED dispatched quantities are summed by the
    36→ *    DATABASE (aggregate/groupBy _sum), never by loading dispatch history
    37→ *    into application memory; the aggregate result is re-validated.
    38→ *  - Audit through the single sanctioned helper INSIDE the same
    39→ *    transaction as every mutation — safe metadata only, never notes
    40→ *    content.
    41→ */

   100→/** Corrupted persisted status → controlled 500 (never Draft). */
   101→function parseDbStatus(value: string): GoodsDispatchStatus {
   102→  try {
   103→    return parseGoodsDispatchStatus(value);
   104→  } catch (error) {
   105→    if (error instanceof Error && error.name === "GoodsDispatchStatusIntegrityError") {
   106→      throw new ApiError(
   107→        500,
   108→        GOODS_DISPATCH_INVALID_PERSISTED_STATUS,
   109→        "Goods dispatch data is in an invalid state. Please contact support.",
   110→      );
   111→    }
   112→    throw error;
   113→  }
   114→}

   116→/**
   117→ * Map pure canonical-math integrity failures to their controlled HTTP
   118→ * result: corrupted dispatch history / corrupt stock is a 500, never a
   119→ * silent clamp. Exported for cross-service mapping.
   120→ */
   121→export function mapPureGoodsDispatchDomainError(error: unknown): never {
   122→  if (error instanceof Error && error.name === "GoodsDispatchStatusIntegrityError") {
   123→    throw new ApiError(
   124→      500,
   125→      GOODS_DISPATCH_INVALID_PERSISTED_STATUS,
   126→      "Goods dispatch data is in an invalid state. Please contact support.",
   127→    );
   128→  }
   129→  if (error instanceof Error && error.name === "GoodsDispatchIntegrityError") {
   130→    // The line-set completeness invariant (over-cap document or a
   131→    // counted/fetched mismatch) is ITS OWN stable integrity code — never
   132→    // misreported as generic history corruption.
   133→    if (error.message.includes(GOODS_DISPATCH_LINE_CAP_INVARIANT)) {
   134→      throw new ApiError(
   135→        500,
   136→        GOODS_DISPATCH_LINE_CAP_INVARIANT,
   137→        "Stored document data exceeds its configured line cap. Please contact support.",
   138→      );
   139→    }
   140→    const corruptStock = error.message.includes("GOODS_DISPATCH_INVALID_PERSISTED_STOCK");
   141→    throw new ApiError(
   142→      500,
   143→      corruptStock
   144→        ? GOODS_DISPATCH_ERROR_CODES.INVALID_PERSISTED_STOCK
   145→        : GOODS_DISPATCH_ERROR_CODES.CORRUPT_HISTORY,
   146→      corruptStock
   147→        ? "Product stock data is in an invalid state. Please contact support."
   148→        : "Goods dispatch data is inconsistent. Please contact support.",
   149→    );
   150→  }
   151→  throw error;
   152→}

   235→function auditInput(
   236→  businessId: string,
   237→  actorId: string,
   238→  action: string,
   239→  goodsDispatchId: string,
   240→  reference: string,
   241→  after: Record<string, unknown>,

   247→    action,
   248→    entityType: "GoodsDispatch",

   255→/**
   256→ * The ONE canonical POSTED dispatched-quantity sum for ONE product.
   257→ * DRAFT/CANCELLED dispatches never count. Database aggregate (_sum),
   258→ * never a findMany of history into memory; the result is re-validated
   259→ * (exact integer, ≥ 0) and corruption surfaces through the ONE mapping,
   260→ * never clamped.
   261→ */
   262→export async function sumPostedDispatchedQtyForProduct(
   263→  tx: DbLike,
   264→  productId: string,
   265→): Promise<number> {
   266→  const agg = await tx.goodsDispatchLine.aggregate({
   267→    where: {
   268→      productId,
   269→      goodsDispatch: { status: "POSTED" }, // POSTED only
   270→    },
   271→    _sum: { qty: true },
   272→  });
   273→  return validateDispatchedSum(agg._sum.qty ?? 0);
   274→}
   275→
   276→/** Aggregate result re-validation (SAFE integer, ≥ 0, no clamp). */
   277→function validateDispatchedSum(sum: number): number {
   278→  if (!Number.isSafeInteger(sum) || sum < 0) {
   279→    throw mapPureGoodsDispatchDomainError(
   280→      new Error("GOODS_DISPATCH_CORRUPT_HISTORY: negative or unsafe aggregate"),
   281→    );
   282→  }
   283→  return sum;
   284→}

   286→/** Defensive line-count validation (add line). */
   287→async function assertLineCap(tx: Tx, goodsDispatchId: string, cap: number): Promise<number> {
   288→  const lineCount = await tx.goodsDispatchLine.count({ where: { goodsDispatchId } });
   289→  if (lineCount > cap) {
   290→    // An ALREADY-corrupted document (more lines than its own cap) is
   291→    // database corruption: 500 integrity, never disguised as the ordinary
   292→    // 422 user validation failure.
   293→    throw new ApiError(
   294→      500,
   295→      GOODS_DISPATCH_LINE_CAP_INVARIANT,
   296→      "Stored document data exceeds its configured line cap. Please contact support.",
   297→    );
   298→  }
   299→  if (lineCount === cap) {
   300→    // The NORMAL write-time rejection for a legitimately full draft.
   301→    throw new ApiError(
   302→      422,
   303→      GOODS_DISPATCH_ERROR_CODES.LINE_LIMIT,
   304→      `A goods dispatch can contain at most ${cap} lines.`,
   305→    );
   306→  }
   307→  return lineCount;
   308→}

   312→export async function createGoodsDispatch(input: {
   313→  businessId: string;
   314→  userId: string;
   315→  customerId?: string;
   316→  reference?: string;
   317→  notes?: string | null;
   318→  ip?: string | null;
   319→  userAgent?: string | null;
   320→}) {
   321→  return withWriteTx(async (tx) => {
   322→    // Optional customer link: tenant-scoped, not soft-deleted; the name
   323→    // snapshot is frozen at creation (deletion/renames never rewrite it).
   324→    let customerNameAtDispatch = "";
   325→    if (input.customerId) {
   326→      customerNameAtDispatch = await loadActiveCustomerName(
   327→        tx,
   328→        input.businessId,
   329→        input.customerId,
   330→      );
   331→    }
   332→
   333→    // Reference: user-supplied (normalized) or server-generated.
   334→    let reference = input.reference;
   335→    if (reference === undefined) {
   336→      for (let attempt = 0; attempt < 5; attempt += 1) {
   337→        const candidate = generateGoodsDispatchReference(new Date(), (max) => randomInt(max));
   338→        const clash = await tx.goodsDispatch.findFirst({
   339→          where: { businessId: input.businessId, reference: candidate },
   340→          select: { id: true },
   341→        });
   342→        if (!clash) {
   343→          reference = candidate;
   344→          break;
   345→        }
   346→      }
   347→      if (reference === undefined) {
   348→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   349→        throw new ApiError(
   350→          409,
   351→          GOODS_DISPATCH_ERROR_CODES.DUPLICATE_REFERENCE,
   352→          "Could not generate a unique reference. Please try again.",
   353→        );
   354→      }
   355→    }
   356→
   357→    try {
   358→      const row = await tx.goodsDispatch.create({
   359→        data: {
   360→          businessId: input.businessId, // server-derived, never client
   361→          customerId: input.customerId ?? null,
   362→          customerNameAtDispatch, // frozen snapshot ("" when no customer)
   363→          reference,
   364→          status: "DRAFT", // preparation only
   365→          notes: input.notes ?? null, // document data only
   366→          createdByUserId: input.userId, // server-derived
   367→        },
   368→        select: { id: true, reference: true, status: true },
   369→      });
   370→
   371→      await audit(
   372→        auditInput(input.businessId, input.userId, "goods_dispatch.create", row.id, row.reference, {
   373→          customerId: input.customerId ?? null,
   374→          customerNameAtDispatch,
   375→          status: row.status,
   376→        }),
   377→        tx,
   378→      );
   379→
   380→      return row;
   381→    } catch (error) {
   382→      // Database uniqueness stays authoritative for duplicates.
   383→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   384→        throw new ApiError(
   385→          409,
   386→          GOODS_DISPATCH_ERROR_CODES.DUPLICATE_REFERENCE,
   387→          "A goods dispatch with this reference already exists.",
   388→        );
   389→      }
   390→      throw error;
   391→    }
   392→  });
   393→}

   397→export async function listGoodsDispatches(input: {
   398→  businessId: string;
   399→  q?: string;
   400→  status?: string; // ALL → undefined
   401→  customerId?: string;
   402→  from?: string;
   403→  to?: string;
   404→  timezone?: string; // AUTHENTICATED business timezone (never client authority)
   405→  page: number;
   406→  pageSize: number;
   407→}): Promise<{
   408→  items: GoodsDispatchListItemDto[];
   409→  total: number;
   410→  page: number;
   411→  pageSize: number;
   412→}> {
   413→  const cap = goodsDispatchLineCap(process.env.GOODS_DISPATCH_MAX_LINES);
   414→  const where: Prisma.GoodsDispatchWhereInput = {
   415→    businessId: input.businessId, // tenant scope
   416→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   417→    ...(input.customerId ? { customerId: input.customerId } : {}),
   418→    ...(input.from || input.to
   419→      ? (() => {
   420→          // The from/to calendar dates are interpreted in the
   421→          // AUTHENTICATED BUSINESS's timezone (inclusive local start of
   422→          // `from`, EXCLUSIVE local start of the day after `to`). A record
   423→          // at 00:15 local belongs to its local date even when its UTC
   424→          // date is the previous day. Unresolvable bounds → controlled 422.
   425→          const window = goodsDispatchDayWindow(input.from, input.to, input.timezone);
   426→          if (!window) {
   427→            throw new ApiError(422, "VALIDATION_ERROR", "Invalid date filter.");
   428→          }
   429→          return { createdAt: window };
   430→        })()
   431→      : {}),
   432→    ...(input.q
   433→      ? {
   434→          OR: [
   435→            { reference: { contains: input.q } }, // dispatch reference
   436→            { customerNameAtDispatch: { contains: input.q } }, // customer snapshot
   437→            {
   438→              lines: {
   439→                some: {
   440→                  OR: [
   441→                    { productNameAtDispatch: { contains: input.q } }, // product snapshot
   442→                    { skuAtDispatch: { contains: input.q } }, // SKU snapshot
   443→                  ],
   444→                },
   445→              },
   446→            },
   447→          ],
   448→        }
   449→      : {}),
   450→  };
   451→
   452→  const [total, rows] = await Promise.all([
   453→    db.goodsDispatch.count({ where }),
   454→    // DB-level pagination; never findMany + slice.
   455→    db.goodsDispatch.findMany({
   456→      where,
   457→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   458→      skip: (input.page - 1) * input.pageSize,
   459→      take: input.pageSize,
   460→      select: {
   461→        id: true,
   462→        reference: true,
   463→        status: true,
   464→        customerId: true,
   465→        customerNameAtDispatch: true,
   466→        postedAt: true,
   467→        cancelledAt: true,
   468→        createdAt: true,
   469→        lines: {
   470→          // Bounded aggregate relation, not one query per row.
   471→          // take cap+1 PROVES the slice is the whole line set: a dispatch
   472→          // over its own cap is corruption and fails the read below —
   473→          // never a silently truncated total.
   474→          select: { qty: true },
   475→          take: cap + 1,
   476→        },
   477→        _count: { select: { lines: true } },
   478→      },
   479→    }),
   480→  ]);
   481→
   482→  const items = rows.map((row) => {
   483→    try {
   484→      // The THREE explicit completeness invariants (fetched ≤ cap,
   485→      // counted ≤ cap, counted === fetched) are enforced by the ONE
   486→      // canonical pure validator: a count mismatch fails the read EVEN
   487→      // WHEN both numbers are below the cap — a subset total must never
   488→      // masquerade as complete. Then the proven-complete slice feeds the
   489→      // canonical sum, which re-validates every persisted quantity
   490→      // (safe integer ≥ 1 — a stored zero is corruption).
   491→      assertGoodsDispatchLineSetComplete(row.lines.length, row._count.lines, cap);
   492→      const totalDispatchedQty = computeDispatchedQty(row.lines.map((l) => l.qty));
   493→      return goodsDispatchListItemDto({
   494→        id: row.id,
   495→        reference: row.reference,
   496→        status: row.status,
   497→        customerId: row.customerId,
   498→        customerNameAtDispatch: row.customerNameAtDispatch,
   499→        lineCount: row._count.lines,
   500→        totalDispatchedQty,
   501→        postedAt: row.postedAt,
   502→        cancelledAt: row.cancelledAt,
   503→        createdAt: row.createdAt,
   504→        createdByUserId: "",
   505→        createdByName: null,
   506→        notes: null,
   507→      });
   508→    } catch (error) {
   509→      // A corrupted quantity inside history surfaces as the stable
   510→      // integrity 500, never a partial row and never a generic 500.
   511→      return mapPureGoodsDispatchDomainError(error);
   512→    }
   513→  });
   514→  return { items, total, page: input.page, pageSize: input.pageSize };
   515→}

   519→export async function getGoodsDispatch(input: {
   520→  businessId: string;
   521→  id: string;
   522→}): Promise<GoodsDispatchDetailDto> {
   523→  const cap = goodsDispatchLineCap(process.env.GOODS_DISPATCH_MAX_LINES);
   524→  // Lines are bounded by the document's own cap; one bounded read.
   525→  // take cap+1 proves completeness: the +1 row is corruption and fails the
   526→  // read with the stable invariant code below — never a detail DTO
   527→  // silently rendered from only the first cap lines.
   528→  const row = await db.goodsDispatch.findFirst({
   529→    where: { id: input.id, businessId: input.businessId }, // tenant scope
   530→    select: {
   531→      id: true,
   532→      reference: true,
   533→      status: true,
   534→      notes: true,
   535→      customerId: true,
   536→      customerNameAtDispatch: true,
   537→      createdByUserId: true,
   538→      postedAt: true,
   539→      cancelledAt: true,
   540→      createdAt: true,
   541→      createdByUser: { select: { name: true } }, // name only
   542→      lines: {
   543→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   544→        take: cap + 1,
   545→        select: {
   546→          id: true,
   547→          productId: true,
   548→          productNameAtDispatch: true,
   549→          skuAtDispatch: true,
   550→          unitAtDispatch: true,
   551→          qty: true,
   552→          reasonCode: true,
   553→          createdAt: true,
   554→          updatedAt: true,
   555→        },
   556→      },
   557→    },
   558→  });
   559→  if (!row) throw new ApiError(404, "NOT_FOUND", "Goods dispatch not found.");
   560→
   561→  try {
   562→    // Never render a partial document: cap+1 rows fetched, so a fetched
   563→    // count above the cap proves corruption (the counted arm is identical
   564→    // here — no separate _count exists on this read).
   565→    assertGoodsDispatchLineSetComplete(row.lines.length, row.lines.length, cap);
   566→    return goodsDispatchDetailDto(
   567→      {
   568→        id: row.id,
   569→        reference: row.reference,
   570→        status: row.status,
   571→        notes: row.notes,
   572→        customerId: row.customerId,
   573→        customerNameAtDispatch: row.customerNameAtDispatch,
   574→        createdByUserId: row.createdByUserId,
   575→        createdByName: row.createdByUser?.name ?? null,
   576→        postedAt: row.postedAt,
   577→        cancelledAt: row.cancelledAt,
   578→        createdAt: row.createdAt,
   579→      },
   580→      row.lines,
   581→    );
   582→  } catch (error) {
   583→    return mapPureGoodsDispatchDomainError(error);
   584→  }
   585→}

   647→/* -------------------------------- line add --------------------------------- */
   648→
   649→export async function addGoodsDispatchLine(input: {
   650→  businessId: string;
   651→  userId: string;
   652→  id: string;
   653→  productId: string;
   654→  qty: number;
   655→  reasonCode: string;
   656→  ip?: string | null;
   657→  userAgent?: string | null;
   658→}) {
   659→  return withWriteTx(async (tx) => {
   660→    const header = await loadDispatchHeader(tx, input.businessId, input.id);
   661→    requireDraft(header.status); // DRAFT-only
   662→
   663→    // The product IS the authoritative outbound source: it must exist
   664→    // inside the SAME tenant and be ACTIVE (a new dispatch line cannot
   665→    // target a soft-deleted product). A foreign product is
   666→    // indistinguishable from an unknown one (404, no existence leak).
   667→    const product = await tx.product.findFirst({
   668→      where: { id: input.productId, businessId: input.businessId, deletedAt: null },
   669→      select: {
   670→        id: true,
   671→        name: true,
   672→        sku: true,
   673→        unit: true,
   674→        stockQty: true,
   675→        isActive: true,
   676→      },
   677→    });
   678→    if (!product || !product.isActive) {
   679→      throw new ApiError(
   680→        404,
   681→        GOODS_DISPATCH_ERROR_CODES.PRODUCT_NOT_FOUND,
   682→        "Product not found.",
   683→      );
   684→    }
   685→
   686→    const cap = goodsDispatchLineCap(process.env.GOODS_DISPATCH_MAX_LINES);
   687→    await assertLineCap(tx, header.id, cap); // defensive count
   688→
   689→    // ADVISORY availability check at line-add time: qty ≤ current stock.
   690→    // Draft lines of OTHER dispatches are deliberately NOT subtracted —
   691→    // over-allocation across drafts is the documented stale-draft
   692→    // scenario and only POST is authoritative (inside the write tx).
   693→    if (exceedsAvailableStock(product.stockQty, input.qty)) {
   694→      throw new ApiError(
   695→        409,
   696→        GOODS_DISPATCH_ERROR_CODES.INSUFFICIENT_STOCK,
   697→        `Insufficient stock for ${product.name}: ${product.stockQty} on hand, ${input.qty} to dispatch.`,
   698→      );
   699→    }
   700→
   701→    try {
   702→      const line = await tx.goodsDispatchLine.create({
   703→        data: {
   704→          goodsDispatchId: header.id,
   705→          productId: product.id,
   706→          qty: input.qty,
   707→          reasonCode: input.reasonCode, // closed vocabulary (zod)
   708→          // Snapshots copied from the PRODUCT at add time, then frozen —
   709→          // never re-read live afterwards.
   710→          productNameAtDispatch: product.name,
   711→          skuAtDispatch: product.sku,
   712→          unitAtDispatch: product.unit,
   713→        },
   714→        select: { id: true },
   715→      });
   716→
   717→      await audit(
   718→        auditInput(
   719→          input.businessId,
   720→          input.userId,
   721→          "goods_dispatch.line_add",
   722→          header.id,
   723→          header.reference,
   724→          {
   725→            lineId: line.id, // safe per-line metadata
   726→            productId: product.id,
   727→            productNameAtDispatch: product.name,
   728→            qty: input.qty,
   729→            reasonCode: input.reasonCode,
   730→          },
   731→        ),
   732→        tx,
   733→      );
   734→
   735→      return { id: line.id };
   736→    } catch (error) {
   737→      // DB uniqueness is authoritative for duplicate lines (one line per
   738→      // product per dispatch).
   739→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   740→        throw new ApiError(
   741→          409,
   742→          GOODS_DISPATCH_ERROR_CODES.DUPLICATE_LINE,
   743→          "This product is already on the dispatch.",
   744→        );
   745→      }
   746→      throw error;
   747→    }
   748→  });
   749→}

   882→/* ---------------------------------- post ----------------------------------- */
   883→
   884→/**
   885→ * THE atomic stock posting transaction. All steps in ONE database
   886→ * transaction; a failure at ANY step rolls back EVERYTHING — no stock
   887→ * change, no adjustment, no status, no audit.
   888→ */
   889→export async function postGoodsDispatch(input: {
   890→  businessId: string;
   891→  userId: string;
   892→  id: string;
   893→  ip?: string | null;
   894→  userAgent?: string | null;
   895→}) {
   896→  return withWriteTx(async (tx) => {
   897→    // Reload dispatch INSIDE the transaction; never trust an earlier GET.
   898→    // Corrupted status is surfaced, never Draft. Only DRAFT → POSTED (at
   899→    // most one successful transition; concurrent posts serialize on the
   900→    // shared write transaction).
   901→    const header = await loadDispatchHeader(tx, input.businessId, input.id);
   902→    assertTransition(header.status, "POSTED");
   903→
   904→    // Load the dispatch's own lines (bounded by the document cap; cap+1
   905→    // proves completeness).
   906→    const cap = goodsDispatchLineCap(process.env.GOODS_DISPATCH_MAX_LINES);
   907→    const lines = await tx.goodsDispatchLine.findMany({
   908→      where: { goodsDispatchId: header.id },
   909→      orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   910→      take: cap + 1, // cap+1 proves completeness
   911→      select: {
   912→        id: true,
   913→        productId: true,
   914→        qty: true,
   915→        reasonCode: true,
   916→        productNameAtDispatch: true,
   917→      },
   918→    });
   919→    if (lines.length > cap) {
   920→      // A PERSISTED document holding more lines than its own invariant
   921→      // permits is DATA CORRUPTION → 500 GOODS_DISPATCH_LINE_CAP_INVARIANT.
   922→      // Never the 422 write-time LINE_LIMIT error (that classification is
   923→      // for a newly submitted over-limit request). No status change, no
   924→      // stock mutation, no adjustment, no audit.
   925→      throw new ApiError(
   926→        500,
   927→        GOODS_DISPATCH_LINE_CAP_INVARIANT,
   928→        "Stored document data exceeds its configured line cap. Please contact support.",
   929→      );
   930→    }
   931→
   932→    // A dispatch with zero lines can never post.
   933→    if (lines.length === 0) {
   934→      throw new ApiError(
   935→        422,
   936→        GOODS_DISPATCH_ERROR_CODES.EMPTY,
   937→        "Add at least one line before posting this dispatch.",
   938→      );
   939→    }
   940→
   941→    // Validate EVERY persisted line BEFORE any mutation: a corrupted
   942→    // persisted line (zero, negative, fractional, unsafe, over the domain
   943→    // max, or an unknown reason code) must NEVER become a zero-delta or
   944→    // positive-stock adjustment. Controlled integrity error — no stock
   945→    // mutation, no InventoryAdjustment, no status mutation, no post audit.
   946→    for (const line of lines) {
   947→      if (!isValidPersistedDispatchQty(line.qty)) {
   948→        throw new ApiError(
   949→          500,
   950→          GOODS_DISPATCH_ERROR_CODES.CORRUPT_HISTORY,
   951→          "Goods dispatch data is inconsistent. Please contact support.",
   952→        );
   953→      }
   954→      if (!isGoodsDispatchReasonCode(line.reasonCode)) {
   955→        throw new ApiError(
   956→          500,
   957→          GOODS_DISPATCH_ERROR_CODES.CORRUPT_HISTORY,
   958→          "Goods dispatch data is inconsistent. Please contact support.",
   959→        );
   960→      }
   961→    }
   962→
   963→    const adjustmentIds: string[] = [];
   964→    const stockChanges: { productId: string; from: number; to: number }[] = [];
   965→
   966→    // Group products once (bounded by the line cap — no N+1 per line).
   967→    const productIds = Array.from(new Set(lines.map((l) => l.productId)));
   968→    const products = await tx.product.findMany({
   969→      where: { id: { in: productIds }, businessId: input.businessId }, // tenant scope
   970→      select: { id: true, name: true, stockQty: true },
   971→    });
   972→    const productById = new Map(products.map((p) => [p.id, p]));
   973→
   974→    for (const line of lines) {
   975→      const product = productById.get(line.productId);
   976→      if (!product) {
   977→        // A hard-deleted product fails the post safely (never fabrication).
   978→        throw new ApiError(
   979→          422,
   980→          GOODS_DISPATCH_ERROR_CODES.PRODUCT_NOT_FOUND,
   981→          "A product on this dispatch no longer exists.",
   982→        );
   983→      }
   984→
   985→      // Corrupt/negative persisted stock is surfaced as the controlled
   986→      // integrity failure BEFORE any sufficiency comparison: a negative
   987→      // stock reading is corruption, not "insufficient stock", and it is
   988→      // NEVER silently repaired or zeroed.
   989→      if (!Number.isSafeInteger(product.stockQty) || product.stockQty < 0) {
   990→        throw new ApiError(
   991→          500,
   992→          GOODS_DISPATCH_ERROR_CODES.INVALID_PERSISTED_STOCK,
   993→          "Product stock data is in an invalid state. Please contact support.",
   994→        );
   995→      }
   996→
   997→      // CURRENT STOCK CHECK (final authority): the physical decrement
   998→      // needs enough aggregate stock NOW (no lot/batch provenance — the
   999→      // documented aggregate-stock trade-off).
  1000→      if (exceedsAvailableStock(product.stockQty, line.qty)) {
  1001→        throw new ApiError(
  1002→          409,
  1003→          GOODS_DISPATCH_ERROR_CODES.INSUFFICIENT_STOCK,
  1004→          `Insufficient stock for ${line.productNameAtDispatch}: ${product.stockQty} on hand, ${line.qty} to dispatch.`,
  1005→        );
  1006→      }
  1007→
  1008→      // Exact integer arithmetic; stockAfterDispatch re-validates both
  1009→      // operands and the non-negative result (defense in depth).
  1010→      const newStock = stockAfterDispatch(product.stockQty, line.qty);
  1011→
  1012→      // Decrement stock.
  1013→      await tx.product.update({
  1014→        where: { id: product.id },
  1015→        data: { stockQty: newStock },
  1016→        select: { id: true },
  1017→      });
  1018→      stockChanges.push({ productId: product.id, from: product.stockQty, to: newStock });
  1019→
  1020→      // ONE negative InventoryAdjustment per line (delta = -qty),
  1021→      // server-derived actor/business/product.
  1022→      const adjustment = await tx.inventoryAdjustment.create({
  1023→        data: {
  1024→          businessId: input.businessId, // server-derived
  1025→          productId: product.id, // derived from the dispatch line
  1026→          delta: -line.qty, // always negative
  1027→          reason: DISPATCH_ADJUSTMENT_REASON(header.reference), // stable identifier
  1028→          actorId: input.userId, // server-derived
  1029→        },
  1030→        select: { id: true },
  1031→      });
  1032→      adjustmentIds.push(adjustment.id);
  1033→    }
  1034→
  1035→    // Complete the posting (server timestamps only).
  1036→    const postedAt = new Date();
  1037→    const updated = await tx.goodsDispatch.update({
  1038→      where: { id: header.id },
  1039→      data: { status: "POSTED", postedAt },
  1040→      select: { id: true, reference: true, status: true, postedAt: true },
  1041→    });
  1042→
  1043→    // ONE audit record for the posting, written in the SAME transaction;
  1044→    // if it fails, everything above rolls back.
  1045→    const totalDispatchedQty = computeDispatchedQty(lines.map((l) => l.qty));
  1046→    await audit(
  1047→      auditInput(input.businessId, input.userId, "goods_dispatch.post", header.id, header.reference, {
  1048→        customerId: header.customerId,
  1049→        customerNameAtDispatch: header.customerNameAtDispatch,
  1050→        lineCount: lines.length,
  1051→        totalDispatchedQty, // quantity facts only; never notes
  1052→        adjustmentIds,
  1053→        previousStatus: header.status,
  1054→        newStatus: updated.status,
  1055→        postedAt: updated.postedAt?.toISOString() ?? null,
  1056→      }),
  1057→      tx,
  1058→    );
  1059→
  1060→    return {
  1061→      id: updated.id,
  1062→      reference: updated.reference,
  1063→      status: updated.status,
  1064→      postedAt: updated.postedAt,
  1065→      lineCount: lines.length,
  1066→      totalDispatchedQty,
  1067→      stockChanges,
  1068→    };
  1069→  });
  1070→}

  1115→export async function listRecentDispatchesForProduct(input: {

  1117→  productId: string;
  1118→  take?: number;
  1119→}): Promise<ProductRecentDispatchDto[]> {
  1120→  const take = Math.min(Math.max(input.take ?? 10, 1), 10); // bounded
  1121→  const rows = await db.goodsDispatchLine.findMany({
  1122→    where: {
  1123→      productId: input.productId,
  1124→      goodsDispatch: { businessId: input.businessId },
  1125→    },
  1126→    orderBy: [{ createdAt: "desc" }, { id: "desc" }],
  1127→    take,
  1128→    select: {
  1129→      qty: true,
  1130→      reasonCode: true,
  1131→      goodsDispatch: {
  1132→        select: {
  1133→          id: true,
  1134→          reference: true,
  1135→          status: true,
  1136→          customerNameAtDispatch: true,
  1137→          postedAt: true,
  1138→        },
  1139→      },
  1140→    },
  1141→  });
  1142→  return rows.map((r) => {
  1143→    try {
  1144→      // A persisted line quantity of zero (or any invalid value) and a
  1145→      // corrupted reason code are NEVER treated as legitimate history:
  1146→      // the widget fails with the stable integrity code instead.
  1147→      computeDispatchedQty([r.qty]); // safe integer ≥ 1
  1148→      if (!isGoodsDispatchReasonCode(r.reasonCode)) {
  1149→        throw new GoodsDispatchIntegrityError(
  1150→          `${GOODS_DISPATCH_CORRUPT_HISTORY}: invalid persisted reason code`,
  1151→        );
  1152→      }

  1154→        id: r.goodsDispatch.id,
  1155→        reference: r.goodsDispatch.reference,
  1156→        status: r.goodsDispatch.status,
  1157→        customerNameAtDispatch: r.goodsDispatch.customerNameAtDispatch,
  1158→        qty: r.qty,
  1159→        reasonCode: r.reasonCode,
  1160→        postedAt: r.goodsDispatch.postedAt,
  1161→      });
  1162→    } catch (error) {
  1163→      return mapPureGoodsDispatchDomainError(error);
  1164→    }
  1165→  });
  1166→}
