    17→type AuthErrors = {
    18→  invalidCredentials: string;
    19→  rateLimited: string;
    20→  emailTaken: string;
    21→  passwordMismatch: string;
    22→  passwordPolicy: string;
    23→  invalidData: string;
    24→  network: string;
    25→  generic: string;
    26→};
    27→
    28→/**
    29→ * Server 422 envelopes carry the zod issues in `details`. When the failed
    30→ * field is the password we can give the user the REAL reason instead of a
    31→ * generic "try again" (BUGFIX P51-R1: signup showed a useless generic error
    32→ * when the password policy rejected the submitted password).
    33→ */
    34→function hasValidationPath(details: unknown, field: string): boolean {
    35→  if (!Array.isArray(details)) return false;
    36→  return details.some(
    37→    (issue) =>
    38→      issue !== null &&
    39→      typeof issue === "object" &&
    40→      Array.isArray((issue as { path?: unknown }).path) &&
    41→      (issue as { path: unknown[] }).path[0] === field,
    42→  );
    43→}
    44→
    45→function mapSignInError(error: string | null | undefined, t: AuthErrors): string {
    46→  if (error === "CredentialsSignin") return t.invalidCredentials;
    47→  return t.generic;
    48→}
    49→
    50→function mapApiError(err: unknown, t: AuthErrors): string {
    51→  if (err instanceof ApiClientError) {
    52→    if (err.status === 0) return t.network;
    53→    if (err.code === "EMAIL_TAKEN") return t.emailTaken;
    54→    if (err.code === "RATE_LIMITED") return t.rateLimited;
    55→    if (err.code === "VALIDATION_ERROR") {
    56→      if (hasValidationPath(err.details, "password")) return t.passwordPolicy;
    57→      return t.invalidData;
    58→    }
    59→  }
    60→  return t.generic;
    61→}

   189→/**
   190→ * "Reset password" flow — consumes a single-use token from the deep link
   191→ * (#/reset-password/<token>). On success every previous session is dead:
   192→ * any stale local session is cleared immediately and the user is sent to
   193→ * sign-in via the success card button.
   194→ */
   195→function ResetPasswordCard({
   196→  token,
   197→  onGoToSignIn,
   198→  onRequestNewLink,
   199→}: {
   200→  token: string;
   201→  onGoToSignIn: () => void;
   202→  onRequestNewLink: () => void;
   203→}) {
   204→  const t = useTranslations("auth");
   205→  const queryClient = useQueryClient();
   206→  const [newPassword, setNewPassword] = useState("");
   207→  const [confirmPassword, setConfirmPassword] = useState("");
   208→  const [error, setError] = useState<string | null>(null);
   209→  const [tokenInvalid, setTokenInvalid] = useState(false);
   210→  const [doneMessage, setDoneMessage] = useState<string | null>(null);
   211→
   212→  const resetMutation = useMutation({
   213→    mutationFn: async () =>
   214→      api<{ message?: string }>("/api/v1/auth/password-reset/confirm", {
   215→        method: "POST",
   216→        body: JSON.stringify({ token, newPassword, confirmPassword }),
   217→      }),
   218→    onSuccess: () => {
   219→      setError(null);
   220→      setTokenInvalid(false);
   221→      setDoneMessage(t("resetSuccessBody"));
   222→      // The server killed every session for this user — clear any stale
   223→      // local one right away (the success card below routes to sign-in).
   224→      void signOut({ redirect: false });
   225→      void queryClient.invalidateQueries();
   226→    },
   227→    onError: (err) => {
   228→      if (err instanceof ApiClientError) {
   229→        if (err.code === "INVALID_RESET_TOKEN") {
   230→          setTokenInvalid(true);
   231→          setError(null);
   232→        } else if (err.status === 0) {
   233→          setTokenInvalid(false);
   234→          setError(t("errors.network"));
   235→        } else if (err.code === "RATE_LIMITED") {
   236→          setTokenInvalid(false);
   237→          setError(t("errors.rateLimited"));
   238→        } else {
   239→          setTokenInvalid(false);
   240→          setError(t("errors.generic"));
   241→        }
   242→      } else {
   243→        setTokenInvalid(false);
   244→        setError(t("errors.generic"));
   245→      }
   246→    },
   247→  });
   248→
   249→  return (
   250→    <Card className="w-full max-w-md shadow-sm">
   251→      <CardHeader className="space-y-1.5">
   252→        <CardTitle className="text-2xl">
   253→          {doneMessage ? t("resetSuccessTitle") : t("resetTitle")}
   254→        </CardTitle>
   255→        {doneMessage ? null : <CardDescription>{t("resetSubtitle")}</CardDescription>}
   256→      </CardHeader>
   257→      <CardContent>
   258→        {doneMessage ? (
   259→          <div className="space-y-5 py-2 text-center">
   260→            <div
   261→              className="bg-primary/10 mx-auto flex size-12 items-center justify-center rounded-full"
   262→              aria-hidden="true"
   263→            >
   264→              <KeyRound className="text-primary size-6" />
   265→            </div>
   266→            <p className="text-foreground text-sm" role="status">
   267→              {doneMessage}
   268→            </p>
   269→            <Button type="button" className="w-full" onClick={onGoToSignIn}>
   270→              <LogIn className="size-4" aria-hidden="true" />
   271→              {t("goToSignIn")}
   272→            </Button>
   273→          </div>
   274→        ) : (
   275→          <form
   276→            onSubmit={(e) => {
   277→              e.preventDefault();
   278→              setError(null);
   279→              setTokenInvalid(false);
   280→              if (newPassword !== confirmPassword) {
   281→                setError(t("errors.passwordMismatch"));
   282→                return;
   283→              }
   284→              resetMutation.mutate();
   285→            }}
   286→            className="space-y-4"
   287→            noValidate
   288→          >
   289→            <div className="space-y-2">
   290→              <Label htmlFor="reset-password">{t("newPassword")}</Label>
   291→              <Input
   292→                id="reset-password"
   293→                type="password"
   294→                autoComplete="new-password"
   295→                dir="ltr"
   296→                placeholder={t("passwordPlaceholder")}
   297→                value={newPassword}
   298→                onChange={(e) => setNewPassword(e.target.value)}
   299→                required
   300→                minLength={8}
   301→              />
   302→              <p className="text-muted-foreground text-xs">{t("passwordHint")}</p>
   303→            </div>
   304→
   305→            <div className="space-y-2">
   306→              <Label htmlFor="reset-confirm">{t("confirmNewPassword")}</Label>
   307→              <Input
   308→                id="reset-confirm"
   309→                type="password"
   310→                autoComplete="new-password"
   311→                dir="ltr"
   312→                value={confirmPassword}
   313→                onChange={(e) => setConfirmPassword(e.target.value)}
   314→                required
   315→              />
   316→            </div>
   317→
   318→            {tokenInvalid ? (
   319→              <div className={errorBoxClass} role="alert">
   320→                <p>{t("errors.invalidResetToken")}</p>
   321→                <button
   322→                  type="button"
   323→                  className="text-primary mt-1.5 font-medium underline-offset-4 hover:underline"
   324→                  onClick={onRequestNewLink}
   325→                >
   326→                  {t("sendResetLink")}
   327→                </button>
   328→              </div>
   329→            ) : error ? (
   330→              <p className={errorBoxClass} role="alert">
   331→                {error}
   332→              </p>
   333→            ) : null}
   334→
   335→            <Button type="submit" className="w-full" disabled={resetMutation.isPending}>
   336→              {resetMutation.isPending ? (
   337→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   338→              ) : (
   339→                <KeyRound className="size-4" aria-hidden="true" />
   340→              )}
   341→              {t("resetSubmit")}
   342→            </Button>
   343→          </form>
   344→        )}
   345→      </CardContent>
   346→    </Card>
   347→  );
   348→}

   350→export function AuthView() {
   351→  const t = useTranslations("auth");
   352→  const tCommon = useTranslations("common");
   353→  const view = useAppStore((s) => s.view);
   354→  const navigate = useAppStore((s) => s.navigate);
   355→  // next-intl's t is a function — nested property access (t.errors) is
   356→  // undefined at runtime, which crashed the onError handlers. Build the
   357→  // message map with explicit t() calls instead (BUGFIX r8).
   358→  const authErrorMessages: AuthErrors = {
   359→    invalidCredentials: t("errors.invalidCredentials"),
   360→    rateLimited: t("errors.rateLimited"),
   361→    emailTaken: t("errors.emailTaken"),
   362→    passwordMismatch: t("errors.passwordMismatch"),
   363→    passwordPolicy: t("errors.passwordPolicy"),
   364→    invalidData: t("errors.invalidData"),
   365→    network: t("errors.network"),
   366→    generic: t("errors.generic"),
   367→  };
   368→  const { status } = useSession();
   369→  const queryClient = useQueryClient();
   370→  const [mode, setMode] = useState<"signin" | "signup">("signin");
   371→  const [name, setName] = useState("");
   372→  const [email, setEmail] = useState("");
   373→  const [password, setPassword] = useState("");
   374→  const [confirmPassword, setConfirmPassword] = useState("");
   375→  const [error, setError] = useState<string | null>(null);
   376→
   377→  // Prompt 30 §46/§109 — the MFA login step machine. Password success with
   378→  // MFA enabled NEVER creates a session (authorize throws the fixed
   379→  // MFA_REQUIRED code); the client then obtains a SHORT-LIVED challenge,
   380→  // collects the TOTP/recovery code, and exchanges it for a ONE-TIME
   381→  // assertion that completes the real NextAuth login.
   382→  const [mfaStage, setMfaStage] = useState<"idle" | "challenge" | "code">("idle");
   383→  const [mfaChallenge, setMfaChallenge] = useState<string | null>(null);
   384→  const [mfaMethod, setMfaMethod] = useState<"TOTP" | "RECOVERY_CODE">("TOTP");
   385→  const [mfaCode, setMfaCode] = useState("");
   386→  // Prompt 27 §53/§58: explicit legal acceptance — UNCHECKED by default, with
   387→  // links to the actual documents BEFORE acceptance. Required by the signup UI.
   388→  const [acceptedLegal, setAcceptedLegal] = useState(false);
   389→  const tLegal = useTranslations("legal.auth");
   390→
   391→  const signInMutation = useMutation({
   392→    mutationFn: async () => {
   393→      const res = await signIn("credentials", { redirect: false, email, password });
   394→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   395→      return res;
   396→    },
   397→    onSuccess: () => {
   398→      setError(null);
   399→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   400→      void queryClient.invalidateQueries({ queryKey: ["me"] });
   401→    },
   402→    onError: (err) => {
   403→      // Prompt 30 §110 — MFA_REQUIRED arrives ONLY after the password was
   404→      // verified inside authorize(); switch to the bounded MFA step.
   405→      if (err.message === "MFA_REQUIRED") {
   406→        void startMfaChallenge();
   407→        return;
   408→      }
   409→      setError(
   410→        err instanceof ApiClientError
   411→          ? mapApiError(err, authErrorMessages)
   412→          : mapSignInError(err.message, authErrorMessages),
   413→      );
   414→    },
   415→  });
   416→
   417→  // Step 2 of the MFA flow (§46): re-verify the password at the dedicated
   418→  // endpoint to obtain the challenge token (rate-limited, hash-only server
   419→  // state, bound to the user's sessionVersion).
   420→  const startMfaChallenge = () => {
   421→    setMfaStage("challenge");
   422→    mfaChallengeMutation.mutate();
   423→  };
   424→
   425→  const mfaChallengeMutation = useMutation({
   426→    mutationFn: async () =>
   427→      api<{ mfaRequired: boolean; challenge: string; expiresAt: string }>(
   428→        "/api/v1/auth/mfa/challenge",
   429→        { method: "POST", body: JSON.stringify({ email, password }), businessId: null },
   430→      ),
   431→    onSuccess: (res) => {
   432→      setMfaChallenge(res.challenge);
   433→      setMfaStage("code");
   434→      setMfaCode("");
   435→      setError(null);
   436→    },
   437→    onError: (err) => {
   438→      setMfaStage("idle");
   439→      if (err instanceof ApiClientError) {
   440→        if (err.code === "MFA_NOT_ENABLED") {
   441→          // The account has no MFA (state changed mid-flow) — retry plain.
   442→          signInMutation.mutate();
   443→          return;
   444→        }
   445→        if (err.code === "RATE_LIMITED") {
   446→          setError(t("mfaRateLimited"));
   447→          return;
   448→        }
   449→        if (err.code === "INVALID_CREDENTIALS") {
   450→          setError(authErrorMessages.invalidCredentials);
   451→          return;
   452→        }
   453→      }
   454→      setError(authErrorMessages.generic);
   455→    },
   456→  });
   457→
   458→  // Step 3 (§51): exchange challenge + code for the one-time assertion,
   459→  // then complete the REAL NextAuth sign-in with it (§52 — the final JWT
   460→  // is still issued by NextAuth only).
   461→  const mfaCompleteMutation = useMutation({
   462→    mutationFn: async () => {
   463→      const assertion = await api<{ assertion: string }>("/api/v1/auth/mfa/complete", {
   464→        method: "POST",
   465→        body: JSON.stringify({ challenge: mfaChallenge, code: mfaCode, method: mfaMethod }),
   466→        businessId: null,
   467→      });
   468→      const res = await signIn("credentials", {
   469→        redirect: false,
   470→        email,
   471→        mfaAssertion: assertion.assertion,
   472→      });
   473→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   474→      return res;
   475→    },
   476→    onSuccess: () => {
   477→      setMfaStage("idle");
   478→      setMfaChallenge(null);
   479→      setMfaCode("");
   480→      setError(null);
   481→      void queryClient.invalidateQueries({ queryKey: ["businesses"] });
   482→      void queryClient.invalidateQueries({ queryKey: ["me"] });
   483→    },
   484→    onError: (err) => {
   485→      if (err instanceof ApiClientError) {
   486→        if (err.code === "MFA_INVALID_CODE") {
   487→          setError(t("mfaInvalidCode"));
   488→          setMfaCode("");
   489→          return;
   490→        }
   491→        if (err.code === "MFA_CHALLENGE_EXPIRED" || err.code === "MFA_CHALLENGE_INVALID") {
   492→          setError(t("mfaChallengeExpired"));
   493→          return;
   494→        }
   495→        if (err.code === "MFA_RATE_LIMITED" || err.code === "RATE_LIMITED") {
   496→          setError(t("mfaRateLimited"));
   497→          return;
   498→        }
   499→        if (err.code === "MFA_RECOVERY_CODE_INVALID" || err.code === "MFA_RECOVERY_CODE_USED") {
   500→          setError(t("mfaInvalidCode"));
   501→          setMfaCode("");
   502→          return;
   503→        }
   504→      }
   505→      // Assertion replay/expiry inside authorize(): restart the MFA step.
   506→      if (err.message === "MFA_REQUIRED") {
   507→        setError(t("mfaChallengeExpired"));
   508→        setMfaStage("idle");
   509→        setMfaChallenge(null);
   510→        return;
   511→      }
   512→      setError(authErrorMessages.generic);
   513→    },
   514→  });
   515→
   516→  const registerMutation = useMutation({
   517→    mutationFn: async () => {
   518→      await api("/api/v1/auth/register", {
   519→        method: "POST",
   520→        body: JSON.stringify({
   521→          name,
   522→          email,
   523→          password,
   524→          // Server resolves the accepted document versions (Prompt 27 §163).
   525→          acceptedLegal: acceptedLegal ? { terms: true, privacy: true } : undefined,
   526→        }),
   527→      });
   528→      const res = await signIn("credentials", { redirect: false, email, password });
   529→      if (!res || res.error) throw new Error(res?.error ?? "CredentialsSignin");
   530→      return res;
   531→    },
   532→    onSuccess: () => {
   533→      setError(null);
   534→      void queryClient.invalidateQueries();
   535→    },
   536→    onError: (err) => {
   537→      setError(
   538→        err instanceof ApiClientError
   539→          ? mapApiError(err, authErrorMessages)
   540→          : mapSignInError(err.message, authErrorMessages),
   541→      );
   542→    },
   543→  });
   544→
   545→  const busy = signInMutation.isPending || registerMutation.isPending || status === "loading";
   546→
   547→  // Store-driven mode switches (hash updates): password flows always route
   548→  // through navigate() so deep links, back/forward and refresh all work.
   549→  function goBackToSignIn() {
   550→    setMode("signin");
   551→    setError(null);
   552→    navigate({ kind: "dashboard" });
   553→  }
   554→
   555→  function onSubmit(e: React.FormEvent) {
   556→    e.preventDefault();
   557→    setError(null);
   558→    if (mode === "signup") {
   559→      if (password !== confirmPassword) {
   560→        setError(t("errors.passwordMismatch"));
   561→        return;
   562→      }
   563→      // Client-side mirror of the server password policy (auth-security
   564→      // passwordSchema): min 8 chars, at least one letter and one digit.
