   133→// ---- Markdown-lite renderer (no deps, no raw HTML — safe by construction) --
   134→
   135→function inline(text: string, keyPrefix: string): ReactNode[] {
   136→  const parts = text.split(/(\*\*[^*]+\*\*|`[^`]+`)/g);
   137→  return parts.map((part, i) => {
   138→    const key = `${keyPrefix}-${i}`;
   139→    if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
   140→      return (
   141→        <strong key={key} className="font-semibold">
   142→          {part.slice(2, -2)}
   143→        </strong>
   144→      );
   145→    }
   146→    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
   147→      return (
   148→        <code
   149→          key={key}
   150→          dir="ltr"
   151→          className="bg-muted rounded px-1 py-0.5 font-mono text-[0.85em]"
   152→        >
   153→          {part.slice(1, -1)}
   154→        </code>
   155→      );
   156→    }
   157→    return <span key={key}>{part}</span>;
   158→  });
   159→}

   161→function MarkdownLite({ text, streaming }: { text: string; streaming?: boolean }) {
   162→  const nodes: ReactNode[] = [];
   163→  let bullets: string[] = [];
   164→  let para: string[] = [];
   165→
   166→  const flushBullets = () => {
   167→    if (bullets.length === 0) return;
   168→    nodes.push(
   169→      <ul
   170→        key={`ul-${nodes.length}`}
   171→        className="marker:text-muted-foreground/50 my-1 list-disc space-y-1 ps-5"
   172→      >
   173→        {bullets.map((b, i) => (
   174→          <li key={i}>{inline(b, `b${nodes.length}-${i}`)}</li>
   175→        ))}
   176→      </ul>,
   177→    );
   178→    bullets = [];
   179→  };
   180→  const flushPara = () => {
   181→    if (para.length === 0) return;
   182→    nodes.push(
   183→      <p key={`p-${nodes.length}`} className="leading-relaxed whitespace-pre-wrap">
   184→        {inline(para.join("\n"), `p${nodes.length}`)}
   185→      </p>,
   186→    );
   187→    para = [];
   188→  };
   189→
   190→  for (const raw of text.split("\n")) {
   191→    const line = raw.trimEnd();
   192→    if (/^\s*[-*•]\s+/.test(line)) {
   193→      flushPara();
   194→      bullets.push(line.replace(/^\s*[-*•]\s+/, ""));
   195→    } else if (line === "") {
   196→      flushBullets();
   197→      flushPara();
   198→    } else {
   199→      flushBullets();
   200→      para.push(line);
   201→    }
   202→  }
   203→  flushBullets();
   204→  flushPara();
   205→
   206→  return (
   207→    <div className="space-y-1.5 text-sm">
   208→      {nodes}
   209→      {streaming ? (
   210→        <span
   211→          aria-hidden="true"
   212→          className="ml-0.5 inline-block h-4 w-0.5 animate-pulse rounded-full bg-emerald-600 align-middle dark:bg-emerald-400"
   213→        />
   214→      ) : null}
   215→    </div>
   216→  );
   217→}

   309→// ---- Follow-up suggestions ---------------------------------------------------
   310→
   311→/**
   312→ * Context-aware follow-up pool. The chips are UI suggestions only — they
   313→ * never touch data. Deterministic: keyword match on the answer text picks
   314→ * the lead chip; the rest rotate by conversation length so repeated
   315→ * answers surface different prompts.
   316→ */
   317→const FOLLOW_UP_KEYS = [
   318→  "followUpOutstanding",
   319→  "followUpLowStock",
   320→  "followUpExpenses",
   321→  "followUpSales",
   322→  "followUpCash",
   323→  "followUpProducts",
   324→  "followUpPayments",
   325→  "followUpSummary",
   326→] as const;
   327→
   328→type FollowUpKey = (typeof FOLLOW_UP_KEYS)[number];
   329→
   330→const FOLLOW_UP_MATCHERS: Array<{ key: FollowUpKey; re: RegExp }> = [
   331→  { key: "followUpOutstanding", re: /owe|outstanding|receivable|بدهکار|بدهی|طلب/i },
   332→  { key: "followUpLowStock", re: /low.?stock|restock|inventory|موجودی|انبار|شارژ/i },
   333→  { key: "followUpExpenses", re: /expense|spend|cost|هزینه|خرج/i },
   334→  { key: "followUpSales", re: /sale|revenue|invoice|فروش|درآمد|فاکتور/i },
   335→  { key: "followUpCash", re: /cash|payment|collected|نقد|پرداخت|دریافتی/i },
   336→  { key: "followUpProducts", re: /product|best.?seller|top|محصول|پرفروش/i },
   337→];
   338→
   339→/** Live server counts (same deterministic aggregates as the dashboard)
   340→ *  used to RANK suggestions — a chip is only proposed when its subject
   341→ *  actually exists in the data (r10: "only propose restock when there is
   342→ *  something to restock"). */
   343→type SummaryLite = {
   344→  kpis?: { salesCount?: number; expensesCount?: number };
   345→  outstanding?: { receivableCount?: number };
   346→  inventory?: { lowStockCount?: number };
   347→  topProducts?: unknown[];
   348→};
   349→
   350→function followUpRelevant(
   351→  key: FollowUpKey,
   352→  counts: {
   353→    receivableCount: number;
   354→    lowStockCount: number;
   355→    salesCount: number;
   356→    expensesCount: number;
   357→    topProducts: number;
   358→  } | null,
   359→): boolean {
   360→  if (!counts) return true; // counts not loaded yet — keep the neutral default
   361→  switch (key) {
   362→    case "followUpOutstanding":
   363→      return counts.receivableCount > 0;
   364→    case "followUpLowStock":
   365→      return counts.lowStockCount > 0;
   366→    case "followUpExpenses":
   367→      return counts.expensesCount > 0;
   368→    case "followUpProducts":
   369→      return counts.topProducts > 0;
   370→    case "followUpSales":
   371→      return true;
   372→    case "followUpCash":
   373→    case "followUpPayments":
   374→    case "followUpSummary":
   375→      return true;
   376→  }
   377→}
   378→
   379→function pickFollowUps(
   380→  answer: string,
   381→  conversationLength: number,
   382→  counts: Parameters<typeof followUpRelevant>[1] = null,
   383→): FollowUpKey[] {
   384→  const lead = FOLLOW_UP_MATCHERS.find((m) => m.re.test(answer))?.key;
   385→  const pool = FOLLOW_UP_KEYS.filter(
   386→    (k) => k !== lead && followUpRelevant(k, counts),
   387→  );
   388→  const rotate = conversationLength % Math.max(1, pool.length);
   389→  const rotated = [...pool.slice(rotate), ...pool.slice(0, rotate)];
   390→  const picked = rotated.slice(0, lead ? 2 : 3);
   391→  return lead ? [lead, ...picked] : picked;
   392→}

   562→    setSessions((prev) => [session, ...prev]);

   564→  }, []);
   565→
   566→  /**
   567→   * Mutate the ACTIVE session's messages. Accepts a value or updater so the
   568→   * existing send/stream/patch code (and its functional setState calls)
   569→   * keeps working unchanged.
   570→   */
   571→  const setMessages = useCallback(
   572→    (update: ChatMessage[] | ((prev: ChatMessage[]) => ChatMessage[])) => {
   573→      const targetId = activeSessionIdRef.current;
   574→      setSessions((prevSessions) => {
   575→        const idx = prevSessions.findIndex((s) => s.id === targetId);
   576→        if (idx === -1) {
   577→          // No live session — create one from the resulting messages.
   578→          if (!targetId) return prevSessions;
   579→          const msgs = typeof update === "function" ? update([]) : update;
   580→          if (msgs.length === 0) return prevSessions;
   581→          const firstUser = msgs.find((m) => m.role === "user");
   582→          const session: CopilotSession = {
   583→            ...createEmptySession(firstUser ? titleFromQuestion(firstUser.content) : ""),
   584→            id: targetId,
   585→            messages: msgs,
   586→            updatedAt: Date.now(),
   587→          };
   588→          return [session, ...prevSessions];
   589→        }
   590→        const cur = prevSessions[idx];
   591→        const msgs =
   592→          typeof update === "function"
   593→            ? update(cur.messages as ChatMessage[])
   594→            : update;
   595→        const firstUser = msgs.find((m) => m.role === "user");
   596→        const next: CopilotSession = {
   597→          ...cur,
   598→          title:
   599→            cur.title ||
   600→            (firstUser ? titleFromQuestion(firstUser.content) : cur.title),
   601→          messages: msgs,
   602→          updatedAt: Date.now(),
   603→        };
   604→        return [next, ...prevSessions.filter((s) => s.id !== targetId)];
   605→      });
   606→    },
   607→    [],
   608→  );
   609→
   610→  /** Best-effort server mirror of one appended message (Phase 07). */
   611→  const syncAppend = useCallback(
   612→    (
   613→      sessionId: string | null,
   614→      msg: { role: "user" | "assistant"; content: string; meta?: unknown },
   615→      title?: string,
   616→    ) => {
   617→      if (!sessionId) return;
   618→      void serverAppendMessage(business.id, sessionId, msg, title);
   619→    },
   620→    [business.id],
   621→  );
   622→
   623→  const switchSession = useCallback((id: string) => {
   624→    if (activeStreamId !== null) return; // never mid-stream
   625→    activeSessionIdRef.current = id;
   626→    setActiveSessionId(id);
   627→  }, [activeStreamId]);
   628→
   629→  const startNewConversation = useCallback(() => {
   630→    if (activeStreamId !== null) return;
   631→    if ((sessions.find((s) => s.id === activeSessionIdRef.current)?.messages ?? []).length === 0) {
   632→      textareaRef.current?.focus();
   633→      return; // already on a fresh conversation
   634→    }
   635→    const session = createEmptySession();
   636→    activeSessionIdRef.current = session.id;
   637→    setActiveSessionId(session.id);
   638→    setSessions((prev) => [session, ...prev]);
   639→    textareaRef.current?.focus();
   640→  }, [activeStreamId, sessions]);
   641→
   642→  const deleteSession = useCallback(
   643→    (id: string) => {
   644→      if (activeStreamId !== null) return;
   645→      setSessions((prev) => {
   646→        const next = prev.filter((s) => s.id !== id);
   647→        if (activeSessionIdRef.current === id) {
   648→          const fallback = next[0]?.id ?? null;
   649→          activeSessionIdRef.current = fallback;
   650→          setActiveSessionId(fallback);
   651→        }
   652→        return next;
   653→      });
   654→      void serverDeleteSession(business.id, id);
   655→      toast({ title: t("sessionDeletedToast") });
   656→    },
   657→    [activeStreamId, business.id, t],
   658→  );
   659→
   660→  /** Clear EVERY transcript for this business — local cache + server copy. */
   661→  const clearAllSessions = useCallback(() => {
   662→    if (activeStreamId !== null) return;
   663→    setSessions([]);
   664→    activeSessionIdRef.current = null;
   665→    setActiveSessionId(null);
   666→    clearCopilotState(business.id);
   667→    void serverClearSessions(business.id);
   668→    toast({ title: t("clearAllToast") });
   669→  }, [activeStreamId, business.id, t]);
   670→
   671→  const patchAssistant = useCallback(
   672→    (id: string, patch: (m: ChatMessage) => ChatMessage) => {
   673→      setMessages((prev) => prev.map((m) => (m.id === id ? patch(m) : m)));
   674→    },
   675→    [],
   676→  );
   677→
   678→  const localizedError = useCallback(
   679→    (code: string, retryAfterMs?: number) => {
   680→      switch (code) {
   681→        case "RATE_LIMITED":
   682→          return t("errorRateLimited", {
   683→            seconds: String(Math.max(1, Math.ceil((retryAfterMs ?? 0) / 1000))),
   684→          });
   685→        case "PERMISSION_DENIED":
   686→          return t("errorForbidden");
   687→        case "AI_UNAVAILABLE":
   688→          return t("errorAiUnavailable");
   689→        default:
   690→          return tErrors("generic");
   691→      }
   692→    },
   693→    [t, tErrors],
   694→  );
   695→
   696→  const send = useCallback(
   697→    async (question: string, history: ChatMessage[]) => {
   698→      const trimmed = question.trim();
   699→      if (!trimmed || trimmed.length > 500 || activeStreamId !== null) return;
   700→      lastQuestionRef.current = trimmed;
   701→      setAtBottom(true);
   702→
   703→      const assistantId = nextId();
   704→      setMessages((prev) => [
   705→        ...prev,
   706→        { id: assistantId, role: "assistant", content: "" },
   707→      ]);
   708→      setActiveStreamId(assistantId);
   709→
   710→      const controller = new AbortController();
   711→      abortRef.current = controller;
   712→
   713→      // Stream accumulators — also used to mirror the finished answer to the
   714→      // server transcript (Phase 07 cross-device persistence).
   715→      let answerText = "";
   716→      let answerMeta: CopilotMeta | undefined;
   717→
   718→      try {
   719→        const res = await fetch("/api/v1/copilot/stream", {
   720→          method: "POST",
   721→          credentials: "same-origin",
   722→          headers: {
   723→            "content-type": "application/json",
   724→            "x-business-id": business.id,
   725→          },
   726→          body: JSON.stringify({
   727→            question: trimmed,
   728→            history: history
   729→              .filter((m) => !m.failed)
   730→              .slice(-6)
   731→              .map((m) => ({ role: m.role, content: m.content })),
   732→            days,
   733→          }),
   734→          signal: controller.signal,
   735→        });
   736→
   737→        if (!res.ok || !res.body) {
   738→          let code = "INTERNAL_ERROR";
   739→          let retryAfterMs: number | undefined;
   740→          try {
   741→            const body = (await res.json()) as {
   742→              error?: { code?: string; details?: { retryAfterMs?: number } };
   743→            };
   744→            if (body?.error?.code) code = body.error.code;
   745→            retryAfterMs = body?.error?.details?.retryAfterMs;
   746→          } catch {
   747→            // keep defaults
   748→          }
   749→          patchAssistant(assistantId, (m) => ({
   750→            ...m,
   751→            content: localizedError(code, retryAfterMs),
   752→            failed: true,
   753→          }));
   754→          return;
   755→        }
   756→
   757→        // Parse the SSE stream incrementally.
   758→        const reader = res.body.getReader();
   759→        const decoder = new TextDecoder();
   760→        let buffer = "";
   761→        let streamError: { code?: string; message?: string } | null = null;
   762→        let sawDone = false;
   763→
   764→        while (true) {
   765→          const { done, value } = await reader.read();
   766→          if (done) break;
   767→          buffer += decoder.decode(value, { stream: true });
   768→          const events = buffer.split("\n\n");
   769→          buffer = events.pop() ?? "";
   770→          for (const evt of events) {
   771→            for (const line of evt.split("\n")) {
   772→              if (!line.startsWith("data: ")) continue;
   773→              let payload: {
   774→                meta?: CopilotMeta;
   775→                delta?: string;
   776→                done?: boolean;
   777→                error?: { code?: string; message?: string };
   778→              };
   779→              try {
   780→                payload = JSON.parse(line.slice(6));
   781→              } catch {
   782→                continue;
   783→              }
   784→              if (payload.meta) {
   785→                // First event — dataset provenance for this answer.
   786→                const meta = payload.meta;
   787→                answerMeta = meta;
   788→                patchAssistant(assistantId, (m) => ({ ...m, meta }));
   789→              } else if (typeof payload.delta === "string" && payload.delta.length > 0) {
   790→                const delta = payload.delta;
   791→                answerText += delta;
   792→                patchAssistant(assistantId, (m) => ({
   793→                  ...m,
   794→                  content: m.content + delta,
   795→                }));
   796→              } else if (payload.error) {
   797→                streamError = payload.error;
   798→              } else if (payload.done) {
   799→                sawDone = true;
   800→              }
   801→            }
   802→          }
   803→        }
   804→
   805→        if (streamError) {
   806→          const code = streamError.code ?? "INTERNAL_ERROR";
   807→          if (code === "RATE_LIMITED" || code === "AI_UNAVAILABLE" || !sawDone) {
   808→            patchAssistant(assistantId, (m) =>
   809→              m.content.trim().length === 0
   810→                ? { ...m, content: localizedError(code), failed: true }
   811→                : { ...m, content: `${m.content}\n\n⚠ ${localizedError(code)}` },
   812→            );
   813→          }
   814→        } else if (answerText.trim().length > 0) {
   815→          // Completed answer → mirror to the server transcript (with its
   816→          // citation provenance). Failed answers stay local-only by design.
   817→          syncAppend(
   818→            activeSessionIdRef.current,
   819→            {
   820→              role: "assistant",
   821→              content: answerText,
   822→              ...(answerMeta ? { meta: answerMeta } : {}),
   823→            },
   824→            titleFromQuestion(lastQuestionRef.current ?? ""),
   825→          );
   826→        }
   827→      } catch (error) {
   828→        if ((error as Error)?.name === "AbortError") {
   829→          // User pressed Stop — keep the partial answer (and mirror it), drop it if empty.
   830→          if (answerText.trim().length > 0) {
   831→            syncAppend(
   832→              activeSessionIdRef.current,
   833→              {
   834→                role: "assistant",
   835→                content: answerText,
   836→                ...(answerMeta ? { meta: answerMeta } : {}),
   837→              },
   838→              titleFromQuestion(lastQuestionRef.current ?? ""),
   839→            );
   840→          }
   841→          setMessages((prev) =>
   842→            prev.flatMap((m) =>
   843→              m.id === assistantId ? (m.content.trim() ? [m] : []) : [m],
   844→            ),
   845→          );
   846→        } else if (error instanceof ApiClientError) {
   847→          patchAssistant(assistantId, (m) => ({
   848→            ...m,
   849→            content: localizedError(error.code),
   850→            failed: true,
   851→          }));
   852→        } else {
   853→          patchAssistant(assistantId, (m) => ({
   854→            ...m,
   855→            content: localizedError("NETWORK_ERROR"),
   856→            failed: true,
   857→          }));
   858→        }
   859→      } finally {
   860→        abortRef.current = null;
   861→        setActiveStreamId((cur) => (cur === assistantId ? null : cur));
   862→      }
   863→    },
   864→    [activeStreamId, business.id, days, localizedError, patchAssistant, syncAppend],
   865→  );
   866→
   867→  const handleSubmit = () => {
   868→    const trimmed = input.trim();
   869→    if (!trimmed || pending) return;
   870→    setInput("");
   871→    if (textareaRef.current) textareaRef.current.style.height = "auto";
   872→    const sessionId = ensureActiveSession(trimmed);
   873→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: trimmed }]);
   874→    syncAppend(sessionId, { role: "user", content: trimmed }, titleFromQuestion(trimmed));
   875→    void send(trimmed, messages);
   876→  };
   877→
   878→  const handleStop = () => {
   879→    abortRef.current?.abort();
   880→  };
   881→
   882→  const handleRetry = () => {
   883→    const q = lastQuestionRef.current;
   884→    if (!q || pending) return;
   885→    const sessionId = ensureActiveSession();
   886→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: q }]);
   887→    syncAppend(sessionId, { role: "user", content: q });
   888→    void send(q, messages);
   889→  };
   890→
   891→  const copyAnswer = async (m: ChatMessage) => {
   892→    try {
   893→      await navigator.clipboard.writeText(m.content);
   894→      setCopiedId(m.id);
   895→      window.setTimeout(() => setCopiedId((c) => (c === m.id ? null : c)), 1500);
   896→    } catch {
   897→      /* clipboard unavailable — silently ignore */
   898→    }
   899→  };
   900→
   901→  /** Download the Q&A transcript as a shareable Markdown file. */
   902→  const exportTranscript = () => {
   903→    if (messages.length === 0 || pending) return;
   904→    try {
   905→      const localeTag = business.locale === "fa" ? "fa-IR" : "en-GB";
   906→      const nowLabel = new Intl.DateTimeFormat(localeTag, {
   907→        dateStyle: "long",
   908→        timeStyle: "short",
   909→      }).format(new Date());
   910→      const windowLabel = t(`window_${days}` as "window_7");
   911→
   912→      const lines: string[] = [];
   913→      lines.push(`# ${t("title")} — ${t("exportTranscript")}`);
   914→      lines.push("");
   915→      lines.push(`- **${business.name}**`);
   916→      lines.push(`- ${t("windowLabel")}: ${windowLabel}`);
   917→      lines.push(`- ${t("citationsGenerated", { time: nowLabel })}`);
   918→      lines.push(`- ${t("groundedNote")}`);
   919→      lines.push("");
   920→      lines.push("---");
   921→
   922→      let qIndex = 0;
   923→      for (let i = 0; i < messages.length; i++) {
   924→        const m = messages[i];
   925→        if (m.role !== "user" || m.failed) continue;
   926→        const answer = messages[i + 1];
   927→        if (!answer || answer.role !== "assistant" || answer.failed) continue;
   928→        qIndex += 1;
   929→        lines.push("");
   930→        lines.push(`## ${qIndex}. ${m.content}`);
   931→        lines.push("");
   932→        lines.push(answer.content);
   933→        const meta = answer.meta;
   934→        if (meta) {
   935→          const c = meta.counts;
   936→          const bits = [
   937→            t(`window_${meta.days === 7 ? "7" : meta.days === 30 ? "30" : "90"}` as "window_7"),
   938→            `${meta.rangeFrom} → ${meta.rangeTo}`,
   939→            ...(c
   940→              ? [
   941→                  t("citationsSales", { count: c.sales }),
   942→                  t("citationsExpenses", { count: c.expenses }),
   943→                  t("citationsOpenInvoices", { count: c.openInvoices }),
   944→                  t("citationsPayments", { count: c.recentPayments }),
   945→                ]
   946→              : []),
   947→          ];
   948→          try {
   949→            const time = new Intl.DateTimeFormat(localeTag, {
   950→              hour: "2-digit",
   951→              minute: "2-digit",
   952→            }).format(new Date(meta.generatedAt));
   953→            bits.push(t("citationsGenerated", { time }));
   954→          } catch {
   955→            /* timestamp is decorative in the export */
   956→          }
   957→          lines.push("");
   958→          lines.push(`> ${t("citationsTitle")}: ${bits.join(" · ")}`);
   959→        }
   960→        lines.push("");
   961→        lines.push("---");
   962→      }
   963→
   964→      if (qIndex === 0) return; // nothing answerable to export
   965→
   966→      const md = lines.join("\n");
   967→      const blob = new Blob([md], { type: "text/markdown;charset=utf-8" });
   968→      const url = URL.createObjectURL(blob);
   969→      const a = document.createElement("a");
   970→      a.href = url;
   971→      a.download = `copilot-transcript-${new Date().toISOString().slice(0, 10)}.md`;
   972→      document.body.appendChild(a);
   973→      a.click();
   974→      a.remove();
   975→      URL.revokeObjectURL(url);
   976→      toast({ title: t("exportedToast") });
   977→    } catch {
   978→      toast({ title: t("exportFailed"), variant: "destructive" });
   979→    }
   980→  };
   981→
   982→  const askFollowUp = (question: string) => {
   983→    if (pending) return;
   984→    const sessionId = ensureActiveSession();
   985→    setMessages((prev) => [...prev, { id: nextId(), role: "user", content: question }]);
   986→    syncAppend(sessionId, { role: "user", content: question });
   987→    void send(question, messages);
   988→  };
   989→
   990→  // Sessions that hold an actual transcript (newest first — setSessions keeps
   991→  // the mutated session at the front, hydration sorts by updatedAt).
   992→  // NOTE: hooks must run before the `!canUse` early return below.
   993→  const savedSessions = useMemo(
   994→    () =>
   995→      sessions
   996→        .filter((s) => s.messages.length > 0)
   997→        .sort((a, b) => b.updatedAt - a.updatedAt),
   998→    [sessions],
   999→  );
  1000→  const recentSessions = savedSessions.slice(0, 3);
  1001→
  1002→  if (!canUse) {
  1003→    return (
  1004→      <div className="space-y-6" data-view="copilot">
  1005→        <PageHeader title={t("title")} subtitle={t("subtitle")} />
  1006→        <Card className="mx-auto max-w-md p-8 text-center">
  1007→          <div className="bg-muted mx-auto flex size-12 items-center justify-center rounded-full">
  1008→            <Lock className="text-muted-foreground size-5" aria-hidden="true" />
  1009→          </div>
  1010→          <h2 className="mt-4 text-base font-semibold">{t("lockedTitle")}</h2>
  1011→          <p className="text-muted-foreground mt-1.5 text-sm">{t("lockedDesc")}</p>
  1012→        </Card>
  1013→      </div>
  1014→    );
  1015→  }
  1016→
  1017→  const empty = messages.length === 0;
  1018→  const lastFailed = !pending && messages[messages.length - 1]?.failed === true;
  1019→  const waitingForFirstToken =
  1020→    pending && (messages[messages.length - 1]?.content ?? "").length === 0;
  1021→
  1022→  const formatRelative = (ts: number) => {
  1023→    try {
  1024→      const rtf = new Intl.RelativeTimeFormat(uiLocale === "fa" ? "fa" : "en", {
  1025→        numeric: "auto",
  1026→      });
  1027→      const diffMs = ts - Date.now();
  1028→      const mins = Math.round(diffMs / 60_000);
  1029→      if (Math.abs(mins) < 60) return rtf.format(mins, "minute");
  1030→      const hours = Math.round(diffMs / 3_600_000);
  1031→      if (Math.abs(hours) < 24) return rtf.format(hours, "hour");
  1032→      return rtf.format(Math.round(diffMs / 86_400_000), "day");
  1033→    } catch {
  1034→      return "";
  1035→    }
  1036→  };
  1037→
  1038→  return (
  1039→    <div className="space-y-4" data-view="copilot">
  1040→      <PageHeader
  1041→        title={t("title")}
  1042→        subtitle={t("subtitle")}
  1043→        actions={
  1044→          <span className="text-muted-foreground border-emerald-500/30 bg-emerald-500/5 inline-flex items-center gap-1.5 rounded-full border px-3 py-1 text-xs font-medium text-emerald-700 dark:text-emerald-400">
  1045→            <ShieldCheck className="size-3.5" aria-hidden="true" />
  1046→            {t("readOnly")}
  1047→          </span>
  1048→        }
  1049→      />
  1050→
  1051→      <Card className="flex h-[calc(100svh-15rem)] min-h-[520px] flex-col overflow-hidden">
  1052→        {/* Toolbar */}
  1053→        <div className="flex flex-wrap items-center justify-between gap-2 border-b px-4 py-2.5">
  1054→          <div className="flex items-center gap-2" role="group" aria-label={t("windowLabel")}>
  1055→            <CalendarClock className="text-muted-foreground size-4" aria-hidden="true" />
  1056→            <span className="text-muted-foreground text-xs font-medium">{t("windowLabel")}</span>
  1057→            <div className="bg-muted flex overflow-hidden rounded-lg p-0.5">
  1058→              {WINDOWS.map((w) => (
  1059→                <button
  1060→                  key={w}
  1061→                  type="button"
  1062→                  onClick={() => setDays(w)}
  1063→                  aria-pressed={days === w}
  1064→                  className={cn(
  1065→                    "rounded-md px-2.5 py-1 text-xs font-medium transition-colors",
  1066→                    days === w
  1067→                      ? "bg-background text-foreground shadow-sm"
  1068→                      : "text-muted-foreground hover:text-foreground",
  1069→                  )}
  1070→                >
  1071→                  {t(`window_${w}` as "window_7")}
  1072→                </button>

  1074→            </div>
  1075→          </div>
  1076→          <div className="flex items-center gap-1">
  1077→            <Collapsible open={dataOpen} onOpenChange={setDataOpen}>
  1078→              <CollapsibleTrigger asChild>
  1079→                <Button variant="ghost" size="sm" className="text-muted-foreground h-8 gap-1.5 text-xs">
  1080→                  <ListTree className="size-3.5" aria-hidden="true" />
  1081→                  {t("dataIncludedTitle")}
  1082→                  <ChevronDown className={cn("size-3.5 transition-transform", dataOpen && "rotate-180")} aria-hidden="true" />
  1083→                </Button>
  1084→              </CollapsibleTrigger>
  1085→            </Collapsible>
  1086→            {/* Conversation history — switch / resume / delete transcripts. */}
  1087→            <DropdownMenu>
  1088→              <DropdownMenuTrigger asChild disabled={pending}>
  1089→                <Button

  1095→                >
  1096→                  <History className="size-3.5" aria-hidden="true" />
  1097→                  <span className="hidden sm:inline">{t("sessionHistory")}</span>
  1098→                  {savedSessions.length > 0 ? (

  1100→                      {savedSessions.length}

  1103→                </Button>
  1104→              </DropdownMenuTrigger>
  1105→              <DropdownMenuContent align="end" className="w-80">

  1108→                </DropdownMenuLabel>
  1109→                <DropdownMenuSeparator />
  1110→                {savedSessions.length === 0 ? (
  1111→                  <p className="text-muted-foreground px-2 py-3 text-xs">
  1112→                    {t("noSessions")}
  1113→                  </p>
  1114→                ) : (
  1115→                  <>
  1116→                    <div className="max-h-72 overflow-y-auto">
  1117→                      {savedSessions.map((s) => (
  1118→                        <DropdownMenuItem
