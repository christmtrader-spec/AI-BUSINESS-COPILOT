    21→ *   the short-lived in-memory cache (§67) serves UI read endpoints only
    22→ *   and is invalidated on every mutation.

    58→/**
    59→ * Prompt 32 §72/§76/§84 — post-commit billing notification queue.
    60→ * Domain transaction → commit → notification (NEVER inside the tx): each
    61→ * mutation collects the durable SubscriptionEvent ids it created and the
    62→ * mapper runs only after `db.$transaction` has resolved successfully. A
    63→ * rolled-back transaction therefore emits NO notification (the event rows
    64→ * it queued are discarded with it).
    65→ */
    66→type PendingBillingNotification = {
    67→  eventId: string;
    68→  type: SubscriptionEventType;
    69→  toPlanCode: string | null;
    70→};
    71→
    72→function flushBillingNotifications(
    73→  businessId: string,
    74→  queue: readonly PendingBillingNotification[],
    75→): void {
    76→  if (queue.length === 0) return;
    77→  void (async () => {
    78→    const { notifySubscriptionEvent } = await import("@/lib/notifications");
    79→    for (const item of queue) {
    80→      notifySubscriptionEvent({
    81→        businessId,
    82→        eventId: item.eventId,
    83→        type: item.type,
    84→        toPlanCode: item.toPlanCode,
    85→      });
    86→    }
    87→  })();
    88→}

   228→/**
   229→ * Get-or-create the deterministic FREE subscription for a business
   230→ * (§13/§14/§118). Idempotent, transactional, auditable — used at business
   231→ * creation, by the seed backfill and as the lazy safety net on read.
   232→ */
   233→export async function ensureSubscription(
   234→  businessId: string,
   235→  opts: { actorId?: string | null; source?: string; tx?: Tx } = {},
   236→): Promise<SubRow> {
   237→  const run = async (tx: Tx): Promise<SubRow> => {
   238→    const existing = await loadSubscription(tx, businessId);
   239→    if (existing) return existing;
   240→
   241→    const business = await tx.business.findUnique({ where: { id: businessId } });
   242→    if (!business) {
   243→      throw new ApiError(403, "FORBIDDEN", "You do not have access to this business.");
   244→    }
   245→
   246→    const free = await tx.plan.findUnique({ where: { code: FREE_PLAN_CODE } });
   247→    if (!free || !free.active) {
   248→      // Catalog corruption — fail closed loudly, never invent entitlements.
   249→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   250→    }
   251→
   252→    const now = new Date();
   253→    const created = await tx.subscription.create({
   254→      data: {
   255→        businessId,
   256→        planId: free.id,
   257→        status: "ACTIVE",
   258→        billingPeriod: "MONTHLY",
   259→        currentPeriodStart: now,
   260→        currentPeriodEnd: periodEnd(now, "MONTHLY"),
   261→      },
   262→      include: { plan: true },
   263→    });
   264→    await tx.subscriptionEvent.create({
   265→      data: {
   266→        businessId,
   267→        subscriptionId: created.id,
   268→        type: "CREATED",
   269→        toStatus: "ACTIVE",
   270→        toPlanCode: free.code,
   271→        dataJson: JSON.stringify({ source: opts.source ?? "ensure" }),
   272→        actorId: opts.actorId ?? null,
   273→      },
   274→    });
   275→    await audit(
   276→      {
   277→        businessId,
   278→        actorId: opts.actorId ?? null,
   279→        action: "billing.subscription_created",
   280→        entityType: "Subscription",
   281→        entityId: created.id,
   282→        after: { planCode: free.code, status: "ACTIVE", source: opts.source ?? "ensure" },
   283→      },
   284→      tx,
   285→    );
   286→    return created as SubRow;
   287→  };
   288→
   289→  if (opts.tx) return run(opts.tx);
   290→  return db.$transaction(run, { isolationLevel: "Serializable" });
   291→}
   292→
   293→/**
   294→ * Lazy expiry (§17/§91): when a read finds a subscription whose grace/
   295→ * trial/period has passed, transition it to EXPIRED with an event —
   296→ * transactional, only if still in an expirable state.
   297→ */
   298→async function expireSubscriptionIfDue(sub: SubRow): Promise<SubRow> {
   299→  const decision = decidePlan(sub);
   300→  if (!("expired" in decision) || !decision.expired) return sub;
   301→
   302→  const updated = await db.$transaction(
   303→    async (tx) => {
   304→      const current = await loadSubscription(tx, sub.businessId);
   305→      if (!current) return current;
   306→      const re = decidePlan(current);
   307→      if (!("expired" in re) || !re.expired) return current;
   308→      const row = await tx.subscription.update({
   309→        where: { id: current.id },
   310→        data: { status: "EXPIRED", endedAt: new Date() },
   311→        include: { plan: true },
   312→      });
   313→      const expiredEventId = await tx.subscriptionEvent.create({
   314→        data: {
   315→          businessId: current.businessId,
   316→          subscriptionId: current.id,
   317→          type: "EXPIRED",
   318→          fromStatus: current.status,
   319→          toStatus: "EXPIRED",
   320→          fromPlanCode: current.plan.code,
   321→        },
   322→        select: { id: true },
   323→      });
   324→      await audit(
   325→        {
   326→          businessId: current.businessId,
   327→          action: "billing.subscription_expired",
   328→          entityType: "Subscription",
   329→          entityId: current.id,
   330→          before: { status: current.status },
   331→          after: { status: "EXPIRED" },
   332→        },
   333→        tx,
   334→      );
   335→      return { row, expiredEventId: expiredEventId.id, planCode: current.plan.code };
   336→    },
   337→    { isolationLevel: "Serializable" },
   338→  );
   339→  invalidateBillingCache(sub.businessId);
   340→  if (updated && typeof updated === "object" && "expiredEventId" in updated) {
   341→    // Prompt 32 §86 — EXPIRED is a customer-visible state (post-commit).
   342→    flushBillingNotifications(sub.businessId, [
   343→      { eventId: updated.expiredEventId, type: "EXPIRED", toPlanCode: updated.planCode },
   344→    ]);
   345→  }
   346→  return updated && typeof updated === "object" && "row" in updated ? updated.row : updated;
   347→}

   349→/**
   350→ * Resolve the effective plan for a subscription row (§17 + §13 fallback).
   351→ * Returns the plan whose entitlements apply.
   352→ */
   353→async function resolveEffectivePlan(sub: SubRow, tx: Tx | typeof db) {
   354→  const decision = decidePlan(sub);
   355→  if (decision.kind === "own-plan") return { plan: sub.plan, entitled: true };
   356→  const free = await tx.plan.findUnique({ where: { code: FREE_PLAN_CODE } });
   357→  if (!free) {
   358→    throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   359→  }
   360→  return { plan: free, entitled: decision.entitled };
   361→}

   363→/** Resolve a plan's full entitlement map (fail-closed on corruption). */
   364→function entitlementsFromRows(
   365→  rows: Array<{ entitlement: { code: string; type: string; unit: string | null; description: string }; value: number | null; boolValue: boolean }>,
   366→): Record<EntitlementCode, ResolvedEntitlement> {
   367→  const map = {} as Record<EntitlementCode, ResolvedEntitlement>;
   368→  // Fail-closed defaults first (§68/§93): a MISSING catalog row can never
   369→  // behave as "allowed/unlimited". The seed guarantees completeness; if a
   370→  // row is missing anyway the gate blocks loudly.
   371→  for (const spec of ENTITLEMENTS) {
   372→    map[spec.code] = {
   373→      code: spec.code,
   374→      type: spec.type,
   375→      unit: spec.unit,
   376→      value: spec.type === "LIMIT" ? 0 : false,
   377→      description: spec.description,
   378→    };
   379→  }
   380→  for (const row of rows) {
   381→    if (!isEntitlementCode(row.entitlement.code)) continue;
   382→    const spec = ENTITLEMENTS.find((s) => s.code === row.entitlement.code)!;
   383→    map[spec.code] = {
   384→      code: spec.code,
   385→      type: spec.type,
   386→      unit: spec.unit,
   387→      value:
   388→        spec.type === "LIMIT"
   389→          ? row.value // number, or null = UNLIMITED
   390→          : row.boolValue,
   391→      description: spec.description,
   392→    };
   393→  }
   394→  return map;
   395→}

   397→/** Fresh (uncached) entitlement context — the gate-enforcement loader. */
   398→export async function getEntitlementContextFresh(businessId: string): Promise<EntitlementContext> {
   399→  try {
   400→    let sub = await loadSubscription(db, businessId);
   401→    if (!sub) {
   402→      sub = await ensureSubscription(businessId, { source: "lazy-backfill" });
   403→      invalidateBillingCache(businessId);
   404→    }
   405→    sub = await expireSubscriptionIfDue(sub);
   406→    const { plan, entitled } = await resolveEffectivePlan(sub, db);
   407→    const rows = await db.planEntitlement.findMany({
   408→      where: { planId: plan.id },
   409→      include: { entitlement: true },
   410→    });
   411→    return {
   412→      businessId,
   413→      subscription: subscriptionDto(sub, plan.code, entitled),
   414→      entitlements: entitlementsFromRows(rows),
   415→    };
   416→  } catch (error) {
   417→    if (error instanceof ApiError) throw error;
   418→    console.error("[billing] context lookup failed:", error);
   419→    throw new BillingError("BILLING_UNAVAILABLE", "Billing state is temporarily unavailable. Please try again.");
   420→  }
   421→}

   423→/** Same as getEntitlementContextFresh but inside an existing transaction. */
   424→export async function getEntitlementContextTx(tx: Tx, businessId: string): Promise<EntitlementContext> {
   425→  let sub = await loadSubscription(tx, businessId);
   426→  if (!sub) {
   427→    sub = await ensureSubscription(businessId, { source: "lazy-backfill", tx });
   428→  }
   429→  const { plan, entitled } = await resolveEffectivePlan(sub, tx);
   430→  const rows = await tx.planEntitlement.findMany({
   431→    where: { planId: plan.id },
   432→    include: { entitlement: true },
   433→  });
   434→  return {
   435→    businessId,
   436→    subscription: subscriptionDto(sub, plan.code, entitled),
   437→    entitlements: entitlementsFromRows(rows),
   438→  };
   439→}

   441→/** Cached context for UI read endpoints (§67); 15s TTL, mutation-invalidated. */
   442→export async function getEntitlementContext(businessId: string): Promise<EntitlementContext> {
   443→  const hit = cache.get(businessId);
   444→  if (hit && Date.now() - hit.at < CACHE_TTL_MS) return hit.ctx;
   445→  const ctx = await getEntitlementContextFresh(businessId);
   446→  cache.set(businessId, { at: Date.now(), ctx });
   447→  return ctx;
   448→}

   454→export function hasEntitlementIn(ctx: EntitlementContext, code: EntitlementCode): boolean {
   455→  const e = ctx.entitlements[code];
   456→  return e !== undefined && e.type === "BOOLEAN" && e.value === true;
   457→}
   458→
   459→export function getLimitIn(ctx: EntitlementContext, code: EntitlementCode): number | null {
   460→  const e = ctx.entitlements[code];
   461→  if (!e || e.type !== "LIMIT") return 0; // unknown/misconfigured → fail closed
   462→  return e.value === null ? null : (e.value as number);
   463→}

   465→/** §25/§27 — throws the structured error; `currentUsage` counted by caller. */
   466→export function assertWithinLimitIn(
   467→  ctx: EntitlementContext,
   468→  code: EntitlementCode,
   469→  currentUsage: number,
   470→): void {
   471→  const limit = getLimitIn(ctx, code);
   472→  if (limit === null) return; // unlimited (e.g. ENTERPRISE)
   473→  if (currentUsage >= limit) throw limitReachedError(code, limit, currentUsage);
   474→}

   512→// ---------------------------------------------------------------------------
   513→// Usage counting (§105/§106 — documented counting policies)
   514→// ---------------------------------------------------------------------------
   515→
   516→export async function countEntitlementUsage(
   517→  businessId: string,
   518→  ownerId: string,
   519→  codes: readonly EntitlementCode[],
   520→): Promise<Record<string, { used: number; limit: number | null }>> {
   521→  const ctx = await getEntitlementContext(businessId);
   522→  const out: Record<string, { used: number; limit: number | null }> = {};
   523→  for (const code of codes) {
   524→    const limit = getLimitIn(ctx, code);
   525→    let used = 0;
   526→    switch (code) {
   527→      case "CUSTOMERS_MAX":
   528→        used = await db.customer.count({ where: { businessId, deletedAt: null } });
   529→        break;
   530→      case "PRODUCTS_MAX":
   531→        // Deactivation (isActive=false) is a catalog state, NOT deletion —
   532→        // both count (documented in entitlements.md).
   533→        used = await db.product.count({ where: { businessId, deletedAt: null } });
   534→        break;
   535→      case "TEAM_MEMBERS_MAX":
   536→        // Active memberships only; owner included (§108); pending invites
   537→        // never count (§31).
   538→        used = await db.businessMember.count({ where: { businessId } });
   539→        break;
   540→      case "BUSINESSES_MAX":
   541→        // Businesses the user OWNS (§107) — staff memberships don't count.
   542→        used = await db.business.count({ where: { ownerId } });
   543→        break;
   544→      default:
   545→        used = 0;
   546→    }
   547→    out[code] = { used, limit };
   548→  }
   549→  return out;
   550→}

   552→// ---------------------------------------------------------------------------
   553→// Business-creation limit (§30/§107) — a USER-level check
   554→// ---------------------------------------------------------------------------
   555→
   556→/**
   557→ * BUSINESSES_MAX is special: it caps how many businesses a USER may OWN,
   558→ * so it cannot be read from one business's subscription. Deterministic
   559→ * policy (documented in entitlements.md): the effective limit is the
   560→ * MAXIMUM BUSINESSES_MAX across the plans of businesses the user already
   561→ * owns — upgrading any owned business lifts the cap; a user with no
   562→ * businesses gets the FREE limit. Count + check run inside the caller's
   563→ * transaction (race-safe on SQLite, §112).
   564→ */
   565→export async function assertWithinBusinessesLimitTx(tx: Tx, userId: string): Promise<void> {
   566→  const owned = await tx.business.findMany({
   567→    where: { ownerId: userId },
   568→    select: {
   569→      subscription: {
   570→        include: {
   571→          plan: { include: { planEntitlements: { include: { entitlement: true } } } },
   572→        },
   573→      },
   574→    },
   575→  });
   576→
   577→  let limit = 0;
   578→  for (const b of owned) {
   579→    const sub = b.subscription;
   580→    if (!sub) continue; // grandfathered mid-backfill rows contribute nothing
   581→    const row = sub.plan.planEntitlements.find((pe) => pe.entitlement.code === "BUSINESSES_MAX");
   582→    if (!row) continue; // fail-closed: corrupt plan contributes nothing
   583→    const planLimit = row.value; // null = unlimited
   584→    if (planLimit === null) return; // any unlimited plan lifts the cap
   585→    limit = Math.max(limit, planLimit);
   586→  }
   587→  if (limit === 0) {
   588→    // No owned business (or none with a valid subscription) → FREE limit.
   589→    const free = await tx.plan.findUnique({
   590→      where: { code: FREE_PLAN_CODE },
   591→      include: { planEntitlements: { include: { entitlement: true } } },
   592→    });
   593→    const row = free?.planEntitlements.find((pe) => pe.entitlement.code === "BUSINESSES_MAX");
   594→    if (!row) {
   595→      // BUGFIX P51-R1-USER-BUG3: an EMPTY/CORRUPT catalog is an OPERATIONAL
   596→      // failure, not a user-facing plan limit. The previous `row?.value ?? 0`
   597→      // made every fresh user (0 owned businesses) hit `0 >= 0` and receive a
   598→      // FALSE "plan limit reached" alert on their very first business. Honest
   599→      // fail-closed per §68 — same semantics as the other catalog guards above.
   600→      throw new BillingError("BILLING_UNAVAILABLE", "Billing catalog is not initialized.");
   601→    }
   602→    if (row.value === null) return; // FREE configured unlimited — mirrors getLimitIn (§472)
   603→    limit = row.value;
   604→  }
   605→
   606→  if (owned.length >= limit) {
   607→    throw limitReachedError("BUSINESSES_MAX", limit, owned.length);
   608→  }
   609→}

   615→export type SubscriptionMutationResult = {
   616→  subscription: SubscriptionDto;
   617→  /** false when the call was an idempotent no-op (§48). */
   618→  changed: boolean;
   619→};
   620→
   621→function assertTransition(from: SubscriptionStatus, to: SubscriptionStatus): void {
   622→  if (from === to) return;
   623→  if (!isValidTransition(from, to)) {
   624→    throw new BillingError(
   625→      "INVALID_SUBSCRIPTION_STATE",
   626→      `Transition ${from} → ${to} is not allowed.`,
   627→      { from, to, allowed: VALID_TRANSITIONS[from] },
   628→    );
   629→  }
   630→}

   632→async function eventAndAudit(
   633→  tx: Tx,
   634→  input: {
   635→    businessId: string;
   636→    subscriptionId: string;
   637→    type: SubscriptionEventType;
   638→    fromStatus?: string | null;
   639→    toStatus?: string | null;
   640→    fromPlanCode?: string | null;
   641→    toPlanCode?: string | null;
   642→    data?: Record<string, unknown>;
   643→    actorId?: string | null;
   644→    auditAction: string;
   645→    auditBefore?: unknown;
   646→    auditAfter?: unknown;
   647→  },
   648→): Promise<string> {
   649→  const created = await tx.subscriptionEvent.create({
   650→    data: {
   651→      businessId: input.businessId,
   652→      subscriptionId: input.subscriptionId,
   653→      type: input.type,
   654→      fromStatus: input.fromStatus ?? null,
   655→      toStatus: input.toStatus ?? null,
   656→      fromPlanCode: input.fromPlanCode ?? null,
   657→      toPlanCode: input.toPlanCode ?? null,
   658→      dataJson: input.data ? JSON.stringify(input.data) : null,
   659→      actorId: input.actorId ?? null,
   660→    },
   661→    select: { id: true },
   662→  });
   663→  await audit(
   664→    {
   665→      businessId: input.businessId,
   666→      actorId: input.actorId ?? null,
   667→      action: input.auditAction,
   668→      entityType: "Subscription",
   669→      entityId: input.subscriptionId,
   670→      before: input.auditBefore,
   671→      after: input.auditAfter,
   672→    },
   673→    tx,
   674→  );
   675→  // Prompt 32 — the durable event row id feeds the post-commit notification
   676→  // mapper (§72: domain transaction → commit → notification; the caller
   677→  // flushes AFTER this transaction resolves, §332).
   678→  return created.id;
   679→}

   681→/**
   682→ * Internal plan change (§41/§43/§82/§83): validates the target plan,
   683→ * verifies state transitions, records PLAN_CHANGED (+ REACTIVATED when it
   684→ * doubles as reactivation), all in ONE transaction. NO money changes hands
   685→ * and NOTHING is labeled as payment (§83). Downgrades never delete data —
   686→ * they only cap future growth (§42).
   687→ */
   688→export async function changePlan(
   689→  businessId: string,
   690→  targetPlanCode: string,
   691→  opts: { actorId?: string | null; source?: string } = {},
   692→): Promise<SubscriptionMutationResult> {
   693→  const billingNotifyQueue: PendingBillingNotification[] = [];
   694→  const result = await db.$transaction(
   695→    async (tx) => {
   696→      let sub = await loadSubscription(tx, businessId);
   697→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   698→
   699→      const target = await tx.plan.findUnique({ where: { code: targetPlanCode } });
   700→      if (!target || !target.active) {
   701→        throw new BillingError("INVALID_PLAN", "The requested plan does not exist or is inactive.", {
   702→          planCode: targetPlanCode,
   703→        });
   704→      }
   705→
   706→      if (sub.planId === target.id) {
   707→        // §48 idempotency — same plan → no event, no mutation.
   708→        return { sub, changed: false as const };
   709→      }
   710→
   711→      const fromStatus = asSubscriptionStatus(sub.status);
   712→      const wasCanceled = fromStatus === "CANCELED";
   713→      const reactivating = wasCanceled || fromStatus === "EXPIRED" || fromStatus === "PAUSED";
   714→      // Any plan change re-engages the subscription (a deliberate renewal
   715→      // decision by an authorized user — never automatic, §45).
   716→      assertTransition(fromStatus, "ACTIVE");
   717→
   718→      if (reactivating) {
   719→        billingNotifyQueue.push({
   720→          eventId: await eventAndAudit(tx, {
   721→            businessId,
   722→            subscriptionId: sub.id,
   723→            type: "REACTIVATED",
   724→            fromStatus,
   725→            toStatus: "ACTIVE",
   726→            fromPlanCode: sub.plan.code,
   727→            toPlanCode: target.code,
   728→            actorId: opts.actorId ?? null,
   729→            auditAction: "billing.subscription_reactivated",
   730→            auditBefore: { status: fromStatus, planCode: sub.plan.code },
   731→            auditAfter: { status: "ACTIVE", planCode: target.code },
   732→          }),
   733→          type: "REACTIVATED",
   734→          toPlanCode: target.code,
   735→        });
   736→      }
   737→
   738→      const now = new Date();
   739→      const period =
   740→        sub.currentPeriodEnd && sub.currentPeriodEnd.getTime() > now.getTime()
   741→          ? { start: sub.currentPeriodStart ?? now, end: sub.currentPeriodEnd }
   742→          : { start: now, end: periodEnd(now, (isBillingPeriod(sub.billingPeriod) ? sub.billingPeriod : "MONTHLY") as BillingPeriod) };
   743→
   744→      const updated = await tx.subscription.update({
   745→        where: { id: sub.id },
   746→        data: {
   747→          planId: target.id,
   748→          status: "ACTIVE",
   749→          cancelAtPeriodEnd: false,
   750→          canceledAt: null,
   751→          endedAt: null,
   752→          currentPeriodStart: period.start,
   753→          currentPeriodEnd: period.end,
   754→        },
   755→        include: { plan: true },
   756→      });
   757→
   758→      billingNotifyQueue.push({
   759→        eventId: await eventAndAudit(tx, {
   760→          businessId,
   761→          subscriptionId: sub.id,
   762→          type: "PLAN_CHANGED",
   763→          fromStatus,
   764→          toStatus: "ACTIVE",
   765→          fromPlanCode: sub.plan.code,
   766→          toPlanCode: target.code,
   767→          data: { source: opts.source ?? "settings" },
   768→          actorId: opts.actorId ?? null,
   769→          auditAction: "billing.plan_changed",
   770→          auditBefore: { planCode: sub.plan.code, status: fromStatus },
   771→          auditAfter: { planCode: target.code, status: "ACTIVE" },
   772→        }),
   773→        type: "PLAN_CHANGED",
   774→        toPlanCode: target.code,
   775→      });
   776→
   777→      return { sub: updated as SubRow, changed: true as const };
   778→    },
   779→    { isolationLevel: "Serializable" },
   780→  );
   781→
   782→  invalidateBillingCache(businessId);
   783→  flushBillingNotifications(businessId, billingNotifyQueue);
   784→  return {
   785→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
   786→    changed: result.changed,
   787→  };
   788→}
   789→
   790→/**
   791→ * Cancellation (§44): sets cancelAtPeriodEnd + status CANCELED — access
   792→ * continues until currentPeriodEnd, then lazy expiry moves it to EXPIRED.
   793→ * Idempotent (§48): a second cancel is a no-op. Data is never touched.
   794→ */
   795→export async function cancelSubscription(
   796→  businessId: string,
   797→  opts: { actorId?: string | null } = {},
   798→): Promise<SubscriptionMutationResult> {
   799→  const billingNotifyQueue: PendingBillingNotification[] = [];
   800→  const result = await db.$transaction(
   801→    async (tx) => {
   802→      const sub = await loadSubscription(tx, businessId);
   803→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   804→
   805→      const fromStatus = asSubscriptionStatus(sub.status);
   806→      if (fromStatus === "CANCELED" && sub.cancelAtPeriodEnd) {
   807→        return { sub, changed: false as const };
   808→      }
   809→      assertTransition(fromStatus, "CANCELED");
   810→
   811→      const updated = await tx.subscription.update({
   812→        where: { id: sub.id },
   813→        data: { status: "CANCELED", cancelAtPeriodEnd: true, canceledAt: new Date() },
   814→        include: { plan: true },
   815→      });
   816→      billingNotifyQueue.push({
   817→        eventId: await eventAndAudit(tx, {
   818→          businessId,
   819→          subscriptionId: sub.id,
   820→          type: "CANCELED",
   821→          fromStatus,
   822→          toStatus: "CANCELED",
   823→          fromPlanCode: sub.plan.code,
   824→          actorId: opts.actorId ?? null,
   825→          auditAction: "billing.subscription_canceled",
   826→          auditBefore: { status: fromStatus, planCode: sub.plan.code },
   827→          auditAfter: { status: "CANCELED", cancelAtPeriodEnd: true },
   828→        }),
   829→        type: "CANCELED",
   830→        toPlanCode: sub.plan.code,
   831→      });
   832→      return { sub: updated as SubRow, changed: true as const };
   833→    },
   834→    { isolationLevel: "Serializable" },
   835→  );
   836→  invalidateBillingCache(businessId);
   837→  flushBillingNotifications(businessId, billingNotifyQueue);
   838→  const decision = decidePlan(result.sub);
   839→  return {
   840→    subscription: subscriptionDto(
   841→      result.sub,
   842→      decision.kind === "own-plan" ? result.sub.plan.code : FREE_PLAN_CODE,
   843→      decision.entitled,
   844→    ),
   845→    changed: result.changed,
   846→  };
   847→}

   849→/**
   850→ * Reactivation (§45): only a CANCELED subscription with remaining grace can
   851→ * return to ACTIVE. EXPIRED requires the explicit changePlan renewal path.
   852→ * Idempotent: reactivating an ACTIVE subscription is a no-op.
   853→ */
   854→export async function reactivateSubscription(
   855→  businessId: string,
   856→  opts: { actorId?: string | null } = {},
   857→): Promise<SubscriptionMutationResult> {
   858→  const billingNotifyQueue: PendingBillingNotification[] = [];
   859→  const result = await db.$transaction(
   860→    async (tx) => {
   861→      const sub = await loadSubscription(tx, businessId);
   862→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
   863→
   864→      const fromStatus = asSubscriptionStatus(sub.status);
   865→      if (fromStatus === "ACTIVE") return { sub, changed: false as const };
   866→      // §45/§46: reactivation is ONLY the CANCELED → ACTIVE path (undo the
   867→      // cancellation while the period still runs). An EXPIRED subscription
   868→      // returns exclusively through changePlan — a deliberate, authorized
   869→      // renewal — and PAUSED resumes through a dedicated admin flow that
   870→      // does not exist yet. Arbitrary hopping is refused.
   871→      if (fromStatus === "EXPIRED" || fromStatus === "PAUSED") {
   872→        throw new BillingError(
   873→          "INVALID_SUBSCRIPTION_STATE",
   874→          "This subscription state cannot be reactivated. Use a plan change to renew.",
   875→          { from: fromStatus, allowed: VALID_TRANSITIONS[fromStatus] },
   876→        );
   877→      }
   878→      assertTransition(fromStatus, "ACTIVE");
   879→
   880→      const updated = await tx.subscription.update({
   881→        where: { id: sub.id },
   882→        data: { status: "ACTIVE", cancelAtPeriodEnd: false, canceledAt: null, endedAt: null },
   883→        include: { plan: true },
   884→      });
   885→      billingNotifyQueue.push({
   886→        eventId: await eventAndAudit(tx, {
   887→          businessId,
   888→          subscriptionId: sub.id,
   889→          type: "REACTIVATED",
   890→          fromStatus,
   891→          toStatus: "ACTIVE",
   892→          fromPlanCode: sub.plan.code,
   893→          actorId: opts.actorId ?? null,
   894→          auditAction: "billing.subscription_reactivated",
   895→          auditBefore: { status: fromStatus },
   896→          auditAfter: { status: "ACTIVE" },
   897→        }),
   898→        type: "REACTIVATED",
   899→        toPlanCode: sub.plan.code,
   900→      });
   901→      return { sub: updated as SubRow, changed: true as const };
   902→    },
   903→    { isolationLevel: "Serializable" },
   904→  );
   905→  invalidateBillingCache(businessId);
   906→  flushBillingNotifications(businessId, billingNotifyQueue);
   907→  return {
   908→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
   909→    changed: result.changed,
   910→  };
   911→}

  1105→      const sub = await loadSubscription(tx, businessId);
  1106→      if (!sub) throw new BillingError("SUBSCRIPTION_NOT_FOUND", "No subscription exists for this business.");
  1107→      const fromStatus = asSubscriptionStatus(sub.status);
  1108→      if (fromStatus !== "TRIALING") {
  1109→        throw new BillingError("INVALID_SUBSCRIPTION_STATE", "The subscription is not in a trial.", {
  1110→          status: fromStatus,
  1111→        });
  1112→      }
  1113→      const updated = await tx.subscription.update({
  1114→        where: { id: sub.id },
  1115→        data: { status: "ACTIVE" },
  1116→        include: { plan: true },
  1117→      });
  1118→      billingNotifyQueue.push({
  1119→        eventId: await eventAndAudit(tx, {
  1120→          businessId,
  1121→          subscriptionId: sub.id,
  1122→          type: "TRIAL_ENDED",
  1123→          fromStatus,
  1124→          toStatus: "ACTIVE",
  1125→          fromPlanCode: sub.plan.code,
  1126→          actorId: opts.actorId ?? null,
  1127→          auditAction: "billing.trial_ended",
  1128→          auditBefore: { status: fromStatus },
  1129→          auditAfter: { status: "ACTIVE" },
  1130→        }),
  1131→        type: "TRIAL_ENDED",
  1132→        toPlanCode: sub.plan.code,
  1133→      });
  1134→      return { sub: updated as SubRow, changed: true as const };
  1135→    },
  1136→    { isolationLevel: "Serializable" },
  1137→  );
  1138→  invalidateBillingCache(businessId);
  1139→  flushBillingNotifications(businessId, billingNotifyQueue);
  1140→  return {
  1141→    subscription: subscriptionDto(result.sub, result.sub.plan.code, true),
  1142→    changed: result.changed,
  1143→  };
  1144→}

  1150→export async function listSubscriptionEvents(
  1151→  businessId: string,
  1152→  opts: { page?: number; pageSize?: number } = {},
  1153→): Promise<{ items: SubscriptionEventDto[]; total: number; page: number; pageSize: number }> {
  1154→  const page = Math.max(1, opts.page ?? 1);
  1155→  const pageSize = Math.min(100, Math.max(1, opts.pageSize ?? 20));
  1156→  const [rows, total] = await Promise.all([
  1157→    db.subscriptionEvent.findMany({
  1158→      where: { businessId },
  1159→      orderBy: { createdAt: "desc" },
  1160→      skip: (page - 1) * pageSize,
  1161→      take: pageSize,
  1162→      include: { actor: { select: { name: true, email: true } } },
  1163→    }),
  1164→    db.subscriptionEvent.count({ where: { businessId } }),
  1165→  ]);
  1166→  return {
  1167→    items: rows.map((r) => ({
  1168→      id: r.id,
  1169→      type: (isSubscriptionEventTypeSafe(r.type) ? r.type : "CREATED") as SubscriptionEventType,
  1170→      fromStatus: r.fromStatus,
  1171→      toStatus: r.toStatus,
  1172→      fromPlanCode: r.fromPlanCode,
  1173→      toPlanCode: r.toPlanCode,
  1174→      data: r.dataJson ? safeParse(r.dataJson) : null,
  1175→      actor: r.actor ? { name: r.actor.name, email: r.actor.email } : null,
  1176→      createdAt: r.createdAt.toISOString(),
  1177→    })),
  1178→    total,
  1179→    page,
  1180→    pageSize,
  1181→  };
  1182→}
  1183→
  1184→function isSubscriptionEventTypeSafe(value: string): boolean {
  1185→  return [
  1186→    "CREATED",
  1187→    "PLAN_CHANGED",
  1188→    "TRIAL_STARTED",
  1189→    "TRIAL_ENDED",
  1190→    "CANCELED",
  1191→    "REACTIVATED",
  1192→    "EXPIRED",
  1193→    "PAUSED",
  1194→    "RESUMED",
  1195→  ].includes(value);
  1196→}
  1197→
  1198→function safeParse(json: string): unknown {
  1199→  try {
  1200→    return JSON.parse(json);
  1201→  } catch {
  1202→    return null;
  1203→  }
  1204→}
