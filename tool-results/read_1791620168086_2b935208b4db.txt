   133→   133→export function mapPureGoodsReturnDomainError(error: unknown): never {

   139→   139→    );

   142→   142→    // §66 — corrupted SOURCE receipt status uses P42's own stable code.
   143→   143→    throw new ApiError(
   144→   144→      500,
   145→   145→      GOODS_RECEIPT_INVALID_PERSISTED_STATUS,
   146→   146→      "Goods receipt data is in an invalid state. Please contact support.",
   147→   147→    );

   151→   151→    throw new ApiError(
   152→   152→      500,

   154→   154→        ? GOODS_RETURN_ERROR_CODES.INVALID_PERSISTED_STOCK
   155→   155→        : GOODS_RETURN_ERROR_CODES.CORRUPT_HISTORY,

   295→   295→ * receipt line. DRAFT/CANCELLED returns never count (§9/§10). Database
   296→   296→ * aggregate (_sum), never a findMany of history into memory; the result
   297→   297→ * is re-validated (exact integer, ≥ 0) and corruption surfaces through
   298→   298→ * the ONE mapping, never clamped.

   300→   300→export async function sumPostedReturnedQtyForReceiptLine(
   301→   301→  tx: Tx,
   302→   302→  goodsReceiptLineId: string,
   303→   303→): Promise<number> {

   305→   305→    where: {
   306→   306→      goodsReceiptLineId,
   307→   307→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only

   309→   309→    _sum: { returnedQty: true },

   317→   317→ * line). Database aggregate with a join filter — no receipt-line

   335→   335→/** §38 — aggregate result re-validation (exact integer, ≥ 0, no clamp). */
   336→   336→function validateReturnedSum(sum: number): number {
   337→   337→  if (!Number.isInteger(sum) || sum < 0) {
   338→   338→    throw mapPureGoodsReturnDomainError(
   339→   339→      new Error("GOODS_RETURN_CORRUPT_HISTORY: negative aggregate"),
   340→   340→    );
   341→   341→  }

   344→   344→
   345→   345→/**
   346→   346→ * §141 — grouped returned-quantity sums for MANY receipt lines (bounded
   347→   347→ * batch, no N+1): one groupBy + (only when needed) one bounded id-in
   348→   348→ * mapping fetch. Used by the receipt-detail return facts.
   349→   349→ */
   350→   350→export async function sumPostedReturnedQtyForReceiptLines(
   351→   351→  db: DbLike,
   352→   352→  goodsReceiptLineIds: string[],
   353→   353→): Promise<Map<string, number>> {
   354→   354→  const result = new Map(goodsReceiptLineIds.map((id) => [id, 0]));
   355→   355→  if (goodsReceiptLineIds.length === 0) return result;
   356→   356→  const grouped = await db.goodsReturnLine.groupBy({
   357→   357→    by: ["goodsReceiptLineId"],
   358→   358→    where: {
   359→   359→      goodsReceiptLineId: { in: goodsReceiptLineIds },
   360→   360→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   361→   361→    },
   362→   362→    _sum: { returnedQty: true },
   363→   363→  });
   364→   364→  for (const g of grouped) {
   365→   365→    result.set(g.goodsReceiptLineId, validateReturnedSum(g._sum.returnedQty ?? 0));
   366→   366→  }
   367→   367→  return result;
   368→   368→}
   369→   369→
   370→   370→/**
   371→   371→ * §77/§141 — grouped returned-quantity sums for MANY PO lines (bounded
   372→   372→ * batch, no N+1): one groupBy (by source receipt line) + one bounded
   373→   373→ * id-in mapping fetch (only receipt lines that actually HAVE returns —
   374→   374→ * never the PO's whole receipt history). Used by the PO net receiving
   375→   375→ * summary and the PO detail aggregate.
   376→   376→ */
   377→   377→export async function sumPostedReturnedQtyForPoLines(
   378→   378→  db: DbLike,
   379→   379→  purchaseOrderLineIds: string[],
   380→   380→): Promise<Map<string, number>> {
   381→   381→  const result = new Map(purchaseOrderLineIds.map((id) => [id, 0]));
   382→   382→  if (purchaseOrderLineIds.length === 0) return result;
   383→   383→  const grouped = await db.goodsReturnLine.groupBy({
   384→   384→    by: ["goodsReceiptLineId"],
   385→   385→    where: {
   386→   386→      goodsReturn: { status: "POSTED" }, // §9 — POSTED only
   387→   387→      goodsReceiptLine: { purchaseOrderLineId: { in: purchaseOrderLineIds } },
   388→   388→    },
   389→   389→    _sum: { returnedQty: true },
   390→   390→  });
   391→   391→  if (grouped.length === 0) return result;
   392→   392→  // §141 — bounded batch: only the receipt lines that appear in return
   393→   393→  // history, fetched by indexed id (NOT the PO's whole receipt set).
   394→   394→  const receiptLineIds = grouped.map((g) => g.goodsReceiptLineId);
   395→   395→  const mapping = await db.goodsReceiptLine.findMany({
   396→   396→    where: { id: { in: receiptLineIds } },
   397→   397→    select: { id: true, purchaseOrderLineId: true },
   398→   398→  });
   399→   399→  const poLineByReceiptLine = new Map(mapping.map((m) => [m.id, m.purchaseOrderLineId]));
   400→   400→  for (const g of grouped) {
   401→   401→    const poLineId = poLineByReceiptLine.get(g.goodsReceiptLineId);
   402→   402→    if (!poLineId) continue; // defensive — FK guarantees existence
   403→   403→    result.set(poLineId, (result.get(poLineId) ?? 0) + validateReturnedSum(g._sum.returnedQty ?? 0));
   404→   404→  }
   405→   405→  return result;
   406→   406→}

   423→   423→export async function createGoodsReturn(input: {
   424→   424→  businessId: string;
   425→   425→  userId: string;
   426→   426→  goodsReceiptId: string;
   427→   427→  reference?: string;
   428→   428→  notes?: string | null;
   429→   429→  ip?: string | null;
   430→   430→  userAgent?: string | null;
   431→   431→}) {
   432→   432→  return withWriteTx(async (tx) => {
   433→   433→    // §13/§16/§17/§18 — the receipt must be tenant-scoped and POSTED; all
   434→   434→    // snapshots derive from ITS frozen columns (never client input, never
   435→   435→    // live Supplier/Product rows).
   436→   436→    const receipt = await loadPostedSourceReceipt(tx, input.businessId, input.goodsReceiptId);
   437→   437→
   438→   438→    // §20 — reference: user-supplied (normalized) or server-generated.
   439→   439→    let reference = input.reference;
   440→   440→    if (reference === undefined) {
   441→   441→      for (let attempt = 0; attempt < 5; attempt += 1) {
   442→   442→        const candidate = generateGoodsReturnReference(new Date(), (max) => randomInt(max));
   443→   443→        const clash = await tx.goodsReturn.findFirst({
   444→   444→          where: { businessId: input.businessId, reference: candidate },
   445→   445→          select: { id: true },
   446→   446→        });
   447→   447→        if (!clash) {
   448→   448→          reference = candidate;
   449→   449→          break;
   450→   450→        }
   451→   451→      }
   452→   452→      if (reference === undefined) {
   453→   453→        // Astronomically unlikely (31^6); fail honestly rather than loop.
   454→   454→        throw new ApiError(
   455→   455→          409,
   456→   456→          GOODS_RETURN_ERROR_CODES.DUPLICATE_REFERENCE,
   457→   457→          "Could not generate a unique reference. Please try again.",
   458→   458→        );
   459→   459→      }
   460→   460→    }
   461→   461→
   462→   462→    try {
   463→   463→      const row = await tx.goodsReturn.create({
   464→   464→        data: {
   465→   465→          businessId: input.businessId, // §25 — server-derived, never client
   466→   466→          goodsReceiptId: receipt.id,
   467→   467→          reference,
   468→   468→          status: "DRAFT", // §10 — preparation only
   469→   469→          supplierNameAtReturn: receipt.supplierNameAtReceipt, // §16 — frozen identity
   470→   470→          goodsReceiptReferenceAtReturn: receipt.reference, // §17 — frozen identity
   471→   471→          purchaseOrderReferenceAtReturn: receipt.purchaseOrderReferenceAtReceipt, // §18 — frozen identity
   472→   472→          notes: input.notes ?? null, // §22 — document data only
   473→   473→          createdByUserId: input.userId, // §56 — server-derived
   474→   474→        },
   475→   475→        select: { id: true, reference: true, status: true },
   476→   476→      });
   477→   477→
   478→   478→      await audit(
   479→   479→        auditInput(input.businessId, input.userId, "goods_return.create", row.id, row.reference, {
   480→   480→          goodsReceiptId: receipt.id,
   481→   481→          goodsReceiptReferenceAtReturn: receipt.reference,
   482→   482→          purchaseOrderReferenceAtReturn: receipt.purchaseOrderReferenceAtReceipt,
   483→   483→          supplierNameAtReturn: receipt.supplierNameAtReceipt,
   484→   484→          status: row.status,
   485→   485→        }),
   486→   486→        tx,
   487→   487→      );
   488→   488→
   489→   489→      return row;
   490→   490→    } catch (error) {
   491→   491→      // §21 — database uniqueness stays authoritative for duplicates.
   492→   492→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   493→   493→        throw new ApiError(
   494→   494→          409,
   495→   495→          GOODS_RETURN_ERROR_CODES.DUPLICATE_REFERENCE,
   496→   496→          "A goods return with this reference already exists.",
   497→   497→        );
   498→   498→      }
   499→   499→      throw error;
   500→   500→    }
   501→   501→  });
   502→   502→}

   506→   506→export async function listGoodsReturns(input: {
   507→   507→  businessId: string;
   508→   508→  q?: string;
   509→   509→  status?: string; // ALL → undefined
   510→   510→  goodsReceiptId?: string;
   511→   511→  from?: string;
   512→   512→  to?: string;
   513→   513→  page: number;
   514→   514→  pageSize: number;
   515→   515→}): Promise<{
   516→   516→  items: GoodsReturnListItemDto[];
   517→   517→  total: number;
   518→   518→  page: number;
   519→   519→  pageSize: number;
   520→   520→}> {
   521→   521→  const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
   522→   522→  const where: Prisma.GoodsReturnWhereInput = {
   523→   523→    businessId: input.businessId, // §67 — tenant scope
   524→   524→    ...(input.status && input.status !== "ALL" ? { status: input.status } : {}),
   525→   525→    ...(input.goodsReceiptId ? { goodsReceiptId: input.goodsReceiptId } : {}),
   526→   526→    ...(input.from || input.to ? { createdAt: dayWindow(input.from, input.to) } : {}),
   527→   527→    ...(input.q
   528→   528→      ? {
   529→   529→          OR: [
   530→   530→            { reference: { contains: input.q } }, // return reference (§73)
   531→   531→            { goodsReceiptReferenceAtReturn: { contains: input.q } }, // receipt reference
   532→   532→            { purchaseOrderReferenceAtReturn: { contains: input.q } }, // PO reference snapshot
   533→   533→            { supplierNameAtReturn: { contains: input.q } }, // supplier snapshot
   534→   534→            {
   535→   535→              lines: {
   536→   536→                some: {
   537→   537→                  OR: [
   538→   538→                    { productNameAtReturn: { contains: input.q } }, // product snapshot
   539→   539→                    { skuAtReturn: { contains: input.q } }, // SKU snapshot
   540→   540→                  ],
   541→   541→                },
   542→   542→              },
   543→   543→            },
   544→   544→          ],
   545→   545→        }
   546→   546→      : {}),
   547→   547→  };
   548→   548→
   549→   549→  const [total, rows] = await Promise.all([
   550→   550→    db.goodsReturn.count({ where }),
   551→   551→    // §72 — DB-level pagination; never findMany + slice.
   552→   552→    db.goodsReturn.findMany({
   553→   553→      where,
   554→   554→      orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   555→   555→      skip: (input.page - 1) * input.pageSize,
   556→   556→      take: input.pageSize,
   557→   557→      select: {
   558→   558→        id: true,
   559→   559→        reference: true,
   560→   560→        status: true,
   561→   561→        goodsReceiptId: true,
   562→   562→        goodsReceiptReferenceAtReturn: true,
   563→   563→        purchaseOrderReferenceAtReturn: true,
   564→   564→        supplierNameAtReturn: true,
   565→   565→        postedAt: true,
   566→   566→        cancelledAt: true,
   567→   567→        createdAt: true,
   568→   568→        lines: {
   569→   569→          // §76 — bounded aggregate relation, not one query per row.
   570→   570→          // take cap+1 PROVES the slice is the whole line set: a return
   571→   571→          // over its own cap is corruption and fails the read below —
   572→   572→          // never a silently truncated total.
   573→   573→          select: { returnedQty: true },
   574→   574→          take: cap + 1,
   575→   575→        },
   576→   576→        _count: { select: { lines: true } },
   577→   577→      },
   578→   578→    }),
   579→   579→  ]);
   580→   580→
   581→   581→  const items = rows.map((row) => {
   582→   582→    // §76 — the capped slice is only trustworthy when the database really
   583→   583→    // contains ≤ cap rows: the +1 row or a _count mismatch means corruption
   584→   584→    // — fail the read honestly, never compute a partial total.
   585→   585→    if (row.lines.length > cap || row._count.lines !== row.lines.length) {
   586→   586→      assertLineCapInvariant(Math.max(row.lines.length, row._count.lines), cap);
   587→   587→    }
   588→   588→    try {
   589→   589→      // §76 — proven-complete slice → the canonical sum is exact.
   590→   590→      const totalReturnedQty = computeReturnedQty(row.lines.map((l) => l.returnedQty));
   591→   591→      return goodsReturnListItemDto({
   592→   592→        id: row.id,
   593→   593→        reference: row.reference,
   594→   594→        status: row.status,
   595→   595→        goodsReceiptId: row.goodsReceiptId,
   596→   596→        goodsReceiptReferenceAtReturn: row.goodsReceiptReferenceAtReturn,
   597→   597→        purchaseOrderReferenceAtReturn: row.purchaseOrderReferenceAtReturn,
   598→   598→        supplierNameAtReturn: row.supplierNameAtReturn,
   599→   599→        lineCount: row._count.lines,
   600→   600→        totalReturnedQty,
   601→   601→        postedAt: row.postedAt,
   602→   602→        cancelledAt: row.cancelledAt,
   603→   603→        createdAt: row.createdAt,
   604→   604→        createdByUserId: "",
   605→   605→        createdByName: null,
   606→   606→        notes: null,
   607→   607→      });
   608→   608→    } catch (error) {
   609→   609→      // §38 — a corrupted quantity inside history surfaces as the stable
   610→   610→      // integrity 500, never a partial row and never a generic 500.
   611→   611→      return mapPureGoodsReturnDomainError(error);
   612→   612→    }
   613→   613→  });
   614→   614→  return { items, total, page: input.page, pageSize: input.pageSize };
   615→   615→}

   751→   751→
   752→   752→export async function addGoodsReturnLine(input: {
   753→   753→  businessId: string;
   754→   754→  userId: string;
   755→   755→  id: string;
   756→   756→  goodsReceiptLineId: string;
   757→   757→  returnedQty: number;
   758→   758→  reasonCode: string;
   759→   759→  ip?: string | null;
   760→   760→  userAgent?: string | null;
   761→   761→}) {
   762→   762→  return withWriteTx(async (tx) => {
   763→   763→    const header = await loadReturnHeader(tx, input.businessId, input.id);
   764→   764→    requireDraft(header.status); // §62 — DRAFT-only
   765→   765→
   766→   766→    // §62/§28 — the source receipt line must belong to THIS return's
   767→   767→    // tenant-scoped receipt. One tenant-scoped join validates
   768→   768→    // return→receipt→line ownership; a foreign line is indistinguishable
   769→   769→    // from an unknown one (§68 → 404, no existence leak).
   770→   770→    const receiptLine = await tx.goodsReceiptLine.findFirst({
   771→   771→      where: { id: input.goodsReceiptLineId, goodsReceiptId: header.goodsReceiptId },
   772→   772→      select: {
   773→   773→        id: true,
   774→   774→        productId: true,
   775→   775→        receivedQty: true,
   776→   776→        productNameAtReceipt: true, // §30 — snapshots copied from the receipt line
   777→   777→        skuAtReceipt: true,
   778→   778→        unitAtReceipt: true,
   779→   779→        product: { select: { id: true, businessId: true } },
   780→   780→      },
   781→   781→    });
   782→   782→    if (!receiptLine) {
   783→   783→      throw new ApiError(
   784→   784→        404,
   785→   785→        GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
   786→   786→        "Goods receipt line not found.",
   787→   787→      );
   788→   788→    }
   789→   789→
   790→   790→    // §62 — the product must exist inside the SAME tenant (belt-and-braces:
   791→   791→    // the FK is Restrict so a hard-deleted product is nearly impossible,
   792→   792→    // but the safe controlled failure is demanded, never fabrication).
   793→   793→    if (receiptLine.product.businessId !== input.businessId) {
   794→   794→      throw new ApiError(
   795→   795→        404,
   796→   796→        GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
   797→   797→        "Goods receipt line not found.",
   798→   798→      );
   799→   799→    }
   800→   800→
   801→   801→    // §62 — the source receipt must STILL be POSTED at line-add time (a
   802→   802→    // receipt cannot leave POSTED, but the defensive re-check keeps the
   803→   803→    // invariant local and fail-loud if that ever changed).
   804→   804→    await loadPostedSourceReceipt(tx, input.businessId, header.goodsReceiptId);
   805→   805→
   806→   806→    const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
   807→   807→    await assertLineCap(tx, header.id, cap); // §62 — defensive count
   808→   808→
   809→   809→    // §62/§35 — ADVISORY cumulative check at line-add time: returnedQty ≤
   810→   810→    // (received − POSTED returned). Draft lines of OTHER returns are
   811→   811→    // deliberately NOT subtracted — over-allocation across drafts is the
   812→   812→    // documented §63 stale-draft scenario and only POST is authoritative.
   813→   813→    const alreadyReturned = await sumPostedReturnedQtyForReceiptLine(tx, receiptLine.id);
   814→   814→    const returnable = computeReturnableQty(receiptLine.receivedQty, alreadyReturned); // §36
   815→   815→    if (exceedsReturnable(receiptLine.receivedQty, alreadyReturned, input.returnedQty)) {
   816→   816→      throw new ApiError(
   817→   817→        409,
   818→   818→        GOODS_RETURN_ERROR_CODES.EXCEEDS_RETURNABLE,
   819→   819→        `Only ${returnable} unit(s) remain returnable on this receipt line.`,
   820→   820→      );
   821→   821→    }
   822→   822→
   823→   823→    try {
   824→   824→      const line = await tx.goodsReturnLine.create({
   825→   825→        data: {
   826→   826→          goodsReturnId: header.id,
   827→   827→          goodsReceiptLineId: receiptLine.id,
   828→   828→          productId: receiptLine.productId, // §28/§47 — derived, never client authority
   829→   829→          returnedQty: input.returnedQty,
   830→   830→          reasonCode: input.reasonCode, // §23 — closed vocabulary (zod)
   831→   831→          // §30 — snapshots copied from the RECEIPT LINE, not live Product.
   832→   832→          productNameAtReturn: receiptLine.productNameAtReceipt,
   833→   833→          skuAtReturn: receiptLine.skuAtReceipt,
   834→   834→          unitAtReturn: receiptLine.unitAtReceipt,
   835→   835→        },
   836→   836→        select: { id: true },
   837→   837→      });
   838→   838→
   839→   839→      await audit(
   840→   840→        auditInput(input.businessId, input.userId, "goods_return.line_add", header.id, header.reference, {
   841→   841→          lineId: line.id, // §132 — safe per-line metadata
   842→   842→          goodsReceiptLineId: receiptLine.id,
   843→   843→          productId: receiptLine.productId,
   844→   844→          productNameAtReturn: receiptLine.productNameAtReceipt,
   845→   845→          returnedQty: input.returnedQty,
   846→   846→          reasonCode: input.reasonCode,
   847→   847→        }),
   848→   848→        tx,
   849→   849→      );
   850→   850→
   851→   851→      return { id: line.id };
   852→   852→    } catch (error) {
   853→   853→      // §29/§75 — DB uniqueness is authoritative for duplicate lines.
   854→   854→      if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002") {
   855→   855→        throw new ApiError(
   856→   856→          409,
   857→   857→          GOODS_RETURN_ERROR_CODES.DUPLICATE_LINE,
   858→   858→          "This goods receipt line is already on the return.",
   859→   859→        );
   860→   860→      }
   861→   861→      throw error;
   862→   862→    }
   863→   863→  });
   864→   864→}

   914→   914→      );
   915→   915→    }
   916→   916→
   917→   917→    // §58-analog — re-read the persisted row after mutation: the response
   918→   918→    // carries the REAL stored snapshot fields and timestamps.
   919→   919→    const updated = await tx.goodsReturnLine.update({
   920→   920→      where: { id: line.id },
   921→   921→      data: { returnedQty: input.returnedQty, reasonCode: input.reasonCode },
   922→   922→      select: {
   923→   923→        id: true,
   924→   924→        goodsReceiptLineId: true,
   925→   925→        productId: true,
   926→   926→        productNameAtReturn: true,
   927→   927→        skuAtReturn: true,
   928→   928→        unitAtReturn: true,
   929→   929→        returnedQty: true,
   930→   930→        reasonCode: true,
   931→   931→        createdAt: true,
   932→   932→        updatedAt: true,
   933→   933→      },
   934→   934→    });
   935→   935→
   936→   936→    await audit(
   937→   937→      auditInput(input.businessId, input.userId, "goods_return.line_update", header.id, header.reference, {
   938→   938→        lineId: line.id, // §132 — before/after quantity + reason only
   939→   939→        returnedQty: { from: line.returnedQty, to: updated.returnedQty },
   940→   940→        reasonCode: { from: line.reasonCode, to: updated.reasonCode },
   941→   941→      }),
   942→   942→      tx,
   943→   943→    );
   944→   944→
   945→   945→    return goodsReturnLineDto(updated);
   946→   946→  });

   987→   987→/* ---------------------------------- post ----------------------------------- */
   988→   988→
   989→   989→/**
   990→   990→ * §43 — THE atomic stock posting transaction. Steps 1-21 in ONE
   991→   991→ * database transaction (§48); a failure at ANY step rolls back EVERYTHING
   992→   992→ * (§52) — no stock change, no adjustment, no status, no audit.
   993→   993→ */
   994→   994→export async function postGoodsReturn(input: {
   995→   995→  businessId: string;
   996→   996→  userId: string;
   997→   997→  id: string;
   998→   998→  ip?: string | null;
   999→   999→  userAgent?: string | null;
  1000→  1000→}) {
  1001→  1001→  return withWriteTx(async (tx) => {
  1002→  1002→    // §43 steps 1-3 — reload return INSIDE the transaction; never trust an
  1003→  1003→    // earlier GET. Corrupted status is surfaced, never Draft. Only
  1004→  1004→    // DRAFT → POSTED (§9 — at most one successful transition §53).
  1005→  1005→    const header = await loadReturnHeader(tx, input.businessId, input.id);
  1006→  1006→    assertTransition(header.status, "POSTED");
  1007→  1007→
  1008→  1008→    // §43 steps 5-6 — the source receipt must STILL exist in this tenant
  1009→  1009→    // and be POSTED (corrupted receipt status never becomes POSTED, §66).
  1010→  1010→    const receipt = await loadPostedSourceReceipt(tx, input.businessId, header.goodsReceiptId);
  1011→  1011→
  1012→  1012→    // §43 steps 7-8 — load the return's own lines (bounded by the document
  1013→  1013→    // cap; cap+1 proves completeness).
  1014→  1014→    const cap = goodsReturnLineCap(process.env.GOODS_RETURN_MAX_LINES);
  1015→  1015→    const lines = await tx.goodsReturnLine.findMany({
  1016→  1016→      where: { goodsReturnId: header.id },
  1017→  1017→      orderBy: [{ createdAt: "asc" }, { id: "asc" }],
  1018→  1018→      take: cap + 1, // cap+1 proves completeness
  1019→  1019→      select: {
  1020→  1020→        id: true,
  1021→  1021→        goodsReceiptLineId: true,
  1022→  1022→        productId: true,
  1023→  1023→        returnedQty: true,
  1024→  1024→        productNameAtReturn: true,
  1025→  1025→      },
  1026→  1026→    });
  1027→  1027→    if (lines.length > cap) {
  1028→  1028→      // Impossible by §33/§71 — if it ever happens, fail honestly.
  1029→  1029→      throw new ApiError(
  1030→  1030→        422,
  1031→  1031→        GOODS_RETURN_ERROR_CODES.LINE_LIMIT,
  1032→  1032→        "Goods return line limit exceeded.",
  1033→  1033→      );
  1034→  1034→    }
  1035→  1035→
  1036→  1036→    // §61 — a return with zero lines can never post.
  1037→  1037→    if (lines.length === 0) {
  1038→  1038→      throw new ApiError(
  1039→  1039→        422,
  1040→  1040→        GOODS_RETURN_ERROR_CODES.EMPTY,
  1041→  1041→        "Add at least one line before posting this return.",
  1042→  1042→      );
  1043→  1043→    }
  1044→  1044→
  1045→  1045→    // §43 step 9 — every line's source receipt line must belong to THIS
  1046→  1046→    // receipt (defensive re-verification of the document chain), with the
  1047→  1047→    // immutable receivedQty as the quantity authority (§34).
  1048→  1048→    const receiptLineIds = lines.map((l) => l.goodsReceiptLineId);
  1049→  1049→    const receiptLines = await tx.goodsReceiptLine.findMany({
  1050→  1050→      where: { id: { in: receiptLineIds } },
  1051→  1051→      select: { id: true, goodsReceiptId: true, receivedQty: true },
  1052→  1052→    });
  1053→  1053→    const receiptLineById = new Map(receiptLines.map((rl) => [rl.id, rl]));
  1054→  1054→
  1055→  1055→    // §43 step 10 / §37 — ONE grouped aggregate for ALL prior POSTED
  1056→  1056→    // returns of these source lines (never per-line queries, never an
  1057→  1057→    // unbounded history read).
  1058→  1058→    const returnedByLine = await sumPostedReturnedQtyForReceiptLines(tx, receiptLineIds);
  1059→  1059→
  1060→  1060→    const adjustmentIds: string[] = [];
  1061→  1061→    const stockChanges: { productId: string; from: number; to: number }[] = [];
  1062→  1062→
  1063→  1063→    for (const line of lines) {
  1064→  1064→      const sourceLine = receiptLineById.get(line.goodsReceiptLineId);
  1065→  1065→      if (!sourceLine || sourceLine.goodsReceiptId !== receipt.id) {
  1066→  1066→        // §43 step 9 / §142 — a detached source line is corruption; fail
  1067→  1067→        // safely (no partial posting), no existence leak.
  1068→  1068→        throw new ApiError(
  1069→  1069→          404,
  1070→  1070→          GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
  1071→  1071→          "Goods receipt line not found.",
  1072→  1072→        );
  1073→  1073→      }
  1074→  1074→
  1075→  1075→      // §43 step 11 / §35 — the HARD cumulative return cap, final
  1076→  1076→      // authority. Prior POSTED returns + this line ≤ receivedQty.
  1077→  1077→      const alreadyReturned = returnedByLine.get(line.goodsReceiptLineId) ?? 0;
  1078→  1078→      const returnable = computeReturnableQty(sourceLine.receivedQty, alreadyReturned); // §36
  1079→  1079→      if (exceedsReturnable(sourceLine.receivedQty, alreadyReturned, line.returnedQty)) {
  1080→  1080→        throw new ApiError(
  1081→  1081→          409,
  1082→  1082→          GOODS_RETURN_ERROR_CODES.EXCEEDS_RETURNABLE,
  1083→  1083→          `Only ${returnable} unit(s) remain returnable on this receipt line.`,
  1084→  1084→        );
  1085→  1085→      }
  1086→  1086→
  1087→  1087→      // §43 step 12 — reload the product (tenant-scoped; the product id
  1088→  1088→      // derives from the receipt line, §47 — never client authority).
  1089→  1089→      const product = await tx.product.findFirst({
  1090→  1090→        where: { id: line.productId, businessId: input.businessId }, // §67
  1091→  1091→        select: { id: true, stockQty: true },
  1092→  1092→      });
  1093→  1093→      if (!product) {
  1094→  1094→        // A hard-deleted product fails the post safely (never fabrication).
  1095→  1095→        throw new ApiError(
  1096→  1096→          422,
  1097→  1097→          GOODS_RETURN_ERROR_CODES.SOURCE_LINE_NOT_FOUND,
  1098→  1098→          "A product on this return no longer exists.",
  1099→  1099→        );
  1100→  1100→      }
  1101→  1101→
  1102→  1102→      // §41/§164 — corrupt/negative persisted stock is surfaced as the
  1103→  1103→      // controlled integrity failure BEFORE any sufficiency comparison:
  1104→  1104→      // a negative stock reading is corruption, not "insufficient stock",
  1105→  1105→      // and it is NEVER silently repaired or zeroed.
  1106→  1106→      if (!Number.isInteger(product.stockQty) || product.stockQty < 0) {
  1107→  1107→        throw new ApiError(
  1108→  1108→          500,
  1109→  1109→          GOODS_RETURN_ERROR_CODES.INVALID_PERSISTED_STOCK,
  1110→  1110→          "Product stock data is in an invalid state. Please contact support.",
  1111→  1111→        );
  1112→  1112→      }
  1113→  1113→
  1114→  1114→      // §39 — CURRENT STOCK CHECK: the physical decrement needs enough
  1115→  1115→      // aggregate stock NOW (the application has no lot/batch provenance —
  1116→  1116→      // documented limitation §40/§197).
  1117→  1117→      if (product.stockQty < line.returnedQty) {
  1118→  1118→        throw new ApiError(
  1119→  1119→          409,
  1120→  1120→          GOODS_RETURN_ERROR_CODES.INSUFFICIENT_STOCK,
  1121→  1121→          `Insufficient stock for ${line.productNameAtReturn}: ${product.stockQty} on hand, ${line.returnedQty} to return.`,
  1122→  1122→        );
  1123→  1123→      }
  1124→  1124→
  1125→  1125→      // §42 — exact integer arithmetic; stockAfterReturn re-validates both
  1126→  1126→      // operands and the non-negative result (defense in depth).
  1127→  1127→      const newStock = stockAfterReturn(product.stockQty, line.returnedQty);
  1128→  1128→
  1129→  1129→      // §43 steps 16 — decrement stock.
  1130→  1130→      await tx.product.update({
  1131→  1131→        where: { id: product.id },
  1132→  1132→        data: { stockQty: newStock },
  1133→  1133→        select: { id: true },
  1134→  1134→      });
  1135→  1135→      stockChanges.push({ productId: product.id, from: product.stockQty, to: newStock });
  1136→  1136→
  1137→  1137→      // §43 step 17 / §44-§47 — ONE negative InventoryAdjustment per line
  1138→  1138→      // (delta = -returnedQty), server-derived actor/business/product.
  1139→  1139→      const adjustment = await tx.inventoryAdjustment.create({
  1140→  1140→        data: {
  1141→  1141→          businessId: input.businessId, // §46 — server-derived
  1142→  1142→          productId: product.id, // §47 — derived from the receipt line chain
  1143→  1143→          delta: -line.returnedQty, // §44 — always negative
  1144→  1144→          reason: RETURN_ADJUSTMENT_REASON(header.reference), // §44 — stable identifier
  1145→  1145→          actorId: input.userId, // §45 — server-derived
  1146→  1146→        },
  1147→  1147→        select: { id: true },
  1148→  1148→      });
  1149→  1149→      adjustmentIds.push(adjustment.id);
  1150→  1150→    }
  1151→  1151→
  1152→  1152→    // §43 steps 18-19 — complete the posting (server timestamps only).
  1153→  1153→    const postedAt = new Date();
  1154→  1154→    const updated = await tx.goodsReturn.update({
  1155→  1155→      where: { id: header.id },
  1156→  1156→      data: { status: "POSTED", postedAt },
  1157→  1157→      select: { id: true, reference: true, status: true, postedAt: true },
  1158→  1158→    });
  1159→  1159→
  1160→  1160→    // §43 step 20 / §48/§133 — ONE audit record for the posting, written
  1161→  1161→    // in the SAME transaction; if it fails, everything above rolls back.
  1162→  1162→    const totalReturnedQty = computeReturnedQty(lines.map((l) => l.returnedQty));
  1163→  1163→    await audit(
  1164→  1164→      auditInput(input.businessId, input.userId, "goods_return.post", header.id, header.reference, {
  1165→  1165→        goodsReceiptId: receipt.id,
  1166→  1166→        goodsReceiptReferenceAtReturn: header.goodsReceiptReferenceAtReturn,
  1167→  1167→        lineCount: lines.length,
  1168→  1168→        totalReturnedQty, // §132 — quantity facts only; never notes
  1169→  1169→        adjustmentIds,
  1170→  1170→        previousStatus: header.status,
  1171→  1171→        newStatus: updated.status,
  1172→  1172→        postedAt: updated.postedAt?.toISOString() ?? null,
  1173→  1173→      }),
  1174→  1174→      tx,
  1175→  1175→    );
  1176→  1176→
  1177→  1177→    return {
  1178→  1178→      id: updated.id,
  1179→  1179→      reference: updated.reference,
  1180→  1180→      status: updated.status,
  1181→  1181→      postedAt: updated.postedAt,
  1182→  1182→      lineCount: lines.length,
  1183→  1183→      totalReturnedQty,
  1184→  1184→      stockChanges,
  1185→  1185→    };
  1186→  1186→  });
  1187→  1187→}

  1231→  1231→/**
  1232→  1232→ * §86/§87 — ADDITIVE per-receipt-line return facts for the GoodsReceipt
  1233→  1233→ * detail: receivedQty (immutable gross, §34), returnedQty (POSTED only,
  1234→  1234→ * §36) and returnableQty = received − returned. One grouped aggregate —
  1235→  1235→ * never one query per line (§141). `returnable` reports whether ANY line
  1236→  1236→ * still has returnableQty > 0 (the §87 CTA gate; client-advisory only —
  1237→  1237→ * the server revalidates at post time).
  1238→  1238→ */
  1239→  1239→export async function getGoodsReceiptReturnFacts(input: {
  1240→  1240→  businessId: string;
  1241→  1241→  receiptId: string;
  1242→  1242→  receivedLines: ReadonlyArray<{ id: string; receivedQty: number }>;
  1243→  1243→}): Promise<{
  1244→  1244→  returnable: boolean;
  1245→  1245→  lines: ReceiptLineReturnFactsDto[];
  1246→  1246→}> {
  1247→  1247→  const returnedByLine = await sumPostedReturnedQtyForReceiptLines(
  1248→  1248→    db,
  1249→  1249→    input.receivedLines.map((l) => l.id),
  1250→  1250→  );
  1251→  1251→  let returnable = false;
  1252→  1252→  const lines = input.receivedLines.map((l) => {
  1253→  1253→    const returnedQty = returnedByLine.get(l.id) ?? 0;
  1254→  1254→    // §36 — returned > received is corruption; surface it, never clamp.
  1255→  1255→    if (returnedQty > l.receivedQty) {
  1256→  1256→      return mapPureGoodsReturnDomainError(
  1257→  1257→        new GoodsReturnIntegrityError(
  1258→  1258→          `${GOODS_RETURN_CORRUPT_HISTORY}: returned ${returnedQty} > received ${l.receivedQty}`,
  1259→  1259→        ),
  1260→  1260→      );
  1261→  1261→    }
  1262→  1262→    const returnableQty = l.receivedQty - returnedQty;
  1263→  1263→    if (returnableQty > 0) returnable = true;
  1264→  1264→    return receiptLineReturnFactsDto({
  1265→  1265→      goodsReceiptLineId: l.id,
  1266→  1266→      receivedQty: l.receivedQty,
  1267→  1267→      returnedQty,
  1268→  1268→    });
  1269→  1269→  });
  1270→  1270→  return { returnable, lines };
  1271→  1271→}
