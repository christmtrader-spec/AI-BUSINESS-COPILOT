    67→function formatDate(iso: string | null, locale: string): string {
    68→  if (!iso) return "—";
    69→  try {
    70→    return new Intl.DateTimeFormat(locale === "fa" ? "fa-IR" : locale, {
    71→      dateStyle: "medium",
    72→      timeStyle: "short",
    73→    }).format(new Date(iso));
    74→  } catch {
    75→    return iso.slice(0, 10);
    76→  }
    77→}

    79→/** Display-only remaining trial days (server state stays the authority). */
    80→function trialDaysRemaining(trialEndIso: string): number {
    81→  const end = Date.parse(trialEndIso);
    82→  if (Number.isNaN(end)) return 0;
    83→  return Math.max(0, Math.ceil((end - Date.now()) / 86_400_000));
    84→}

    86→function StatusBadge({ status }: { status: string }) {
    87→  const t = useTranslations("billing");
    88→  const tone =
    89→    status === "ACTIVE"
    90→      ? "border-emerald-500/30 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400"
    91→      : status === "TRIALING"
    92→        ? "border-amber-500/30 bg-amber-500/15 text-amber-700 dark:text-amber-400"
    93→        : status === "CANCELED"
    94→          ? "border-orange-500/30 bg-orange-500/15 text-orange-700 dark:text-orange-400"
    95→          : status === "PAST_DUE" || status === "PAUSED"
    96→            ? "border-yellow-500/30 bg-yellow-500/15 text-yellow-700 dark:text-yellow-400"
    97→            : "text-muted-foreground border-border bg-muted"; // EXPIRED
    98→  return (
    99→    <Badge variant="outline" className={cn("font-medium", tone)} data-testid="billing-status">
   100→      {t(`status_${status}` as "status_ACTIVE")}
   101→    </Badge>
   102→  );
   103→}

   105→function usagePercent(used: number, limit: number | null): number | null {
   106→  if (limit === null || limit <= 0) return null;
   107→  return Math.min(100, Math.round((used / limit) * 100));
   108→}

   115→function UsageMetricRow({
   116→  metric,
   117→  m,
   118→  periodEnd,
   119→  locale,
   120→}: {
   121→  metric: string;
   122→  m: UsageMetricPayload;
   123→  periodEnd: string | null;
   124→  locale: string;
   125→}) {
   126→  const tUsage = useTranslations("billing.usage");
   127→  const pct = m.percentageUsed;
   128→  const tone =
   129→    m.level === "reached"
   130→      ? "border-destructive/40 bg-destructive/10 text-destructive"
   131→      : m.level === "critical"
   132→        ? "border-orange-500/40 bg-orange-500/10 text-orange-700 dark:text-orange-400"
   133→        : m.level === "warning"
   134→          ? "border-amber-500/40 bg-amber-500/10 text-amber-700 dark:text-amber-400"
   135→          : null;
   136→  const barTone =
   137→    m.level === "reached"
   138→      ? "[&>div]:bg-destructive"
   139→      : m.level === "critical"
   140→        ? "[&>div]:bg-orange-500"
   141→        : m.level === "warning"
   142→          ? "[&>div]:bg-amber-500"
   143→          : "";
   144→  const badge =
   145→    m.level === "reached"
   146→      ? tUsage("reachedBadge")
   147→      : m.level === "critical"
   148→        ? tUsage("criticalBadge")
   149→        : m.level === "warning"
   150→          ? tUsage("warningBadge")
   151→          : null;
   152→  return (
   153→    <div className="space-y-1.5" data-testid={`usage-${metric}`}>
   154→      <div className="flex flex-wrap items-center justify-between gap-2 text-sm">
   155→        <span className="flex items-center gap-2">
   156→          {tUsage(`metric_${metric}` as "metric_AI_REQUESTS")}
   157→          {badge ? (
   158→            <Badge variant="outline" className={cn("font-medium", tone)}>
   159→              {badge}
   160→            </Badge>
   161→          ) : null}
   162→        </span>
   163→        {m.limit === null ? (
   164→          <span className="text-muted-foreground font-medium">
   165→            {tUsage("usedOnly", { used: m.used })} · {tUsage("unlimitedMetric")}
   166→          </span>
   167→        ) : (
   168→          <span className="text-muted-foreground font-medium">
   169→            {tUsage("usedOf", { used: m.used, limit: m.limit })} ·{" "}
   170→            <span>
   171→              {m.remaining === 0
   172→                ? tUsage("remainingNone")
   173→                : tUsage("remaining", { count: m.remaining ?? 0 })}
   174→            </span>
   175→          </span>
   176→        )}
   177→      </div>
   178→      {pct !== null ? (
   179→        <Progress
   180→          value={pct}
   181→          aria-label={tUsage(`metric_${metric}` as "metric_AI_REQUESTS")}
   182→          className={cn("h-1.5", barTone)}
   183→        />
   184→      ) : null}
   185→      {m.level === "reached" ? (
   186→        <p className="text-destructive flex items-start gap-1.5 text-xs">
   187→          <Info className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   188→          {periodEnd
   189→            ? tUsage("reachedNotice", { date: formatDate(periodEnd, locale) })
   190→            : tUsage("reachedNoticeNoDate")}
   191→        </p>
   192→      ) : null}
   193→    </div>
   194→  );
   195→}

   551→                        >
   552→                          {t("overLimit")}
   553→                        </Badge>
   554→                      ) : at ? (
   555→                        <Badge
   556→                          variant="outline"
   557→                          className="border-amber-500/40 bg-amber-500/10 text-amber-700 dark:text-amber-400"
   558→                        >
   559→                          {t("atLimit")}
   560→                        </Badge>
   561→                      ) : null}
   562→                    </span>
   563→                    {e.type === "BOOLEAN" ? (
   564→                      e.value === true ? (
   565→                        <Badge
   566→                          variant="outline"
   567→                          className="border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   568→                        >
   569→                          <BadgeCheck className="size-3.5" aria-hidden="true" />
   570→                          {t("included")}
   571→                        </Badge>
   572→                      ) : (
   573→                        <Badge variant="outline" className="text-muted-foreground">
   574→                          {t("notIncluded")}
   575→                        </Badge>
   576→                      )
   577→                    ) : (
   578→                      <span className="text-muted-foreground font-medium">
   579→                        {e.value === null
   580→                          ? t("unlimited")
   581→                          : u
   582→                            ? t("limitOf", { used: u.used, limit: e.value as number })
   583→                            : t("limitOf", { used: 0, limit: e.value as number })}
   584→                      </span>
   585→                    )}
   586→                  </div>
   587→                  {pct !== null ? (
   588→                    <Progress
   589→                      value={pct}
   590→                      aria-label={t(`entitlement.${e.code}` as "entitlement.CUSTOMERS_MAX")}
   591→                      className={cn("h-1.5", over && "[&>div]:bg-destructive")}

   594→                </div>

   602→          </p>

   626→                  <span
   627→                    className="bg-primary/10 text-primary mt-1 size-2 shrink-0 rounded-full"
   628→                    aria-hidden="true"
   629→                  />
   630→                  <div className="min-w-0 flex-1">
   631→                    <p className="font-medium">
   632→                      {t(`event_${ev.type}` as "event_CREATED")}

   635→                          {" "}

   638→                      ) : null}
   639→                    </p>
   640→                    <p className="text-muted-foreground text-xs">
   641→                      {formatDate(ev.createdAt, locale)}
   642→                      {ev.actor?.name ? ` · ${ev.actor.name}` : ""}
   643→                    </p>
   644→                  </div>
   645→                </li>
   646→              ))}
   647→            </ol>
   648→          )}
   649→        </CardContent>
   650→      </Card>
   651→
   652→      {/* ---- Plan dialog (Prompt 22: prices + payment path + internal switch) -- */}
   653→      <AlertDialog open={planDialogOpen} onOpenChange={setPlanDialogOpen}>
   654→        <AlertDialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">

   657→            <AlertDialogDescription>{t("changePlanDesc")}</AlertDialogDescription>
   658→          </AlertDialogHeader>
   659→          <div className="space-y-2" data-testid="billing-plan-options">
   660→            {(catalogQ.data?.plans ?? [])
   661→              .filter((p) => p.selfService)

   663→                const isCurrent = p.code === sub.plan.code;
   664→                const priceLine = p.pricing
   665→                  ? {

   670→                return (
   671→                  <div
   672→                    key={p.code}
   673→                    data-testid={`plan-option-${p.code}`}
   674→                    className={cn(
   675→                      "w-full rounded-lg border p-3 text-start transition-colors",
   676→                      isCurrent
   677→                        ? "border-primary/40 bg-primary/5"
   678→                        : "hover:bg-accent/50",
   679→                    )}
   680→                  >
   681→                    <div className="flex items-center justify-between gap-2">
   682→                      <span className="text-sm font-semibold">{p.name}</span>
   683→                      {isCurrent ? (

   685→                      ) : priceLine ? (

   688→                            {t("checkout.perMonth", { amount: priceLine.monthly })}
   689→                          </span>
   690→                          <span aria-hidden="true">·</span>
   691→                          <span>
   692→                            {t("checkout.perYear", { amount: priceLine.yearly })} (
   693→                            {t("checkout.yearlySaving")})
   694→                          </span>
   695→                        </span>
   696→                      ) : (
   697→                        <span className="text-muted-foreground text-xs font-medium">
   698→                          {t("checkout.free")}
   699→                        </span>
   700→                      )}
   701→                    </div>
   702→                    <p className="text-muted-foreground mt-1 text-xs">{p.description}</p>
   703→                    {!isCurrent && canManage ? (
   704→                      <div className="mt-2 flex flex-wrap items-center gap-2">
   705→                        {priceLine ? (

   707→                            <Button

   709→                              onClick={() => startCheckout(p.code, p.name, "MONTHLY")}

   719→                              size="sm"
   720→                              variant="outline"
   721→                              onClick={() => startCheckout(p.code, p.name, "YEARLY")}
   722→                              disabled={createCheckout.isPending}
   723→                              data-testid={`checkout-yearly-${p.code}`}
   724→                            >
   725→                              {t("checkout.periodYearly")}
   726→                            </Button>
   727→                          </>
   728→                        ) : null}
   729→                        <Button
   730→                          size="sm"
   731→                          variant="ghost"
   732→                          className="text-muted-foreground"
   733→                          onClick={() => onSwitch(p.code, p.name)}

   737→                          {t("changePlan")}
   738→                        </Button>
   739→                      </div>
   740→                    ) : null}
   741→                  </div>
   742→                );
   743→              })}
   744→          </div>
   745→          <p className="text-muted-foreground flex items-center gap-1.5 text-xs">
   746→            <Info className="size-3.5 shrink-0" aria-hidden="true" />
   747→            {t("priceNote")}
   748→          </p>
   749→          <AlertDialogFooter>
   750→            <AlertDialogCancel disabled={changePlan.isPending || createCheckout.isPending}>
   751→              {tCommon("cancel")}
   752→            </AlertDialogCancel>
   753→          </AlertDialogFooter>
   754→        </AlertDialogContent>
   755→      </AlertDialog>
   756→
   757→      {/* ---- Prompt 22: sandbox checkout + payment history ---------------------- */}
   758→      <CheckoutDialog
   759→        checkoutId={checkout?.id ?? null}
   760→        planName={checkout?.planName ?? ""}
   761→        onClose={() => setCheckout(null)}
   762→      />
   763→      <PaymentsHistoryCard businessId={business.id} />
   764→
   765→      {/* ---- Cancel confirmation ------------------------------------------------ */}
   766→      <AlertDialog open={cancelDialogOpen} onOpenChange={setCancelDialogOpen}>
   767→        <AlertDialogContent>
   768→          <AlertDialogHeader>
   769→            <AlertDialogTitle>{t("cancelTitle")}</AlertDialogTitle>

   771→              {t("cancelDesc", { date: formatDate(sub.currentPeriodEnd, locale) })}
   772→            </AlertDialogDescription>
   773→          </AlertDialogHeader>
   774→          <AlertDialogFooter>
   775→            <AlertDialogCancel disabled={cancelSub.isPending}>{tCommon("cancel")}</AlertDialogCancel>
   776→            <AlertDialogAction
   777→              onClick={(e) => {
   778→                e.preventDefault();
   779→                onCancel();
   780→              }}
   781→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   782→              data-testid="billing-cancel-confirm"
   783→            >
   784→              {cancelSub.isPending ? (
   785→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
