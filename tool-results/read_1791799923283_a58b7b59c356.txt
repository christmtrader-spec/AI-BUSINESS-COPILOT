    94→async function safe<T>(fn: () => Promise<T>): Promise<{ status: number; code: string; value?: T }> {
    95→  try {
    96→    const value = await fn();
    97→    return { status: 200, code: "", value };
    98→  } catch (error) {
    99→    const e = error as { status?: number; code?: string };
   100→    return { status: e.status ?? 500, code: e.code ?? String(error) };
   101→  }
   102→}

   366→      expireRes.status === "rejected" && ["SALES_QUOTATION_INVALID_TRANSITION", "SALES_QUOTATION_NOT_EXPIRED"].includes(codeOf((expireRes as PromiseRejectedResult).reason)),

   375→  /* -------- 8. concurrent duplicate references (§10) -------- */

   387→  /* ------- 9. conversion failure rollback (§13/H) ------- */

   391→  // in-transaction conversion guard then fails AFTER the transaction began
